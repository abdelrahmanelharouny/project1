   123→function ProviderStatusDot({ provider }: { provider: ProviderHealth }) {
   124→  const t = useTranslations('observations')
   125→  const probe = provider.probe?.status
   126→  let dotClass = 'bg-emerald-600'
   127→  let label = t('statusAvailable')
   128→  if (provider.status === 'optional' || provider.status === 'degraded') {
   129→    dotClass = 'bg-amber-500'
   130→    label = provider.status
   131→  }
   132→  if (probe === 'unreachable') {
   133→    dotClass = 'bg-red-500'
   134→    label = t('probeUnreachable')
   135→  } else if (probe === 'degraded') {
   136→    dotClass = 'bg-amber-500'
   137→    label = t('probeDegraded')
   138→  } else if (probe === 'ok') {
   139→    dotClass = 'bg-emerald-600'
   140→    label = t('probeLive', { ms: provider.probe?.latency_ms ?? '?' })
   141→  }
   142→  return (
   143→    <span className="inline-flex items-center gap-1.5 text-[11px] text-stone-500">
   144→      <span className={`inline-block h-2 w-2 rounded-full ${dotClass}`} aria-hidden />
   145→      {label}
   146→    </span>
   147→  )
