   334→# -------------------------------------------------------- background jobs
   335→class BackgroundJob(Base, TimestampMixin):
   336→    """Queue-level job record for work executed outside the HTTP request.
   337→
   338→    One row per submitted task. Domain tables (e.g. ``ingestion_jobs``) keep
   339→    their own domain-specific state; a BackgroundJob row references the domain
   340→    row via ``ref_kind``/``ref_id`` so clients can poll a single unified
   341→    endpoint (``GET /api/v1/jobs/background``) regardless of job kind.
   342→
   343→    Lifecycle: queued -> running -> succeeded | failed. Rows are never
   344→    hard-deleted (audit/provenance). Jobs left 'running' by a process crash
   345→    are marked failed('interrupted') on the next startup — only 'queued'
   346→    jobs are re-enqueued.
   347→    """
   348→
   349→    __tablename__ = "background_jobs"
   350→
   351→    id: Mapped[uuid.UUID] = mapped_column(UUID(as_uuid=True), primary_key=True, default=_uuid)
   352→    kind: Mapped[str] = mapped_column(String(60), nullable=False)
   353→    status: Mapped[str] = mapped_column(String(20), nullable=False, default="queued")
   354→    progress: Mapped[int] = mapped_column(Integer, nullable=False, default=0)
   355→    stage: Mapped[str | None] = mapped_column(String(200), nullable=True)
   356→    ref_kind: Mapped[str | None] = mapped_column(String(60), nullable=True)
   357→    ref_id: Mapped[uuid.UUID | None] = mapped_column(UUID(as_uuid=True), nullable=True)
   358→    label: Mapped[str | None] = mapped_column(Text, nullable=True)
   359→    result: Mapped[dict[str, Any] | None] = mapped_column(JSONB, nullable=True)
   360→    error: Mapped[str | None] = mapped_column(Text, nullable=True)
   361→    request_id: Mapped[str | None] = mapped_column(String(64), nullable=True)
   362→    created_by: Mapped[uuid.UUID | None] = mapped_column(UUID(as_uuid=True), nullable=True)
   363→    started_at: Mapped[datetime | None] = mapped_column(DateTime(timezone=True), nullable=True)
   364→    finished_at: Mapped[datetime | None] = mapped_column(DateTime(timezone=True), nullable=True)
   365→
   366→    __table_args__ = (
   367→        CheckConstraint(
   368→            "status IN ('queued','running','succeeded','failed','cancelled')",
   369→            name="ck_background_job_status",
   370→        ),
   371→        CheckConstraint("progress >= 0 AND progress <= 100", name="ck_background_job_progress"),
   372→        Index("ix_background_jobs_status", status),
   373→        Index("ix_background_jobs_ref", ref_kind, ref_id),
   374→        Index("ix_background_jobs_created_by", created_by),
   375→    )
