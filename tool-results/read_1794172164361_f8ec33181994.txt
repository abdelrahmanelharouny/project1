     1→"""Identity domain: registration, login, token auth, tenant authorization.
     2→
     3→Tenant rule (Master rule 17 / §17.1): every farm/field/data resource access is
     4→authorized against the requesting user's organization membership.
     5→
     6→Auth hardening (P0-2): account lockout after repeated failed logins, email
     7→verification gate, refresh-token rotation with family reuse detection,
     8→single-use credential challenge tokens (password reset / email verification)
     9→and role-based membership management (viewer < agronomist < manager < owner).

    51→# RBAC hierarchy (Master §17): the numeric order defines "at least as
    52→# privileged as" comparisons for role gates.
    53→ROLE_ORDER: dict[str, int] = {"viewer": 0, "agronomist": 1, "manager": 2, "owner": 3}
    54→
    55→
    56→def role_at_least(role_code: str, minimum: str) -> bool:
    57→    """True when `role_code` grants at least the privileges of `minimum`."""
    58→    if role_code not in ROLE_ORDER or minimum not in ROLE_ORDER:
    59→        return False
    60→    return ROLE_ORDER[role_code] >= ROLE_ORDER[minimum]

    63→class Principal:
    64→    """Authenticated user + their memberships, resolved once per request."""
    65→
    66→    def __init__(self, user: User, memberships: list[Membership]) -> None:
    67→        self.user = user
    68→        self.memberships = memberships
    69→        self._org_ids = {m.organization_id for m in memberships if m.status == "active"}
    70→
    71→    @property
    72→    def id(self) -> uuid.UUID:
    73→        return self.user.id
    74→
    75→    def is_member_of(self, organization_id: uuid.UUID) -> bool:
    76→        return organization_id in self._org_ids
    77→
    78→    def require_member_of(self, organization_id: uuid.UUID) -> None:
    79→        if not self.is_member_of(organization_id):
    80→            raise Forbidden("You do not have access to this organization's resources.")
    81→
    82→    def org_ids(self) -> set[uuid.UUID]:
    83→        return set(self._org_ids)
    84→
    85→    def membership_in(self, organization_id: uuid.UUID) -> Membership | None:
    86→        """The principal's ACTIVE membership in the organization, or None."""
    87→        for membership in self.memberships:
    88→            if membership.organization_id == organization_id and membership.status == "active":
    89→                return membership
    90→        return None
    91→
    92→    def require_org_role(self, organization_id: uuid.UUID, minimum: str) -> Membership:
    93→        """Return the active membership after verifying the minimum role."""
    94→        membership = self.membership_in(organization_id)
    95→        if membership is None:
    96→            raise Forbidden("You do not have access to this organization's resources.")
    97→        if not role_at_least(membership.role_code, minimum):
    98→            raise Forbidden(
    99→                f"This action requires the '{minimum}' role or higher on this organization."
   100→            )
   101→        return membership
   102→
   103→    def require_any_org_role(self, minimum: str) -> Membership:
   104→        """Role gate for GLOBAL-context mutations (no single organization
   105→        scope, e.g. platform-level knowledge packs): require at least one
   106→        active membership with the minimum role in ANY of the principal's
   107→        organizations. Tenant-scoped endpoints should use require_org_role
   108→        or require_farm_role instead."""
   109→        for membership in self.memberships:
   110→            if membership.status == "active" and role_at_least(membership.role_code, minimum):
   111→                return membership
   112→        raise Forbidden(
   113→            f"This action requires the '{minimum}' role or higher on at least one organization."
   114→        )

   117→def get_principal(
   118→    credentials: HTTPAuthorizationCredentials | None = Depends(bearer_scheme),
   119→    db: Session = Depends(get_db),
   120→) -> Principal:
   121→    if credentials is None or credentials.scheme.lower() != "bearer":
   122→        raise Unauthorized("Missing bearer token.")
   123→    token_hash = hash_token(credentials.credentials)
   124→    row = db.execute(
   125→        select(AuthToken, User)
   126→        .join(User, AuthToken.user_id == User.id)
   127→        .where(AuthToken.token_hash == token_hash, AuthToken.kind == "access")
   128→    ).first()
   129→    if row is None:
   130→        raise Unauthorized("Invalid or expired token.")
   131→    token, user = row
   132→    now = datetime.now(UTC)
   133→    if token.revoked_at is not None or token.expires_at < now:
   134→        raise Unauthorized("Token expired or revoked.")
   135→    if not user.is_active:
   136→        raise Unauthorized("Account disabled.")
   137→    memberships = list(
   138→        db.scalars(select(Membership).where(Membership.user_id == user.id)).all()
   139→    )
   140→    return Principal(user=user, memberships=memberships)

   167→# ------------------------------------------------------- credential tokens
   168→def _issue_credential_token(db: Session, user_id: uuid.UUID, kind: str) -> tuple[str, datetime]:
   169→    """Create a single-use CredentialToken row; returns (plain_token, expires_at).
   170→
   171→    Only the SHA-256 hash is stored; the plaintext is delivered out-of-band
   172→    by the caller (email in production, dev convenience field in sandboxes).
   173→    """
   174→    ttl = (
   175→        settings.password_reset_ttl_seconds
   176→        if kind == "password_reset"
   177→        else settings.email_verification_ttl_seconds
   178→    )
   179→    plain, token_hash, expires_at = generate_token(ttl_seconds=ttl)
   180→    db.add(
   181→        CredentialToken(user_id=user_id, kind=kind, token_hash=token_hash, expires_at=expires_at)
   182→    )
   183→    return plain, expires_at
   184→
   185→
   186→def _issue_token_pair(db: Session, user_id: uuid.UUID, family_id: uuid.UUID | None) -> tuple[str, datetime, str, datetime]:
   187→    """Issue an access+refresh AuthToken pair; returns (access, access_exp, refresh, refresh_exp)."""
   188→    access, access_hash, access_expires_at = generate_token()
   189→    db.add(
   190→        AuthToken(
   191→            user_id=user_id,
   192→            token_hash=access_hash,
   193→            expires_at=access_expires_at,
   194→            kind="access",
   195→            family_id=family_id,
   196→        )
   197→    )
   198→    refresh, refresh_hash, refresh_expires_at = generate_refresh_token()
   199→    db.add(
   200→        AuthToken(
   201→            user_id=user_id,
   202→            token_hash=refresh_hash,
   203→            expires_at=refresh_expires_at,
   204→            kind="refresh",
   205→            family_id=family_id,
   206→        )
   207→    )
   208→    return access, access_expires_at, refresh, refresh_expires_at

   210→
   211→def _revoke_family(db: Session, family_id: uuid.UUID | None, now: datetime) -> None:
   212→    """Revoke every not-yet-revoked token in a refresh rotation family."""
   213→    if family_id is None:
   214→        return
   215→    rows = db.scalars(
   216→        select(AuthToken).where(
   217→            AuthToken.family_id == family_id, AuthToken.revoked_at.is_(None)
   218→        )
   219→    ).all()
   220→    for token in rows:
   221→        token.revoked_at = now

   224→def register_user(
   225→    db: Session, *, email: str, password: str, display_name: str
   226→) -> tuple[User, Organization, str, datetime, str, datetime]:
   227→    """Create a user + personal organization and issue an access/refresh pair.
   228→
   229→    Returns (user, org, access_token, access_expires_at, refresh_token,
   230→    refresh_expires_at). When NOT dev_mode an email-verification challenge is
   231→    created; its plaintext token is never returned to clients.
   232→    """
   233→    email_normalized = email.strip().lower()
   234→    existing = db.scalar(select(User).where(func.lower(User.email) == email_normalized))
   235→    if existing is not None:
   236→        raise Conflict("An account with this email already exists.")
   237→    user = User(
   238→        email=email_normalized,
   239→        password_hash=hash_password(password),
   240→        display_name=display_name.strip() or email_normalized.split("@")[0],
   241→        email_verified=settings.dev_mode,
   242→    )
   243→    db.add(user)
   244→    org = Organization(name=f"{user.display_name}'s organization")
   245→    db.add(org)
   246→    db.flush()
   247→    db.add(Membership(organization_id=org.id, user_id=user.id, role_code="owner", status="active"))
   248→    db.flush()
   249→    family_id = uuid.uuid4()
   250→    access, access_expires_at, refresh, refresh_expires_at = _issue_token_pair(
   251→        db, user.id, family_id
   252→    )
   253→    if not settings.dev_mode:
   254→        # Unverified deployments: create the verification challenge. The plain
   255→        # token goes out-of-band (email); it is NEVER returned to clients.
   256→        _issue_credential_token(db, user.id, "email_verification")
   257→    record_auth_audit(
   258→        db,
   259→        actor_id=user.id,
   260→        organization_id=org.id,
   261→        action="auth.register",
   262→        entity_id=str(user.id),
   263→        meta={"email": user.email, "email_verified": user.email_verified},
   264→    )
   265→    db.commit()
   266→    return user, org, access, access_expires_at, refresh, refresh_expires_at

   269→def login_user(
   270→    db: Session, *, email: str, password: str
   271→) -> tuple[User, str, datetime, str, datetime]:
   272→    """Authenticate a user; returns (user, access, access_exp, refresh, refresh_exp).
   273→
   274→    Hardening: temporary lockout after `auth_max_failed_logins` consecutive
   275→    failures, email-verification gate, audit trail on every outcome.
   276→    """
   277→    email_normalized = email.strip().lower()
   278→    user = db.scalar(select(User).where(func.lower(User.email) == email_normalized))
   279→    now = datetime.now(UTC)
   280→    if user is not None and user.locked_until is not None and user.locked_until > now:
   281→        retry_after = max(int((user.locked_until - now).total_seconds()), 1)
   282→        record_auth_audit(
   283→            db,
   284→            actor_id=user.id,
   285→            action="auth.login.locked",
   286→            entity_id=email_normalized,
   287→            meta={"retry_after_seconds": retry_after},
   288→        )
   289→        db.commit()
   290→        raise AccountLocked(
   291→            "Account temporarily locked due to failed sign-in attempts.",
   292→            retry_after_seconds=retry_after,
   293→        )
   294→    if user is None or not verify_password(password, user.password_hash):
   295→        if user is not None:
   296→            user.failed_login_count = (user.failed_login_count or 0) + 1
   297→            if user.failed_login_count >= settings.auth_max_failed_logins:
   298→                user.locked_until = now + timedelta(seconds=settings.auth_lockout_seconds)
   299→                user.failed_login_count = 0
   300→            record_auth_audit(
   301→                db,
   302→                actor_id=user.id,
   303→                action="auth.login.failure",
   304→                entity_id=email_normalized,
   305→            )
   306→        else:
   307→            # Unknown email: audit with no actor (no account enumeration side
   308→            # channel — identical client-visible response either way).
   309→            record_auth_audit(
   310→                db,
   311→                actor_id=None,
   312→                action="auth.login.failure",
   313→                entity_id=email_normalized,
   314→            )
   315→        db.commit()
   316→        raise Unauthorized("Invalid email or password.")
   317→    # Correct credentials: reset the lockout counters.
   318→    user.failed_login_count = 0
   319→    user.locked_until = None
   320→    if not user.is_active:
   321→        db.commit()
   322→        raise Unauthorized("Account disabled.")
   323→    if not user.email_verified:
   324→        record_auth_audit(
   325→            db,
   326→            actor_id=user.id,
   327→            action="auth.login.unverified",
   328→            entity_id=email_normalized,
   329→        )
   330→        db.commit()
   331→        raise EmailNotVerified("Verify your email before signing in.")
   332→    family_id = uuid.uuid4()
   333→    access, access_expires_at, refresh, refresh_expires_at = _issue_token_pair(
   334→        db, user.id, family_id
   335→    )
   336→    first_membership = db.scalar(
   337→        select(Membership)
   338→        .where(Membership.user_id == user.id, Membership.status == "active")
   339→        .order_by(Membership.created_at)
   340→    )
   341→    record_auth_audit(
   342→        db,
   343→        actor_id=user.id,
   344→        organization_id=first_membership.organization_id if first_membership else None,
   345→        action="auth.login.success",
   346→        entity_id=str(user.id),
   347→    )
   348→    db.commit()
   349→    return user, access, access_expires_at, refresh, refresh_expires_at

   352→def rotate_refresh_token(
   353→    db: Session, refresh_plain: str
   354→) -> tuple[User, str, datetime, str, datetime]:
   355→    """Rotate a refresh token: returns (user, access, access_exp, refresh, refresh_exp).
   356→
   357→    Reuse of an already-revoked refresh token is treated as a compromised
   358→    session family: every token in the family is revoked and the request is
   359→    rejected. Refresh tokens never authenticate API calls (kind=access filter
   360→    in get_principal) — they only mint new token pairs here.
   361→    """
   362→    token_hash = hash_token(refresh_plain)
   363→    token = db.scalar(
   364→        select(AuthToken).where(
   365→            AuthToken.token_hash == token_hash, AuthToken.kind == "refresh"
   366→        )
   367→    )
   368→    if token is None:
   369→        raise Unauthorized("Invalid refresh token.")
   370→    user = db.get(User, token.user_id)
   371→    if user is None:  # pragma: no cover — FK guarantees the user row
   372→        raise Unauthorized("Invalid refresh token.")
   373→    now = datetime.now(UTC)
   374→    if token.revoked_at is not None:
   375→        # REUSE DETECTED: a rotated (revoked) token came back. Revoke the
   376→        # whole family — all tokens minted from this login/rotation chain.
   377→        _revoke_family(db, token.family_id, now)
   378→        record_auth_audit(
   379→            db,
   380→            actor_id=user.id,
   381→            action="auth.token.reuse_detected",
   382→            entity_id=str(token.family_id) if token.family_id else None,
   383→        )
   384→        db.commit()
   385→        raise Unauthorized("Refresh token reuse detected — session family revoked.")
   386→    if token.expires_at < now:
   387→        token.revoked_at = now
   388→        db.commit()
   389→        raise Unauthorized("Refresh token expired.")
   390→    family_id = token.family_id or uuid.uuid4()
   391→    token.revoked_at = now
   392→    access, access_expires_at, refresh, refresh_expires_at = _issue_token_pair(
   393→        db, user.id, family_id
   394→    )
   395→    record_auth_audit(
   396→        db,
   397→        actor_id=user.id,
   398→        action="auth.token.refresh",
   399→        entity_id=str(family_id),
   400→    )
   401→    db.commit()
   402→    return user, access, access_expires_at, refresh, refresh_expires_at

   405→# ------------------------------------------------------- password reset
   406→def request_password_reset(db: Session, email: str) -> str | None:
   407→    """Create a password-reset challenge; returns the plaintext token or None.
   408→
   409→    Unknown emails return None WITHOUT creating a token or audit row — no
   410→    account enumeration side channel. The audit trail only records that a
   411→    reset was requested for an existing account.
   412→    """
   413→    email_normalized = email.strip().lower()
   414→    user = db.scalar(select(User).where(func.lower(User.email) == email_normalized))
   415→    if user is None:
   416→        return None
   417→    plain, _expires_at = _issue_credential_token(db, user.id, "password_reset")
   418→    record_auth_audit(
   419→        db,
   420→        actor_id=user.id,
   421→        action="auth.password.reset_requested",
   422→        entity_id=email_normalized,
   423→    )
   424→    db.commit()
   425→    return plain

   428→def confirm_password_reset(db: Session, token_plain: str, new_password: str) -> User:
   429→    """Consume a password-reset challenge, set the new password and revoke
   430→    every live session token of the user."""
   431→    token_hash = hash_token(token_plain)
   432→    row = db.scalar(
   433→        select(CredentialToken).where(
   434→            CredentialToken.token_hash == token_hash,
   435→            CredentialToken.kind == "password_reset",
   436→        )
   437→    )
   438→    now = datetime.now(UTC)
   439→    if row is None or row.used_at is not None or row.expires_at < now:
   440→        raise ValidationFailed("Invalid or expired reset token.")
   441→    row.used_at = now
   442→    user = db.get(User, row.user_id)
   443→    user.password_hash = hash_password(new_password)
   444→    live_tokens = db.scalars(
   445→        select(AuthToken).where(
   446→            AuthToken.user_id == user.id, AuthToken.revoked_at.is_(None)
   447→        )
   448→    ).all()
   449→    for token in live_tokens:
   450→        token.revoked_at = now
   451→    record_auth_audit(
   452→        db,
   453→        actor_id=user.id,
   454→        action="auth.password.reset_confirmed",
   455→        entity_id=str(user.id),
   456→    )
   457→    db.commit()
   458→    return user

   461→# ---------------------------------------------------- email verification
   462→def request_email_verification(db: Session, email: str) -> str | None:
   463→    """Create an email-verification challenge; returns the plaintext token.
   464→
   465→    Unknown emails and already-verified accounts return None (no new
   466→    challenge, no enumeration side channel).
   467→    """
   468→    email_normalized = email.strip().lower()
   469→    user = db.scalar(select(User).where(func.lower(User.email) == email_normalized))
   470→    if user is None or user.email_verified:
   471→        return None
   472→    plain, _expires_at = _issue_credential_token(db, user.id, "email_verification")
   473→    record_auth_audit(
   474→        db,
   475→        actor_id=user.id,
   476→        action="auth.email.verification_requested",
   477→        entity_id=email_normalized,
   478→    )
   479→    db.commit()
   480→    return plain

   483→def confirm_email_verification(db: Session, token_plain: str) -> User:
   484→    """Consume an email-verification challenge and mark the account verified."""
   485→    token_hash = hash_token(token_plain)
   486→    row = db.scalar(
   487→        select(CredentialToken).where(
   488→            CredentialToken.token_hash == token_hash,
   489→            CredentialToken.kind == "email_verification",

   493→    if row is None or row.used_at is not None or row.expires_at < now:

   499→        db,
   500→        actor_id=user.id,
   501→        action="auth.email.verified",
   502→        entity_id=str(user.id),

   504→    db.commit()

   545→def _count_other_active_owners(
   546→    db: Session, organization_id: uuid.UUID, excluded_membership_id: uuid.UUID
   547→) -> int:
   548→    return int(
   549→        db.scalar(
   550→            select(func.count())
   551→            .select_from(Membership)
   552→            .where(
   553→                Membership.organization_id == organization_id,
   554→                Membership.role_code == "owner",
   555→                Membership.status == "active",
   556→                Membership.id != excluded_membership_id,
   557→            )
   558→        )
   559→        or 0

   563→def add_member(
   564→    db: Session, principal: Principal, farm: Farm, *, email: str, role_code: str
   565→) -> Membership:
   566→    """Add an existing FieldShift account to the farm's organization (owner-only)."""
   567→    require_farm_role(db, principal, farm, "owner")
   568→    email_normalized = email.strip().lower()
   569→    user = db.scalar(select(User).where(func.lower(User.email) == email_normalized))
   570→    if user is None:
   571→        raise NotFound("No FieldShift account exists with this email.")
   572→    existing = db.scalar(
   573→        select(Membership).where(
   574→            Membership.organization_id == farm.organization_id,
   575→            Membership.user_id == user.id,
   576→        )
   577→    )
   578→    if existing is not None:
   579→        raise Conflict("This user is already a member of this organization.")
   580→    if role_code not in ROLE_ORDER:
   581→        raise ValidationFailed(
   582→            f"Unknown role '{role_code}'.",
   583→            details=[{"field": "role_code", "reason": role_code}],
   584→        )
   585→    membership = Membership(
   586→        organization_id=farm.organization_id,
   587→        user_id=user.id,
   588→        role_code=role_code,
   589→        status="active",
   590→    )
   591→    db.add(membership)
   592→    db.flush()
   593→    record_auth_audit(
   594→        db,
   595→        actor_id=principal.id,
   596→        organization_id=farm.organization_id,
   597→        action="membership.create",
   598→        entity_type="membership",
   599→        entity_id=str(membership.id),
   600→        meta={"user_id": str(user.id), "email": user.email, "role_code": role_code},
   601→    )
   602→    db.commit()
   603→    return membership

   606→def update_member_role(
   607→    db: Session, principal: Principal, farm: Farm, membership_id: uuid.UUID, new_role: str
   608→) -> Membership:
   609→    """Change a member's role (owner-only) with a last-owner guard."""
   610→    require_farm_role(db, principal, farm, "owner")
   611→    if new_role not in ROLE_ORDER:
   612→        raise ValidationFailed(
   613→            f"Unknown role '{new_role}'.",
   614→            details=[{"field": "role_code", "reason": new_role}],
   615→        )
   616→    membership = db.scalar(
   617→        select(Membership).where(
   618→            Membership.id == membership_id,
   619→            Membership.organization_id == farm.organization_id,
   620→        )
   621→    )
   622→    if membership is None:
   623→        raise NotFound("Membership not found.")
   624→    # LAST-OWNER GUARD: the organization must always keep at least one
   625→    # active owner, otherwise no one could manage members again.
   626→    if membership.role_code == "owner" and new_role != "owner":
   627→        if _count_other_active_owners(db, farm.organization_id, membership.id) < 1:
   628→            raise Conflict(
   629→                "Cannot demote the last owner of this organization — promote another "
   630→                "member to owner first."
   631→            )
   632→    membership.role_code = new_role
   633→    record_auth_audit(
   634→        db,
   635→        actor_id=principal.id,
   636→        organization_id=farm.organization_id,
   637→        action="membership.update",
   638→        entity_type="membership",
   639→        entity_id=str(membership.id),
   640→        meta={"user_id": str(membership.user_id), "new_role": new_role},
   641→    )
   642→    db.commit()
   643→    return membership

   646→def remove_member(db: Session, principal: Principal, farm: Farm, membership_id: uuid.UUID) -> None:
   647→    """Remove a member from the farm's organization (owner-only, last-owner guard)."""
   648→    require_farm_role(db, principal, farm, "owner")
   649→    membership = db.scalar(
   650→        select(Membership).where(
   651→            Membership.id == membership_id,
   652→            Membership.organization_id == farm.organization_id,
   653→        )
   654→    )
   655→    if membership is None:
   656→        raise NotFound("Membership not found.")
   657→    if membership.role_code == "owner":
   658→        if _count_other_active_owners(db, farm.organization_id, membership.id) < 1:
   659→            raise Conflict(
   660→                "Cannot remove the last owner of this organization — promote another "
   661→                "member to owner first."
   662→            )
   663→    audit_meta = {"user_id": str(membership.user_id), "role_code": membership.role_code}
   664→    db.delete(membership)
   665→    record_auth_audit(
   666→        db,
   667→        actor_id=principal.id,
   668→        organization_id=farm.organization_id,
   669→        action="membership.delete",
   670→        entity_type="membership",
   671→        entity_id=str(membership_id),
   672→        meta=audit_meta,
   673→    )
   674→    db.commit()
