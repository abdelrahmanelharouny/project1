     1→"""Report job lifecycle service (P3 W5) — synchronous, Session-based.
     2→
     3→Flow: ``create_report_job`` validates access + entitlements and queues a
     4→``report_jobs`` row → an executor (inline for small formats, or the
     5→integrator's background worker via ``core.jobs``) runs
     6→``execute_report_job`` which gathers data section-by-section (each section
     7→wrapped — a failing data source degrades to ``provenance: "missing"``
     8→instead of failing the whole report), renders bytes with the pure renderers,
     9→writes the artifact under ``settings.reports_dir`` and finalizes the row.
    10→
    11→Honesty contract (P3 §6.2): every section/KPI carries provenance; missing
    12→data is reported as missing, never fabricated. The ``title`` argument on
    13→create is recorded on the activity timeline (the current ``report_jobs``
    14→schema has no title column); the rendered title comes from the report-type
    15→registry (localized).
    16→
    17→Error hygiene: failed jobs store ``Type: message`` (truncated) — no SQL,
    18→no environment, no secrets.
    19→"""

   112→# ------------------------------------------------------------------ creation
   113→def create_report_job(
   114→    session: Session,
   115→    principal: Principal,
   116→    *,
   117→    report_type: str,
   118→    output_format: str,
   119→    locale: str = "en",
   120→    farm_id: uuid.UUID | None = None,
   121→    field_id: uuid.UUID | None = None,
   122→    organization_id: uuid.UUID | None = None,
   123→    date_from: date | None = None,
   124→    date_to: date | None = None,
   125→    title: str | None = None,
   126→) -> ReportJob:
   127→    """Validate, gate and queue a report job.
   128→
   129→    Access: the caller needs an ACTIVE membership with role >= viewer on the
   130→    target organization (explicitly via farm/field/org scope, or implicitly
   131→    when the principal belongs to exactly one organization).
   132→
   133→    Entitlements: when the target organization has a LIVE subscription the
   134→    ``exports_enabled`` flag and the ``reports_per_month`` quota are
   135→    enforced; without a subscription the gate is a pass-through (same rule
   136→    as the other P3 gates — free/unsubscribed orgs are not billed meters).
   137→    """
   138→    if report_type not in contracts.REPORT_TYPES:
   139→        raise ValidationFailed(
   140→            f"Unknown report type '{report_type}'.",
   141→            details=[{"field": "report_type", "reason": sorted(contracts.REPORT_TYPES)}],
   142→        )
   143→    if output_format not in contracts.REPORT_FORMATS:
   144→        raise ValidationFailed(
   145→            f"Unknown report format '{output_format}'.",
   146→            details=[{"field": "output_format", "reason": list(contracts.REPORT_FORMATS)}],
   147→        )
   148→    if locale not in contracts.REPORT_LOCALES:
   149→        raise ValidationFailed(
   150→            f"Unknown locale '{locale}'.",
   151→            details=[{"field": "locale", "reason": list(contracts.REPORT_LOCALES)}],
   152→        )
   153→    if title is not None and len(title) > 300:
   154→        raise ValidationFailed("title must be at most 300 characters.")
   155→
   156→    # ---- scope resolution + viewer-level access
   157→    resolved_org_id: uuid.UUID
   158→    farm: Farm | None = None
   159→    if farm_id is not None:
   160→        farm = get_farm_for_principal(session, principal, farm_id)
   161→        principal.require_org_role(farm.organization_id, "viewer")
   162→        resolved_org_id = farm.organization_id
   163→    elif field_id is not None:
   164→        field = get_field_for_principal(session, principal, field_id)
   165→        farm = session.get(Farm, field.farm_id)
   166→        principal.require_org_role(farm.organization_id, "viewer")
   167→        resolved_org_id = farm.organization_id
   168→    elif organization_id is not None:
   169→        principal.require_org_role(organization_id, "viewer")
   170→        resolved_org_id = organization_id
   171→    else:
   172→        org_ids = principal.org_ids()
   173→        if len(org_ids) == 1:
   174→            resolved_org_id = next(iter(org_ids))
   175→        else:
   176→            raise ValidationFailed(
   177→                "Provide farm_id, field_id or organization_id to scope the report."
   178→            )
   179→
   180→    # ---- date range (model check: date_to >= date_from)
   181→    if date_from is None and date_to is None:
   182→        date_from, date_to = contracts.default_date_range()
   183→    elif date_from is None or date_to is None:
   184→        raise ValidationFailed("Provide both date_from and date_to, or neither.")
   185→    if date_from > date_to:
   186→        raise ValidationFailed(
   187→            "date_from must be on or before date_to.",
   188→            details=[{"field": "date_from", "reason": str(date_from)}],
   189→        )
   190→
   191→    # ---- entitlement gate (only when the org carries a live subscription)
   192→    subscription = entitlements.get_live_subscription(session, resolved_org_id)
   193→    if subscription is not None:
   194→        entitlements.require_flag(session, resolved_org_id, entitlements.KEY_EXPORTS_ENABLED)
   195→        entitlements.check_quota(session, resolved_org_id, entitlements.KEY_REPORTS_PER_MONTH)
   196→
   197→    job = ReportJob(
   198→        organization_id=resolved_org_id,
   199→        farm_id=farm.id if farm is not None else None,
   200→        requested_by=principal.id,
   201→        report_type=report_type,
   202→        output_format=output_format,
   203→        locale=locale,
   204→        date_from=date_from,
   205→        date_to=date_to,
   206→        status=contracts.REPORT_STATUS_QUEUED,
   207→    )
   208→    session.add(job)
   209→    session.flush()
   210→
   211→    activity.record_event(
   212→        session,
   213→        organization_id=resolved_org_id,
   214→        event_type=activity.EVENT_REPORT_CREATED,
   215→        summary=f"Report '{report_type}' queued ({output_format}, {locale}).",
   216→        farm_id=job.farm_id,
   217→        actor_user_id=principal.id,
   218→        entity_kind="report_job",
   219→        entity_id=str(job.id),
   220→        details={
   221→            "report_type": report_type,
   222→            "output_format": output_format,
   223→            "locale": locale,
   224→            "date_from": date_from.isoformat(),
   225→            "date_to": date_to.isoformat(),
   226→            "title": title,
   227→        },
   228→    )
   229→    session.commit()
   230→    return job

   233→# ----------------------------------------------------------------- execution
   234→def execute_report_job(
   235→    session: Session,
   236→    job_id: uuid.UUID,
   237→    *,
   238→    now: datetime | None = None,
   239→) -> tuple[bool, str | None]:
   240→    """Render one queued (or failed) report job. Returns (success, error).
   241→
   242→    Never raises for data problems — a section that cannot be gathered is
   243→    rendered with ``provenance: "missing"`` and a note. Only infrastructural
   244→    failures (renderer crash, unwritable storage) mark the job failed.
   245→    """
   246→    job = session.get(ReportJob, job_id)
   247→    if job is None:
   248→        raise NotFound("Report job not found.")
   249→    if job.status == contracts.REPORT_STATUS_RUNNING:
   250→        raise Conflict("Report job is already running.")
   251→    if job.status == contracts.REPORT_STATUS_EXPIRED:
   252→        raise Conflict("Report job has expired; regenerate it first.")
   253→
   254→    now = now or datetime.now(UTC)
   255→    job.status = contracts.REPORT_STATUS_RUNNING
   256→    job.error = None
   257→    session.commit()
   258→
   259→    try:
   260→        spec = contracts.get_report_type(job.report_type)
   261→        payload = _build_payload(session, job, spec, now)
   262→        blob = renderers.render(payload, job.output_format, job.locale)
   263→        artifact_path = _write_artifact(job, blob)
   264→        job.file_path = str(artifact_path)
   265→        job.file_size_bytes = len(blob)
   266→        job.status = contracts.REPORT_STATUS_SUCCEEDED
   267→        job.error = None
   268→        job.retention_until = contracts.retention_until_for(now)
   269→        # Usage accounting ONLY for subscribed orgs (same rule as the gate).
   270→        if entitlements.get_live_subscription(session, job.organization_id) is not None:
   271→            entitlements.increment_usage(
   272→                session, job.organization_id, entitlements.KEY_REPORTS_PER_MONTH
   273→            )
   274→        session.commit()
   275→        if payload.get("rendering_notes"):
   276→            logger.info(
   277→                "report %s rendering notes: %s", job.id, payload["rendering_notes"]
   278→            )
   279→        # P3 W6: notify external systems (webhook deliveries are signed and
   280→        # dispatched by the retry sweep; emission here must never fail the job).
   281→        try:
   282→            from app.publicapi.emitters import emit_event
   283→
   284→            emit_event(
   285→                session,
   286→                job.organization_id,
   287→                "report.ready",
   288→                payload_id=str(job.id),
   289→                data={
   290→                    "report_job_id": str(job.id),
   291→                    "report_type": job.report_type,
   292→                    "format": job.output_format,
   293→                    "locale": job.locale,
   294→                },
   295→            )
   296→            session.commit()
   297→        except Exception:  # noqa: BLE001
   298→            session.rollback()
   299→        return True, None
   300→    except Exception as exc:  # noqa: BLE001 — job failure must never raise
   301→        session.rollback()
   302→        job = session.get(ReportJob, job_id)
   303→        if job is None:  # pragma: no cover — row deleted mid-flight
   304→            return False, str(exc)
   305→        job.status = contracts.REPORT_STATUS_FAILED
   306→        job.error = _safe_error(exc)
   307→        job.retention_until = contracts.retention_until_for(now, contracts.FAILED_JOB_RETENTION_DAYS)
   308→        session.commit()
   309→        logger.warning("report job %s failed: %s", job_id, job.error)
   310→        return False, job.error
   311→
   312→
   313→def _safe_error(exc: Exception) -> str:
   314→    """Storage-safe error line: type + message only, truncated."""
   315→    message = f"{type(exc).__name__}: {exc}"
   316→    return message[:2000]
   317→
   318→
   319→def _write_artifact(job: ReportJob, blob: bytes) -> Path:
   320→    base = Path(settings.reports_dir)
   321→    base.mkdir(parents=True, exist_ok=True)
   322→    path = base / f"{job.id}.{job.output_format}"
   323→    tmp = path.with_suffix(path.suffix + ".tmp")
   324→    tmp.write_bytes(blob)
   325→    tmp.replace(path)  # atomic-ish publication
   326→    return path

   329→# ------------------------------------------------------------- data gathering
   330→def _build_payload(
   331→    session: Session, job: ReportJob, spec: contracts.ReportTypeSpec, now: datetime
   332→) -> dict[str, Any]:
   333→    organization = session.get(Organization, job.organization_id)
   334→    farm = session.get(Farm, job.farm_id) if job.farm_id else None
   335→    fields = _scope_fields(session, farm, job.organization_id)
   336→    locale = job.locale
   337→    if farm is not None:
   338→        farm_count = 1
   339→    else:
   340→        farm_count = int(
   341→            session.scalar(
   342→                select(func.count())
   343→                .select_from(Farm)
   344→                .where(Farm.organization_id == job.organization_id)
   345→            ) or 0
   346→        )
   347→
   348→    sections: list[dict[str, Any]] = []
   349→    sections.append(_overview_section(job, spec, farm, fields, farm_count))
   350→
   351→    builder_by_section: dict[str, Any] = {
   352→        "field_breakdown": lambda: _field_breakdown_section(
   353→            session, fields, job.date_from, job.date_to, locale
   354→        ),
   355→        "water": lambda: _water_section(session, fields, job.date_from, job.date_to, locale),
   356→        "soil": lambda: _soil_section(session, fields, locale),
   357→        "recommendations": lambda: _recommendations_section(session, fields, locale),
   358→        "crop_cycles": lambda: _crop_cycles_section(
   359→            session, fields, job.date_from, job.date_to, locale
   360→        ),
   361→        "seasonality": lambda: _seasonality_section(
   362→            session, fields, job.date_from, job.date_to, locale
   363→        ),
   364→        "risk": lambda: _risk_section(session, fields, locale),
   365→        "data_quality": lambda: _data_quality_section(
   366→            session, fields, job.date_from, job.date_to, locale
   367→        ),
   368→    }
   369→    for key in spec.default_sections:
   370→        if key == "overview":
   371→            continue
   372→        builder = builder_by_section.get(key)
   373→        if builder is None:
   374→            sections.append(
   375→                contracts.missing_section(_heading(key, locale), "Section not implemented.")
   376→            )
   377→            continue
   378→        try:
   379→            sections.append(builder())
   380→        except Exception as exc:  # noqa: BLE001 — degrade the section, not the report
   381→            logger.warning("report %s section '%s' failed: %s", job.id, key, exc)
   382→            sections.append(
   383→                contracts.missing_section(
   384→                    _heading(key, locale),
   385→                    f"Section could not be gathered ({type(exc).__name__}); "
   386→                    "shown as missing rather than estimated.",
   387→                )
   388→            )
   389→
   390→    field_ids = [f.id for f in fields]
   391→    limitations = [
   392→        "Yield, input costs and profit are not recorded by the platform and "
   393→        "are never estimated in this report.",
   394→        "Irrigation volumes are not metered platform-wide; water figures "
   395→        "derive from precipitation observations only.",
   396→        "PDF v1 contains no charts — tables and key figures only.",
   397→        "This report summarizes platform data; it is not an agronomic or "
   398→        "financial certification.",
   399→    ]
   400→    if locale == "ar":
   401→        limitations.append(
   402→            "v1 يترجم عناوين التقرير والأقسام؛ تظل رؤوس أعمدة الجداول "
   403→            "بأسمائها الإنجليزية الآلية."
   404→        )
   405→
   406→    return {
   407→        "title": spec.title_ar if locale == "ar" else spec.title_en,
   408→        "report_type": spec.key,
   409→        "primary_section": spec.primary_section,
   410→        "org_name": organization.name if organization else "",
   411→        "farm_name": farm.name if farm else None,
   412→        "generated_at": now.isoformat(),
   413→        "date_range": {"from": job.date_from.isoformat(), "to": job.date_to.isoformat()},
   414→        "sections": sections,
   415→        "provenance_legend": list(renderers.DEFAULT_LEGEND),
   416→        "assumptions": [
   417→            "Water volumes derive from precipitation observations × registered "
   418→            "field area (1 mm over 1 ha = 10 m³); effective rainfall is NOT deducted.",
   419→            "Period filters use observation timestamps compared in UTC.",
   420→            "Field areas are the user-registered geometry areas, not surveyed values.",
   421→        ],
   422→        "limitations": limitations,
   423→        "sources": [
   424→            "observations — data-platform provider ingests",
   425→            "crop_history — farmer records",
   426→            "AI recommendations — platform engines (rules / rules+llm)",
   427→            "water plans — seasonal water intelligence snapshots",
   428→            "soil measurements — laboratory and farmer-reported samples",
   429→        ] + ([f"fields in scope: {len(field_ids)}"] if field_ids else []),
   430→        "rendering_notes": [],
   431→    }

   434→def _scope_fields(session: Session, farm: Farm | None, organization_id: uuid.UUID) -> list[Field]:
   435→    stmt = select(Field).join(Farm, Field.farm_id == Farm.id)
   436→    if farm is not None:
   437→        stmt = stmt.where(Field.farm_id == farm.id)
   438→    else:
   439→        stmt = stmt.where(Farm.organization_id == organization_id)
   440→    return list(session.scalars(stmt.order_by(Farm.name, Field.name)).all())

   443→def _overview_section(
   444→    job: ReportJob,
   445→    spec: contracts.ReportTypeSpec,
   446→    farm: Farm | None,
   447→    fields: list[Field],
   448→    farm_count: int | None = None,
   449→) -> dict[str, Any]:
   450→    locale = job.locale
   451→    total_area = round(sum(f.area_ha for f in fields), 2) if fields else None
   452→    period_days = (job.date_to - job.date_from).days + 1
   453→    return contracts.make_section(
   454→        _heading("overview", locale),
   455→        prose=spec.description,
   456→        kpis=[
   457→            contracts.make_kpi(
   458→                "Farms in scope",
   459→                1 if farm is not None else farm_count,
   460→                "count", PROVENANCE_OBSERVED,
   461→                note="Farm-scoped report." if farm is not None else
   462→                "Organization-scoped report (all farms).",
   463→            ),
   464→            contracts.make_kpi(
   465→                "Fields in scope", len(fields) or None, "count", PROVENANCE_OBSERVED
   466→            ),
   467→            contracts.make_kpi(
   468→                "Total area", total_area, "ha", PROVENANCE_OBSERVED,
   469→                note="Sum of registered field areas.",
   470→            ),
   471→            contracts.make_kpi(
   472→                "Period length", period_days, "days", PROVENANCE_OBSERVED
   473→            ),
   474→        ],
   475→        provenance=PROVENANCE_OBSERVED,
   476→    )

   479→def _field_breakdown_section(
   480→    session: Session, fields: list[Field], date_from: date, date_to: date, locale: str
   481→) -> dict[str, Any]:
   482→    start = datetime(date_from.year, date_from.month, date_from.day, tzinfo=UTC)
   483→    end = datetime(date_to.year, date_to.month, date_to.day, 23, 59, 59, tzinfo=UTC)
   484→    rows: list[list[Any]] = []
   485→    for field in fields:
   486→        obs = int(
   487→            session.scalar(
   488→                select(func.count())
   489→                .select_from(Observation)
   490→                .where(
   491→                    Observation.field_id == field.id,
   492→                    Observation.observed_at >= start,
   493→                    Observation.observed_at <= end,
   494→                )
   495→            ) or 0
   496→        )
   497→        recs = int(
   498→            session.scalar(
   499→                select(func.count())
   500→                .select_from(AIRecommendation)
   501→                .where(
   502→                    AIRecommendation.field_id == field.id,
   503→                    AIRecommendation.status == "active",
   504→                )
   505→            ) or 0
   506→        )
   507→        cycles = int(
   508→            session.scalar(
   509→                select(func.count())
   510→                .select_from(CropHistory)
   511→                .where(
   512→                    CropHistory.field_id == field.id,
   513→                    CropHistory.sowing_date >= date_from,
   514→                    CropHistory.sowing_date <= date_to,
   515→                )
   516→            ) or 0
   517→        )
   518→        rows.append([field.name, round(field.area_ha, 2), obs, recs, cycles])
   519→    return contracts.make_section(
   520→        _heading("field_breakdown", locale),
   521→        prose="Per-field activity counts for the selected period.",
   522→        table=contracts.make_table(
   523→            ["Field", "Area", "Observations", "Active recommendations", "Crop cycles"],
   524→            rows,

   532→def _water_section(
   533→    session: Session, fields: list[Field], date_from: date, date_to: date, locale: str
   534→) -> dict[str, Any]:
   535→    field_ids = [f.id for f in fields]
   536→    start = datetime(date_from.year, date_from.month, date_from.day, tzinfo=UTC)
   537→    end = datetime(date_to.year, date_to.month, date_to.day, 23, 59, 59, tzinfo=UTC)
   538→    area_by_field = {f.id: f for f in fields}
   539→    field_names = {f.id: f.name for f in fields}
   540→
   541→    plan_rows: list[list[Any]] = []
   542→    if field_ids:
   543→        plans = list(
   544→            session.scalars(
   545→                select(WaterPlan)
   546→                .where(
   547→                    WaterPlan.field_id.in_(field_ids),
   548→                    WaterPlan.created_at >= start,
   549→                    WaterPlan.created_at <= end,
   550→                )
   551→                .order_by(WaterPlan.created_at.desc())
   552→                .limit(SECTION_ROW_CAP)
   553→            ).all()
   554→        )
   555→        for plan in plans:
   556→            plan_rows.append([
   557→                field_names.get(plan.field_id, str(plan.field_id)),
   558→                plan.label,
   559→                plan.status,
   560→                len(plan.seasons or []),
   561→                plan.water_allocation_m3,
   562→                plan.created_at.date().isoformat() if plan.created_at else None,
   563→            ])
   564→
   565→    # Precipitation-derived volumes (same formula as app.analytics: honest,
   566→    # model_derived, effective rainfall NOT deducted).
   567→    volume_m3: float | None = None
   568→    if field_ids:
   569→        precip = session.execute(
   570→            select(Observation.field_id, func.sum(Observation.value))
   571→            .where(
   572→                Observation.field_id.in_(field_ids),
   573→                Observation.variable_code == "precipitation",
   574→                Observation.unit.in_(("mm", "kg/m^2")),
   575→                Observation.observed_at >= start,
   576→                Observation.observed_at <= end,
   577→            )
   578→            .group_by(Observation.field_id)
   579→        ).all()
   580→        total = 0.0
   581→        for field_id, mm_sum in precip:
   582→            if mm_sum is None or field_id not in area_by_field:
   583→                continue
   584→            total += float(mm_sum) * 10.0 * area_by_field[field_id].area_ha
   585→        volume_m3 = round(total, 1) if precip else None
   586→
   587→    has_plans = bool(plan_rows)
   588→    return contracts.make_section(
   589→        _heading("water", locale),
   590→        prose=(
   591→            "Water plans created in the period and precipitation-derived "
   592→            "volumes. Irrigation metering is not implemented platform-wide."
   593→        ),
   594→        table=contracts.make_table(
   595→            ["Field", "Plan", "Status", "Seasons", "Allocation", "Created"],
   596→            plan_rows,
   597→            units=[None, None, None, "count", "m3", None],
   598→        )
   599→        if plan_rows
   600→        else None,
   601→        kpis=[
   602→            contracts.make_kpi(
   603→                "Precipitation volume (model-derived)", volume_m3, "m3", PROVENANCE_MODEL,
   604→                note="1 mm over 1 ha = 10 m³; effective rainfall NOT deducted.",
   605→            ),
   606→            contracts.make_kpi(
   607→                "Irrigation volume", None, "m3", PROVENANCE_MISSING,
   608→                note="Irrigation is not metered by the platform yet.",
   609→            ),
   610→        ],
   611→        provenance=PROVENANCE_MODEL if volume_m3 is not None or has_plans else PROVENANCE_MISSING,
   612→        note=None if (volume_m3 is not None or has_plans) else
   613→        "No water plans and no precipitation observations in this period.",

   807→def _risk_section(session: Session, fields: list[Field], locale: str) -> dict[str, Any]:
   808→    field_ids = [f.id for f in fields]
   809→    rows: list[list[Any]] = []
   810→    synthetic_total = total = 0
   811→    if field_ids:
   812→        now = datetime.now(UTC)
   813→        for variable in _VARIABLE_FAMILY:
   814→            count = int(
   815→                session.scalar(
   816→                    select(func.count())
   817→                    .select_from(Observation)
   818→                    .where(
   819→                        Observation.field_id.in_(field_ids),
   820→                        Observation.variable_code == variable,
   821→                    )
   822→                ) or 0
   823→            )
   824→            last = session.scalar(
   825→                select(func.max(Observation.observed_at)).where(
   826→                    Observation.field_id.in_(field_ids),
   827→                    Observation.variable_code == variable,
   828→                )
   829→            )
   830→            if count == 0 or last is None:
   831→                rows.append([variable, 0, None, None, "missing"])
   832→                continue
   833→            age_days = round((now - last).total_seconds() / 86400, 1)
   834→            status = "fresh" if age_days <= 3 else ("stale" if age_days <= 14 else "very_stale")
   835→            rows.append([variable, count, last.date().isoformat(), age_days, status])
   836→        total = int(
   837→            session.scalar(
   838→                select(func.count())
   839→                .select_from(Observation)
   840→                .where(Observation.field_id.in_(field_ids))
   841→            ) or 0
   842→        )
   843→        synthetic_total = int(
   844→            session.scalar(
   845→                select(func.count())
   846→                .select_from(Observation)
   847→                .where(
   848→                    Observation.field_id.in_(field_ids),
   849→                    Observation.is_synthetic.is_(True),
   850→                )
   851→            ) or 0
   852→        )
   853→    synthetic_share = round(synthetic_total / total, 4) if total else None
   854→    return contracts.make_section(
   855→        _heading("risk", locale),
   856→        prose=(
   857→            "Data-driven exposure view: how fresh and how complete the "
   858→            "underlying observations are. NOT an agronomic risk assessment."
   859→        ),
   860→        table=contracts.make_table(
   861→            ["Variable", "Observations", "Last observed", "Age (days)", "Status"],
   862→            rows,
   863→            units=[None, "count", None, "days", None],
   864→        ),
   865→        kpis=[
   866→            contracts.make_kpi(
   867→                "Synthetic observation share", synthetic_share, "ratio", PROVENANCE_MODEL,
   868→                note="Fraction of observations flagged synthetic in the scope.",
   869→            ),
   870→        ],
   871→        provenance=PROVENANCE_OBSERVED if rows else PROVENANCE_MISSING,
   872→        note=None if rows else "No observations in scope.",
   873→    )

   913→def list_report_jobs(
   914→    session: Session,
   915→    principal: Principal,
   916→    *,
   917→    status: str | None = None,
   918→    report_type: str | None = None,
   919→    farm_id: uuid.UUID | None = None,
   920→    limit: int = 50,
   921→    offset: int = 0,
   922→) -> tuple[list[ReportJob], int]:
   923→    """Org-scoped library listing, newest first, with hard limits."""
   924→    org_ids = principal.org_ids()
   925→    if not org_ids:
   926→        return [], 0
   927→    if status is not None and status not in contracts.REPORT_STATUSES:
   928→        raise ValidationFailed(
   929→            f"Unknown status '{status}'.",
   930→            details=[{"field": "status", "reason": list(contracts.REPORT_STATUSES)}],
   931→        )
   932→    if report_type is not None and report_type not in contracts.REPORT_TYPES:
   933→        raise ValidationFailed(
   934→            f"Unknown report type '{report_type}'.",
   935→            details=[{"field": "report_type", "reason": sorted(contracts.REPORT_TYPES)}],
   936→        )
   937→    limit = max(1, min(limit, 200))
   938→    offset = max(0, offset)
   939→    conditions = [ReportJob.organization_id.in_(org_ids)]
   940→    if status is not None:
   941→        conditions.append(ReportJob.status == status)
   942→    if report_type is not None:
   943→        conditions.append(ReportJob.report_type == report_type)
   944→    if farm_id is not None:
   945→        conditions.append(ReportJob.farm_id == farm_id)
   946→    total = int(
   947→        session.scalar(select(func.count()).select_from(ReportJob).where(*conditions)) or 0
   948→    )
   949→    rows = list(
   950→        session.scalars(
   951→            select(ReportJob)
   952→            .where(*conditions)
   953→            .order_by(ReportJob.created_at.desc(), ReportJob.id.desc())
   954→            .limit(limit)
   955→            .offset(offset)
   956→        ).all()
   957→    )
   958→    return rows, total

   961→# ------------------------------------------------------------------ download
   962→def read_report_file(session: Session, principal: Principal, job_id: uuid.UUID) -> Path:
   963→    """Resolve the artifact path for download — traversal-safe.
   964→
   965→    The stored path must resolve INSIDE ``settings.reports_dir`` and carry
   966→    the job's id as its stem; anything else is treated as not found. Expired
   967→    artifacts return 404 (the sweep has usually removed them already).
   968→    """
   969→    job = get_report_job(session, principal, job_id)
   970→    if job.status != contracts.REPORT_STATUS_SUCCEEDED or not job.file_path:
   971→        raise NotFound("Report artifact is not available for this job.")
   972→    now = datetime.now(UTC)
   973→    if job.retention_until is not None and job.retention_until <= now:
   974→        raise NotFound("Report artifact has expired and is no longer downloadable.")
   975→    base = Path(settings.reports_dir).resolve()
   976→    candidate = Path(job.file_path)
   977→    if not candidate.is_absolute():
   978→        candidate = base / candidate
   979→    try:
   980→        candidate = candidate.resolve()
   981→    except OSError as exc:  # pragma: no cover — defensive
   982→        raise NotFound("Report artifact path could not be resolved.") from exc
   983→    if candidate.parent != base or candidate.stem != str(job.id):
   984→        raise NotFound("Report artifact path is invalid.")
   985→    if not candidate.is_file():
   986→        raise NotFound("Report artifact file is missing from storage.")
   987→    return candidate

   991→def delete_report_job(session: Session, principal: Principal, job_id: uuid.UUID) -> None:
   992→    """Manager+ deletion: removes the artifact and the library row."""
   993→    job = session.get(ReportJob, job_id)
   994→    if job is None:
   995→        raise NotFound("Report job not found.")
   996→    principal.require_org_role(job.organization_id, "manager")
   997→    if job.file_path:
   998→        try:
   999→            path = Path(job.file_path)
  1000→            if path.is_file() and Path(settings.reports_dir).resolve() in path.resolve().parents:
  1001→                path.unlink()
  1002→        except OSError:  # pragma: no cover — storage hiccup must not block
  1003→            logger.warning("could not delete artifact for report %s", job.id)
  1004→    session.delete(job)
  1005→    session.commit()

  1008→# ----------------------------------------------------------------- retention
  1009→def purge_expired_reports(session: Session, now: datetime | None = None) -> int:
  1010→    """Retention sweep: expired artifacts lose their file and the job row is
  1011→    marked ``expired`` (kept as a library stub). Returns rows transitioned."""
  1012→    now = now or datetime.now(UTC)
  1013→    due = list(
  1014→        session.scalars(
  1015→            select(ReportJob).where(
  1016→                ReportJob.status.in_(
  1017→                    (contracts.REPORT_STATUS_SUCCEEDED, contracts.REPORT_STATUS_FAILED)
  1018→                ),
  1019→                ReportJob.retention_until.is_not(None),
  1020→                ReportJob.retention_until <= now,
  1021→            )
  1022→        ).all()
  1023→    )
  1024→    # Rows that never got a retention stamp (e.g. written by older code):
  1025→    # terminal and older than the default window are also purge-eligible.
  1026→    threshold = now - timedelta(days=contracts.REPORT_RETENTION_DAYS_DEFAULT)
  1027→    unstamped = list(
  1028→        session.scalars(
  1029→            select(ReportJob).where(
  1030→                ReportJob.status.in_(
  1031→                    (contracts.REPORT_STATUS_SUCCEEDED, contracts.REPORT_STATUS_FAILED)
  1032→                ),
  1033→                ReportJob.retention_until.is_(None),
  1034→                ReportJob.updated_at <= threshold,
  1035→            )
  1036→        ).all()
  1037→    )
  1038→    purged = 0
  1039→    base = Path(settings.reports_dir).resolve()
  1040→    for job in due + unstamped:
  1041→        if job.file_path:
  1042→            try:
  1043→                path = Path(job.file_path)
  1044→                if path.is_file() and base in path.resolve().parents:
  1045→                    path.unlink()
  1046→            except OSError:  # pragma: no cover
  1047→                logger.warning("retention sweep could not delete %s", job.file_path)
  1048→        job.file_path = None
  1049→        job.file_size_bytes = None
  1050→        job.status = contracts.REPORT_STATUS_EXPIRED
  1051→        purged += 1
  1052→    if purged:
  1053→        session.commit()
  1054→    return purged
  1055→
  1056→
  1057→# ----------------------------------------------------------------- schedule
  1058→def set_report_schedule(
  1059→    session: Session,
  1060→    principal: Principal,
  1061→    job_id: uuid.UUID,
  1062→    *,
  1063→    interval_days: int | None,
  1064→) -> ReportJob:
  1065→    """Attach (or clear) a recurrence on a report template (manager+).
  1066→
  1067→    Setting an interval arms ``next_run_at`` = now + interval; clearing
  1068→    (``interval_days=None``) makes the job one-off again.
  1069→    """
  1070→    job = session.get(ReportJob, job_id)
  1071→    if job is None:
  1072→        raise NotFound("Report job not found.")
  1073→    principal.require_org_role(job.organization_id, "manager")
  1074→    if interval_days is not None:
  1075→        if (
  1076→            interval_days < contracts.MIN_INTERVAL_DAYS
  1077→            or interval_days > contracts.MAX_INTERVAL_DAYS
  1078→        ):
  1079→            raise ValidationFailed(
  1080→                f"interval_days must be between {contracts.MIN_INTERVAL_DAYS} "
  1081→                f"and {contracts.MAX_INTERVAL_DAYS}.",
  1082→                details=[{"field": "interval_days", "reason": interval_days}],
  1083→            )
  1084→        job.interval_days = interval_days
  1085→        job.next_run_at = datetime.now(UTC) + timedelta(days=interval_days)
  1086→    else:
  1087→        job.interval_days = None
  1088→        job.next_run_at = None
  1089→    session.commit()
  1090→    return job
  1091→
  1092→
  1093→def advance_recurring(session: Session, now: datetime | None = None) -> list[ReportJob]:
  1094→    """Recurring sweep: for every due template job, roll its ``next_run_at``
  1095→    forward and enqueue a ONE-OFF copy carrying the next contiguous date
  1096→    window. The template keeps the schedule; copies never re-schedule.
  1097→
  1098→    Missed periods are caught up by advancing ``next_run_at`` interval-by-
  1099→    interval, but at most ONE catch-up copy is created per template per sweep
  1100→    (a delayed worker must not fan out unbounded backfills silently).
  1101→    """
  1102→    now = now or datetime.now(UTC)
  1103→    due = list(
  1104→        session.scalars(
  1105→            select(ReportJob).where(
  1106→                ReportJob.interval_days.is_not(None),
  1107→                ReportJob.next_run_at.is_not(None),
  1108→                ReportJob.next_run_at <= now,
  1109→            )
  1110→        ).all()
  1111→    )
  1112→    created: list[ReportJob] = []
  1113→    for template in due:
  1114→        interval = max(1, int(template.interval_days or 1))
  1115→        window_days = max(1, (template.date_to - template.date_from).days + 1)
  1116→        copy = ReportJob(
  1117→            organization_id=template.organization_id,
  1118→            farm_id=template.farm_id,
  1119→            requested_by=template.requested_by,
  1120→            report_type=template.report_type,
  1121→            output_format=template.output_format,
  1122→            locale=template.locale,
  1123→            # Rolling contiguous window immediately after the previous one.
  1124→            date_from=template.date_to + timedelta(days=1),
  1125→            date_to=template.date_to + timedelta(days=window_days),
  1126→            status=contracts.REPORT_STATUS_QUEUED,
  1127→            interval_days=None,
  1128→            next_run_at=None,
  1129→        )
  1130→        session.add(copy)
  1131→        # Catch-up: advance past every missed period without fanning out.
  1132→        next_run = template.next_run_at
  1133→        while next_run is not None and next_run <= now:
  1134→            next_run = next_run + timedelta(days=interval)
  1135→        template.next_run_at = next_run
  1136→        activity.record_event(
  1137→            session,
  1138→            organization_id=template.organization_id,
  1139→            event_type=activity.EVENT_REPORT_CREATED,
  1140→            summary=f"Recurring report '{template.report_type}' re-enqueued.",
  1141→            farm_id=template.farm_id,
  1142→            actor_user_id=template.requested_by,
  1143→            entity_kind="report_job",
  1144→            entity_id=str(copy.id),
  1145→            details={
  1146→                "template_job_id": str(template.id),
  1147→                "interval_days": interval,
  1148→                "report_type": template.report_type,
  1149→            },
  1150→        )
  1151→        created.append(copy)
  1152→    if created:
  1153→        session.commit()
  1154→    return created
