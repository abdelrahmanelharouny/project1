    60→function InlineError({ err, t }: { err: unknown; t: T }) {
    61→  const forbidden = err instanceof ApiError && err.status === 403
    62→  return (
    63→    <div className="rounded-md border border-amber-200 bg-amber-50 p-3 text-sm text-amber-900">
    64→      <p className="font-semibold">{forbidden ? t('errors.forbiddenTitle') : t('common.loadError')}</p>
    65→      <p className="mt-1 text-xs leading-relaxed">
    66→        {forbidden ? t('errors.forbiddenBody') : describeError(err).title}
    67→      </p>
    68→    </div>
    69→  )

    89→function provenanceClass(p: Provenance): string {
    90→  switch (p) {
    91→    case 'observed':
    92→      return 'border-emerald-200 bg-emerald-50 text-emerald-800'
    93→    case 'model_derived':
    94→      return 'border-stone-300 bg-stone-100 text-stone-600'
    95→    case 'simulated':
    96→      return 'border-amber-200 bg-amber-50 text-amber-800'
    97→    default:
    98→      return 'border-stone-200 bg-stone-50 text-stone-400'
    99→  }
   100→}
   101→
   102→function provenanceLabel(t: T, p: Provenance): string {
   103→  switch (p) {
   104→    case 'observed':
   105→      return t('provenance.observed')
   106→    case 'model_derived':
   107→      return t('provenance.modelDerived')
   108→    case 'simulated':
   109→      return t('provenance.simulated')
   110→    default:
   111→      return t('provenance.missing')
   112→  }
   113→}
   114→
   115→function provenanceTip(t: T, p: Provenance): string {
   116→  switch (p) {
   117→    case 'observed':
   118→      return t('provenance.observedTip')
   119→    case 'model_derived':
   120→      return t('provenance.modelDerivedTip')
   121→    case 'simulated':
   122→      return t('provenance.simulatedTip')
   123→    default:
   124→      return t('provenance.missingTip')
   125→  }
   126→}

   128→function metricLabel(t: T, metric: AnalyticsMetric): string {
   129→  const map: Record<string, string> = {
   130→    total_area_ha: 'metricTotalAreaHa',
   131→    farm_count: 'metricFarmCount',
   132→    field_count: 'metricFieldCount',
   133→    active_crop_cycles: 'metricActiveCropCycles',
   134→    observations_in_period: 'metricObservationsInPeriod',
   135→    data_coverage: 'metricDataCoverage',
   136→    water_volume_m3: 'metricWaterVolumeM3',
   137→    water_cost: 'metricWaterCost',
   138→    cost_per_hectare: 'metricCostPerHectare',
   139→    yield: 'metricYield',
   140→  }
   141→  const key = map[metric.key]
   142→  return key ? t(`insights.${key}`) : metric.label
   143→}
   144→
   145→function formatMetricValue(value: number, unit: string, locale: string): string {
   146→  if (unit === 'ratio') return pct(value)
   147→  const decimals = unit === 'count' ? 0 : 2
   148→  return value.toLocaleString(locale, { maximumFractionDigits: decimals })
   149→}

   178→  )

   181→function MetricTile({ t, metric, locale }: { t: T; metric: AnalyticsMetric; locale: string }) {
   182→  const missing = metric.value === null || metric.provenance === 'missing'
   183→  const period =
   184→    metric.period.from || metric.period.to
   185→      ? `${metric.period.from ? formatDay(metric.period.from) : '…'} — ${metric.period.to ? formatDay(metric.period.to) : '…'}`
   186→      : null
   187→  return (
   188→    <div className="rounded-lg border border-stone-200 bg-white p-3">
   189→      <div className="flex items-center justify-between gap-2">
   190→        <p className="text-xs font-medium text-stone-600">{metricLabel(t, metric)}</p>
   191→        {metric.note ? (
   192→          <Tooltip>
   193→            <TooltipTrigger asChild>
   194→              <button type="button" aria-label={t('common.note')} className="text-stone-400 hover:text-stone-600">
   195→                <Info className="h-3.5 w-3.5" aria-hidden />
   196→              </button>
   197→            </TooltipTrigger>
   198→            <TooltipContent className="max-w-72">
   199→              <p>{metric.note}</p>
   200→            </TooltipContent>
   201→          </Tooltip>
   202→        ) : null}
   203→      </div>
   204→      <div className="mt-1.5 flex flex-wrap items-center gap-2">
   205→        {missing ? (
   206→          <NoDataChip t={t} />
   207→        ) : (
   208→          <span className="text-xl font-semibold tabular-nums text-stone-900">
   209→            {formatMetricValue(metric.value as number, metric.unit, locale)}
   210→          </span>
   211→        )}
   212→        {!missing && metric.unit !== 'count' && metric.unit !== 'ratio' ? (
   213→          <span className="text-xs text-stone-500">{metric.unit}</span>
   214→        ) : null}
   215→        <ProvenanceBadge t={t} p={missing ? 'missing' : metric.provenance} />
   216→      </div>
   217→      {period ? <p className="mt-1 text-[11px] text-stone-400">{period}</p> : null}
   218→    </div>
   219→  )
   220→}

   222→function defaultFrom(): string {
   223→  const d = new Date()
   224→  d.setDate(d.getDate() - 89)
   225→  return d.toISOString().slice(0, 10)
   226→}
   227→
   228→function defaultTo(): string {
   229→  return new Date().toISOString().slice(0, 10)
   230→}

   336→      <div className="space-y-4">

   339→        <Skeleton className="h-64 w-full" />

   341→    )

   346→
   347→  return (
   348→    <div className="space-y-4">
   349→      <header>
   350→        <h2 className="text-lg font-semibold text-stone-900">{t('insights.title')}</h2>
   351→        <p className="mt-0.5 text-sm text-stone-500">{t('insights.intro')}</p>
   352→      </header>

   358→            <Label className="text-xs text-stone-600" htmlFor="p3-ins-farm">
