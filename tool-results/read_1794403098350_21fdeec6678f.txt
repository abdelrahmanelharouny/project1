    51→def _season_climate_evidence(
    52→    run: dbm.FeatureRun,
    53→    season_label: str,
    54→    start: date,
    55→    end: date,
    56→) -> tuple[WaterEvidence | None, list[str]]:
    57→    """Month-of-year climatology of ET0 and precipitation for one season window
    58→    from a single feature run (the same historical-proxy basis as rotation)."""
    59→    import calendar as cal
    60→
    61→    monthly: dict[str, dict[int, list[tuple[int, float]]]] = {}
    62→    for row in run.values:
    63→        if row.aggregation_level != "monthly" or row.value is None:
    64→            continue
    65→        monthly.setdefault(row.feature_code, {}).setdefault(
    66→            row.period_start.month, []
    67→        ).append((row.period_start.year, row.value))
    68→
    69→    def collect_month_means(code: str) -> list[tuple[int, int, float]]:
    70→        table = monthly.get(code, {})
    71→        out: list[tuple[int, int, float]] = []
    72→        y, m = start.year, start.month
    73→        while (y, m) <= (end.year, end.month):
    74→            entries = table.get(m)
    75→            if entries:
    76→                out.append((y, m, sum(v for _yr, v in entries) / len(entries)))
    77→            m += 1
    78→            if m == 13:
    79→                y, m = y + 1, 1
    80→        return out
    81→
    82→    et0 = collect_month_means("reference_et0_total_monthly")
    83→    precip = collect_month_means("precipitation_total_monthly")
    84→
    85→    warnings: list[str] = []
    86→    if not et0:
    87→        warnings.append(
    88→            f"Season '{season_label}': no monthly ET0 feature values in the evidence "
    89→            "run — crop ETc will not be evaluable (missing evidence is never zero)."
    90→        )
    91→    if not precip:
    92→        warnings.append(
    93→            f"Season '{season_label}': no monthly precipitation feature values in the "
    94→            "evidence run — the rainfall credit will be unknown (conservative "
    95→            "requirement, disclosed)."
    96→        )
    97→
    98→    years_used = sorted({
    99→        y
   100→        for table in monthly.values()
   101→        for entries in table.values()
   102→        for y, _v in entries
   103→    })
   104→    evidence = WaterEvidence(
   105→        feature_run_id=str(run.id),
   106→        is_synthetic=bool(run.is_synthetic),
   107→        monthly_et0=tuple(et0),
   108→        monthly_precip=tuple(precip),
   109→        coverage_ratio=1.0 if et0 else 0.0,
   110→        years_used=tuple(years_used),
   111→        basis=(
   112→            f"month-of-year climatology from feature run {str(run.id)[:8]} "
   113→            f"({f'years {years_used[0]}-{years_used[-1]}' if years_used else 'no years'}) "
   114→            "— a historical proxy, NOT a forecast for the planning season."
   115→        ),
   116→    )
   117→    return evidence, warnings

   120→def _efficiency_from_payload(
   121→    payload_eff: dict[str, Any] | None,
   122→) -> EfficiencyInput | None:
   123→    """Efficiency from an explicit farmer value or a method class (the FAO
   124→    TM1 Annex I Table 8 indicative values). A method class value is a
   125→    REFERENCE choice, recorded with its citation."""
   126→    if payload_eff is None:
   127→        return None
   128→    if not isinstance(payload_eff, dict):
   129→        raise ValidationFailed(
   130→            "irrigation_efficiency must be an object.",
   131→            details=[{"field": "irrigation_efficiency"}],
   132→        )
   133→    value = payload_eff.get("value")
   134→    method_class = payload_eff.get("method_class")
   135→    if value is not None:
   136→        try:
   137→            value = float(value)
   138→        except (TypeError, ValueError):
   139→            raise ValidationFailed(
   140→                "irrigation_efficiency.value must be a number.",
   141→                details=[{"field": "irrigation_efficiency"}],
   142→            ) from None
   143→        if not 0 < value <= 1:
   144→            raise ValidationFailed(
   145→                "irrigation_efficiency.value must be within (0, 1].",
   146→                details=[{"field": "irrigation_efficiency", "reason": str(value)}],
   147→            )
   148→        return EfficiencyInput(
   149→            value=value,
   150→            evidence_class="farmer_reported_record",
   151→            citation_text="Farmer-supplied field application efficiency.",
   152→            method_class=method_class,
   153→        )
   154→    if method_class is not None:
   155→        for entry in IRRIGATION_EFFICIENCY_CLASSES:
   156→            if entry["method_class"] == method_class:
   157→                return EfficiencyInput(
   158→                    value=float(entry["efficiency"]),
   159→                    evidence_class=entry["evidence_class"],
   160→                    citation_text=entry["citation_text"],
   161→                    method_class=method_class,
   162→                )
   163→        raise ValidationFailed(
   164→            f"Unknown irrigation method class '{method_class}'.",
   165→            details=[{
   166→                "field": "irrigation_efficiency",
   167→                "reason": ", ".join(e["method_class"] for e in IRRIGATION_EFFICIENCY_CLASSES),
   168→            }],
   169→        )
   170→    raise ValidationFailed(
   171→        "irrigation_efficiency needs 'value' (farmer-measured) or 'method_class'.",
   172→        details=[{"field": "irrigation_efficiency"}],
   173→    )

   176→def build_soil_water_inputs(
   177→    db: Session,
   178→    field: dbm.Field,
   179→    payload: dict[str, Any],
   180→) -> tuple[SoilWaterInputs | None, list[str], dict[str, Any]]:
   181→    """Assemble soil water-holding inputs from the requested source.
   182→
   183→    - farmer_measured: explicit θFC/θWP (+ optional Zr/p) from the request
   184→      (evidence class farmer_reported_record);
   185→    - ptf_texture: Saxton & Rawls (2006) over the field's latest soil
   186→      observations (sand/clay/SOC), with the single-layer disclosure;
   187→    - none: no soil inputs — the FAO-33 no-storage fallback applies."""
   188→    source = payload.get("soil_source") or "none"
   189→    warnings: list[str] = []
   190→    meta: dict[str, Any] = {"soil_source": source}
   191→    if source not in SOIL_SOURCES:
   192→        raise ValidationFailed(
   193→            f"Unknown soil_source '{source}'.",
   194→            details=[{"field": "soil_source", "reason": ", ".join(SOIL_SOURCES)}],
   195→        )
   196→    if source == "none":
   197→        return None, warnings, meta
   198→
   199→    if source == "farmer_measured":
   200→        fs = payload.get("farmer_soil") or {}
   201→        theta_fc = fs.get("theta_fc")
   202→        theta_wp = fs.get("theta_wp")
   203→        if theta_fc is None or theta_wp is None:
   204→            raise ValidationFailed(
   205→                "soil_source 'farmer_measured' requires farmer_soil.theta_fc and "
   206→                "farmer_soil.theta_wp (m3/m3).",
   207→                details=[{"field": "farmer_soil"}],
   208→            )
   209→        try:
   210→            theta_fc, theta_wp = float(theta_fc), float(theta_wp)
   211→        except (TypeError, ValueError):
   212→            raise ValidationFailed(
   213→                "farmer_soil.theta_fc/theta_wp must be numbers.",
   214→                details=[{"field": "farmer_soil"}],
   215→            ) from None
   216→        if not (0 < theta_wp < theta_fc <= 0.6):
   217→            raise ValidationFailed(
   218→                "farmer_soil contents must satisfy 0 < θWP < θFC ≤ 0.6 (m3/m3).",
   219→                details=[{"field": "farmer_soil", "reason": f"{theta_wp} !< {theta_fc}"}],
   220→            )
   221→        zr = fs.get("zr_m")
   222→        p_table = fs.get("p_table")
   223→        inputs = SoilWaterInputs(
   224→            theta_fc=theta_fc,
   225→            theta_fc_evidence_class="farmer_reported_record",
   226→            theta_wp=theta_wp,
   227→            theta_wp_evidence_class="farmer_reported_record",
   228→            zr_m=float(zr) if zr is not None else None,
   229→            zr_evidence_class="farmer_reported_record" if zr is not None else None,
   230→            p_table=float(p_table) if p_table is not None else None,
   231→            p_evidence_class="farmer_reported_record" if p_table is not None else None,
   232→            source_label="farmer_measured",
   233→            source_notes=(
   234→                "Soil water-holding characteristics supplied by the farmer "
   235→                "(farmer_reported_record) — not independently verified.",
   236→            ),
   237→        )
   238→        meta["farmer_soil"] = fs
   239→        return inputs, warnings, meta
   240→
   241→    # ptf_texture
   242→    rows = db.execute(
   243→        select(dbm.Observation)
   244→        .where(
   245→            dbm.Observation.field_id == field.id,
   246→            dbm.Observation.variable_code.in_(SOIL_VARIABLES),
   247→            dbm.Observation.value.is_not(None),
   248→        )
   249→        .order_by(dbm.Observation.observed_at.desc())
   250→        .limit(50)
   251→    ).scalars().all()
   252→    latest: dict[str, dbm.Observation] = {}
   253→    for obs in rows:
   254→        latest.setdefault(obs.variable_code, obs)
   255→    sand = latest.get("soil_sand_content")
   256→    clay = latest.get("soil_clay_content")
   257→    soc = latest.get("soil_organic_carbon")
   258→    if sand is None or clay is None:
   259→        warnings.append(
   260→            "soil_source 'ptf_texture' requested but the field has no sand/clay "
   261→            "observations — the PTF estimate is refused (never extrapolated) and "
   262→            "the plan falls back to the no-storage FAO-33 method."
   263→        )
   264→        return None, warnings, {**meta, "ptf": "refused_missing_texture"}
   265→    sand_frac = float(sand.value) / 1000.0  # g/kg → fraction
   266→    clay_frac = float(clay.value) / 1000.0
   267→    om_percent = (
   268→        soil_mod.organic_carbon_to_om_percent(float(soc.value)) if soc is not None else None
   269→    )
   270→    notes = [
   271→        "θFC/θWP ESTIMATED by the Saxton & Rawls (2006) pedotransfer function over "
   272→        "texture (evidence class global_soil_prediction; PTF version "
   273→        f"{PTF_VERSION}); a texture-based model estimate, not a measurement.",
   274→        "The PTF inputs are the field's latest soil observations. If they come from "
   275→        "SoilGrids, they describe the 0-5 cm layer ASSUMED representative of the "
   276→        "root zone — a single-layer extrapolation explicitly disclosed (backlog S-6).",
   277→    ]
   278→    if om_percent is None:
   279→        notes.append(
   280→            "No organic-carbon observation available — the PTF OM term is set to 0 "
   281→            "(a disclosed simplification, typically a small effect)."
   282→        )
   283→        om_used = 0.0
   284→    else:
   285→        om_used = om_percent
   286→        notes.append(
   287→            "Organic matter derived from soil organic carbon via the van Bemmelen "
   288→            "factor (OM ≈ SOC × 1.724)."
   289→        )
   290→    try:
   291→        est = soil_mod.saxton_rawls_2006(sand_frac, clay_frac, om_used)
   292→    except soil_mod.PTFDomainError as exc:
   293→        warnings.append(
   294→            f"The field's texture is outside the Saxton-Rawls validity domain "
   295→            f"({exc}) — the PTF estimate is refused and the plan falls back to the "
   296→            "no-storage FAO-33 method."
   297→        )
   298→        return None, warnings, {**meta, "ptf": "refused_domain"}
   299→    meta["ptf"] = {
   300→        "ptf_version": PTF_VERSION,
   301→        "sand_fraction": sand_frac,
   302→        "clay_fraction": clay_frac,
   303→        "om_percent": round(om_used, 4),
   304→        "theta_fc": est["theta_fc_33kPa"],
   305→        "theta_wp": est["theta_wp_1500kPa"],
   306→        "input_evidence_classes": {
   307→            "sand": sand.evidence_class,
   308→            "clay": clay.evidence_class,
   309→            "organic_carbon": soc.evidence_class if soc else None,
   310→        },
   311→    }
   312→    inputs = SoilWaterInputs(
   313→        theta_fc=est["theta_fc_33kPa"],
   314→        theta_fc_evidence_class="global_soil_prediction",
   315→        theta_wp=est["theta_wp_1500kPa"],
   316→        theta_wp_evidence_class="global_soil_prediction",
   317→        zr_m=None,   # Zr comes from the crop knowledge (pack) in the planner caller
   318→        zr_evidence_class=None,
   319→        p_table=None,  # p comes from the crop knowledge (pack)
   320→        p_evidence_class=None,
   321→        source_label="ptf_texture",
   322→        source_notes=tuple(notes),
   323→    )
   324→    return inputs, warnings, meta

   327→def start_water_plan(
   328→    db: Session,
   329→    *,
   330→    field: dbm.Field,
   331→    payload: dict[str, Any],
   332→    created_by: Any = None,
   333→) -> dbm.WaterPlan:
   334→    """Validate, execute inline and persist a water plan (202-style job)."""
   335→    idempotency_key = payload.get("idempotency_key")
   336→    if idempotency_key:
   337→        existing = db.scalar(
   338→            select(dbm.WaterPlan).where(
   339→                dbm.WaterPlan.idempotency_key == idempotency_key,
   340→                dbm.WaterPlan.field_id == field.id,
   341→            )
   342→        )
   343→        if existing is not None:
   344→            return existing
   345→
   346→    raw_seasons = payload.get("seasons") or []
   347→    if not raw_seasons:
   348→        raise ValidationFailed(
   349→            "At least one season is required.",
   350→            details=[{"field": "seasons"}],
   351→        )
   352→    season_specs = _validate_seasons(raw_seasons)
   353→    if len(season_specs) > MAX_SEASONS:
   354→        raise ValidationFailed(
   355→            f"At most {MAX_SEASONS} seasons per water plan.",
   356→            details=[{"field": "seasons"}],
   357→        )
   358→
   359→    # crop assignments: per-season crop_code required (knowledge-base crops)
   360→    assignments_in = payload.get("crop_assignments") or []
   361→    if not isinstance(assignments_in, list) or len(assignments_in) != len(season_specs):
   362→        raise ValidationFailed(
   363→            "crop_assignments must list one crop_code per season.",
   364→            details=[{"field": "crop_assignments"}],
   365→        )
   366→    sowings_in = payload.get("sowing_dates") or {}
   367→
   368→    pe_method = payload.get("effective_rainfall_method") or "fao_aglw_1986"
   369→    if pe_method not in EFFECTIVE_RAINFALL_METHODS:
   370→        raise ValidationFailed(
   371→            f"Unknown effective_rainfall_method '{pe_method}'.",
   372→            details=[{"field": "effective_rainfall_method"}],
   373→        )
   374→    pe_fraction = payload.get("pe_fraction")
   375→    if pe_method == "constant_fraction":
   376→        if pe_fraction is None:
   377→            raise ValidationFailed(
   378→                "effective_rainfall_method 'constant_fraction' requires pe_fraction.",
   379→                details=[{"field": "pe_fraction"}],
   380→            )
   381→        pe_fraction = float(pe_fraction)
   382→        if not 0 <= pe_fraction <= 1:
   383→            raise ValidationFailed(
   384→                "pe_fraction must be within [0, 1].",
   385→                details=[{"field": "pe_fraction"}],
   386→            )
   387→    elif pe_fraction is not None:
   388→        raise ValidationFailed(
   389→            "pe_fraction is only valid with the constant_fraction method.",
   390→            details=[{"field": "pe_fraction"}],
   391→        )
   392→
   393→    efficiency = _efficiency_from_payload(payload.get("irrigation_efficiency"))
   394→
   395→    allocation = payload.get("water_allocation_m3")
   396→    if allocation is not None:
   397→        try:
   398→            allocation = float(allocation)
   399→        except (TypeError, ValueError):
   400→            raise ValidationFailed(
   401→                "water_allocation_m3 must be a number.",
   402→                details=[{"field": "water_allocation_m3"}],
   403→            ) from None
   404→        if allocation <= 0:
   405→            raise ValidationFailed(
   406→                "water_allocation_m3 must be positive.",
   407→                details=[{"field": "water_allocation_m3"}],
   408→            )
   409→
   410→    knowledge = load_knowledge(db)
   411→    if not knowledge.crops:
   412→        raise ValidationFailed(
   413→            "No active knowledge pack with crops — import a knowledge pack first.",
   414→            details=[{"field": "knowledge"}],
   415→        )
   416→    wk_map = water_knowledge_map(knowledge)
   417→
   418→    soil_inputs, soil_warnings, soil_meta = build_soil_water_inputs(db, field, payload)
   419→
   420→    area_ha = float(field.area_ha)
   421→    label = str(payload.get("label") or "Seasonal water plan")[:120]
   422→
   423→    run = dbm.WaterPlan(
   424→        field_id=field.id,
   425→        status="running",
   426→        label=label,
   427→        # M7.1 H-01: pin the boundary version in effect (plan results are
   428→        # computed over the field's geometry — area conversions, PTF texture
   429→        # aggregation). NULL only for pre-M7.1 rows.
   430→        boundary_version=field.boundary_version,
   431→        started_at=datetime.now(UTC),
   432→        seasons=[
   433→            {"label": s.label, "start_date": s.start_date.isoformat(), "end_date": s.end_date.isoformat()}
   434→            for s in season_specs
   435→        ],
   436→        crop_assignments=list(assignments_in),
   437→        methods={
   438→            "water_engine_version": WATER_ENGINE_VERSION,
   439→            "methods_version": WATER_METHODS_VERSION,
   440→            "effective_rainfall_method": pe_method,
   441→            "pe_fraction": pe_fraction,
   442→            "irrigation_efficiency": (
   443→                {
   444→                    "value": efficiency.value,
   445→                    "evidence_class": efficiency.evidence_class,
   446→                    "method_class": efficiency.method_class,
   447→                }
   448→                if efficiency
   449→                else None
   450→            ),
   451→            "soil_source": soil_meta.get("soil_source", "none"),
   452→            "ptf": soil_meta.get("ptf"),
   453→        },
   454→        soil_inputs={
   455→            "theta_fc": soil_inputs.theta_fc if soil_inputs else None,
   456→            "theta_wp": soil_inputs.theta_wp if soil_inputs else None,
   457→            "zr_m": soil_inputs.zr_m if soil_inputs else None,
   458→            "p_table": soil_inputs.p_table if soil_inputs else None,
   459→            "source_label": soil_inputs.source_label if soil_inputs else "none",
   460→            "evidence_classes": (
   461→                {
   462→                    "theta_fc": soil_inputs.theta_fc_evidence_class,
   463→                    "theta_wp": soil_inputs.theta_wp_evidence_class,
   464→                    "zr": soil_inputs.zr_evidence_class,
   465→                    "p": soil_inputs.p_evidence_class,
   466→                }
   467→                if soil_inputs
   468→                else {}
   469→            ),
   470→        },
   471→        water_engine_version=WATER_ENGINE_VERSION,
   472→        methods_version=WATER_METHODS_VERSION,
   473→        water_allocation_m3=allocation,
   474→        request_parameters=dict(payload.get("request_parameters") or {}),
   475→        idempotency_key=idempotency_key,
   476→        created_by=getattr(created_by, "id", None),
   477→    )
   478→    db.add(run)
   479→    db.flush()
   480→
   481→    try:
   482→        evidence_run = _pick_evidence_run(db, field.id, payload.get("evidence_run_id"))
   483→        warnings: list[str] = []
   484→        if evidence_run is None:
   485→            warnings.append(
   486→                "No succeeded feature run exists for this field — every "
   487→                "climate-informed dimension stays unevaluated."
   488→            )
   489→        if evidence_run is not None and evidence_run.is_synthetic:
   490→            warnings.append(
   491→                "The evidence feature run used SYNTHETIC data — every climate-informed "
   492→                "outcome is a demonstration, never real farm evidence."
   493→            )
   494→
   495→        requests: list[tuple[WaterPlanRequest, Any, WaterEvidence]] = []
   496→        not_evaluated: dict[str, list[str]] = {}
   497→        season_results: list[dict[str, Any]] = []
   498→        feature_run_ids: list[str] = []
   499→        fallback_soil = None
   500→        if soil_inputs is not None and soil_inputs.zr_m is None:
   501→            # PTF/farmer path without explicit Zr: the crop's pack rooting depth
   502→            # is used per season (disclosed); store the decision here.
   503→            fallback_soil = soil_inputs
   504→
   505→        for season_spec, crop_assignment in zip(season_specs, assignments_in):
   506→            crop_code = (crop_assignment or {}).get("crop_code") if isinstance(crop_assignment, dict) else crop_assignment
   507→            if not crop_code:
   508→                raise ValidationFailed(
   509→                    f"Season '{season_spec.label}' needs a crop_code.",
   510→                    details=[{"field": "crop_assignments"}],
   511→                )
   512→            crop = knowledge.crops.get(crop_code)
   513→            if crop is None:
   514→                raise ValidationFailed(
   515→                    f"Unknown crop_code '{crop_code}' — the knowledge base has no such crop.",
   516→                    details=[{"field": "crop_assignments", "reason": crop_code}],
   517→                )
   518→            wk = wk_map[crop_code]
   519→            if evidence_run is None:
   520→                season_evidence = WaterEvidence(
   521→                    feature_run_id="none", is_synthetic=False,
   522→                    basis="no feature run available — no climate evidence",
   523→                )
   524→                warnings.append(
   525→                    f"Season '{season_spec.label}': planned without climate evidence; "
   526→                    "all water quantities not evaluated."
   527→                )
   528→            else:
   529→                season_evidence, ev_warnings = _season_climate_evidence(
   530→                    evidence_run, season_spec.label, season_spec.start_date, season_spec.end_date
   531→                )
   532→                warnings.extend(ev_warnings)
   533→                if str(evidence_run.id) not in feature_run_ids:
   534→                    feature_run_ids.append(str(evidence_run.id))
   535→
   536→            sowing_raw = sowings_in.get(season_spec.label) if isinstance(sowings_in, dict) else None
   537→            sowing_date = None
   538→            if sowing_raw:
   539→                try:
   540→                    sowing_date = date.fromisoformat(str(sowing_raw))
   541→                except ValueError:
   542→                    raise ValidationFailed(
   543→                        f"sowing_dates['{season_spec.label}'] must be an ISO date.",
   544→                        details=[{"field": "sowing_dates"}],
   545→                    ) from None
   546→
   547→            soil_for_season = None
   548→            if fallback_soil is not None:
   549→                # Zr from the crop's verified pack range midpoint (disclosed)
   550→                zr = wk.zr_reference_m
   551→                if zr is None and fallback_soil.zr_m is not None:
   552→                    zr = fallback_soil.zr_m
   553→                soil_for_season = SoilWaterInputs(
   554→                    theta_fc=fallback_soil.theta_fc,
   555→                    theta_fc_evidence_class=fallback_soil.theta_fc_evidence_class,
   556→                    theta_wp=fallback_soil.theta_wp,
   557→                    theta_wp_evidence_class=fallback_soil.theta_wp_evidence_class,
   558→                    zr_m=zr,
   559→                    zr_evidence_class="knowledge_pack_table22_midpoint" if zr is not None else None,
   560→                    p_table=wk.p_table if wk.p_table is not None else fallback_soil.p_table,
   561→                    p_evidence_class=wk.p_source.get("evidence_class") if (wk.p_table is not None and wk.p_source) else None,
   562→                    source_label=fallback_soil.source_label,
   563→                    source_notes=fallback_soil.source_notes,
   564→                )
   565→            elif soil_inputs is not None:
   566→                soil_for_season = soil_inputs
   567→
   568→            request = WaterPlanRequest(
   569→                season_label=season_spec.label,
   570→                start_date=season_spec.start_date,
   571→                end_date=season_spec.end_date,
   572→                crop_code=crop_code,
   573→                sowing_date=sowing_date,
   574→                pe_method=pe_method,
   575→                pe_fraction=pe_fraction,
   576→                soil=soil_for_season,
   577→                efficiency=efficiency,
   578→                water_allocation_m3=allocation,
   579→                area_ha=area_ha,
   580→            )
   581→            requests.append((request, wk, season_evidence))
   582→
   583→        plan_result = plan_rotation_water(requests)
   584→        season_results = plan_result["seasons"]
   585→
   586→        for sp in season_results:
   587→            if sp["missing_evidence"]:
   588→                not_evaluated[sp["season_label"]] = sp["missing_evidence"]
   589→
   590→        run.status = "succeeded"
   591→        run.result = plan_result
   592→        run.warnings = warnings
   593→        run.not_evaluated = not_evaluated
   594→        run.feature_run_ids = feature_run_ids
   595→        run.knowledge_pack_refs = [dict(r) for r in knowledge.pack_refs]
   596→        run.checksum_sha256 = _plan_snapshot_checksum(plan_result, run)
   597→        run.completed_at = datetime.now(UTC)
   598→        db.flush()
   599→    except ValidationFailed:
   600→        run.status = "failed"
   601→        run.error_summary = "validation failed"
   602→        run.completed_at = datetime.now(UTC)
   603→        raise
   604→    except Exception as exc:  # noqa: BLE001 — job failure accounting
   605→        run.status = "failed"
   606→        run.error_summary = f"{type(exc).__name__}: {exc}"
   607→        run.completed_at = datetime.now(UTC)
   608→        db.flush()
   609→        raise
   610→    return run

   613→def _plan_snapshot_checksum(plan_result: dict[str, Any], run: dbm.WaterPlan) -> str:
   614→    import hashlib
   615→    import json
   616→
   617→    payload = {
   618→        "engine": run.water_engine_version,
   619→        "methods": run.methods_version,
   620→        "seasons": run.seasons,
   621→        "assignments": run.crop_assignments,
   622→        "methods_cfg": run.methods,
   623→        "soil": run.soil_inputs,
   624→        "allocation": run.water_allocation_m3,
   625→        "totals": [
   626→            s["totals"] for s in plan_result.get("seasons", [])
   627→        ],
   628→        "packs": run.knowledge_pack_refs,
   629→    }
   630→    canonical = json.dumps(payload, sort_keys=True, separators=(",", ":"), default=str)
   631→    return hashlib.sha256(canonical.encode("utf-8")).hexdigest()
