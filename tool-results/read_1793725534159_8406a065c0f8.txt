    70→/** Thrown for any non-2xx API response; carries the parsed error envelope. */
    71→export class ApiError extends Error {
    72→  readonly status: number
    73→  readonly code: string
    74→  readonly details: ApiErrorDetail[]
    75→  readonly requestId: string | undefined
    76→
    77→  constructor(params: {
    78→    status: number
    79→    code: string
    80→    message: string
    81→    details?: ApiErrorDetail[]
    82→    requestId?: string
    83→  }) {
    84→    super(params.message)
    85→    this.name = 'ApiError'
    86→    this.status = params.status
    87→    this.code = params.code
    88→    this.details = params.details ?? []
    89→    this.requestId = params.requestId
    90→  }
    91→}
    92→
    93→/** Human-readable summary of any thrown error for toasts. */
    94→export function describeError(err: unknown): { title: string; description?: string } {
    95→  if (err instanceof ApiError) {
    96→    const firstDetail = err.details[0]
    97→    const description = firstDetail
    98→      ? `${firstDetail.field}: ${firstDetail.reason}${err.requestId ? ` · request ${err.requestId.slice(0, 8)}` : ''}`
    99→      : err.requestId
   100→        ? `request ${err.requestId.slice(0, 8)}`
   101→        : undefined
   102→    return { title: err.message, description }
   103→  }
   104→  if (err instanceof Error) {
   105→    return { title: 'Network error', description: err.message }
   106→  }
   107→  return { title: 'Unexpected error' }
   108→}

   110→function withPort(path: string): string {
   111→  const joiner = path.includes('?') ? '&' : '?'
   112→  return `${path}${joiner}${PORT_QUERY}`
   113→}

   115→export function getToken(): string | null {
   116→  if (typeof window === 'undefined') return null
   117→  return window.localStorage.getItem(TOKEN_STORAGE_KEY)
   118→}
   119→
   120→export function setToken(token: string | null): void {
   121→  if (typeof window === 'undefined') return
   122→  if (token) {
   123→    window.localStorage.setItem(TOKEN_STORAGE_KEY, token)
   124→  } else {
   125→    window.localStorage.removeItem(TOKEN_STORAGE_KEY)
   126→  }
   127→}

   129→interface RequestOptions {
   130→  method?: 'GET' | 'POST' | 'PATCH' | 'PUT' | 'DELETE'
   131→  body?: unknown
   132→  /** Multipart form body (file uploads). Content-Type is set by the browser. */
   133→  formData?: FormData
   134→  /** Skip attaching the bearer token (health, dev/seed-demo). */
   135→  anonymous?: boolean
   136→}
   137→
   138→async function request<T>(path: string, options: RequestOptions = {}): Promise<T> {
   139→  const headers: Record<string, string> = { Accept: 'application/json' }
   140→  if (options.body !== undefined) headers['Content-Type'] = 'application/json'
   141→  if (!options.anonymous) {
   142→    const token = getToken()
   143→    if (token) headers.Authorization = `Bearer ${token}`
   144→  }
   145→
   146→  let response: Response
   147→  try {
   148→    response = await fetch(withPort(path), {
   149→      method: options.method ?? 'GET',
   150→      headers,
   151→      body:
   152→        options.formData !== undefined
   153→          ? options.formData
   154→          : options.body !== undefined
   155→            ? JSON.stringify(options.body)
   156→            : undefined,
   157→      cache: 'no-store',
   158→    })
   159→  } catch (err) {
   160→    throw new ApiError({
   161→      status: 0,
   162→      code: 'NETWORK_ERROR',
   163→      message: 'Could not reach the FieldShift API.',
   164→      details: [],
   165→    })
   166→  }
   167→
   168→  const text = await response.text()
   169→  let payload: unknown = null
   170→  if (text) {
   171→    try {
   172→      payload = JSON.parse(text)
   173→    } catch {
   174→      payload = null
   175→    }
   176→  }
   177→
   178→  if (!response.ok) {
   179→    const envelope = payload as ApiErrorBody | null
   180→    const error = envelope && typeof envelope === 'object' && 'error' in envelope ? envelope.error : undefined
   181→    throw new ApiError({
   182→      status: response.status,
   183→      code: error?.code ?? `HTTP_${response.status}`,
   184→      message: error?.message ?? `Request failed with status ${response.status}.`,
   185→      details: error?.details ?? [],
   186→      requestId: error?.request_id,
   187→    })
   188→  }
   189→
   190→  return payload as T
   191→}

   358→export type GeometryParseResult =
   359→  | { ok: true; geometry: import('./types').GeoJSONGeometry }
   360→  | { ok: false; error: string }
   361→
   362→/** Parse + shallow-validate a GeoJSON Polygon / MultiPolygon from user input. */
   363→export function parseGeometryInput(text: string): GeometryParseResult {
   364→  let parsed: unknown
   365→  try {
   366→    parsed = JSON.parse(text)
   367→  } catch {
   368→    return { ok: false, error: 'The text is not valid JSON. Check for missing quotes, brackets or trailing commas.' }
   369→  }
   370→  if (typeof parsed !== 'object' || parsed === null || Array.isArray(parsed)) {
   371→    return { ok: false, error: 'Expected a GeoJSON object like {"type":"Polygon","coordinates":[...]}' }
   372→  }
   373→  const obj = parsed as Record<string, unknown>
   374→  if (obj.type === 'Polygon') {
   375→    if (!Array.isArray(obj.coordinates) || obj.coordinates.length === 0) {
   376→      return { ok: false, error: 'Polygon coordinates must be a non-empty array of rings.' }
   377→    }
   378→    return { ok: true, geometry: parsed as import('./types').GeoJSONPolygon }
   379→  }
   380→  if (obj.type === 'MultiPolygon') {
   381→    if (!Array.isArray(obj.coordinates) || obj.coordinates.length === 0) {
   382→      return { ok: false, error: 'MultiPolygon coordinates must be a non-empty array of polygons.' }
   383→    }
   384→    return { ok: true, geometry: parsed as import('./types').GeoJSONMultiPolygon }
   385→  }
   386→  return { ok: false, error: `Unsupported geometry type "${String(obj.type)}". Use Polygon or MultiPolygon.` }
   387→}

   406→/** Client-side guard mirroring backend job validation. Returns an error message or null. */
   407→export function validateJobWindow(
   408→  start: string,
   409→  end: string,
   410→  bounds?: { min?: string; max?: string; maxYears?: number },
   411→): string | null {
   412→  const dateRe = /^\d{4}-\d{2}-\d{2}$/
   413→  if (!dateRe.test(start) || !dateRe.test(end)) return 'Enter both dates as YYYY-MM-DD.'
   414→  const min = bounds?.min ?? COVERAGE_START
   415→  const max = bounds?.max ?? COVERAGE_END
   416→  if (start < min || end > max) {
   417→    return `Dates must fall inside the dataset coverage ${min} → ${max}.`
   418→  }
   419→  if (end < start) return 'The end date must be on or after the start date.'
   420→  const years = (Date.parse(end) - Date.parse(start)) / (365.25 * 24 * 3600 * 1000)
   421→  const maxYears = bounds?.maxYears ?? MAX_JOB_YEARS
   422→  if (years > maxYears) return `A single job covers at most ${maxYears} years. Split the range.`
   423→  return null
   424→}
