   107→/** Thrown for any non-2xx API response; carries the parsed error envelope. */
   108→export class ApiError extends Error {
   109→  readonly status: number
   110→  readonly code: string
   111→  readonly details: ApiErrorDetail[]
   112→  readonly requestId: string | undefined
   113→
   114→  constructor(params: {
   115→    status: number
   116→    code: string
   117→    message: string
   118→    details?: ApiErrorDetail[]
   119→    requestId?: string
   120→  }) {
   121→    super(params.message)
   122→    this.name = 'ApiError'
   123→    this.status = params.status
   124→    this.code = params.code
   125→    this.details = params.details ?? []
   126→    this.requestId = params.requestId
   127→  }
   128→}

   130→/** Human-readable summary of any thrown error for toasts. */
   131→export function describeError(err: unknown): { title: string; description?: string } {
   132→  if (err instanceof ApiError) {
   133→    const firstDetail = err.details[0]
   134→    const description = firstDetail
   135→      ? `${firstDetail.field}: ${firstDetail.reason}${err.requestId ? ` · request ${err.requestId.slice(0, 8)}` : ''}`
   136→      : err.requestId
   137→        ? `request ${err.requestId.slice(0, 8)}`
   138→        : undefined
   139→    return { title: err.message, description }
   140→  }
   141→  if (err instanceof Error) {
   142→    return { title: 'Network error', description: err.message }
   143→  }
   144→  return { title: 'Unexpected error' }
   145→}

   147→function withPort(path: string): string {
   148→  const joiner = path.includes('?') ? '&' : '?'
   149→  return `${path}${joiner}${PORT_QUERY}`
   150→}
   151→
   152→/**
   153→ * Exchange the HttpOnly refresh cookie for a fresh access+refresh cookie pair
   154→ * (rotating refresh: the presented token is revoked server-side; reusing a
   155→ * superseded token revokes the whole session family).
   156→ *
   157→ * Single-flight: several concurrent 401s share ONE rotation call — parallel
   158→ * rotations would present already-rotated tokens and revoke the family.
   159→ */
   160→let refreshInFlight: Promise<boolean> | null = null
   161→
   162→export async function refreshSession(): Promise<boolean> {
   163→  if (!refreshInFlight) {
   164→    refreshInFlight = authRequest('/refresh')
   165→      .then(() => true)
   166→      .catch(() => false)
   167→      .finally(() => {
   168→        refreshInFlight = null
   169→      })
   170→  }
   171→  return refreshInFlight
   172→}

   174→/** POST to a Next.js /api/auth/* route handler (session cookies live here). */
   175→async function authRequest<T>(path: string, body?: unknown): Promise<T> {
   176→  const response = await fetch(`${AUTH_BASE}${path}`, {
   177→    method: 'POST',
   178→    headers: body !== undefined ? { 'Content-Type': 'application/json' } : undefined,
   179→    body: body !== undefined ? JSON.stringify(body) : undefined,
   180→    cache: 'no-store',
   181→  })
   182→  const text = await response.text()
   183→  let payload: unknown = null
   184→  if (text) {
   185→    try {
   186→      payload = JSON.parse(text)
   187→    } catch {
   188→      payload = null
   189→    }
   190→  }
   191→  if (!response.ok) {
   192→    const envelope = payload as ApiErrorBody | null
   193→    const error =
   194→      envelope && typeof envelope === 'object' && 'error' in envelope ? envelope.error : undefined
   195→    const code = error?.code ?? `HTTP_${response.status}`
   196→    const message = error?.message ?? `Request failed with status ${response.status}.`
   197→    void notifyPaymentRequired(response.status, code, message)
   198→    throw new ApiError({
   199→      status: response.status,
   200→      code,
   201→      message,
   202→      details: error?.details ?? [],
   203→      requestId: error?.request_id,
   204→    })
   205→  }
   206→  return payload as T
   207→}
   208→
   209→/**
   210→ * P3 entitlement guard: the backend answers 402 PAYMENT_REQUIRED when a
   211→ * plan limit blocks an action. Fan the signal out globally (single toast +
   212→ * Billing navigation) instead of every component re-implementing it.
   213→ * Lazy import keeps this module SSR-safe and dependency-light.
   214→ */
   215→async function notifyPaymentRequired(status: number, code: string, message: string): Promise<void> {
   216→  if (status !== 402 && code !== 'PAYMENT_REQUIRED') return
   217→  const match = /'([a-z_]+)'/.exec(message)
   218→  const { emitPaymentRequired } = await import('./entitlement-guard')
   219→  emitPaymentRequired({ entitlement: match?.[1], message })
   220→}

   222→interface RequestOptions {
   223→  method?: 'GET' | 'POST' | 'PATCH' | 'PUT' | 'DELETE'
   224→  body?: unknown
   225→  /** Multipart form body (file uploads). Content-Type is set by the browser. */
   226→  formData?: FormData
   227→  /** Skip the 401→refresh retry (health checks, anonymous endpoints). */
   228→  anonymous?: boolean
   229→}
   230→
   231→interface RequestInternals {
   232→  /** Prevent recursive 401→refresh retries (used by refreshSession itself). */
   233→  skipAuthRetry?: boolean
   234→}
   235→
   236→async function request<T>(
   237→  path: string,
   238→  options: RequestOptions = {},
   239→  internals: RequestInternals = {},
   240→): Promise<T> {
   241→  const headers: Record<string, string> = { Accept: 'application/json' }
   242→  if (options.body !== undefined) headers['Content-Type'] = 'application/json'
   243→
   244→  let response: Response
   245→  try {
   246→    response = await fetch(withPort(path), {
   247→      method: options.method ?? 'GET',
   248→      headers,
   249→      body:
   250→        options.formData !== undefined
   251→          ? options.formData
   252→          : options.body !== undefined
   253→            ? JSON.stringify(options.body)
   254→            : undefined,
   255→      cache: 'no-store',
   256→    })
   257→  } catch (err) {
   258→    throw new ApiError({
   259→      status: 0,
   260→      code: 'NETWORK_ERROR',
   261→      message: 'Could not reach the CropTwist API.',
   262→      details: [],
   263→    })
   264→  }
   265→
   266→  const text = await response.text()
   267→  let payload: unknown = null
   268→  if (text) {
   269→    try {
   270→      payload = JSON.parse(text)
   271→    } catch {
   272→      payload = null
   273→    }
   274→  }
   275→
   276→  if (!response.ok) {
   277→    const envelope = payload as ApiErrorBody | null
   278→    const error = envelope && typeof envelope === 'object' && 'error' in envelope ? envelope.error : undefined
   279→    // Rotating-refresh retry (P0-2/P1): an access token may have simply
   280→    // expired while the session is still valid. Try ONE cookie refresh, then
   281→    // retry the original request once. Never recurse from the refresh call
   282→    // itself (it goes through authRequest, not here).
   283→    if (response.status === 401 && !options.anonymous && !internals.skipAuthRetry) {
   284→      const refreshed = await refreshSession()
   285→      if (refreshed) {
   286→        return request<T>(path, options, { skipAuthRetry: true })
   287→      }
   288→    }
   289→    const code = error?.code ?? `HTTP_${response.status}`
   290→    const message = error?.message ?? `Request failed with status ${response.status}.`
   291→    void notifyPaymentRequired(response.status, code, message)
   292→    throw new ApiError({
   293→      status: response.status,
   294→      code,
   295→      message,
   296→      details: error?.details ?? [],
   297→      requestId: error?.request_id,
   298→    })
   299→  }
   300→
   301→  return payload as T
   302→}

   451→const JOB_TERMINAL_STATUSES = new Set(['succeeded', 'failed', 'cancelled'])
   452→
   453→/**
   454→ * Poll an ingestion/import job until it reaches a terminal state.
   455→ *
   456→ * The backend executes data jobs on a background worker pool (202 + status
   457→ * 'queued'); when workers are disabled it still executes inline (the response
   458→ * is already terminal, returned as-is). Progress callbacks receive every
   459→ * intermediate snapshot (status/progress for UI feedback).
   460→ */
   461→export async function pollJobUntilDone(
   462→  jobId: string,
   463→  opts: { onProgress?: (job: DataJob) => void; intervalMs?: number; timeoutMs?: number } = {},
   464→): Promise<DataJob> {
   465→  const interval = opts.intervalMs ?? 1000
   466→  const deadline = Date.now() + (opts.timeoutMs ?? 180_000)
   467→  let current = await getJob(jobId)
   468→  while (!JOB_TERMINAL_STATUSES.has(current.status)) {
   469→    opts.onProgress?.(current)
   470→    if (Date.now() > deadline) {
   471→      throw new Error('Timed out waiting for the job to finish — check its status later.')
   472→    }
   473→    await new Promise((resolve) => setTimeout(resolve, interval))
   474→    current = await getJob(jobId)
   475→  }
   476→  return current
   477→}

   508→export type GeometryParseResult =
   509→  | { ok: true; geometry: import('./types').GeoJSONGeometry }
   510→  | { ok: false; error: string }
   511→
   512→/** Parse + shallow-validate a GeoJSON Polygon / MultiPolygon from user input. */
   513→export function parseGeometryInput(text: string): GeometryParseResult {
   514→  let parsed: unknown
   515→  try {
   516→    parsed = JSON.parse(text)
   517→  } catch {
   518→    return { ok: false, error: 'The text is not valid JSON. Check for missing quotes, brackets or trailing commas.' }
   519→  }
   520→  if (typeof parsed !== 'object' || parsed === null || Array.isArray(parsed)) {
   521→    return { ok: false, error: 'Expected a GeoJSON object like {"type":"Polygon","coordinates":[...]}' }
   522→  }
   523→  const obj = parsed as Record<string, unknown>
   524→  if (obj.type === 'Polygon') {
   525→    if (!Array.isArray(obj.coordinates) || obj.coordinates.length === 0) {
   526→      return { ok: false, error: 'Polygon coordinates must be a non-empty array of rings.' }
   527→    }
   528→    return { ok: true, geometry: parsed as import('./types').GeoJSONPolygon }
   529→  }
   530→  if (obj.type === 'MultiPolygon') {
   531→    if (!Array.isArray(obj.coordinates) || obj.coordinates.length === 0) {
   532→      return { ok: false, error: 'MultiPolygon coordinates must be a non-empty array of polygons.' }
   533→    }
   534→    return { ok: true, geometry: parsed as import('./types').GeoJSONMultiPolygon }
   535→  }
   536→  return { ok: false, error: `Unsupported geometry type "${String(obj.type)}". Use Polygon or MultiPolygon.` }
   537→}

   556→/** Client-side guard mirroring backend job validation. Returns an error message or null. */
   557→export function validateJobWindow(
   558→  start: string,
   559→  end: string,
   560→  bounds?: { min?: string; max?: string; maxYears?: number },
   561→): string | null {
   562→  const dateRe = /^\d{4}-\d{2}-\d{2}$/
   563→  if (!dateRe.test(start) || !dateRe.test(end)) return 'Enter both dates as YYYY-MM-DD.'
   564→  const min = bounds?.min ?? COVERAGE_START
   565→  const max = bounds?.max ?? COVERAGE_END
   566→  if (start < min || end > max) {
   567→    return `Dates must fall inside the dataset coverage ${min} → ${max}.`
   568→  }
   569→  if (end < start) return 'The end date must be on or after the start date.'
   570→  const years = (Date.parse(end) - Date.parse(start)) / (365.25 * 24 * 3600 * 1000)
   571→  const maxYears = bounds?.maxYears ?? MAX_JOB_YEARS
   572→  if (years > maxYears) return `A single job covers at most ${maxYears} years. Split the range.`
   573→  return null
   574→}

  1009→// --- laboratory CSV import ------------------------------------------------------
  1010→// Backend contract: POST /soil-samples/import (org-level) requires a `mapping`
  1011→// form field whose JSON carries default_field_id — the field binding arrives
  1012→// there, so the component's (fieldId, file) signature is preserved.
  1013→
  1014→export async function importSoilLabCsv(fieldId: string, file: File): Promise<SoilLabImportResponse> {
  1015→  const formData = new FormData()
  1016→  formData.append('file', file)
  1017→  formData.append('mapping', JSON.stringify({ default_field_id: fieldId }))
  1018→  const data = await request<SoilLabImportResponse>('/api/v1/soil-samples/import', {
  1019→    method: 'POST',
  1020→    formData,
  1021→  })
  1022→  // The backend's idempotent "already_imported" reply omits the counters;
  1023→  // normalize so callers always see numbers (re-import of identical content
  1024→  // creates nothing — 0 is the truthful value there).
  1025→  return {
  1026→    ...data,
  1027→    created_samples: data.created_samples ?? 0,
  1028→    created_measurements: data.created_measurements ?? 0,
  1029→    errors: data.errors ?? [],
  1030→  }
  1031→}
