    48→} from '@/components/ui/select'

    66→const PROFILES: { value: SyntheticProfile; label: string }[] = [
    67→  { value: 'temperate_seasonal', label: 'Temperate seasonal' },
    68→  { value: 'mediterranean_dry_summer', label: 'Mediterranean — dry summer' },
    69→  { value: 'semi_arid', label: 'Semi-arid' },

    71→  { value: 'arid_irrigated', label: 'Arid — irrigated' },
    72→]
    73→
    74→const DEFAULT_START = '2023-01-01'
    75→const DEFAULT_END = '2023-12-31'
    76→

    78→  title,
    79→  unit,
    80→  note,
    81→  children,
    82→}: {
    83→  title: string
    84→  unit: string
    85→  note?: string
    86→  children: React.ReactNode
    87→}) {
    88→  return (
    89→    <Card className="border-stone-200 bg-white shadow-sm">
    90→      <CardHeader className="pb-1">
    91→        <CardTitle className="text-sm font-medium text-stone-800">
    92→          {title} <span className="font-normal text-stone-500">({unit})</span>
    93→        </CardTitle>
    94→        {note ? <CardDescription className="text-xs">{note}</CardDescription> : null}
    95→      </CardHeader>
    96→      <CardContent className="pt-0">{children}</CardContent>
    97→    </Card>
    98→  )
    99→}
   100→
   101→export function Observations() {
   102→  const queryClient = useQueryClient()
   103→  const selectedField = useFieldShiftUi((s) => s.selectedField)
   104→  const setSelectedField = useFieldShiftUi((s) => s.setSelectedField)
   105→
   106→  const catalog = useCatalog()
   107→
   108→  // ---- ingestion form state (job + errors remember their field) ----
   109→  const [selectedVariables, setSelectedVariables] = useState<string[]>(['precipitation'])
   110→  const [profile, setProfile] = useState<SyntheticProfile>('mediterranean_dry_summer')
   111→  const [seed, setSeed] = useState(42)
   112→  const [jobStart, setJobStart] = useState(DEFAULT_START)
   113→  const [jobEnd, setJobEnd] = useState(DEFAULT_END)
   114→  const [formError, setFormError] = useState<{ fieldId: string; message: string } | null>(null)
   115→  const [job, setJob] = useState<{ fieldId: string; data: DataJob } | null>(null)
   116→
   117→  // ---- loaded observation window ----
   118→  const [windowStart, setWindowStart] = useState(DEFAULT_START)
   119→  const [windowEnd, setWindowEnd] = useState(DEFAULT_END)
   120→  const [appliedWindow, setAppliedWindow] = useState({ start: DEFAULT_START, end: DEFAULT_END })
   121→  const [tableVariable, setTableVariable] = useState<string>('all')
   122→
   123→  const observationsQuery = useObservations(
   124→    selectedField?.id ?? null,
   125→    appliedWindow.start,
   126→    appliedWindow.end,
   127→    Boolean(selectedField),
   128→  )
   129→  const items = useMemo(() => observationsQuery.data?.items ?? [], [observationsQuery.data])
   130→
   131→  const visibleJob = job && selectedField && job.fieldId === selectedField.id ? job.data : null
   132→  const visibleFormError =
   133→    formError && selectedField && formError.fieldId === selectedField.id ? formError.message : null
   134→
   135→  const variableName = (code: string) =>
   136→    catalog.data?.variables.find((v) => v.code === code)?.name ?? humanizeCode(code)
   137→
   138→  const ingestMutation = useMutation({
   139→    mutationFn: (fieldId: string) =>
   140→      createDataJob(fieldId, {
   141→        provider_code: 'synthetic',
   142→        variables: selectedVariables,
   143→        start_date: jobStart,
   144→        end_date: jobEnd,
   145→        profile,
   146→        seed,
   147→      }),
   148→    onSuccess: (jobResult) => {
   149→      if (!selectedField) return
   150→      setJob({ fieldId: selectedField.id, data: jobResult })
   151→      if (jobResult.status === 'succeeded') {
   152→        toast.success('Ingestion job succeeded', {
   153→          description: `${jobResult.records_received} records stored · ${jobResult.records_rejected} rejected.`,
   154→        })
   155→        const params = jobResult.request_parameters as { start_date?: string; end_date?: string }
   156→        const start = params.start_date ?? jobStart
   157→        const end = params.end_date ?? jobEnd
   158→        setWindowStart(start)
   159→        setWindowEnd(end)
   160→        setAppliedWindow({ start, end })
   161→        if (selectedField) {
   162→          queryClient.invalidateQueries({ queryKey: ['observations', selectedField.id] })
   163→          queryClient.invalidateQueries({ queryKey: ['data-passport', selectedField.id] })
   164→        }
   165→        queryClient.invalidateQueries({ queryKey: ['fields'] })
   166→      } else {
   167→        toast.error('Ingestion job failed', { description: jobResult.error_summary ?? 'No error summary returned.' })
   168→      }
   169→    },
   170→    onError: (err) => {
   171→      const { title, description } = describeError(err)
   172→      if (selectedField) setFormError({ fieldId: selectedField.id, message: title })
   173→      toast.error(title, { description })
   174→    },
   175→  })
   176→
   177→  const submitJob = () => {
   178→    setFormError(null)
   179→    if (!selectedField) return
   180→    if (selectedVariables.length === 0) {
   181→      setFormError({ fieldId: selectedField.id, message: 'Pick at least one variable to ingest.' })
   182→      return
   183→    }
   184→    const windowProblem = validateJobWindow(jobStart, jobEnd)
   185→    if (windowProblem) {
   186→      setFormError({ fieldId: selectedField.id, message: windowProblem })
   187→      return
   188→    }
   189→    ingestMutation.mutate(selectedField.id)
   190→  }
   191→
   192→  const loadWindow = () => {
   193→    const problem = validateJobWindow(windowStart, windowEnd)
   194→    if (problem) {
   195→      toast.error(problem)
   196→      return
   197→    }
   198→    setAppliedWindow({ start: windowStart, end: windowEnd })
   199→  }
   200→
   201→  // ---- derived table data ----
   202→  const presentVariables = useMemo(
   203→    () => Array.from(new Set(items.map((item) => item.variable_code))).sort(),
   204→    [items],
   205→  )
   206→  // If the selected filter variable is not present in the loaded window, fall back to "all".
   207→  const effectiveTableVariable =
   208→    tableVariable !== 'all' && !presentVariables.includes(tableVariable) ? 'all' : tableVariable
   209→  const filteredItems = useMemo(
   210→    () =>
   211→      (effectiveTableVariable === 'all'
   212→        ? items
   213→        : items.filter((item) => item.variable_code === effectiveTableVariable)
   214→      )
   215→        .slice()
   216→        .sort((a, b) => (a.observed_at < b.observed_at ? 1 : a.observed_at > b.observed_at ? -1 : 0)),
   217→    [items, effectiveTableVariable],
   218→  )
   219→
   220→  const tempSeries = useMemo(
   221→    () =>
   222→      items
   223→        .filter((o) => o.variable_code === 'air_temperature_mean' && o.value !== null)
   224→        .slice()
   225→        .sort((a, b) => (a.observed_at < b.observed_at ? -1 : 1))
   226→        .map((o) => ({ date: o.observed_at.slice(0, 10), value: o.value })),
   227→    [items],
   228→  )
   229→  const precipSeries = useMemo(
   230→    () =>
   231→      items
   232→        .filter((o) => o.variable_code === 'precipitation' && o.value !== null)
   233→        .slice()
   234→        .sort((a, b) => (a.observed_at < b.observed_at ? -1 : 1))
   235→        .map((o) => ({ date: o.observed_at.slice(0, 10), value: o.value })),
   236→    [items],
   237→  )
   238→
   239→  // ------------------------------------------------------------------ render
   240→
   241→  if (!selectedField) {
   242→    return (
   243→      <div className="space-y-4" aria-label="Observations">
   244→        <h2 className="text-lg font-semibold text-stone-900">Observations</h2>
   245→        <Card className="border-dashed border-stone-300 bg-white">
   246→          <CardContent className="flex flex-col items-center gap-3 p-8 text-center">
   247→            <Database className="h-8 w-8 text-stone-400" aria-hidden />
   248→            <p className="text-sm text-stone-500">Pick a field above to ingest and inspect its observation history.</p>
   249→          </CardContent>
   250→        </Card>
   251→      </div>
   252→    )
   253→  }
   254→
   255→  return (
   256→    <div className="space-y-6" aria-label="Observations">
   257→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   258→        <div>
   259→          <h2 className="text-lg font-semibold text-stone-900">Observations</h2>
   260→          <p className="text-sm text-stone-500">
   261→            Run synthetic ingestion jobs, then browse and chart what landed.
   262→          </p>
   263→        </div>
   264→        <FieldPicker value={selectedField.id} onValueChange={setSelectedField} />
   265→      </div>
   266→
   267→      {catalog.isLoading ? <Skeleton className="h-4 w-64" /> : null}
   268→
   269→      <div className="grid gap-6 lg:grid-cols-2">
   270→        {/* ---------------- ingestion form ---------------- */}
   271→        <Card className="border-stone-200 bg-white shadow-sm">
   272→          <CardHeader className="pb-3">
   273→            <CardTitle className="flex items-center gap-2 text-base">
   274→              <FlaskConical className="h-4 w-4 text-emerald-700" aria-hidden />
   275→              Ingest data — synthetic provider
   276→            </CardTitle>
   277→            <CardDescription>
   278→              The only provider in this slice is <code className="rounded bg-stone-100 px-1">synthetic</code>: software
   279→              fixtures for testing, always labelled SYNTHETIC downstream.
   280→            </CardDescription>
   281→          </CardHeader>
   282→          <CardContent className="space-y-4">
   283→            <div className="space-y-1.5">
   284→              <Label>Provider</Label>
   285→              <Input value="synthetic — software fixtures" disabled aria-readonly />
   286→            </div>
   287→
   288→            <div className="space-y-1.5">
   289→              <Label>Variables</Label>
   290→              {catalog.isLoading ? (
   291→                <div className="space-y-2">
   292→                  <Skeleton className="h-5 w-full" />
   293→                  <Skeleton className="h-5 w-3/4" />
   294→                </div>
   295→              ) : catalog.isError ? (
   296→                <p className="text-sm text-red-700">Could not load the data catalog.</p>
   297→              ) : (
   298→                <div className="grid gap-2 rounded-lg border border-stone-200 p-3 sm:grid-cols-2">
   299→                  {(catalog.data?.variables ?? []).map((variable) => {
   300→                    const checked = selectedVariables.includes(variable.code)
   301→                    return (
   302→                      <label
   303→                        key={variable.code}
   304→                        className="flex cursor-pointer items-start gap-2 text-sm text-stone-700"
   305→                      >
   306→                        <Checkbox
   307→                          checked={checked}
   308→                          onCheckedChange={(value) => {
   309→                            setSelectedVariables((previous) =>
   310→                              value
   311→                                ? [...previous, variable.code]
   312→                                : previous.filter((code) => code !== variable.code),
   313→                            )
   314→                          }}
   315→                          aria-label={`${variable.name} (${variable.code})`}
   316→                        />
   317→                        <span className="leading-tight">
   318→                          {variable.name}
   319→                          <span className="block text-[11px] text-stone-500">
   320→                            {variable.code} · {variable.unit}
   321→                          </span>
   322→                        </span>
   323→                      </label>
   324→                    )
   325→                  })}
   326→                </div>
   327→              )}
   328→            </div>
   329→
   330→            <div className="grid gap-4 sm:grid-cols-2">
   331→              <div className="space-y-1.5">
   332→                <Label htmlFor="obs-profile">Climate profile</Label>
   333→                <Select value={profile} onValueChange={(value) => setProfile(value as SyntheticProfile)}>
   334→                  <SelectTrigger id="obs-profile" className="w-full">
   335→                    <SelectValue />
   336→                  </SelectTrigger>
   337→                  <SelectContent>
   338→                    {PROFILES.map((entry) => (
   339→                      <SelectItem key={entry.value} value={entry.value}>
   340→                        {entry.label}
   341→                      </SelectItem>
   342→                    ))}
   343→                  </SelectContent>
   344→                </Select>
   345→                <p className="text-xs text-stone-500">Shapes the synthetic seasonality.</p>
   346→              </div>
   347→              <div className="space-y-1.5">
   348→                <Label htmlFor="obs-seed">Seed</Label>
   349→                <Input
   350→                  id="obs-seed"
   351→                  type="number"
   352→                  min={0}
   353→                  value={seed}
   354→                  onChange={(event) => setSeed(Number(event.target.value) || 0)}
   355→                />
   356→                <p className="text-xs text-stone-500">Same seed + same inputs → identical series.</p>
   357→              </div>
   358→            </div>
   359→
   360→            <div className="grid gap-4 sm:grid-cols-2">
   361→              <div className="space-y-1.5">
   362→                <Label htmlFor="obs-start">Start date</Label>
   363→                <Input
   364→                  id="obs-start"
   365→                  type="date"
   366→                  min={COVERAGE_START}
   367→                  max={COVERAGE_END}
   368→                  value={jobStart}
   369→                  onChange={(event) => setJobStart(event.target.value)}
   370→                />
   371→              </div>
   372→              <div className="space-y-1.5">
   373→                <Label htmlFor="obs-end">End date</Label>
   374→                <Input
   375→                  id="obs-end"
   376→                  type="date"
   377→                  min={COVERAGE_START}
   378→                  max={COVERAGE_END}
   379→                  value={jobEnd}
   380→                  onChange={(event) => setJobEnd(event.target.value)}
   381→                />
   382→              </div>
   383→            </div>
   384→            <p className="-mt-2 text-xs text-stone-500">
   385→              Dataset coverage {COVERAGE_START} → {COVERAGE_END}; one job spans at most {MAX_JOB_YEARS} years.
   386→            </p>
   387→
   388→            {formError ? (
   389→              <p className="text-sm font-medium text-red-700" role="alert">
   390→                {visibleFormError}
   391→              </p>
   392→            ) : null}
   393→
   394→            <Button
   395→              className="w-full bg-emerald-700 hover:bg-emerald-800 sm:w-auto"
   396→              onClick={submitJob}
   397→              disabled={ingestMutation.isPending}
   398→            >
   399→              {ingestMutation.isPending ? (
   400→                <Loader2 className="h-4 w-4 animate-spin" aria-hidden />
   401→              ) : (
   402→                <Database className="h-4 w-4" aria-hidden />
   403→              )}
   404→              Run ingestion job
   405→            </Button>
   406→          </CardContent>
   407→
   408→          {visibleJob ? (
   409→            <CardFooter className="block border-t border-stone-100 pt-4">
   410→              <div className="space-y-2" aria-live="polite">
   411→                <div className="flex flex-wrap items-center gap-2">
   412→                  <JobStatusBadge status={visibleJob.status} />
   413→                  <span className="text-xs text-stone-500">
   414→                    job {visibleJob.job_id.slice(0, 8)} · {visibleJob.provider_code}
   415→                  </span>
   416→                </div>
   417→                <div className="flex gap-4 text-sm text-stone-700">
   418→                  <span>
   419→                    Records received: <strong className="tabular-nums">{visibleJob.records_received}</strong>
   420→                  </span>
   421→                  <span>
   422→                    Rejected: <strong className="tabular-nums">{visibleJob.records_rejected}</strong>
   423→                  </span>
   424→                </div>
   425→                {visibleJob.error_summary ? (
   426→                  <Alert variant="destructive">
   427→                    <XCircle className="h-4 w-4" />
   428→                    <AlertTitle>Job error</AlertTitle>
   429→                    <AlertDescription>{visibleJob.error_summary}</AlertDescription>
   430→                  </Alert>
   431→                ) : null}
   432→                {visibleJob.warnings.length > 0 ? (
   433→                  <Alert className="border-amber-300 bg-amber-50 [&>svg]:text-amber-600">
   434→                    <AlertTriangle className="h-4 w-4" />
   435→                    <AlertTitle className="text-amber-900">Job warnings</AlertTitle>
   436→                    <AlertDescription>
   437→                      <ul className="list-disc space-y-1 pl-4 text-xs text-amber-900">
   438→                        {visibleJob.warnings.map((warning, index) => (
   439→                          <li key={index}>{warning}</li>
   440→                        ))}
   441→                      </ul>
   442→                    </AlertDescription>
   443→                  </Alert>
   444→                ) : null}
   445→                <p className="text-[11px] text-stone-500">
   446→                  Request: {JSON.stringify(visibleJob.request_parameters)}
   447→                </p>
   448→              </div>
   449→            </CardFooter>
   450→          ) : null}
   451→        </Card>
   452→
   453→        {/* ---------------- window + charts ---------------- */}
   454→        <div className="space-y-4">
   455→          <Card className="border-stone-200 bg-white shadow-sm">
   456→            <CardHeader className="pb-3">
   457→              <CardTitle className="flex items-center gap-2 text-base">
   458→                <CalendarRange className="h-4 w-4 text-emerald-700" aria-hidden />
   459→                Loaded window
   460→              </CardTitle>
   461→              <CardDescription>
   462→                Showing {observationsQuery.data ? `${observationsQuery.data.items.length} of ${observationsQuery.data.total}` : '…'}{' '}
   463→                observations for <strong className="font-medium text-stone-700">{selectedField.name}</strong>.
   464→              </CardDescription>
   465→            </CardHeader>
   466→            <CardContent className="flex flex-col gap-3 sm:flex-row sm:items-end">
   467→              <div className="flex-1 space-y-1.5">
   468→                <Label htmlFor="window-start">From</Label>
   469→                <Input
   470→                  id="window-start"
   471→                  type="date"
   472→                  min={COVERAGE_START}
   473→                  max={COVERAGE_END}
   474→                  value={windowStart}
   475→                  onChange={(event) => setWindowStart(event.target.value)}
   476→                />
   477→              </div>
   478→              <div className="flex-1 space-y-1.5">
   479→                <Label htmlFor="window-end">To</Label>
   480→                <Input
   481→                  id="window-end"
   482→                  type="date"
   483→                  min={COVERAGE_START}
   484→                  max={COVERAGE_END}
   485→                  value={windowEnd}
   486→                  onChange={(event) => setWindowEnd(event.target.value)}
   487→                />
   488→              </div>
   489→              <Button variant="outline" className="border-stone-300" onClick={loadWindow} disabled={observationsQuery.isFetching}>
   490→                {observationsQuery.isFetching ? (
   491→                  <Loader2 className="h-4 w-4 animate-spin" aria-hidden />
   492→                ) : (
   493→                  <RefreshCw className="h-4 w-4" aria-hidden />
   494→                )}
   495→                Load
   496→              </Button>
   497→            </CardContent>
   498→          </Card>
   499→
   500→          {observationsQuery.isLoading ? (
   501→            <div className="grid gap-4 sm:grid-cols-2">
   502→              <Skeleton className="h-56 rounded-lg" />
   503→              <Skeleton className="h-56 rounded-lg" />
   504→            </div>
   505→          ) : (
   506→            <div className="grid gap-4 sm:grid-cols-2">
   507→              <ChartCard
   508→                title="Air temperature (mean)"
   509→                unit="°C"
   510→                note={tempSeries.length === 0 ? 'No temperature_mean data in this window.' : undefined}
   511→              >
   512→                {tempSeries.length > 0 ? (
   513→                  <ResponsiveContainer width="100%" height={190}>
   514→                    <LineChart data={tempSeries} margin={{ top: 6, right: 8, bottom: 0, left: -18 }}>
   515→                      <CartesianGrid stroke="#e7e5e4" strokeDasharray="3 3" />
   516→                      <XAxis dataKey="date" tick={{ fontSize: 10 }} tickFormatter={(d: string) => d.slice(5)} minTickGap={36} />
   517→                      <YAxis tick={{ fontSize: 10 }} domain={['auto', 'auto']} />
   518→                      <ReTooltip labelFormatter={(label) => `Date: ${label}`} />
   519→                      <Line type="monotone" dataKey="value" stroke="#047857" strokeWidth={2} dot={false} />
   520→                    </LineChart>
   521→                  </ResponsiveContainer>
   522→                ) : (
   523→                  <div className="flex h-[190px] items-center justify-center rounded-md border border-dashed border-stone-200 text-xs text-stone-500">
   524→                    Ingest air_temperature_mean to see the line chart.
   525→                  </div>
   526→                )}
   527→              </ChartCard>
   528→              <ChartCard
   529→                title="Precipitation"
   530→                unit="mm"
   531→                note={precipSeries.length === 0 ? 'No precipitation data in this window.' : undefined}
   532→              >
   533→                {precipSeries.length > 0 ? (
   534→                  <ResponsiveContainer width="100%" height={190}>
   535→                    <BarChart data={precipSeries} margin={{ top: 6, right: 8, bottom: 0, left: -18 }}>
   536→                      <CartesianGrid stroke="#e7e5e4" strokeDasharray="3 3" />
   537→                      <XAxis dataKey="date" tick={{ fontSize: 10 }} tickFormatter={(d: string) => d.slice(5)} minTickGap={36} />
   538→                      <YAxis tick={{ fontSize: 10 }} />
   539→                      <ReTooltip labelFormatter={(label) => `Date: ${label}`} />
   540→                      <Bar dataKey="value" fill="#78716c" radius={[2, 2, 0, 0]} />
   541→                    </BarChart>
   542→                  </ResponsiveContainer>
   543→                ) : (
   544→                  <div className="flex h-[190px] items-center justify-center rounded-md border border-dashed border-stone-200 text-xs text-stone-500">
   545→                    Ingest precipitation to see the bar chart.
   546→                  </div>
   547→                )}
   548→              </ChartCard>
   549→            </div>
   550→          )}
   551→        </div>
   552→      </div>
   553→
   554→      {/* ---------------- observations table ---------------- */}
   555→      <Card className="border-stone-200 bg-white shadow-sm">
   556→        <CardHeader className="pb-3">
   557→          <div className="flex flex-wrap items-center justify-between gap-3">
   558→            <div>
   559→              <CardTitle className="flex items-center gap-2 text-base">
   560→                <BarChart3 className="h-4 w-4 text-emerald-700" aria-hidden />
   561→                Observations table
   562→              </CardTitle>
   563→              <CardDescription>
   564→                Window {appliedWindow.start} → {appliedWindow.end}
   565→                {observationsQuery.data && observationsQuery.data.total > observationsQuery.data.items.length
   566→                  ? ` · showing the first ${observationsQuery.data.items.length} of ${observationsQuery.data.total} rows — narrow the window to see everything`
   567→                  : ''}
   568→              </CardDescription>
   569→            </div>
   570→            <div className="w-full sm:w-56">
   571→              <Select value={tableVariable} onValueChange={setTableVariable}>
   572→                <SelectTrigger aria-label="Filter by variable" className="w-full bg-white">
   573→                  <SelectValue placeholder="All variables" />
   574→                </SelectTrigger>
   575→                <SelectContent>
   576→                  <SelectItem value="all">All variables</SelectItem>
   577→                  {presentVariables.map((code) => (
   578→                    <SelectItem key={code} value={code}>
   579→                      {variableName(code)}
   580→                    </SelectItem>
   581→                  ))}
   582→                </SelectContent>
   583→              </Select>
   584→            </div>
   585→          </div>
   586→        </CardHeader>
   587→        <CardContent>
   588→          {observationsQuery.isLoading ? (
   589→            <div className="space-y-2">
   590→              {Array.from({ length: 6 }).map((_, index) => (
   591→                <Skeleton key={index} className="h-9 w-full" />
   592→              ))}
   593→            </div>
   594→          ) : observationsQuery.isError ? (
   595→            <Alert variant="destructive">
   596→              <XCircle className="h-4 w-4" />
   597→              <AlertTitle>Could not load observations</AlertTitle>
   598→              <AlertDescription>
   599→                {observationsQuery.error instanceof Error ? observationsQuery.error.message : 'Try loading again.'}
   600→              </AlertDescription>
   601→            </Alert>
   602→          ) : filteredItems.length === 0 ? (
   603→            <div className="rounded-lg border border-dashed border-stone-300 p-6 text-center">
   604→              <CheckCircle2 className="mx-auto h-6 w-6 text-stone-300" aria-hidden />
   605→              <p className="mt-2 text-sm text-stone-600">
   606→                No observations in this window{tableVariable !== 'all' ? ' for this variable' : ''} yet.
   607→              </p>
   608→              <p className="mt-1 text-xs text-stone-500">
   609→                Run an ingestion job on the left, or widen the loaded window.
   610→              </p>
   611→              <div className="mt-3 flex justify-center">
   612→                <GoToFarmsButton label="Manage fields" />
   613→              </div>
   614→            </div>
   615→          ) : (
   616→            <div className="max-h-96 overflow-y-auto rounded-md border scrollbar-thin">
   617→              <Table>
   618→                <TableHeader className="sticky top-0 z-10 bg-white shadow-[0_1px_0_0_var(--color-stone-200)]">
   619→                  <TableRow className="hover:bg-transparent">
   620→                    <TableHead className="w-28">Date</TableHead>
   621→                    <TableHead>Variable</TableHead>
   622→                    <TableHead className="text-right">Value</TableHead>
   623→                    <TableHead>Quality</TableHead>
   624→                    <TableHead>Provenance</TableHead>
   625→                  </TableRow>
   626→                </TableHeader>
   627→                <TableBody>
   628→                  {filteredItems.map((observation) => (
   629→                    <TableRow key={observation.id}>
   630→                      <TableCell className="whitespace-nowrap tabular-nums text-stone-600">
   631→                        {observation.observed_at.slice(0, 10)}
   632→                      </TableCell>
   633→                      <TableCell>
   634→                        <span className="font-medium text-stone-800">{variableName(observation.variable_code)}</span>
   635→                        <span className="block text-[11px] text-stone-500">{observation.variable_code}</span>
   636→                      </TableCell>
   637→                      <TableCell className="text-right tabular-nums">
   638→                        {observation.value === null ? (
   639→                          <span className="text-stone-400">—</span>
   640→                        ) : (
   641→                          <>
   642→                            {observation.value}
   643→                            <span className="ml-1 text-xs text-stone-500">{observation.unit}</span>
   644→                          </>
   645→                        )}
   646→                      </TableCell>
   647→                      <TableCell>
   648→                        <QualityBadge status={observation.quality_status} />
   649→                      </TableCell>
   650→                      <TableCell>
   651→                        {observation.is_synthetic ? (
   652→                          <SyntheticBadge />
   653→                        ) : (
   654→                          <Badge variant="outline" className="border-stone-300 text-[10px] text-stone-500">
   655→                            external
   656→                          </Badge>
   657→                        )}
   658→                      </TableCell>
   659→                    </TableRow>
   660→                  ))}
   661→                </TableBody>
   662→              </Table>
   663→            </div>
   664→          )}
   665→        </CardContent>
   666→      </Card>
   667→    </div>
   668→  )
   669→}
