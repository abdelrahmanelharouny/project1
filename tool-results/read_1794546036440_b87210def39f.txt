    83→function reportTypeLabel(t: T, reportType: string): string {

    88→function statusLabel(t: T, status: string): string {
    89→  switch (status) {
    90→    case 'queued':
    91→      return t('reports.statusQueued')
    92→    case 'running':
    93→      return t('reports.statusRunning')
    94→    case 'succeeded':
    95→      return t('reports.statusSucceeded')
    96→    case 'failed':
    97→      return t('reports.statusFailed')
    98→    case 'expired':
    99→      return t('reports.statusExpired')
   100→    default:
   101→      return status
   102→  }
   103→}

   105→function statusBadge(t: T, status: string) {
   106→  switch (status) {
   107→    case 'succeeded':
   108→      return (
   109→        <Badge variant="outline" className="border-emerald-300 bg-emerald-50 text-emerald-800">
   110→          {statusLabel(t, status)}

   115→        <Badge variant="outline" className="gap-1.5 border-emerald-200 bg-white text-emerald-800">

   117→          {statusLabel(t, status)}

   120→    case 'queued':

   122→        <Badge variant="outline" className="gap-1.5 border-stone-300 bg-white text-stone-600">
   123→          <span className="inline-block h-1.5 w-1.5 animate-pulse rounded-full bg-stone-400" aria-hidden />
   124→          {statusLabel(t, status)}
   125→        </Badge>
   126→      )
   127→    case 'failed':
   128→      return (
   129→        <Badge variant="outline" className="border-red-200 bg-red-50 text-red-700">
   130→          {statusLabel(t, status)}

   133→    default:
   134→      return (
   135→        <Badge variant="outline" className="border-stone-200 bg-stone-50 text-stone-400">
   136→          {statusLabel(t, status)}
   137→        </Badge>

   139→  }

   142→function formatSize(bytes: number | null): string {
   143→  if (bytes === null || bytes === undefined) return '—'
   144→  if (bytes < 1024) return `${bytes} B`
   145→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
   146→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`
   147→}

   149→function toastMutationError(t: T, err: unknown): void {
   150→  if (err instanceof ApiError && err.status === 403) {
   151→    toast.error(t('errors.forbiddenTitle'), { description: t('errors.forbiddenBody') })
   152→    return
   153→  }
   154→  const d = describeError(err)
   155→  toast.error(d.title, { description: d.description })
   156→}

   175→async function saveBlob(blob: Blob, filename: string): Promise<void> {
   176→  const url = URL.createObjectURL(blob)
   177→  try {
   178→    const a = document.createElement('a')
   179→    a.href = url
   180→    a.download = filename
   181→    document.body.appendChild(a)
   182→    a.click()
   183→    a.remove()
   184→  } finally {
   185→    URL.revokeObjectURL(url)
   186→  }
   187→}

   549→                    }
   550→                    busy={downloadM.isPending || regenM.isPending || deleteM.isPending || schedM.isPending}
   551→                    onDownload={() => downloadM.mutate(job)}
   552→                    onRegen={() => regenM.mutate(job)}
   553→                    onSchedule={() => {
   554→                      setSchedJob(job)
   555→                      setSchedValue(job.interval_days === 7 ? 'weekly' : job.interval_days === 30 ? 'monthly' : 'none')
   556→                    }}
   557→                    onDelete={() => setDeleteJob(job)}
   558→                  />
   559→                ))}
   560→              </div>
   561→            </>
   562→          )}
   563→        </CardContent>
   564→      </Card>
   565→
   566→      {/* Delete confirmation */}
   567→      <AlertDialog open={deleteJob !== null} onOpenChange={(open) => !open && setDeleteJob(null)}>
   568→        <AlertDialogContent>
   569→          <AlertDialogHeader>
   570→            <AlertDialogTitle>{t('reports.delDialogTitle')}</AlertDialogTitle>
   571→            <AlertDialogDescription>{t('reports.delDialogBody')}</AlertDialogDescription>
   572→          </AlertDialogHeader>
   573→          <AlertDialogFooter>
   574→            <AlertDialogCancel>{t('common.cancel')}</AlertDialogCancel>
   575→            <AlertDialogAction
   576→              className="bg-red-700 text-white hover:bg-red-800"
   577→              onClick={(e) => {
   578→                e.preventDefault()
   579→                if (deleteJob) deleteM.mutate(deleteJob)
   580→              }}
   581→            >
   582→              {t('reports.del')}
   583→            </AlertDialogAction>
   584→          </AlertDialogFooter>
   585→        </AlertDialogContent>
   586→      </AlertDialog>
   587→
   588→      {/* Schedule dialog */}
   589→      <Dialog open={schedJob !== null} onOpenChange={(open) => !open && setSchedJob(null)}>
   590→        <DialogContent className="sm:max-w-md">
   591→          <DialogHeader>
   592→            <DialogTitle>{t('reports.schedTitle')}</DialogTitle>
   593→            <DialogDescription>{t('reports.schedDesc')}</DialogDescription>
   594→          </DialogHeader>
   595→          <div className="flex flex-col gap-1">
   596→            <Label className="text-xs text-stone-600" htmlFor="p3-rep-sched-interval">
   597→              {t('reports.schedTitle')}
   598→            </Label>
   599→            <Select
   600→              value={schedValue}
   601→              onValueChange={(v) => setSchedValue(v as 'none' | 'weekly' | 'monthly')}
   602→            >
   603→              <SelectTrigger id="p3-rep-sched-interval" className="h-9 bg-white">
   604→                <SelectValue />
   605→              </SelectTrigger>
   606→              <SelectContent>
   607→                <SelectItem value="none">{t('reports.schedNone')}</SelectItem>
   608→                <SelectItem value="weekly">{t('reports.schedWeekly')}</SelectItem>
   609→                <SelectItem value="monthly">{t('reports.schedMonthly')}</SelectItem>
   610→              </SelectContent>
   611→            </Select>
   612→          </div>
   613→          <DialogFooter>
   614→            <Button variant="outline" onClick={() => setSchedJob(null)}>
   615→              {t('common.cancel')}
   616→            </Button>
   617→            <Button
   618→              className="bg-emerald-700 text-white hover:bg-emerald-800"
   619→              disabled={schedM.isPending}
   620→              onClick={() => {
   621→                if (!schedJob) return
   622→                schedM.mutate({
   623→                  jobId: schedJob.id,
   624→                  interval: schedValue === 'none' ? null : schedValue === 'weekly' ? 7 : 30,
   625→                })
   626→              }}
   627→            >
   628→              {t('reports.schedule')}
   629→            </Button>
   630→          </DialogFooter>
   631→        </DialogContent>
   632→      </Dialog>
   633→    </div>
   634→  )
   635→}
