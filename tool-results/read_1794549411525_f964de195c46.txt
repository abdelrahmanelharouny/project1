   117→   117→export interface FarmWithFields {
   118→   118→  farm: Farm
   119→   119→  fields: Field[]
   120→   120→  fieldsQuery: UseQueryResult<FieldListResponse>
   121→   121→}
   122→   122→
   123→   123→/**
   124→   124→ * Farms with their fields flattened via parallel queries — small workspaces,
   125→   125→ * so this stays cheap. Exposed so tabs that need a global field picker
   126→   126→ * (Observations / Data Passport) can render one list.
   127→   127→ */
   128→   128→export function useFarmsWithFields(enabled = true): {
   129→   129→  farms: FarmWithFields[]
   130→   130→  allFields: Field[]
   131→   131→  isLoading: boolean
   132→   132→  hasError: boolean
   133→   133→} {
   134→   134→  const farmsQuery = useFarms(enabled)
   135→   135→  const farms = useMemo(() => farmsQuery.data?.items ?? [], [farmsQuery.data])
   136→   136→
   137→   137→  const fieldQueries = useQueries({
   138→   138→    queries: farms.map((farm) => ({
   139→   139→      queryKey: ['fields', farm.id],
   140→   140→      queryFn: () => listFields(farm.id),
   141→   141→      enabled,
   142→   142→      staleTime: 15_000,
   143→   143→    })),
   144→   144→  })
   145→   145→
   146→   146→  const farmsWithFields = useMemo<FarmWithFields[]>(() => {
   147→   147→    return farms.map((farm, index) => ({
   148→   148→      farm,
   149→   149→      fields: fieldQueries[index]?.data?.items ?? [],
   150→   150→      fieldsQuery: fieldQueries[index],
   151→   151→    }))
   152→   152→  }, [farms, fieldQueries])
   153→   153→
   154→   154→  const allFields = useMemo(() => farmsWithFields.flatMap((entry) => entry.fields), [farmsWithFields])
   155→   155→  const isLoading = farmsQuery.isLoading || fieldQueries.some((q) => q.isLoading)
   156→   156→  const hasError = farmsQuery.isError || fieldQueries.some((q) => q.isError)
   157→   157→
   158→   158→  return { farms: farmsWithFields, allFields, isLoading, hasError }
   159→   159→}
