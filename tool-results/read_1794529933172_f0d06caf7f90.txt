   113→   113→export interface FarmWithFields {
   114→   114→  farm: Farm
   115→   115→  fields: Field[]
   116→   116→  fieldsQuery: UseQueryResult<FieldListResponse>
   117→   117→}
   118→   118→
   119→   119→/**
   120→   120→ * Farms with their fields flattened via parallel queries — small workspaces,
   121→   121→ * so this stays cheap. Exposed so tabs that need a global field picker
   122→   122→ * (Observations / Data Passport) can render one list.
   123→   123→ */
   124→   124→export function useFarmsWithFields(enabled = true): {
   125→   125→  farms: FarmWithFields[]
   126→   126→  allFields: Field[]
   127→   127→  isLoading: boolean
   128→   128→  hasError: boolean
   129→   129→} {
   130→   130→  const farmsQuery = useFarms(enabled)
   131→   131→  const farms = useMemo(() => farmsQuery.data?.items ?? [], [farmsQuery.data])
   132→   132→
   133→   133→  const fieldQueries = useQueries({
   134→   134→    queries: farms.map((farm) => ({
   135→   135→      queryKey: ['fields', farm.id],
   136→   136→      queryFn: () => listFields(farm.id),
   137→   137→      enabled,
   138→   138→      staleTime: 15_000,
   139→   139→    })),
   140→   140→  })
   141→   141→
   142→   142→  const farmsWithFields = useMemo<FarmWithFields[]>(() => {
   143→   143→    return farms.map((farm, index) => ({
   144→   144→      farm,
   145→   145→      fields: fieldQueries[index]?.data?.items ?? [],
   146→   146→      fieldsQuery: fieldQueries[index],
   147→   147→    }))
   148→   148→  }, [farms, fieldQueries])
   149→   149→
   150→   150→  const allFields = useMemo(() => farmsWithFields.flatMap((entry) => entry.fields), [farmsWithFields])
   151→   151→  const isLoading = farmsQuery.isLoading || fieldQueries.some((q) => q.isLoading)
   152→   152→  const hasError = farmsQuery.isError || fieldQueries.some((q) => q.isError)
   153→   153→
   154→   154→  return { farms: farmsWithFields, allFields, isLoading, hasError }
   155→   155→}
