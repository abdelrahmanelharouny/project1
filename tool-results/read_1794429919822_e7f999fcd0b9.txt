   167→   167→def start_subscription(
   168→   168→    db: Session,
   169→   169→    principal: Principal,
   170→   170→    *,
   171→   171→    organization_id: uuid.UUID,
   172→   172→    plan_id: uuid.UUID,
   173→   173→    billing_period: str,
   174→   174→    actor: str = "self",
   175→   175→) -> dict[str, Any]:
   176→   176→    """Start a trial or paid subscription for the organization.
   177→   177→
   178→   178→    Owner/org_admin only. Creates the checkout session via the provider
   179→   179→    (sandbox) and, for plans with a trial, activates a trialing subscription
   180→   180→    immediately; paid-without-trial starts 'active' with a $0-truth invoice
   181→   181→    in the sandbox (test mode — no money moves).
   182→   182→    """
   183→   183→    principal.require_org_role(organization_id, "org_admin")
   184→   184→    if billing_period not in ("monthly", "yearly"):
   185→   185→        raise ValidationFailed("billing_period must be 'monthly' or 'yearly'.")
   186→   186→    existing = entitlements.get_live_subscription(db, organization_id)
   187→   187→    if existing is not None:
   188→   188→        raise Conflict("This organization already has a live subscription.")
   189→   189→    plan = db.get(Plan, plan_id)
   190→   190→    if plan is None or not plan.is_active:
   191→   191→        raise NotFound("Plan not found.")
   192→   192→
   193→   193→    provider = get_provider(settings.billing_provider)
   194→   194→    checkout = provider.create_checkout(
   195→   195→        organization_id=organization_id,
   196→   196→        plan_code=plan.code,
   197→   197→        billing_period=billing_period,
   198→   198→        success_url="/billing?checkout=success",
   199→   199→        cancel_url="/pricing?checkout=cancelled",
   200→   200→    )
   201→   201→    now = datetime.now(UTC)
   202→   202→    start, end = period_bounds(start=now, billing_period=billing_period)
   203→   203→    trial_ends = (
   204→   204→        now + timedelta(days=plan.trial_days) if plan.trial_days and plan.trial_days > 0 else None
   205→   205→    )
   206→   206→    status = "trialing" if trial_ends is not None else "active"
   207→   207→    amount, currency = _price_for(db, plan, billing_period)
   208→   208→    subscription = Subscription(
   209→   209→        organization_id=organization_id,
   210→   210→        plan_id=plan.id,
   211→   211→        status=status,
   212→   212→        billing_period=billing_period,
   213→   213→        provider=provider.name,
   214→   214→        provider_customer_ref=provider.create_customer(
   215→   215→            organization_id=organization_id,
   216→   216→            email=principal.user.email,
   217→   217→            name=principal.user.display_name,
   218→   218→        ),
   219→   219→        provider_subscription_ref=f"sub_sandbox_{subscription_ref_suffix(checkout.session_id)}",
   220→   220→        current_period_start=start,
   221→   221→        current_period_end=end if trial_ends is None else (now + timedelta(days=plan.trial_days)),
   222→   222→        trial_ends_at=trial_ends,
   223→   223→    )
   224→   224→    db.add(subscription)
   225→   225→    db.flush()
   226→   226→    _record_transition(
   227→   227→        db,
   228→   228→        subscription=subscription,
   229→   229→        event_type="subscription.started",
   230→   230→        to_status=status,
   231→   231→        actor_user_id=principal.id,
   232→   232→        details={
   233→   233→            "plan_code": plan.code,
   234→   234→            "billing_period": billing_period,
   235→   235→            "provider": provider.name,
   236→   236→            "mode": provider.mode,
   237→   237→            "checkout_session": checkout.session_id,
   238→   238→            "actor": actor,
   239→   239→        },
   240→   240→    )
   241→   241→    if status == "active":
   242→   242→        _create_invoice(
   243→   243→            db,
   244→   244→            subscription=subscription,
   245→   245→            period_start=start,
   246→   246→            period_end=end,
   247→   247→            amount_cents=amount or 0,
   248→   248→            currency_code=currency,
   249→   249→        )
   250→   250→    activity.record_event(
   251→   251→        db,
   252→   252→        organization_id=organization_id,
   253→   253→        event_type=activity.EVENT_PLAN_CHANGED,
   254→   254→        summary=f"Subscribed to the {plan.name} plan ({billing_period}, sandbox test mode)",
   255→   255→        actor_user_id=principal.id,
   256→   256→        entity_kind="subscription",
   257→   257→        entity_id=str(subscription.id),
   258→   258→        details={"plan_code": plan.code, "mode": provider.mode},
   259→   259→    )
   260→   260→    db.commit()
   261→   261→    db.refresh(subscription)
   262→   262→    return {"subscription": subscription, "checkout": checkout}

   275→   275→def change_plan(
   276→   276→    db: Session,
   277→   277→    principal: Principal,
   278→   278→    *,
   279→   279→    organization_id: uuid.UUID,
   280→   280→    new_plan_id: uuid.UUID,
   281→   281→    billing_period: str | None = None,
   282→   282→) -> Subscription:
   283→   283→    """Upgrade/downgrade effective immediately (sandbox semantics: proration
   284→   284→    is NOT simulated; the next invoice reflects the new plan price)."""
   285→   285→    principal.require_org_role(organization_id, "org_admin")
   286→   286→    subscription = entitlements.get_live_subscription(db, organization_id)
   287→   287→    if subscription is None:

   289→   289→    plan = db.get(Plan, new_plan_id)
   290→   290→    if plan is None or not plan.is_active:

   294→   294→    if period not in ("monthly", "yearly"):
   295→   295→        raise ValidationFailed("billing_period must be 'monthly' or 'yearly'.")

   298→   298→        subscription.billing_period = billing_period
   299→   299→        start, end = period_bounds(start=subscription.current_period_end, billing_period=period)
   300→   300→        subscription.current_period_start = subscription.current_period_end

   303→   303→        db,
   304→   304→        subscription=subscription,
   305→   305→        event_type="subscription.plan_changed",
   306→   306→        to_status=None,
   307→   307→        actor_user_id=principal.id,
   308→   308→        details={
   309→   309→            "from_plan": old_plan.code if old_plan else None,
   310→   310→            "to_plan": plan.code,
   311→   311→            "billing_period": period,
   312→   312→        },
   313→   313→    )
   314→   314→    activity.record_event(
   315→   315→        db,
   316→   316→        organization_id=organization_id,
   317→   317→        event_type=activity.EVENT_PLAN_CHANGED,
   318→   318→        summary=f"Plan changed to {plan.name} (sandbox test mode)",
   319→   319→        actor_user_id=principal.id,
   320→   320→        entity_kind="subscription",
   321→   321→        entity_id=str(subscription.id),
   322→   322→        details={"to_plan": plan.code},
   323→   323→    )
   324→   324→    db.commit()
   325→   325→    db.refresh(subscription)
   326→   326→    return subscription

   399→   399→# -------------------------------------------------------- webhook pipeline
   400→   400→def process_billing_webhook(
   401→   401→    db: Session,
   402→   402→    *,
   403→   403→    provider: str,
   404→   404→    event_type: str,
   405→   405→    provider_event_id: str,
   406→   406→    payload: dict[str, Any],
   407→   407→    signature_valid: bool,
   408→   408→) -> BillingEvent:
   409→   409→    """Idempotent billing webhook ingestion (P3 §4.4).
   410→   410→
   411→   411→    - Signature-invalid events are REJECTED (status 'failed') and never
   412→   412→      processed.
   413→   413→    - (provider, provider_event_id) unique: replays return the original row
   414→   414→      with status 'duplicate' and never double-process (no duplicate
   415→   415→      invoices/transactions).
   416→   416→    - Handled event types: payment.succeeded, payment.declined,
   417→   417→      subscription.renewal_due, subscription.expired.
   418→   418→    """
   419→   419→    from app.core.errors import Forbidden
   420→   420→
   421→   421→    if not signature_valid:
   422→   422→        event = BillingEvent(
   423→   423→            provider=provider,
   424→   424→            event_type=event_type[:80],
   425→   425→            provider_event_id=provider_event_id[:200],
   426→   426→            signature_valid=False,
   427→   427→            status="failed",
   428→   428→            error="invalid webhook signature",
   429→   429→            payload={},
   430→   430→        )
   431→   431→        db.add(event)
   432→   432→        db.commit()
   433→   433→        raise Forbidden("Webhook signature verification failed.")
   434→   434→
   435→   435→    existing = db.scalar(
   436→   436→        select(BillingEvent).where(
   437→   437→            BillingEvent.provider == provider, BillingEvent.provider_event_id == provider_event_id
   438→   438→        )
   439→   439→    )
   440→   440→    if existing is not None:
   441→   441→        existing.status = "duplicate" if existing.status == "processed" else existing.status
   442→   442→        db.commit()
   443→   443→        return existing
   444→   444→
   445→   445→    event = BillingEvent(
   446→   446→        provider=provider,
   447→   447→        event_type=event_type[:80],
   448→   448→        provider_event_id=provider_event_id[:200],
   449→   449→        signature_valid=True,
   450→   450→        status="received",
   451→   451→        payload=payload,
   452→   452→    )
   453→   453→    db.add(event)
   454→   454→    db.flush()
   455→   455→
   456→   456→    org_ref = payload.get("organization_id")
   457→   457→    try:
   458→   458→        if not isinstance(org_ref, str) or not org_ref:
   459→   459→            raise ValidationFailed("payload.organization_id is required.")
   460→   460→        organization_id = uuid.UUID(org_ref)
   461→   461→        subscription = entitlements.get_live_subscription(db, organization_id)
   462→   462→        if subscription is None:
   463→   463→            raise NotFound("No live subscription for this organization.")
   464→   464→
   465→   465→        if event_type == "payment.succeeded":
   466→   466→            amount = int(payload.get("amount_cents") or 0)
   467→   467→            _settle_payment(db, subscription=subscription, amount_cents=amount)
   468→   468→        elif event_type == "payment.declined":
   469→   469→            _enter_past_due(db, subscription=subscription)
   470→   470→        elif event_type == "subscription.renewal_due":
   471→   471→            _renew(db, subscription=subscription, paid=payload.get("simulate_outcome") != "declined")
   472→   472→        elif event_type == "subscription.ended":
   473→   473→            _record_transition(
   474→   474→                db,
   475→   475→                subscription=subscription,
   476→   476→                event_type="subscription.ended",
   477→   477→                to_status="canceled",
   478→   478→                actor_user_id=None,
   479→   479→                details={"source": "provider_webhook"},
   480→   480→            )
   481→   481→        else:
   482→   482→            raise ValidationFailed(f"Unhandled billing event type '{event_type}'.")
   483→   483→        event.status = "processed"
   484→   484→        event.processed_at = datetime.now(UTC)
   485→   485→    except Exception as exc:
   486→   486→        event.status = "failed"
   487→   487→        event.error = f"{type(exc).__name__}: {exc}"[:2000]
   488→   488→        db.commit()
   489→   489→        raise
   490→   490→    db.commit()
   491→   491→    db.refresh(event)
   492→   492→    return event

   495→   495→def _settle_payment(db: Session, *, subscription: Subscription, amount_cents: int) -> None:
   496→   496→    """Payment success: clear past_due/grace, open invoice → paid."""
   497→   497→    now = datetime.now(UTC)
   498→   498→    open_invoice = db.scalar(
   499→   499→        select(Invoice)
   500→   500→        .where(
   501→   501→            Invoice.organization_id == subscription.organization_id,
   502→   502→            Invoice.status == "open",
   503→   503→        )
   504→   504→        .order_by(Invoice.issued_at.desc())
   505→   505→        .limit(1)
   506→   506→    )
   507→   507→    if open_invoice is not None:
   508→   508→        open_invoice.status = "paid"
   509→   509→        open_invoice.paid_at = now
   510→   510→    if subscription.status in ("past_due", "grace"):
   511→   511→        _record_transition(
   512→   512→            db,
   513→   513→            subscription=subscription,
   514→   514→            event_type="payment.succeeded",
   515→   515→            to_status="active",
   516→   516→            actor_user_id=None,
   517→   517→            details={"amount_cents": amount_cents},
   518→   518→        )
   519→   519→        subscription.grace_ends_at = None
   520→   520→    from app.notifications import service as notifications
   521→   521→
   522→   522→    notifications.create_notification(
   523→   523→        db,
   524→   524→        organization_id=subscription.organization_id,
   525→   525→        category="billing",
   526→   526→        severity="info",
   527→   527→        title="Payment received",
   528→   528→        body="Your latest invoice has been paid. Thank you.",
   529→   529→        dedupe_key=f"billing:paid:{subscription.id}:{now.date().isoformat()}",
   530→   530→    )

   558→   558→def _renew(db: Session, *, subscription: Subscription, paid: bool) -> None:
   559→   559→    """Period rollover: honor cancel_at_period_end, expire trials, renew."""
   560→   560→    now = datetime.now(UTC)
   561→   561→    if subscription.cancel_at_period_end:
   562→   562→        subscription.canceled_at = now
   563→   563→        _record_transition(
   564→   564→            db,
   565→   565→            subscription=subscription,
   566→   566→            event_type="subscription.canceled",
   567→   567→            to_status="canceled",
   568→   568→            actor_user_id=None,
   569→   569→            details={"reason": "cancel_at_period_end"},
   570→   570→        )
   571→   571→        return
   572→   572→    # Trial expiry without payment setup → active (sandbox) unless declined.
   573→   573→    plan = db.get(Plan, subscription.plan_id)
   574→   574→    amount, currency = _price_for(db, plan, subscription.billing_period) if plan else (0, "USD")
   575→   575→    start, end = period_bounds(start=subscription.current_period_end, billing_period=subscription.billing_period)
   576→   576→    subscription.current_period_start = start
   577→   577→    subscription.current_period_end = end
   578→   578→    if subscription.status == "trialing":
   579→   579→        _record_transition(
   580→   580→            db,
   581→   581→            subscription=subscription,
   582→   582→            event_type="trial.ended",
   583→   583→            to_status="active" if paid else "past_due",
   584→   584→            actor_user_id=None,
   585→   585→        )
   586→   586→        if not paid:
   587→   587→            subscription.grace_ends_at = grace_deadline(now)
   588→   588→            return
   589→   589→    _create_invoice(
   590→   590→        db,
   591→   591→        subscription=subscription,
   592→   592→        period_start=start,
   593→   593→        period_end=end,
   594→   594→        amount_cents=amount or 0,
   595→   595→        currency_code=currency,
   596→   596→    )
   597→   597→    if not paid:
   598→   598→        _enter_past_due(db, subscription=subscription)
   599→   599→

   601→   601→def expire_stale_subscriptions(db: Session) -> int:
   602→   602→    """Scheduler sweep: cancel_at_period_end subscriptions past period end
   603→   603→    and grace windows that lapsed → 'expired'/'canceled'."""
   604→   604→    now = datetime.now(UTC)
   605→   605→    changed = 0
   606→   606→    rows = list(
   607→   607→        db.scalars(select(Subscription).where(Subscription.status.in_(LIVE_STATUSES))).all()
   608→   608→    )
   609→   609→    for sub in rows:
   610→   610→        if sub.cancel_at_period_end and sub.current_period_end < now:
   611→   611→            sub.canceled_at = now
   612→   612→            _record_transition(
   613→   613→                db,
   614→   614→                subscription=sub,
   615→   615→                event_type="subscription.canceled",
   616→   616→                to_status="canceled",
   617→   617→                actor_user_id=None,
   618→   618→                details={"reason": "period_ended_after_cancel_request"},
   619→   619→            )
   620→   620→            changed += 1
   621→   621→        elif sub.status == "past_due" and sub.grace_ends_at is not None and sub.grace_ends_at < now:
   622→   622→            _record_transition(
   623→   623→                db,
   624→   624→                subscription=sub,
   625→   625→                event_type="subscription.grace_ended",
   626→   626→                to_status="grace",
   627→   627→                actor_user_id=None,
   628→   628→                details={"note": "grace window lapsed — paid capabilities locked"},
   629→   629→            )
   630→   630→            changed += 1
   631→   631→    if changed:
   632→   632→        db.commit()
   633→   633→    return changed
