    61→def _validate_seasons(raw: list[dict[str, Any]]) -> list[SeasonSpec]:
    62→    if not raw:
    63→        raise ValidationFailed(
    64→            "At least one planning season is required.",
    65→            details=[{"field": "seasons"}],
    66→        )
    67→    if len(raw) > MAX_SEASONS:
    68→        raise ValidationFailed(
    69→            f"At most {MAX_SEASONS} planning seasons per run.",
    70→            details=[{"field": "seasons"}],
    71→        )
    72→    seasons: list[SeasonSpec] = []
    73→    labels: set[str] = set()
    74→    for item in raw:
    75→        label = str(item.get("label") or "").strip()
    76→        if not label or len(label) > 60:
    77→            raise ValidationFailed(
    78→                "Season labels must be 1-60 characters.",
    79→                details=[{"field": "seasons", "reason": str(item)[:80]}],
    80→            )
    81→        if label in labels:
    82→            raise ValidationFailed(
    83→                f"Duplicate season label '{label}'.",
    84→                details=[{"field": "seasons", "reason": label}],
    85→            )
    86→        labels.add(label)
    87→        try:
    88→            start = date.fromisoformat(str(item["start_date"]))
    89→            end = date.fromisoformat(str(item["end_date"]))
    90→        except (KeyError, ValueError):
    91→            raise ValidationFailed(
    92→                "Seasons need start_date and end_date (ISO dates).",
    93→                details=[{"field": "seasons", "reason": label}],
    94→            ) from None
    95→        if start > end:
    96→            raise ValidationFailed(
    97→                f"Season '{label}' start_date is after end_date.",
    98→                details=[{"field": "seasons", "reason": label}],
    99→            )
   100→        if (end - start).days > 400:
   101→            raise ValidationFailed(
   102→                f"Season '{label}' is longer than 400 days — split it.",
   103→                details=[{"field": "seasons", "reason": label}],
   104→            )
   105→        seasons.append(SeasonSpec(label=label, start_date=start, end_date=end))
   106→    return seasons

   109→def _pick_evidence_run(db: Session, field_id: Any, explicit_id: str | None) -> dbm.FeatureRun | None:
   110→    if explicit_id:
   111→        run = db.get(dbm.FeatureRun, uuid.UUID(explicit_id))
   112→        if run is None or str(run.field_id) != str(field_id):
   113→            raise NotFound("Feature run not found for this field.")
   114→        if run.status != "succeeded":
   115→            raise ValidationFailed(
   116→                "The referenced feature run has not succeeded.",
   117→                details=[{"field": "evidence_run_id", "reason": run.status}],
   118→            )
   119→        return run
   120→    return db.scalar(
   121→        select(dbm.FeatureRun)
   122→        .where(dbm.FeatureRun.field_id == field_id, dbm.FeatureRun.status == "succeeded")
   123→        .order_by(dbm.FeatureRun.completed_at.desc())
   124→        .limit(1)

   128→def _season_climate(
   129→    run: dbm.FeatureRun, season: SeasonSpec
   130→) -> ClimateEvidence | None:
   131→    """Aggregate the run's monthly feature values over one planning season as a
   132→    MONTH-OF-YEAR climatology (historical proxy — planning seasons are usually
   133→    in the future; no forecast is claimed).
   134→
   135→    Coverage = fraction of season days whose month-of-year has any covered
   136→    value in the run; below MIN_CLIMATE_COVERAGE the season's climate evidence
   137→    is None (the dimension is unevaluated, never approximated)."""
   138→    monthly: dict[str, dict[int, list[tuple[int, float]]]] = {}  # code -> month -> [(year, value)]
   139→    for row in run.values:
   140→        if row.aggregation_level != "monthly" or row.value is None:
   141→            continue
   142→        monthly.setdefault(row.feature_code, {}).setdefault(
   143→            row.period_start.month, []
   144→        ).append((row.period_start.year, row.value))
   145→
   146→    import calendar as cal
   147→
   148→    def months_in_season(season: SeasonSpec) -> list[tuple[int, int, int]]:
   149→        """(year, month, days of that month inside the season)."""
   150→        out = []
   151→        y, m = season.start_date.year, season.start_date.month
   152→        while (y, m) <= (season.end_date.year, season.end_date.month):
   153→            if y == season.start_date.year and m == season.start_date.month:
   154→                month_start = season.start_date
   155→            else:
   156→                month_start = date(y, m, 1)
   157→            if y == season.end_date.year and m == season.end_date.month:
   158→                month_end = season.end_date
   159→            else:
   160→                month_end = date(y, m, cal.monthrange(y, m)[1])
   161→            out.append((y, m, (month_end - month_start).days + 1))
   162→            m += 1
   163→            if m == 13:
   164→                y, m = y + 1, 1
   165→        return out
   166→
   167→    month_days = months_in_season(season)
   168→    total_days = sum(d for _, _, d in month_days)
   169→
   170→    years_used = sorted(
   171→        {
   172→            y
   173→            for table in monthly.values()
   174→            for entries in table.values()
   175→            for y, _v in entries
   176→        }
   177→    )
   178→
   179→    def collect(code: str, mode: str) -> tuple[float | None, int]:
   180→        """Season aggregate of month-of-year climatological means.
   181→
   182→        mode='total' (precip, ET0, hot-day counts): monthly TOTALS scaled by
   183→        the fraction of that month inside the season, summed.
   184→        mode='mean' (temperature): monthly means day-weighted over the season.
   185→        Returns (value, days covered by present months)."""
   186→        table = monthly.get(code, {})
   187→        acc = 0.0
   188→        covered = 0
   189→        for y, m, days in month_days:
   190→            entries = table.get(m)
   191→            if entries:
   192→                mean = sum(v for _yr, v in entries) / len(entries)
   193→                if mode == "total":
   194→                    frac = days / cal.monthrange(y, m)[1]
   195→                    acc += mean * frac
   196→                else:
   197→                    acc += mean * days
   198→                covered += days
   199→        if mode == "mean":
   200→            acc = acc / total_days if total_days else 0.0
   201→        return acc, covered
   202→
   203→    warnings: list[str] = []
   204→    precip_total, covered = collect(CLIMATE_FEATURE_CODES["precipitation"], "total")
   205→    tmean_wsum, covered_t = collect(CLIMATE_FEATURE_CODES["tmean"], "mean")
   206→    hot_days, covered_h = collect(CLIMATE_FEATURE_CODES["hot_days"], "total")
   207→    et0_total, covered_e = collect(CLIMATE_FEATURE_CODES["et0"], "total")
   208→
   209→    coverage = covered / total_days if total_days else 0.0
   210→
   211→    monthly_et0: list[tuple[int, int, float]] = []
   212→    et0_table = monthly.get(CLIMATE_FEATURE_CODES["et0"], {})
   213→    for y, m, _days in month_days:
   214→        entries = et0_table.get(m)
   215→        if entries:
   216→            monthly_et0.append((y, m, sum(v for _yr, v in entries) / len(entries)))
   217→
   218→    # M6A: month-of-year precipitation climatology for the net irrigation
   219→    # requirement (months absent stay absent — never zero-filled).
   220→    monthly_precip: list[tuple[int, int, float]] = []
   221→    precip_table = monthly.get(CLIMATE_FEATURE_CODES["precipitation"], {})
   222→    for y, m, _days in month_days:
   223→        entries = precip_table.get(m)
   224→        if entries:
   225→            monthly_precip.append((y, m, sum(v for _yr, v in entries) / len(entries)))
   226→
   227→    if coverage < MIN_CLIMATE_COVERAGE:
   228→        return None
   229→    if covered < total_days:
   230→        warnings.append(
   231→            "precipitation months incomplete — the season total includes only the "
   232→            "months present in the evidence run (an under-count, never filled)"
   233→        )
   234→
   235→    tmean = tmean_wsum if covered_t == total_days else (
   236→        tmean_wsum * total_days / covered_t if covered_t else None
   237→    )
   238→    if covered_t < total_days:
   239→        warnings.append("temperature months incomplete — mean weighted by covered days")
   240→    hot = round(hot_days, 1) if covered_h == total_days else None
   241→    if covered_h < total_days and hot is None:
   242→        warnings.append("hot-day counts incomplete — not evaluated for this season")
   243→    et0 = et0_total if covered_e == total_days else None
   244→    if covered_e < total_days and et0 is None:
   245→        warnings.append("ET0 months incomplete — water estimates not evaluated")
   246→
   247→    year_note = (
   248→        f"years {years_used[0]}-{years_used[-1]}" if len(years_used) > 1
   249→        else f"year {years_used[0]}" if years_used else "no years"
   250→    )
   251→    return ClimateEvidence(
   252→        feature_run_id=str(run.id),
   253→        is_synthetic=bool(run.is_synthetic),
   254→        season_precipitation_mm=round(precip_total, 2) if covered else None,
   255→        season_mean_temp_degC=round(tmean, 2) if tmean is not None else None,
   256→        season_mean_tmax_degC=None,
   257→        season_hot_days_35c=hot,
   258→        season_et0_mm=round(et0, 2) if et0 is not None else None,
   259→        monthly_et0=tuple(monthly_et0),
   260→        monthly_precip=tuple(monthly_precip),
   261→        coverage_ratio=round(coverage, 4),
   262→        years_used=tuple(years_used),
   263→        basis=(
   264→            f"month-of-year climatology from feature run {str(run.id)[:8]} "
   265→            f"({year_note}) — a historical proxy, NOT a forecast for the "
   266→            "planning season."
   267→        ),
   268→        warnings=tuple(warnings),
   269→    )

   272→def _soil_evidence(db: Session, field_id: Any) -> tuple[SoilEvidence | None, str | None]:
   273→    rows = db.execute(
   274→        select(dbm.Observation)
   275→        .where(
   276→            dbm.Observation.field_id == field_id,
   277→            dbm.Observation.variable_code.in_(SOIL_VARIABLES),
   278→            dbm.Observation.value.is_not(None),
   279→        )
   280→        .order_by(dbm.Observation.observed_at.desc())
   281→        .limit(50)
   282→    ).scalars().all()
   283→    latest: dict[str, dbm.Observation] = {}
   284→    for obs in rows:
   285→        latest.setdefault(obs.variable_code, obs)
   286→    if not latest:
   287→        return None, None
   288→    evidence_classes = sorted({o.evidence_class or "unknown" for o in latest.values()})
   289→
   290→    def val(code: str) -> tuple[float, str] | None:
   291→        obs = latest.get(code)
   292→        if obs is None or obs.value is None:
   293→            return None
   294→        return (float(obs.value), obs.evidence_class or "unknown")
   295→
   296→    soil = SoilEvidence(
   297→        ph=val("soil_ph"),
   298→        organic_carbon_g_kg=val("soil_organic_carbon"),
   299→        clay_g_kg=val("soil_clay_content"),
   300→        sand_g_kg=val("soil_sand_content"),
   301→        bulk_density=val("soil_bulk_density"),
   302→    )
   303→    return soil, ", ".join(evidence_classes)

   306→def build_farm_context(
   307→    db: Session,
   308→    field: dbm.Field,
   309→    seasons: list[SeasonSpec],
   310→    evidence_run_id: str | None,
   311→    hemisphere: str,
   312→) -> tuple[FarmContext, list[str], list[str], dbm.FeatureRun | None]:
   313→    warnings: list[str] = []
   314→    run = _pick_evidence_run(db, field.id, evidence_run_id)
   315→    climate_by_season: dict[str, ClimateEvidence] = {}
   316→    climate_run_refs: dict[str, str] = {}
   317→    if run is None:
   318→        warnings.append(
   319→            "No succeeded feature run exists for this field — climate dimensions are "
   320→            "unevaluated (missing evidence is never treated as zero)."
   321→        )
   322→    else:
   323→        for season in seasons:
   324→            evidence = _season_climate(run, season)
   325→            if evidence is None:
   326→                warnings.append(
   327→                    f"Season '{season.label}': climate coverage below "
   328→                    f"{MIN_CLIMATE_COVERAGE:.0%} in the evidence run — climate dimensions "
   329→                    "unevaluated for this season."
   330→                )
   331→            else:
   332→                climate_by_season[season.label] = evidence
   333→                climate_run_refs[season.label] = str(run.id)
   334→                for w in evidence.warnings:
   335→                    warnings.append(f"Season '{season.label}': {w}")
   336→        if run.is_synthetic:
   337→            warnings.append(
   338→                "The evidence feature run used SYNTHETIC data — every climate-informed "
   339→                "outcome is a demonstration, never real farm evidence."
   340→            )
   341→
   342→    soil, soil_class = _soil_evidence(db, field.id)
   343→
   344→    history_rows = db.scalars(
   345→        select(dbm.CropHistory)
   346→        .where(dbm.CropHistory.field_id == field.id)
   347→        .order_by(dbm.CropHistory.sowing_date.asc().nullslast(), dbm.CropHistory.created_at.asc())
   348→    ).all()
   349→    crop_code_by_id = {
   350→        c.id: c.crop_code
   351→        for c in db.scalars(select(dbm.Crop)).all()
   352→    }
   353→    history: list[dict[str, Any]] = []
   354→    for h in history_rows:
   355→        history.append({
   356→            "id": str(h.id),
   357→            "season_label": h.season_label,
   358→            "crop_code": crop_code_by_id.get(h.crop_id) if h.crop_id else None,
   359→            "crop_label": h.crop_label,
   360→            "sowing_date": h.sowing_date.isoformat() if h.sowing_date else None,
   361→            "harvest_date": h.harvest_date.isoformat() if h.harvest_date else None,
   362→        })
   363→
   364→    constraints_rows = db.scalars(
   365→        select(dbm.FarmConstraint).where(dbm.FarmConstraint.farm_id == field.farm_id)
   366→    ).all()
   367→    market_exclusions: list[str] = []
   368→    equipment_limits: list[str] = []
   369→    water_allocation: float | None = None
   370→    margin_index: dict[str, float] = {}
   371→    for c in constraints_rows:
   372→        if c.constraint_code == "market_exclusion" and c.value_text:
   373→            market_exclusions.extend(
   374→                [p.strip() for p in c.value_text.split(",") if p.strip()]
   375→            )
   376→        elif c.constraint_code == "equipment_limitation" and c.value_text:
   377→            equipment_limits.extend(
   378→                [p.strip() for p in c.value_text.split(",") if p.strip()]
   379→            )
   380→        elif c.constraint_code == "water_allocation_m3_per_season" and c.value_numeric is not None:
   381→            water_allocation = float(c.value_numeric)
   382→        elif c.constraint_code == "market_margin_index" and c.value_text and c.value_numeric is not None:
   383→            margin_index[c.value_text.strip()] = float(c.value_numeric)
   384→
   385→    farm = FarmContext(
   386→        field_id=str(field.id),
   387→        farm_id=str(field.farm_id),
   388→        area_ha=float(field.area_ha),
   389→        country_code=field.farm.country_code,
   390→        hemisphere=hemisphere,
   391→        seasons=tuple(seasons),
   392→        soil=soil,
   393→        climate_by_season=climate_by_season,
   394→        climate_run_refs=climate_run_refs,
   395→        history=tuple(history),
   396→        market_exclusions=tuple(sorted(set(market_exclusions))),
   397→        equipment_limits=tuple(sorted(set(equipment_limits))),
   398→        water_allocation_m3=water_allocation,
   399→        margin_index_by_crop=margin_index,
   400→        synthetic_evidence_in_use=bool(run.is_synthetic) if run else False,
   401→        soil_evidence_class=soil_class,
   402→    )
   403→    return farm, warnings, [str(h["id"]) for h in history], run

   406→def _baseline_assignment(
   407→    seasons: list[SeasonSpec], history: tuple[dict[str, Any], ...]
   408→) -> tuple[tuple[str, str | None], ...]:
   409→    assignment: list[tuple[str, str | None]] = []
   410→    for season in seasons:
   411→        crop_code = None
   412→        for h in history:
   413→            if h.get("season_label") == season.label:
   414→                crop_code = h.get("crop_code")
   415→                break
   416→        if crop_code is None:
   417→            for h in history:
   418→                sow = h.get("sowing_date")
   419→                harv = h.get("harvest_date")
   420→                if sow and harv:
   421→                    if date.fromisoformat(sow) <= season.end_date and date.fromisoformat(
   422→                        harv
   423→                    ) >= season.start_date:
   424→                        crop_code = h.get("crop_code")
   425→                        break
   426→        assignment.append((season.label, crop_code))
   427→    return tuple(assignment)

   430→def _solution_checksum(
   431→    solution: StrategySolution, problem: RotationProblem
   432→) -> str:
   433→    payload = {
   434→        "strategy": solution.strategy_key,
   435→        "assignment": [[lbl, cc] for lbl, cc in solution.assignment],
   436→        "objective": solution.objective_value,
   437→        "criteria": [
   438→            [c.criterion_code, c.raw_value, c.normalized] for c in solution.criteria
   439→        ],
   440→        "problem": problem.checksum(),
   441→        "engine": ROTATION_ENGINE_VERSION,
   442→    }
   443→    canonical = json.dumps(payload, sort_keys=True, separators=(",", ":"))
   444→    return hashlib.sha256(canonical.encode("utf-8")).hexdigest()

   447→def start_rotation_run(
   448→    db: Session,
   449→    *,
   450→    field: dbm.Field,
   451→    payload: dict[str, Any],
   452→    created_by: Any = None,
   453→) -> dbm.RotationRun:
   454→    """Validate, execute inline and persist a rotation run (202-style job)."""
   455→    idempotency_key = payload.get("idempotency_key")
   456→    if idempotency_key:
   457→        # Deepening F-3: replay lookup scoped to THIS field (cross-tenant
   458→        # replay of a known key must never serialize another tenant's run).
   459→        existing = db.scalar(
   460→            select(dbm.RotationRun).where(
   461→                dbm.RotationRun.idempotency_key == idempotency_key,
   462→                dbm.RotationRun.field_id == field.id,
   463→            )
   464→        )
   465→        if existing is not None:
   466→            return existing
   467→
   468→    seasons = _validate_seasons(payload.get("seasons") or [])
   469→    hemisphere = payload.get("hemisphere") or "north"
   470→    if hemisphere not in ("north", "south"):
   471→        raise ValidationFailed(
   472→            "hemisphere must be 'north' or 'south'.",
   473→            details=[{"field": "hemisphere", "reason": hemisphere}],
   474→        )
   475→
   476→    requested_profiles = payload.get("strategy_profiles") or list(optimizer_mod.DEFAULT_PROFILES)
   477→    if not isinstance(requested_profiles, list) or not requested_profiles:
   478→        raise ValidationFailed(
   479→            "strategy_profiles must be a non-empty list.",
   480→            details=[{"field": "strategy_profiles"}],
   481→        )
   482→    known = set(optimizer_mod.DEFAULT_PROFILES) | {"custom"}
   483→    for p in requested_profiles:
   484→        if p not in known:
   485→            raise ValidationFailed(
   486→                f"Unknown strategy profile '{p}'.",
   487→                details=[{"field": "strategy_profiles", "reason": p}],
   488→            )
   489→    custom_weights = payload.get("custom_weights") or {}
   490→    if "custom" in requested_profiles:
   491→        if not custom_weights:
   492→            raise ValidationFailed(
   493→                "The 'custom' profile needs custom_weights.",
   494→                details=[{"field": "custom_weights"}],
   495→            )
   496→        for code, w in custom_weights.items():
   497→            if code not in objectives_mod.CRITERION_FUNCS:
   498→                raise ValidationFailed(
   499→                    f"Unknown criterion '{code}' in custom_weights.",
   500→                    details=[{"field": "custom_weights", "reason": code}],
   501→                )
   502→            if not isinstance(w, (int, float)) or w < 0:
   503→                raise ValidationFailed(
   504→                    f"Custom weight for '{code}' must be a non-negative number.",
   505→                    details=[{"field": "custom_weights", "reason": code}],
   506→                )
   507→        if sum(float(w) for w in custom_weights.values()) <= 0:
   508→            raise ValidationFailed(
   509→                "custom_weights must sum to a positive value.",
   510→                details=[{"field": "custom_weights"}],
   511→            )
   512→
   513→    water_allocation = payload.get("water_allocation_m3")
   514→    if water_allocation is not None:
   515→        try:
   516→            water_allocation = float(water_allocation)
   517→        except (TypeError, ValueError):
   518→            raise ValidationFailed(
   519→                "water_allocation_m3 must be a number.",
   520→                details=[{"field": "water_allocation_m3"}],
   521→            ) from None
   522→        if water_allocation <= 0:
   523→            raise ValidationFailed(
   524→                "water_allocation_m3 must be positive.",
   525→                details=[{"field": "water_allocation_m3"}],
   526→            )
   527→
   528→    knowledge = load_knowledge(db)
   529→    if not knowledge.crops:
   530→        raise ValidationFailed(
   531→            "No active knowledge pack with crops — import a knowledge pack first.",
   532→            details=[{"field": "knowledge"}],
   533→        )
   534→
   535→    run = dbm.RotationRun(
   536→        field_id=field.id,
   537→        status="running",
   538→        seasons=[
   539→            {"label": s.label, "start_date": s.start_date.isoformat(), "end_date": s.end_date.isoformat()}
   540→            for s in seasons
   541→        ],
   542→        strategy_profiles=list(requested_profiles),
   543→        water_allocation_m3=water_allocation,
   544→        knowledge_pack_refs=[dict(r) for r in knowledge.pack_refs],
   545→        rotation_engine_version=ROTATION_ENGINE_VERSION,
   546→        solver_name=SOLVER_NAME,
   547→        random_seed=RANDOM_SEED,
   548→        request_parameters=dict(payload.get("request_parameters") or {}),
   549→        idempotency_key=idempotency_key,
   550→        created_by=getattr(created_by, "id", None),
   551→        started_at=datetime.now(UTC),
   552→    )
   553→    db.add(run)
   554→    db.flush()
   555→
   556→    try:
   557→        farm, warnings, history_ids, evidence_run = build_farm_context(
   558→            db, field, seasons, payload.get("evidence_run_id"), hemisphere
   559→        )
   560→        run.feature_run_ids = (
   561→            [str(evidence_run.id)] if evidence_run is not None else []
   562→        )
   563→        run.crop_history_ids = history_ids
   564→
   565→        assessments = candidates_mod.generate_candidates(knowledge, farm)
   566→        baseline_assignment = _baseline_assignment(seasons, farm.history)
   567→        problem = RotationProblem(
   568→            field_id=str(field.id),
   569→            seasons=tuple(seasons),
   570→            candidates=assessments,
   571→            knowledge=knowledge,
   572→            farm=farm,
   573→            water_allocation_m3=water_allocation,
   574→            baseline_assignment=baseline_assignment,
   575→            synthetic_evidence=farm.synthetic_evidence_in_use,
   576→        )
   577→        run.knowledge_pack_refs = [dict(r) for r in knowledge.pack_refs]
   578→
   579→        solutions: list[StrategySolution] = []
   580→        infeasible_profiles: list[dict[str, Any]] = []
   581→        solver_failures: list[dict[str, Any]] = []
   582→        baseline = optimizer_mod.build_baseline_solution(problem)
   583→        solutions.append(baseline)
   584→
   585→        for profile in requested_profiles:
   586→            weights = (
   587→                {k: float(v) for k, v in custom_weights.items()}
   588→                if profile == "custom"
   589→                else dict(optimizer_mod.PROFILE_WEIGHTS[profile])
   590→            )
   591→            result = optimizer_mod.optimize_strategy(problem, profile, weights)
   592→            if isinstance(result, optimizer_mod.InfeasibleResult):
   593→                # Deepening F-4: only a PROVEN infeasible model is an infeasible
   594→                # profile; solver failures (model_invalid / unknown) are kept
   595→                # separate so a solver breakdown never reads as an agronomic
   596→                # infeasibility statement.
   597→                entry = {
   598→                    "strategy_key": profile,
   599→                    "conflicts": result.conflicts,
   600→                    "notes": result.notes,
   601→                }
   602→                if result.status == "infeasible":
   603→                    infeasible_profiles.append(entry)
   604→                else:
   605→                    entry["solver_status"] = result.status
   606→                    solver_failures.append(entry)
   607→                continue
   608→            solutions.append(result)
   609→
   610→        # evaluate + normalize + objective + pareto
   611→        evaluated: list[StrategySolution] = []
   612→        for sol in solutions:
   613→            criteria = objectives_mod.evaluate_criteria(problem, dict(sol.assignment))
   614→            checks = constraints_mod.check_all(problem, dict(sol.assignment))
   615→            evaluated.append(
   616→                StrategySolution(
   617→                    strategy_key=sol.strategy_key,
   618→                    strategy_kind=sol.strategy_kind,
   619→                    is_baseline=sol.is_baseline,
   620→                    profile_weights=sol.profile_weights,
   621→                    weights_effective=sol.weights_effective,
   622→                    skipped_weights=sol.skipped_weights,
   623→                    assignment=sol.assignment,
   624→                    objective_value=sol.objective_value,
   625→                    criteria=criteria,
   626→                    constraints_checked=tuple(checks),
   627→                    evidence_coverage=None,
   628→                    solver_status=sol.solver_status,
   629→                    solver_wall_time_ms=sol.solver_wall_time_ms,
   630→                )
   631→            )
   632→        objectives_mod.normalize_within_set(evaluated)
   633→        final: list[StrategySolution] = []
   634→        for sol in evaluated:
   635→            objective, effective, skipped = objectives_mod.weighted_objective(sol)
   636→            final.append(
   637→                StrategySolution(
   638→                    strategy_key=sol.strategy_key,
   639→                    strategy_kind=sol.strategy_kind,
   640→                    is_baseline=sol.is_baseline,
   641→                    profile_weights=sol.profile_weights,
   642→                    weights_effective=effective,
   643→                    skipped_weights=skipped,
   644→                    assignment=sol.assignment,
   645→                    objective_value=objective,
   646→                    criteria=sol.criteria,
   647→                    constraints_checked=sol.constraints_checked,
   648→                    evidence_coverage=objectives_mod.evidence_coverage_of(sol),
   649→                    solver_status=sol.solver_status,
   650→                    solver_wall_time_ms=sol.solver_wall_time_ms,
   651→                )
   652→            )
   653→        objectives_mod.compute_pareto(final)
   654→        solutions = final
   655→
   656→        # persist strategies
   657→        not_evaluated: dict[str, Any] = {}
   658→        for season_label, season_assessments in assessments.items():
   659→            unevaluated = {
   660→                code: [d["code"] for d in a.unevaluated_dimensions]
   661→                for code, a in season_assessments.items()
   662→                if a.status == "unevaluated" or a.unevaluated_dimensions
   663→            }
   664→            if unevaluated:
   665→                not_evaluated[season_label] = unevaluated
   666→
   667→        run.not_evaluated = not_evaluated
   668→        run.warnings = warnings
   669→        solver_status = None
   670→        optimized_solutions = [s for s in solutions if not s.is_baseline]
   671→        statuses = [s.solver_status for s in optimized_solutions]
   672→        if infeasible_profiles and not optimized_solutions:
   673→            solver_status = "infeasible"
   674→        elif solver_failures and not optimized_solutions:
   675→            # mirror the solver failure upward (model_invalid / unknown):
   676→            # a solver breakdown is never reported as an agronomic infeasibility.
   677→            solver_status = solver_failures[0]["solver_status"]
   678→        elif infeasible_profiles or solver_failures:
   679→            # mixed outcome: some profiles solved (possibly only feasibly),
   680→            # some did not — the details block carries the specifics.
   681→            solver_status = "feasible"
   682→            warnings.append(
   683→                "Some strategy profiles did not reach a proven optimum "
   684→                f"({len(infeasible_profiles)} infeasible, {len(solver_failures)} solver "
   685→                "failure(s)); see infeasibility details."
   686→            )
   687→        elif optimized_solutions and all(st == "optimal" for st in statuses):
   688→            # 'optimal' is claimed only when EVERY optimized profile proved
   689→            # optimality (a time-limited FEASIBLE result is not optimality).
   690→            solver_status = "optimal"
   691→        elif optimized_solutions:
   692→            solver_status = "feasible"
   693→        else:
   694→            solver_status = "unknown"
   695→        run.solver_status = solver_status
   696→        run.infeasibility_details = {
   697→            "profiles": infeasible_profiles,
   698→            "solver_failures": solver_failures,
   699→            "note": (
   700→                "Infeasible profiles are reported explicitly; no hard constraint was "
   701→                "relaxed to produce the other strategies."
   702→            )
   703→            if infeasible_profiles
   704→            else {},
   705→        }
   706→
   707→        for sol in solutions:
   708→            explanations = build_explanations(problem, sol, solutions)
   709→            explanations["_weights_effective"] = sol.weights_effective
   710→            explanations["_skipped_weights"] = sol.skipped_weights
   711→            db.add(
   712→                dbm.RotationStrategy(
   713→                    run_id=run.id,
   714→                    strategy_key=sol.strategy_key,
   715→                    strategy_kind=sol.strategy_kind,
   716→                    is_baseline=sol.is_baseline,
   717→                    profile_weights=sol.profile_weights,
   718→                    assignment=[
   719→                        {"season_label": lbl, "crop_code": cc} for lbl, cc in sol.assignment
   720→                    ],
   721→                    objective_value=sol.objective_value,
   722→                    criteria=[
   723→                        {
   724→                            "criterion_code": c.criterion_code,
   725→                            "raw_value": c.raw_value,
   726→                            "unit": c.unit,
   727→                            "direction": c.direction,
   728→                            "normalized": c.normalized,
   729→                            "evaluable": c.evaluable,
   730→                            "missing_evidence": list(c.missing_evidence),
   731→                            "derivation": c.derivation,
   732→                            "source_refs": list(c.source_refs),
   733→                        }
   734→                        for c in sol.criteria
   735→                    ],
   736→                    constraints_checked=[
   737→                        {
   738→                            "constraint_code": c.constraint_code,
   739→                            "status": c.status,
   740→                            "description": c.description,
   741→                            "detail": c.detail,
   742→                            "rule_codes": list(c.rule_codes),
   743→                            "source_refs": list(c.source_refs),
   744→                        }
   745→                        for c in sol.constraints_checked
   746→                    ],
   747→                    pareto={
   748→                        "on_front": sol.on_pareto_front,
   749→                        "dominated_by": list(sol.dominated_by),
   750→                        "dominates": list(sol.dominates),
   751→                    },
   752→                    explanations=explanations,
   753→                    evidence_coverage=sol.evidence_coverage,
   754→                    checksum_sha256=_solution_checksum(sol, problem),
   755→                )
   756→            )
   757→        run.status = "succeeded"
   758→        run.completed_at = datetime.now(UTC)
   759→        # M7.1 F-20: record the resolved solver build per run.
   760→        run.solver_version = solver_version()
   761→        db.flush()
   762→    except ValidationFailed:
   763→        run.status = "failed"
   764→        run.error_summary = "validation failed"
   765→        run.completed_at = datetime.now(UTC)
   766→        raise
   767→    except Exception as exc:  # noqa: BLE001 — job failure accounting
   768→        run.status = "failed"
   769→        run.error_summary = f"{type(exc).__name__}: {exc}"
   770→        run.completed_at = datetime.now(UTC)
   771→        db.flush()
   772→        raise
   773→    return run
