    51→    51→# RBAC hierarchy (Master §17): the numeric order defines "at least as

    56→    56→def role_at_least(role_code: str, minimum: str) -> bool:
    57→    57→    """True when `role_code` grants at least the privileges of `minimum`."""
    58→    58→    if role_code not in ROLE_ORDER or minimum not in ROLE_ORDER:
    59→    59→        return False
    60→    60→    return ROLE_ORDER[role_code] >= ROLE_ORDER[minimum]

    63→    63→class Principal:
    64→    64→    """Authenticated user + their memberships, resolved once per request."""
    65→    65→
    66→    66→    def __init__(self, user: User, memberships: list[Membership]) -> None:
    67→    67→        self.user = user
    68→    68→        self.memberships = memberships
    69→    69→        self._org_ids = {m.organization_id for m in memberships if m.status == "active"}
    70→    70→
    71→    71→    @property
    72→    72→    def id(self) -> uuid.UUID:
    73→    73→        return self.user.id
    74→    74→
    75→    75→    def is_member_of(self, organization_id: uuid.UUID) -> bool:
    76→    76→        return organization_id in self._org_ids
    77→    77→
    78→    78→    def require_member_of(self, organization_id: uuid.UUID) -> None:
    79→    79→        if not self.is_member_of(organization_id):
    80→    80→            raise Forbidden("You do not have access to this organization's resources.")
    81→    81→
    82→    82→    def org_ids(self) -> set[uuid.UUID]:
    83→    83→        return set(self._org_ids)
    84→    84→
    85→    85→    def membership_in(self, organization_id: uuid.UUID) -> Membership | None:
    86→    86→        """The principal's ACTIVE membership in the organization, or None."""
    87→    87→        for membership in self.memberships:
    88→    88→            if membership.organization_id == organization_id and membership.status == "active":
    89→    89→                return membership
    90→    90→        return None
    91→    91→
    92→    92→    def require_org_role(self, organization_id: uuid.UUID, minimum: str) -> Membership:
    93→    93→        """Return the active membership after verifying the minimum role."""
    94→    94→        membership = self.membership_in(organization_id)
    95→    95→        if membership is None:
    96→    96→            raise Forbidden("You do not have access to this organization's resources.")
    97→    97→        if not role_at_least(membership.role_code, minimum):
    98→    98→            raise Forbidden(
    99→    99→                f"This action requires the '{minimum}' role or higher on this organization."
   100→   100→            )
   101→   101→        return membership
   102→   102→
   103→   103→    def require_any_org_role(self, minimum: str) -> Membership:
   104→   104→        """Role gate for GLOBAL-context mutations (no single organization
   105→   105→        scope, e.g. platform-level knowledge packs): require at least one
   106→   106→        active membership with the minimum role in ANY of the principal's
   107→   107→        organizations. Tenant-scoped endpoints should use require_org_role
   108→   108→        or require_farm_role instead."""
   109→   109→        for membership in self.memberships:
   110→   110→            if membership.status == "active" and role_at_least(membership.role_code, minimum):
   111→   111→                return membership
   112→   112→        raise Forbidden(
   113→   113→            f"This action requires the '{minimum}' role or higher on at least one organization."
   114→   114→        )

   117→   117→def get_principal(
   118→   118→    credentials: HTTPAuthorizationCredentials | None = Depends(bearer_scheme),
   119→   119→    db: Session = Depends(get_db),
   120→   120→) -> Principal:
   121→   121→    if credentials is None or credentials.scheme.lower() != "bearer":
   122→   122→        raise Unauthorized("Missing bearer token.")
   123→   123→    token_hash = hash_token(credentials.credentials)
   124→   124→    row = db.execute(
   125→   125→        select(AuthToken, User)
   126→   126→        .join(User, AuthToken.user_id == User.id)
   127→   127→        .where(AuthToken.token_hash == token_hash, AuthToken.kind == "access")
   128→   128→    ).first()
   129→   129→    if row is None:
   130→   130→        raise Unauthorized("Invalid or expired token.")
   131→   131→    token, user = row
   132→   132→    now = datetime.now(UTC)
   133→   133→    if token.revoked_at is not None or token.expires_at < now:
   134→   134→        raise Unauthorized("Token expired or revoked.")
   135→   135→    if not user.is_active:
   136→   136→        raise Unauthorized("Account disabled.")
   137→   137→    memberships = list(
   138→   138→        db.scalars(select(Membership).where(Membership.user_id == user.id)).all()
   139→   139→    )
   140→   140→    return Principal(user=user, memberships=memberships)

   167→   167→# ------------------------------------------------------- credential tokens
   168→   168→def _issue_credential_token(db: Session, user_id: uuid.UUID, kind: str) -> tuple[str, datetime]:
   169→   169→    """Create a single-use CredentialToken row; returns (plain_token, expires_at).
   170→   170→
   171→   171→    Only the SHA-256 hash is stored; the plaintext is delivered out-of-band
   172→   172→    by the caller (email in production, dev convenience field in sandboxes).
   173→   173→    """
   174→   174→    ttl = (
   175→   175→        settings.password_reset_ttl_seconds
   176→   176→        if kind == "password_reset"
   177→   177→        else settings.email_verification_ttl_seconds
   178→   178→    )
   179→   179→    plain, token_hash, expires_at = generate_token(ttl_seconds=ttl)
   180→   180→    db.add(
   181→   181→        CredentialToken(user_id=user_id, kind=kind, token_hash=token_hash, expires_at=expires_at)
   182→   182→    )
   183→   183→    return plain, expires_at

   186→   186→def _issue_token_pair(db: Session, user_id: uuid.UUID, family_id: uuid.UUID | None) -> tuple[str, datetime, str, datetime]:
   187→   187→    """Issue an access+refresh AuthToken pair; returns (access, access_exp, refresh, refresh_exp)."""
   188→   188→    access, access_hash, access_expires_at = generate_token()
   189→   189→    db.add(
   190→   190→        AuthToken(
   191→   191→            user_id=user_id,
   192→   192→            token_hash=access_hash,
   193→   193→            expires_at=access_expires_at,
   194→   194→            kind="access",
   195→   195→            family_id=family_id,
   196→   196→        )
   197→   197→    )
   198→   198→    refresh, refresh_hash, refresh_expires_at = generate_refresh_token()
   199→   199→    db.add(
   200→   200→        AuthToken(
   201→   201→            user_id=user_id,
   202→   202→            token_hash=refresh_hash,
   203→   203→            expires_at=refresh_expires_at,
   204→   204→            kind="refresh",
   205→   205→            family_id=family_id,
   206→   206→        )
   207→   207→    )
   208→   208→    return access, access_expires_at, refresh, refresh_expires_at

   211→   211→def _revoke_family(db: Session, family_id: uuid.UUID | None, now: datetime) -> None:
   212→   212→    """Revoke every not-yet-revoked token in a refresh rotation family."""
   213→   213→    if family_id is None:
   214→   214→        return
   215→   215→    rows = db.scalars(
   216→   216→        select(AuthToken).where(
   217→   217→            AuthToken.family_id == family_id, AuthToken.revoked_at.is_(None)
   218→   218→        )
   219→   219→    ).all()
   220→   220→    for token in rows:
   221→   221→        token.revoked_at = now

   224→   224→def register_user(
   225→   225→    db: Session, *, email: str, password: str, display_name: str
   226→   226→) -> tuple[User, Organization, str, datetime, str, datetime]:
   227→   227→    """Create a user + personal organization and issue an access/refresh pair.
   228→   228→
   229→   229→    Returns (user, org, access_token, access_expires_at, refresh_token,
   230→   230→    refresh_expires_at). When NOT dev_mode an email-verification challenge is
   231→   231→    created; its plaintext token is never returned to clients.
   232→   232→    """
   233→   233→    email_normalized = email.strip().lower()
   234→   234→    existing = db.scalar(select(User).where(func.lower(User.email) == email_normalized))
   235→   235→    if existing is not None:
   236→   236→        raise Conflict("An account with this email already exists.")
   237→   237→    user = User(
   238→   238→        email=email_normalized,
   239→   239→        password_hash=hash_password(password),
   240→   240→        display_name=display_name.strip() or email_normalized.split("@")[0],
   241→   241→        email_verified=settings.dev_mode,
   242→   242→    )
   243→   243→    db.add(user)
   244→   244→    org = Organization(name=f"{user.display_name}'s organization")
   245→   245→    db.add(org)
   246→   246→    db.flush()
   247→   247→    db.add(Membership(organization_id=org.id, user_id=user.id, role_code="owner", status="active"))
   248→   248→    db.flush()
   249→   249→    family_id = uuid.uuid4()
   250→   250→    access, access_expires_at, refresh, refresh_expires_at = _issue_token_pair(
   251→   251→        db, user.id, family_id
   252→   252→    )
   253→   253→    if not settings.dev_mode:
   254→   254→        # Unverified deployments: create the verification challenge. The plain
   255→   255→        # token goes out-of-band (email); it is NEVER returned to clients.
   256→   256→        _issue_credential_token(db, user.id, "email_verification")
   257→   257→    record_auth_audit(
   258→   258→        db,
   259→   259→        actor_id=user.id,
   260→   260→        organization_id=org.id,
   261→   261→        action="auth.register",
   262→   262→        entity_id=str(user.id),
   263→   263→        meta={"email": user.email, "email_verified": user.email_verified},
   264→   264→    )
   265→   265→    db.commit()
   266→   266→    return user, org, access, access_expires_at, refresh, refresh_expires_at

   269→   269→def login_user(
   270→   270→    db: Session, *, email: str, password: str
   271→   271→) -> tuple[User, str, datetime, str, datetime]:
   272→   272→    """Authenticate a user; returns (user, access, access_exp, refresh, refresh_exp).
   273→   273→
   274→   274→    Hardening: temporary lockout after `auth_max_failed_logins` consecutive
   275→   275→    failures, email-verification gate, audit trail on every outcome.
   276→   276→    """
   277→   277→    email_normalized = email.strip().lower()
   278→   278→    user = db.scalar(select(User).where(func.lower(User.email) == email_normalized))
   279→   279→    now = datetime.now(UTC)
   280→   280→    if user is not None and user.locked_until is not None and user.locked_until > now:
   281→   281→        retry_after = max(int((user.locked_until - now).total_seconds()), 1)
   282→   282→        record_auth_audit(
   283→   283→            db,
   284→   284→            actor_id=user.id,
   285→   285→            action="auth.login.locked",
   286→   286→            entity_id=email_normalized,
   287→   287→            meta={"retry_after_seconds": retry_after},
   288→   288→        )
   289→   289→        db.commit()
   290→   290→        raise AccountLocked(
   291→   291→            "Account temporarily locked due to failed sign-in attempts.",
   292→   292→            retry_after_seconds=retry_after,
   293→   293→        )
   294→   294→    if user is None or not verify_password(password, user.password_hash):
   295→   295→        if user is not None:
   296→   296→            user.failed_login_count = (user.failed_login_count or 0) + 1
   297→   297→            if user.failed_login_count >= settings.auth_max_failed_logins:
   298→   298→                user.locked_until = now + timedelta(seconds=settings.auth_lockout_seconds)
   299→   299→                user.failed_login_count = 0
   300→   300→            record_auth_audit(
   301→   301→                db,
   302→   302→                actor_id=user.id,
   303→   303→                action="auth.login.failure",
   304→   304→                entity_id=email_normalized,
   305→   305→            )
   306→   306→        else:
   307→   307→            # Unknown email: audit with no actor (no account enumeration side
   308→   308→            # channel — identical client-visible response either way).
   309→   309→            record_auth_audit(
   310→   310→                db,
   311→   311→                actor_id=None,
   312→   312→                action="auth.login.failure",
   313→   313→                entity_id=email_normalized,
   314→   314→            )
   315→   315→        db.commit()
   316→   316→        raise Unauthorized("Invalid email or password.")
   317→   317→    # Correct credentials: reset the lockout counters.
   318→   318→    user.failed_login_count = 0
   319→   319→    user.locked_until = None
   320→   320→    if not user.is_active:
   321→   321→        db.commit()
   322→   322→        raise Unauthorized("Account disabled.")
   323→   323→    if not user.email_verified:
   324→   324→        record_auth_audit(
   325→   325→            db,
   326→   326→            actor_id=user.id,
   327→   327→            action="auth.login.unverified",
   328→   328→            entity_id=email_normalized,
   329→   329→        )
   330→   330→        db.commit()
   331→   331→        raise EmailNotVerified("Verify your email before signing in.")
   332→   332→    family_id = uuid.uuid4()
   333→   333→    access, access_expires_at, refresh, refresh_expires_at = _issue_token_pair(
   334→   334→        db, user.id, family_id
   335→   335→    )
   336→   336→    first_membership = db.scalar(
   337→   337→        select(Membership)
   338→   338→        .where(Membership.user_id == user.id, Membership.status == "active")
   339→   339→        .order_by(Membership.created_at)
   340→   340→    )
   341→   341→    record_auth_audit(
   342→   342→        db,
   343→   343→        actor_id=user.id,
   344→   344→        organization_id=first_membership.organization_id if first_membership else None,
   345→   345→        action="auth.login.success",
   346→   346→        entity_id=str(user.id),
   347→   347→    )
   348→   348→    db.commit()
   349→   349→    return user, access, access_expires_at, refresh, refresh_expires_at

   352→   352→def rotate_refresh_token(
   353→   353→    db: Session, refresh_plain: str
   354→   354→) -> tuple[User, str, datetime, str, datetime]:
   355→   355→    """Rotate a refresh token: returns (user, access, access_exp, refresh, refresh_exp).
   356→   356→
   357→   357→    Reuse of an already-revoked refresh token is treated as a compromised
   358→   358→    session family: every token in the family is revoked and the request is
   359→   359→    rejected. Refresh tokens never authenticate API calls (kind=access filter
   360→   360→    in get_principal) — they only mint new token pairs here.
   361→   361→    """
   362→   362→    token_hash = hash_token(refresh_plain)
   363→   363→    token = db.scalar(
   364→   364→        select(AuthToken).where(
   365→   365→            AuthToken.token_hash == token_hash, AuthToken.kind == "refresh"
   366→   366→        )
   367→   367→    )
   368→   368→    if token is None:
   369→   369→        raise Unauthorized("Invalid refresh token.")
   370→   370→    user = db.get(User, token.user_id)
   371→   371→    if user is None:  # pragma: no cover — FK guarantees the user row
   372→   372→        raise Unauthorized("Invalid refresh token.")
   373→   373→    now = datetime.now(UTC)
   374→   374→    if token.revoked_at is not None:
   375→   375→        # REUSE DETECTED: a rotated (revoked) token came back. Revoke the
   376→   376→        # whole family — all tokens minted from this login/rotation chain.
   377→   377→        _revoke_family(db, token.family_id, now)
   378→   378→        record_auth_audit(
   379→   379→            db,
   380→   380→            actor_id=user.id,
   381→   381→            action="auth.token.reuse_detected",
   382→   382→            entity_id=str(token.family_id) if token.family_id else None,
   383→   383→        )
   384→   384→        db.commit()
   385→   385→        raise Unauthorized("Refresh token reuse detected — session family revoked.")
   386→   386→    if token.expires_at < now:
   387→   387→        token.revoked_at = now
   388→   388→        db.commit()
   389→   389→        raise Unauthorized("Refresh token expired.")
   390→   390→    family_id = token.family_id or uuid.uuid4()
   391→   391→    token.revoked_at = now
   392→   392→    access, access_expires_at, refresh, refresh_expires_at = _issue_token_pair(
   393→   393→        db, user.id, family_id
   394→   394→    )
   395→   395→    record_auth_audit(
   396→   396→        db,
   397→   397→        actor_id=user.id,
   398→   398→        action="auth.token.refresh",
   399→   399→        entity_id=str(family_id),
   400→   400→    )
   401→   401→    db.commit()
   402→   402→    return user, access, access_expires_at, refresh, refresh_expires_at

   417→   417→    plain, _expires_at = _issue_credential_token(db, user.id, "password_reset")
   418→   418→    record_auth_audit(
   419→   419→        db,
   420→   420→        actor_id=user.id,
   421→   421→        action="auth.password.reset_requested",
   422→   422→        entity_id=email_normalized,
   423→   423→    )
   424→   424→    db.commit()
   425→   425→    return plain

   428→   428→def confirm_password_reset(db: Session, token_plain: str, new_password: str) -> User:
   429→   429→    """Consume a password-reset challenge, set the new password and revoke
   430→   430→    every live session token of the user."""
   431→   431→    token_hash = hash_token(token_plain)
   432→   432→    row = db.scalar(
   433→   433→        select(CredentialToken).where(
   434→   434→            CredentialToken.token_hash == token_hash,
   435→   435→            CredentialToken.kind == "password_reset",
   436→   436→        )
   437→   437→    )
   438→   438→    now = datetime.now(UTC)
   439→   439→    if row is None or row.used_at is not None or row.expires_at < now:
   440→   440→        raise ValidationFailed("Invalid or expired reset token.")
   441→   441→    row.used_at = now
   442→   442→    user = db.get(User, row.user_id)
   443→   443→    user.password_hash = hash_password(new_password)
   444→   444→    live_tokens = db.scalars(
   445→   445→        select(AuthToken).where(
   446→   446→            AuthToken.user_id == user.id, AuthToken.revoked_at.is_(None)
   447→   447→        )
   448→   448→    ).all()
   449→   449→    for token in live_tokens:
   450→   450→        token.revoked_at = now
   451→   451→    record_auth_audit(
   452→   452→        db,
   453→   453→        actor_id=user.id,
   454→   454→        action="auth.password.reset_confirmed",
   455→   455→        entity_id=str(user.id),
   456→   456→    )
   457→   457→    db.commit()
   458→   458→    return user

   461→   461→# ---------------------------------------------------- email verification
   462→   462→def request_email_verification(db: Session, email: str) -> str | None:
   463→   463→    """Create an email-verification challenge; returns the plaintext token.
   464→   464→
   465→   465→    Unknown emails and already-verified accounts return None (no new
   466→   466→    challenge, no enumeration side channel).
   467→   467→    """
   468→   468→    email_normalized = email.strip().lower()
   469→   469→    user = db.scalar(select(User).where(func.lower(User.email) == email_normalized))
   470→   470→    if user is None or user.email_verified:
   471→   471→        return None
   472→   472→    plain, _expires_at = _issue_credential_token(db, user.id, "email_verification")
   473→   473→    record_auth_audit(
   474→   474→        db,
   475→   475→        actor_id=user.id,
   476→   476→        action="auth.email.verification_requested",
   477→   477→        entity_id=email_normalized,
   478→   478→    )
   479→   479→    db.commit()
   480→   480→    return plain

   483→   483→def confirm_email_verification(db: Session, token_plain: str) -> User:
   484→   484→    """Consume an email-verification challenge and mark the account verified."""
   485→   485→    token_hash = hash_token(token_plain)
   486→   486→    row = db.scalar(
   487→   487→        select(CredentialToken).where(
   488→   488→            CredentialToken.token_hash == token_hash,
   489→   489→            CredentialToken.kind == "email_verification",
   490→   490→        )
   491→   491→    )
   492→   492→    now = datetime.now(UTC)
   493→   493→    if row is None or row.used_at is not None or row.expires_at < now:
   494→   494→        raise ValidationFailed("Invalid or expired verification token.")
   495→   495→    row.used_at = now
   496→   496→    user = db.get(User, row.user_id)
   497→   497→    user.email_verified = True
   498→   498→    record_auth_audit(
   499→   499→        db,
   500→   500→        actor_id=user.id,
   501→   501→        action="auth.email.verified",
   502→   502→        entity_id=str(user.id),
   503→   503→    )
   504→   504→    db.commit()
   505→   505→    return user

   545→   545→def _count_other_active_owners(
   546→   546→    db: Session, organization_id: uuid.UUID, excluded_membership_id: uuid.UUID
   547→   547→) -> int:
   548→   548→    return int(
   549→   549→        db.scalar(
   550→   550→            select(func.count())
   551→   551→            .select_from(Membership)
   552→   552→            .where(
   553→   553→                Membership.organization_id == organization_id,
   554→   554→                Membership.role_code == "owner",
   555→   555→                Membership.status == "active",
   556→   556→                Membership.id != excluded_membership_id,
   557→   557→            )
   558→   558→        )
   559→   559→        or 0
   560→   560→    )

   563→   563→def add_member(
   564→   564→    db: Session, principal: Principal, farm: Farm, *, email: str, role_code: str
   565→   565→) -> Membership:
   566→   566→    """Add an existing FieldShift account to the farm's organization (owner-only)."""
   567→   567→    require_farm_role(db, principal, farm, "owner")
   568→   568→    email_normalized = email.strip().lower()
   569→   569→    user = db.scalar(select(User).where(func.lower(User.email) == email_normalized))
   570→   570→    if user is None:
   571→   571→        raise NotFound("No FieldShift account exists with this email.")
   572→   572→    existing = db.scalar(
   573→   573→        select(Membership).where(
   574→   574→            Membership.organization_id == farm.organization_id,
   575→   575→            Membership.user_id == user.id,
   576→   576→        )
   577→   577→    )
   578→   578→    if existing is not None:
   579→   579→        raise Conflict("This user is already a member of this organization.")
   580→   580→    if role_code not in ROLE_ORDER:
   581→   581→        raise ValidationFailed(
   582→   582→            f"Unknown role '{role_code}'.",
   583→   583→            details=[{"field": "role_code", "reason": role_code}],
   584→   584→        )
   585→   585→    membership = Membership(
   586→   586→        organization_id=farm.organization_id,
   587→   587→        user_id=user.id,
   588→   588→        role_code=role_code,
   589→   589→        status="active",
   590→   590→    )
   591→   591→    db.add(membership)
   592→   592→    db.flush()
   593→   593→    record_auth_audit(
   594→   594→        db,
   595→   595→        actor_id=principal.id,
   596→   596→        organization_id=farm.organization_id,
   597→   597→        action="membership.create",
   598→   598→        entity_type="membership",
   599→   599→        entity_id=str(membership.id),
   600→   600→        meta={"user_id": str(user.id), "email": user.email, "role_code": role_code},
   601→   601→    )
   602→   602→    db.commit()
   603→   603→    return membership

   606→   606→def update_member_role(
   607→   607→    db: Session, principal: Principal, farm: Farm, membership_id: uuid.UUID, new_role: str
   608→   608→) -> Membership:
   609→   609→    """Change a member's role (owner-only) with a last-owner guard."""
   610→   610→    require_farm_role(db, principal, farm, "owner")
   611→   611→    if new_role not in ROLE_ORDER:
   612→   612→        raise ValidationFailed(
   613→   613→            f"Unknown role '{new_role}'.",
   614→   614→            details=[{"field": "role_code", "reason": new_role}],
   615→   615→        )
   616→   616→    membership = db.scalar(
   617→   617→        select(Membership).where(
   618→   618→            Membership.id == membership_id,
   619→   619→            Membership.organization_id == farm.organization_id,
   620→   620→        )
   621→   621→    )
   622→   622→    if membership is None:
   623→   623→        raise NotFound("Membership not found.")
   624→   624→    # LAST-OWNER GUARD: the organization must always keep at least one
   625→   625→    # active owner, otherwise no one could manage members again.
   626→   626→    if membership.role_code == "owner" and new_role != "owner":
   627→   627→        if _count_other_active_owners(db, farm.organization_id, membership.id) < 1:
   628→   628→            raise Conflict(
   629→   629→                "Cannot demote the last owner of this organization — promote another "
   630→   630→                "member to owner first."
   631→   631→            )
   632→   632→    membership.role_code = new_role
   633→   633→    record_auth_audit(
   634→   634→        db,
   635→   635→        actor_id=principal.id,
   636→   636→        organization_id=farm.organization_id,
   637→   637→        action="membership.update",
   638→   638→        entity_type="membership",
   639→   639→        entity_id=str(membership.id),
   640→   640→        meta={"user_id": str(membership.user_id), "new_role": new_role},
   641→   641→    )
   642→   642→    db.commit()
   643→   643→    return membership

   646→   646→def remove_member(db: Session, principal: Principal, farm: Farm, membership_id: uuid.UUID) -> None:
   647→   647→    """Remove a member from the farm's organization (owner-only, last-owner guard)."""
   648→   648→    require_farm_role(db, principal, farm, "owner")
   649→   649→    membership = db.scalar(
   650→   650→        select(Membership).where(
   651→   651→            Membership.id == membership_id,
   652→   652→            Membership.organization_id == farm.organization_id,
   653→   653→        )
   654→   654→    )
   655→   655→    if membership is None:
   656→   656→        raise NotFound("Membership not found.")
   657→   657→    if membership.role_code == "owner":
   658→   658→        if _count_other_active_owners(db, farm.organization_id, membership.id) < 1:
   659→   659→            raise Conflict(
   660→   660→                "Cannot remove the last owner of this organization — promote another "
   661→   661→                "member to owner first."
   662→   662→            )
   663→   663→    audit_meta = {"user_id": str(membership.user_id), "role_code": membership.role_code}
   664→   664→    db.delete(membership)
   665→   665→    record_auth_audit(
   666→   666→        db,
   667→   667→        actor_id=principal.id,
   668→   668→        organization_id=farm.organization_id,
   669→   669→        action="membership.delete",
   670→   670→        entity_type="membership",
   671→   671→        entity_id=str(membership_id),
   672→   672→        meta=audit_meta,
   673→   673→    )
   674→   674→    db.commit()
