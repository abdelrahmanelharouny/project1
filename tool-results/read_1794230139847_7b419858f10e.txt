   154→function CreateFarmDialog({ open, onOpenChange }: { open: boolean; onOpenChange: (open: boolean) => void }) {
   155→  const queryClient = useQueryClient()

   157→  const [country, setCountry] = useState('')
   158→  const [adminRegion, setAdminRegion] = useState('')
   159→  const [timezone, setTimezone] = useState('UTC')
   160→  const [currency, setCurrency] = useState('')
   161→  const [unitSystem, setUnitSystem] = useState<UnitSystem>('metric')
   162→  const [language, setLanguage] = useState('en')
   163→  const [formError, setFormError] = useState<string | null>(null)
   164→
   165→  // Reset form state when the dialog closes (handler-based, no effect needed).
   166→  const handleOpenChange = (next: boolean) => {
   167→    if (!next) {
   168→      setName('')
   169→      setCountry('')
   170→      setAdminRegion('')
   171→      setTimezone('UTC')
   172→      setCurrency('')
   173→      setUnitSystem('metric')
   174→      setLanguage('en')
   175→      setFormError(null)
   176→    }
   177→    onOpenChange(next)
   178→  }
   179→
   180→  const mutation = useMutation({
   181→    mutationFn: (input: FarmCreateInput) => createFarm(input),
   182→    onSuccess: (farm) => {
   183→      queryClient.invalidateQueries({ queryKey: ['farms'] })

   207→      name: name.trim(),
   208→      country_code: countryClean,
   209→      admin_region: adminRegion.trim() || undefined,
   210→      timezone,
   211→      currency_code: currencyClean,

   337→function AddFieldDialog({
   338→  farm,
   339→  open,
   340→  onOpenChange,
   341→  onFieldSelected,
   342→}: {
   343→  farm: Farm
   344→  open: boolean
   345→  onOpenChange: (open: boolean) => void
   346→  onFieldSelected: (field: { id: string; name: string; farmId: string }) => void
   347→}) {
   348→  const queryClient = useQueryClient()
   349→  const [name, setName] = useState('')
   350→  const [geometry, setGeometry] = useState('')
   351→  const [parseError, setParseError] = useState<string | null>(null)
   352→  const [apiError, setApiError] = useState<{ title: string; description?: string } | null>(null)
   353→  const [created, setCreated] = useState<FieldCreateResponse | null>(null)
   354→
   355→  // Reset form state when the dialog closes.
   356→  const handleOpenChange = (next: boolean) => {
   357→    if (!next) {
   358→      setName('')
   359→      setGeometry('')
   360→      setParseError(null)
   361→      setApiError(null)
   362→      setCreated(null)
   363→    }
   364→    onOpenChange(next)
   365→  }
   366→
   367→  const mutation = useMutation({
   368→    mutationFn: (payload: { name: string; geometry: GeoJSONGeometry }) =>
   369→      createField(farm.id, { name: payload.name, geometry: payload.geometry }),
   370→    onSuccess: (result) => {
   371→      queryClient.invalidateQueries({ queryKey: ['fields', farm.id] })
   372→      queryClient.invalidateQueries({ queryKey: ['farms'] })
   373→      if (result.geometry_warnings.length > 0) {
   374→        setCreated(result)
   375→        toast.warning('Field created — but review the geometry warnings', {
   376→          description: `${result.geometry_warnings.length} warning(s) from PostGIS validation.`,
   377→        })
   378→      } else {
   379→        toast.success(`Field “${result.name}” created`, {
   380→          description: `${formatArea(result.area_ha)} ha · boundary v${result.boundary_version}`,
   381→        })
   382→        onFieldSelected({ id: result.id, name: result.name, farmId: farm.id })
   383→        handleOpenChange(false)
   384→      }
   385→    },
   386→    onError: (err) => {
   387→      const { title, description } = describeError(err)
   388→      setApiError({ title, description })
   389→      toast.error(title, { description })
   390→    },
   391→  })
   392→
   393→  const submit = (event: FormEvent) => {
   394→    event.preventDefault()
   395→    setApiError(null)
   396→    setCreated(null)
   397→    if (!name.trim()) {
   398→      setParseError('Give the field a name first.')
   399→      return
   400→    }
   401→    const parsed = parseGeometryInput(geometry)
   402→    if (!parsed.ok) {
   403→      setParseError(parsed.error)
   404→      return
   405→    }
   406→    setParseError(null)
   407→    mutation.mutate({ name: name.trim(), geometry: parsed.geometry })
   408→  }
   409→
   410→  return (
   411→    <Dialog open={open} onOpenChange={handleOpenChange}>
   412→      <DialogContent className="max-h-[85vh] overflow-y-auto scrollbar-thin sm:max-w-xl">
   413→        <DialogHeader>
   414→          <DialogTitle>Add a field to “{farm.name}”</DialogTitle>
   415→          <DialogDescription>
   416→            The boundary is validated by PostGIS. If something is off (self-intersections, unclosed rings), the
   417→            response tells you exactly what was fixed.
   418→          </DialogDescription>
   419→        </DialogHeader>
   420→        <form onSubmit={submit} className="space-y-4" aria-label="Add field form">
   421→          <div className="space-y-1.5">
   422→            <Label htmlFor="field-name">Field name</Label>
   423→            <Input
   424→              id="field-name"
   425→              placeholder="North Field"
   426→              value={name}
   427→              onChange={(event) => setName(event.target.value)}
   428→            />
   429→          </div>
   430→          <GeometryInput id="field-geometry" value={geometry} onChange={setGeometry} error={parseError} />
   431→
   432→          {created ? (
   433→            <div className="space-y-2">
   434→              <Alert className="border-emerald-300 bg-emerald-50 [&>svg]:text-emerald-700">
   435→                <MapPin className="h-4 w-4" />
   436→                <AlertTitle className="text-emerald-900">
   437→                  Field “{created.name}” created — {formatArea(created.area_ha)} ha
   438→                </AlertTitle>
   439→                <AlertDescription className="text-emerald-900">
   440→                  Boundary v{created.boundary_version} · passport status: {created.data_passport_status}
   441→                </AlertDescription>
   442→              </Alert>
   443→              <GeometryWarningsAlert
   444→                warnings={created.geometry_warnings}
   445→                repaired={created.geometry_repaired}
   446→                title="Geometry warnings from PostGIS"
   447→              />
   448→              <div className="flex justify-end gap-2">
   449→                <Button
   450→                  type="button"
   451→                  variant="outline"
   452→                  onClick={() => {
   453→                    onFieldSelected({ id: created.id, name: created.name, farmId: farm.id })
   454→                    handleOpenChange(false)
   455→                  }}
   456→                >
   457→                  Open the new field
   458→                </Button>
   459→              </div>
   460→            </div>
   461→          ) : null}
   462→
   463→          {apiError ? (
   464→            <Alert variant="destructive">
   465→              <TriangleAlert className="h-4 w-4" />
   466→              <AlertTitle>{apiError.title}</AlertTitle>
   467→              <AlertDescription>{apiError.description ?? 'Adjust the geometry and try again.'}</AlertDescription>
   468→            </Alert>
   469→          ) : null}
   470→
   471→          <DialogFooter>
   472→            <Button type="button" variant="outline" onClick={() => handleOpenChange(false)}>
   473→              {created ? 'Close' : 'Cancel'}
   474→            </Button>
   475→            <Button type="submit" className="bg-emerald-700 hover:bg-emerald-800" disabled={mutation.isPending}>
   476→              {mutation.isPending ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : <Plus className="h-4 w-4" aria-hidden />}
   477→              Create field
   478→            </Button>
   479→          </DialogFooter>
   480→        </form>
   481→      </DialogContent>
   482→    </Dialog>
   483→  )
   484→}

   490→function RenameFieldDialog({
   491→  field,
   492→  open,
   493→  onOpenChange,
   494→}: {
   495→  field: Field
   496→  open: boolean
   497→  onOpenChange: (open: boolean) => void
   498→}) {
   499→  const queryClient = useQueryClient()
   500→  const renameSelectedField = useFieldShiftUi((s) => s.renameSelectedField)
   501→  const [name, setName] = useState(field.name)
   502→
   503→  // Sync the input with the current field name each time the dialog opens.
   504→  const handleOpenChange = (next: boolean) => {
   505→    if (next) setName(field.name)
   506→    onOpenChange(next)
   507→  }
   508→
   509→  const mutation = useMutation({
   510→    mutationFn: () => patchField(field.id, { name: name.trim() }),
   511→    onSuccess: (updated) => {
   512→      queryClient.invalidateQueries({ queryKey: ['fields'] })
   513→      renameSelectedField(updated.id, updated.name)
   514→      toast.success(`Field renamed to “${updated.name}”`)
   515→      onOpenChange(false)
   516→    },
   517→    onError: (err) => {
   518→      const { title, description } = describeError(err)
   519→      toast.error(title, { description })
   520→    },
   521→  })
   522→
   523→  const submit = (event: FormEvent) => {
   524→    event.preventDefault()
   525→    if (!name.trim() || name.trim() === field.name) {
   526→      onOpenChange(false)
   527→      return
   528→    }
   529→    mutation.mutate()
   530→  }
   531→
   532→  return (
   533→    <Dialog open={open} onOpenChange={handleOpenChange}>
   534→      <DialogContent className="sm:max-w-sm">
   535→        <DialogHeader>
   536→          <DialogTitle>Rename field</DialogTitle>
   537→          <DialogDescription>Boundaries and observations keep their history — only the label changes.</DialogDescription>
   538→        </DialogHeader>
   539→        <form onSubmit={submit} className="space-y-4" aria-label="Rename field form">
   540→          <div className="space-y-1.5">
   541→            <Label htmlFor="rename-field">Field name</Label>
   542→            <Input id="rename-field" value={name} onChange={(event) => setName(event.target.value)} />
   543→          </div>
   544→          <DialogFooter>
   545→            <Button type="button" variant="outline" onClick={() => handleOpenChange(false)}>
   546→              Cancel
   547→            </Button>
   548→            <Button type="submit" className="bg-emerald-700 hover:bg-emerald-800" disabled={mutation.isPending}>
   549→              {mutation.isPending ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : null}
   550→              Save name
   551→            </Button>
   552→          </DialogFooter>
   553→        </form>
   554→      </DialogContent>
   555→    </Dialog>
   556→  )
   557→}

   582→    onOpenChange(next)
   583→  }
   584→
   585→  const mutation = useMutation({
   586→    mutationFn: (payload: { geometry: GeoJSONGeometry; change_reason?: string }) =>
   587→      createBoundaryVersion(field.id, payload),
   588→    onSuccess: (response) => {
   589→      queryClient.invalidateQueries({ queryKey: ['fields'] })
   590→      queryClient.invalidateQueries({ queryKey: ['field-map', field.id] })
   591→      queryClient.invalidateQueries({ queryKey: ['boundary-versions', field.id] })
   592→      queryClient.invalidateQueries({ queryKey: ['data-passport', field.id] })
   593→      if (response.geometry_warnings.length > 0) {
   594→        setResult(response)
   595→        toast.warning(`Boundary v${response.boundary_version} stored with warnings`, {
   596→          description: 'Review the PostGIS warnings below.',
   597→        })
   598→      } else {
   599→        toast.success(`Boundary updated to v${response.boundary_version}`, {
   600→          description: `${formatArea(response.area_ha)} ha · geodesic WGS84`,
   601→        })
   602→        handleOpenChange(false)
   603→      }
   604→    },
   605→    onError: (err) => {
   606→      const { title, description } = describeError(err)
   607→      setParseError(title)
   608→      toast.error(title, { description })
   609→    },
   610→  })
   611→
   612→  const submit = (event: FormEvent) => {
   613→    event.preventDefault()
   614→    setResult(null)
   615→    const parsed = parseGeometryInput(geometry)
   616→    if (!parsed.ok) {
   617→      setParseError(parsed.error)
   618→      return
   619→    }
   620→    setParseError(null)
   621→    mutation.mutate({ geometry: parsed.geometry, change_reason: reason.trim() || undefined })
   622→  }
   623→
   624→  return (
   625→    <Dialog open={open} onOpenChange={handleOpenChange}>
   626→      <DialogContent className="max-h-[85vh] overflow-y-auto scrollbar-thin sm:max-w-xl">
   627→        <DialogHeader>
   628→          <DialogTitle>Update boundary — creates version {field.boundary_version + 1}</DialogTitle>
   629→          <DialogDescription>
   630→            Boundary history is never overwritten: every change is stored as a new version with its recomputed area.
   631→          </DialogDescription>
   632→        </DialogHeader>
   633→        <form onSubmit={submit} className="space-y-4" aria-label="Update boundary form">
   634→          <GeometryInput id="boundary-geometry" value={geometry} onChange={setGeometry} error={parseError} />
   635→          <div className="space-y-1.5">
   636→            <Label htmlFor="boundary-reason">Change reason (optional)</Label>
   637→            <Input
   638→              id="boundary-reason"
   639→              placeholder="Re-measured after land swap"
   640→              value={reason}
   641→              onChange={(event) => setReason(event.target.value)}
   642→            />
   643→          </div>
   644→
   645→          {result ? (
   646→            <div className="space-y-2">
   647→              <Alert className="border-emerald-300 bg-emerald-50 [&>svg]:text-emerald-700">
   648→                <MapPin className="h-4 w-4" />
   649→                <AlertTitle className="text-emerald-900">
   650→                  Boundary v{result.boundary_version} stored — {formatArea(result.area_ha)} ha
   651→                </AlertTitle>
   652→                <AlertDescription className="text-emerald-900">
   653→                  The map and version list will refresh automatically.
   654→                </AlertDescription>
   655→              </Alert>
   656→              <GeometryWarningsAlert
   657→                warnings={result.geometry_warnings}
   658→                repaired={result.geometry_repaired}
   659→                title="Geometry warnings from PostGIS"
   660→              />
   661→            </div>
   662→          ) : null}
   663→
   664→          <DialogFooter>
   665→            <Button type="button" variant="outline" onClick={() => handleOpenChange(false)}>
   666→              {result ? 'Close' : 'Cancel'}
   667→            </Button>
   668→            <Button type="submit" className="bg-emerald-700 hover:bg-emerald-800" disabled={mutation.isPending}>
   669→              {mutation.isPending ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : null}
   670→              Store new version
   671→            </Button>
   672→          </DialogFooter>
   673→        </form>
