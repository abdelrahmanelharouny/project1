   123→/**
   124→ * Farms with their fields flattened via parallel queries — small workspaces,
   125→ * so this stays cheap. Exposed so tabs that need a global field picker
   126→ * (Observations / Data Passport) can render one list.
   127→ */
   128→export function useFarmsWithFields(enabled = true): {
   129→  farms: FarmWithFields[]
   130→  allFields: Field[]
   131→  isLoading: boolean
   132→  hasError: boolean
   133→} {
   134→  const farmsQuery = useFarms(enabled)
   135→  const farms = useMemo(() => farmsQuery.data?.items ?? [], [farmsQuery.data])
   136→
   137→  const fieldQueries = useQueries({
   138→    queries: farms.map((farm) => ({
   139→      queryKey: ['fields', farm.id],
   140→      queryFn: () => listFields(farm.id),
   141→      enabled,
   142→      staleTime: 15_000,
   143→    })),
   144→  })
   145→
   146→  const farmsWithFields = useMemo<FarmWithFields[]>(() => {
   147→    return farms.map((farm, index) => ({
   148→      farm,
   149→      fields: fieldQueries[index]?.data?.items ?? [],
   150→      fieldsQuery: fieldQueries[index],
   151→    }))
   152→  }, [farms, fieldQueries])
   153→
   154→  const allFields = useMemo(() => farmsWithFields.flatMap((entry) => entry.fields), [farmsWithFields])
   155→  const isLoading = farmsQuery.isLoading || fieldQueries.some((q) => q.isLoading)
   156→  const hasError = farmsQuery.isError || fieldQueries.some((q) => q.isError)
   157→
   158→  return { farms: farmsWithFields, allFields, isLoading, hasError }
   159→}

   657→  fieldId: string | null,
   658→  params: { status?: SatelliteSceneStatus; limit?: number; offset?: number } = {},
   659→) {
   660→  return useQuery({
   661→    queryKey: ['satellite-scenes', fieldId, params.status ?? null, params.offset ?? 0],
   662→    queryFn: () => listSatelliteScenes(fieldId!, params),
   663→    enabled: !!fieldId,
   664→    // Live job feedback: refetch while any scene is queued/running.
   665→    refetchInterval: (query) => {
   666→      const data = query.state.data as SatelliteCatalogResponse | undefined
   667→      const busy = data?.scenes.some(
   668→        (s) => s.acquisition_status === 'queued' || s.acquisition_status === 'running',
   669→      )
   670→      return busy ? 3000 : false
   671→    },
   672→  })
   673→}
