    91→function fmt(value: number | null | undefined, digits = 2): string {
    92→  if (value === null || value === undefined) return '—'
    93→  return value.toLocaleString(undefined, { maximumFractionDigits: digits })

    96→function CandidateStatusBadge({ status }: { status: CandidateAssessment['status'] }) {
    97→  const map: Record<CandidateAssessment['status'], { cls: string; label: string }> = {
    98→    eligible: { cls: 'border-emerald-200 bg-emerald-50 text-emerald-800', label: 'Eligible' },
    99→    conditionally_eligible: { cls: 'border-amber-200 bg-amber-50 text-amber-900', label: 'Conditional' },
   100→    excluded: { cls: 'border-red-200 bg-red-50 text-red-800', label: 'Excluded' },
   101→    unevaluated: { cls: 'border-stone-300 bg-stone-100 text-stone-600', label: 'Unevaluated' },
   102→  }
   103→  const it = map[status]
   104→  return (
   105→    <Badge variant="outline" className={cn('font-normal', it.cls)}>
   106→      {it.label}
   107→    </Badge>
   108→  )

   113→function CropHistoryCard({ selectedField }: { selectedField: SelectedField }) {
   114→  const history = useCropHistory(selectedField.id)
   115→  const queryClient = useQueryClient()
   116→  const [cropCode, setCropCode] = useState('')
   117→  const [label, setLabel] = useState('')
   118→  const [sowing, setSowing] = useState('')
   119→  const [harvest, setHarvest] = useState('')
   120→  const [seasonLabel, setSeasonLabel] = useState('')
   121→
   122→  const crops = useKnowledgeCrops()
   123→
   124→  const addMutation = useMutation({
   125→    mutationFn: () =>
   126→      addCropHistory(selectedField.id, {
   127→        crop_code: cropCode || null,
   128→        crop_label: label || cropCode || 'Unnamed crop',
   129→        sowing_date: sowing || null,
   130→        harvest_date: harvest || null,
   131→        season_label: seasonLabel || null,
   132→      }),
   133→    onSuccess: () => {
   134→      toast.success('Crop history entry added.')
   135→      setCropCode('')
   136→      setLabel('')
   137→      setSowing('')
   138→      setHarvest('')
   139→      setSeasonLabel('')
   140→      void queryClient.invalidateQueries({ queryKey: ['crop-history', selectedField.id] })
   141→    },
   142→    onError: (err) => {
   143→      const d = describeError(err)
   144→      toast.error(d.title, { description: d.description })
   145→    },
   146→  })
   147→
   148→  const items = history.data?.items ?? []
   149→
   150→  return (
   151→    <Card>
   152→      <CardHeader className="pb-3">
   153→        <CardTitle className="flex items-center gap-2 text-base">
   154→          <Sprout className="h-4 w-4 text-emerald-700" aria-hidden />
   155→          Crop history (rotation baseline)
   156→        </CardTitle>
   157→        <CardDescription>
   158→          What actually grew here. The current rotation is preserved as the labelled
   159→          baseline in every optimization run.
   160→        </CardDescription>
   161→      </CardHeader>
   162→      <CardContent className="space-y-3">
   163→        {history.isLoading ? (
   164→          <div className="flex items-center gap-2 text-sm text-stone-500">
   165→            <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> Loading…
   166→          </div>
   167→        ) : items.length === 0 ? (
   168→          <p className="text-sm text-stone-500">
   169→            No crop history recorded — the baseline will be empty until you add entries.
   170→          </p>
   171→        ) : (
   172→          <ul className="max-h-40 space-y-1.5 overflow-y-auto pr-1 text-sm scrollbar-thin">
   173→            {items.map((item) => (
   174→              <li key={item.id} className="flex flex-wrap items-center gap-2">
   175→                <Badge variant="outline" className="border-stone-300 font-normal text-stone-700">
   176→                  {item.crop_code ?? item.crop_label}
   177→                </Badge>
   178→                <span className="text-xs text-stone-500">
   179→                  {item.season_label ?? '—'} · {item.sowing_date ?? '?'} → {item.harvest_date ?? '?'}
   180→                </span>
   181→                <Badge variant="outline" className="font-normal text-[10px] text-stone-500">
   182→                  {item.evidence_class}
   183→                </Badge>
   184→              </li>
   185→            ))}
   186→          </ul>
   187→        )}
   188→        <div className="grid grid-cols-2 gap-2 border-t border-stone-100 pt-3 sm:grid-cols-5">
   189→          <div className="col-span-2 sm:col-span-1">
   190→            <Label className="text-xs text-stone-500">Crop</Label>
   191→            <Select value={cropCode} onValueChange={setCropCode}>
   192→              <SelectTrigger className="h-9 text-sm">
   193→                <SelectValue placeholder="Knowledge crop…" />
   194→              </SelectTrigger>
   195→              <SelectContent className="max-h-72">
   196→                {(crops.data?.crops ?? []).map((c) => (
   197→                  <SelectItem key={c.id} value={c.crop_code}>
   198→                    {c.crop_code}
   199→                  </SelectItem>
   200→                ))}
   201→              </SelectContent>
   202→            </Select>
   203→          </div>
   204→          <div className="col-span-2 sm:col-span-1">
   205→            <Label className="text-xs text-stone-500">Label (free text)</Label>
   206→            <Input value={label} onChange={(e) => setLabel(e.target.value)} placeholder="e.g. Maize" className="h-9 text-sm" />
   207→          </div>
   208→          <div>
   209→            <Label className="text-xs text-stone-500">Sowing</Label>
   210→            <Input type="date" value={sowing} onChange={(e) => setSowing(e.target.value)} className="h-9 text-sm" />
   211→          </div>
   212→          <div>
   213→            <Label className="text-xs text-stone-500">Harvest</Label>
   214→            <Input type="date" value={harvest} onChange={(e) => setHarvest(e.target.value)} className="h-9 text-sm" />
   215→          </div>
   216→          <div className="flex items-end gap-2">
   217→            <div className="flex-1">
   218→              <Label className="text-xs text-stone-500">Season</Label>
   219→              <Input value={seasonLabel} onChange={(e) => setSeasonLabel(e.target.value)} placeholder="summer_2023" className="h-9 text-sm" />
   220→            </div>
   221→            <Button size="sm" className="h-9" onClick={() => addMutation.mutate()} disabled={addMutation.isPending || (!cropCode && !label)}>
   222→              {addMutation.isPending ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : 'Add'}
   223→            </Button>
   224→          </div>
   225→        </div>
   226→      </CardContent>
   227→    </Card>
   228→  )
   229→}

   233→function CandidateRow({ assessment }: { assessment: CandidateAssessment }) {
   234→  const [open, setOpen] = useState(false)
   235→  return (
   236→    <li className="rounded-lg border border-stone-200 bg-white">

   244→          <span className="truncate text-sm font-medium text-stone-900">{assessment.crop_code}</span>
   245→          <CandidateStatusBadge status={assessment.status} />

   247→            <Badge variant="outline" className="border-sky-200 bg-sky-50 font-normal text-sky-800">
   248→              ≈{fmt(assessment.water_requirement_mm, 0)} mm water
   249→            </Badge>
   250→          )}
   251→        </span>
   252→        <ChevronRight className={cn('h-4 w-4 shrink-0 text-stone-400 transition-transform', open && 'rotate-90')} aria-hidden />
   253→      </button>
   254→      {open && (
   255→        <div className="space-y-2 border-t border-stone-100 px-3 py-2 text-xs text-stone-600">
   256→          {assessment.reasons.map((r, i) => (
   257→            <p key={i} className="leading-relaxed">
   258→              <span
   259→                className={cn(
   260→                  'mr-1 inline-block h-1.5 w-1.5 rounded-full align-middle',
   261→                  r.severity === 'hard' ? 'bg-red-500' : r.severity === 'conditional' ? 'bg-amber-500' : 'bg-emerald-500',
   262→                )}
   263→                aria-hidden
   264→              />
   265→              {r.message}
   266→              {r.source_refs?.map((ref, j) => {
   267→                const typed = ref as { citation_text?: string; source_key?: string }
   268→                return typed.citation_text ? (
   269→                  <span key={j} className="ml-1 text-stone-400">[{typed.source_key}]</span>
   270→                ) : null
   271→              })}
   272→            </p>
   273→          ))}
   274→          {assessment.unevaluated_dimensions.map((d, i) => (
   275→            <p key={i} className="leading-relaxed text-stone-500">
   276→              <Info className="mr-1 inline h-3 w-3 align-[-2px]" aria-hidden />
   277→              Not evaluated — {d.message}
   278→            </p>
   279→          ))}
   280→          {assessment.water_estimate_assumptions.length > 0 && (
   281→            <p className="text-stone-400">Water estimate assumptions: {assessment.water_estimate_assumptions.join(' ')}</p>
   282→          )}
   283→        </div>
   284→      )}
   285→    </li>
   286→  )
   287→}

   289→function CandidatesCard({
   290→  selectedField,
   291→  seasons,
   292→}: {
   293→  selectedField: SelectedField
   294→  seasons: SeasonWindow[]
   295→}) {
   296→  const [result, setResult] = useState<{
   297→    seasons: CandidateSeason[]
   298→    warnings: string[]
   299→    counts: Record<string, Record<string, number>>
   300→  } | null>(null)
   301→
   302→  const mutation = useMutation({
   303→    mutationFn: () =>
   304→      generateCropCandidates(selectedField.id, {
   305→        seasons: seasons.map((s) => ({ label: s.label, start_date: s.start, end_date: s.end })),
   306→        hemisphere: 'north',
   307→      }),
   308→    onSuccess: (data) => {
   309→      setResult({
   310→        seasons: data.seasons,
   311→        warnings: data.warnings,
   312→        counts: data.counts_by_season,
   313→      })
   314→    },
   315→    onError: (err) => {
   316→      const d = describeError(err)
   317→      toast.error(d.title, { description: d.description })
   318→    },
   319→  })
   320→
   321→  return (
   322→    <Card>
   323→      <CardHeader className="pb-3">
   324→        <CardTitle className="text-base">Candidate crops per season</CardTitle>
   325→        <CardDescription>
   326→          Every knowledge-base crop evaluated against the planning seasons: calendar
   327→          windows, duration, climate ranges, soil, water estimability and rotation
   328→          knowledge. Missing evidence is reported, never filled.
   329→        </CardDescription>
   330→      </CardHeader>
   331→      <CardContent className="space-y-3">
   332→        <Button
   333→          size="sm"
   334→          onClick={() => mutation.mutate()}
   335→          disabled={mutation.isPending || seasons.length === 0}
   336→        >
   337→          {mutation.isPending ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : <Sprout className="h-4 w-4" aria-hidden />}
   338→          Generate candidates
   339→        </Button>
   340→        {result?.warnings.map((w, i) => (
   341→          <Alert key={i} className={w.includes('SYNTHETIC') ? 'border-amber-200 bg-amber-50' : undefined}>
   342→            <AlertTriangle className="h-4 w-4" aria-hidden />
   343→            <AlertDescription className="text-xs">{w}</AlertDescription>
   344→          </Alert>
   345→        ))}
   346→        {result?.seasons.map((season) => {
   347→          const counts = result.counts[season.season_label] ?? {}
   348→          return (
   349→            <div key={season.season_label} className="space-y-2">
   350→              <div className="flex flex-wrap items-center gap-1.5 text-xs text-stone-500">
   351→                <span className="font-semibold text-stone-700">{season.season_label}</span>
   352→                <Badge variant="outline" className="border-emerald-200 font-normal text-emerald-800">{counts.eligible ?? 0} eligible</Badge>
   353→                <Badge variant="outline" className="border-amber-200 font-normal text-amber-900">{counts.conditionally_eligible ?? 0} conditional</Badge>
   354→                <Badge variant="outline" className="border-red-200 font-normal text-red-800">{counts.excluded ?? 0} excluded</Badge>
   355→                <Badge variant="outline" className="border-stone-300 font-normal text-stone-600">{counts.unevaluated ?? 0} unevaluated</Badge>
   356→              </div>
   357→              <ul className="max-h-72 space-y-1.5 overflow-y-auto pr-1 scrollbar-thin">
   358→                {season.candidates.map((c) => (
   359→                  <CandidateRow key={c.crop_code} assessment={c} />
   360→                ))}
   361→              </ul>
   362→            </div>
   363→          )
   364→        })}
   365→      </CardContent>
   366→    </Card>
   367→  )
   368→}
