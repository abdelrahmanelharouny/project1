   337→   337→function AddFieldDialog({
   338→   338→  farm,
   339→   339→  open,
   340→   340→  onOpenChange,
   341→   341→  onFieldSelected,
   342→   342→}: {
   343→   343→  farm: Farm
   344→   344→  open: boolean
   345→   345→  onOpenChange: (open: boolean) => void
   346→   346→  onFieldSelected: (field: { id: string; name: string; farmId: string }) => void
   347→   347→}) {
   348→   348→  const queryClient = useQueryClient()
   349→   349→  const [name, setName] = useState('')
   350→   350→  const [geometry, setGeometry] = useState('')
   351→   351→  const [parseError, setParseError] = useState<string | null>(null)
   352→   352→  const [apiError, setApiError] = useState<{ title: string; description?: string } | null>(null)
   353→   353→  const [created, setCreated] = useState<FieldCreateResponse | null>(null)
   354→   354→
   355→   355→  // Reset form state when the dialog closes.
   356→   356→  const handleOpenChange = (next: boolean) => {
   357→   357→    if (!next) {
   358→   358→      setName('')
   359→   359→      setGeometry('')
   360→   360→      setParseError(null)
   361→   361→      setApiError(null)
   362→   362→      setCreated(null)
   363→   363→    }
   364→   364→    onOpenChange(next)
   365→   365→  }
   366→   366→
   367→   367→  const mutation = useMutation({
   368→   368→    mutationFn: (payload: { name: string; geometry: GeoJSONGeometry }) =>
   369→   369→      createField(farm.id, { name: payload.name, geometry: payload.geometry }),
   370→   370→    onSuccess: (result) => {
   371→   371→      queryClient.invalidateQueries({ queryKey: ['fields', farm.id] })
   372→   372→      queryClient.invalidateQueries({ queryKey: ['farms'] })
   373→   373→      if (result.geometry_warnings.length > 0) {
   374→   374→        setCreated(result)
   375→   375→        toast.warning('Field created — but review the geometry warnings', {
   376→   376→          description: `${result.geometry_warnings.length} warning(s) from PostGIS validation.`,
   377→   377→        })
   378→   378→      } else {
   379→   379→        toast.success(`Field “${result.name}” created`, {
   380→   380→          description: `${formatArea(result.area_ha)} ha · boundary v${result.boundary_version}`,
   381→   381→        })
   382→   382→        onFieldSelected({ id: result.id, name: result.name, farmId: farm.id })
   383→   383→        handleOpenChange(false)
   384→   384→      }
   385→   385→    },
   386→   386→    onError: (err) => {
   387→   387→      const { title, description } = describeError(err)
   388→   388→      setApiError({ title, description })
   389→   389→      toast.error(title, { description })
   390→   390→    },
   391→   391→  })
   392→   392→
   393→   393→  const submit = (event: FormEvent) => {
   394→   394→    event.preventDefault()
   395→   395→    setApiError(null)
   396→   396→    setCreated(null)
   397→   397→    if (!name.trim()) {
   398→   398→      setParseError('Give the field a name first.')
   399→   399→      return
   400→   400→    }
   401→   401→    const parsed = parseGeometryInput(geometry)
   402→   402→    if (!parsed.ok) {
   403→   403→      setParseError(parsed.error)
   404→   404→      return
   405→   405→    }
   406→   406→    setParseError(null)
   407→   407→    mutation.mutate({ name: name.trim(), geometry: parsed.geometry })
   408→   408→  }
   409→   409→
   410→   410→  return (
   411→   411→    <Dialog open={open} onOpenChange={handleOpenChange}>
   412→   412→      <DialogContent className="max-h-[85vh] overflow-y-auto scrollbar-thin sm:max-w-xl">
   413→   413→        <DialogHeader>
   414→   414→          <DialogTitle>Add a field to “{farm.name}”</DialogTitle>
   415→   415→          <DialogDescription>
   416→   416→            The boundary is validated by PostGIS. If something is off (self-intersections, unclosed rings), the
   417→   417→            response tells you exactly what was fixed.
   418→   418→          </DialogDescription>
   419→   419→        </DialogHeader>
   420→   420→        <form onSubmit={submit} className="space-y-4" aria-label="Add field form">
   421→   421→          <div className="space-y-1.5">
   422→   422→            <Label htmlFor="field-name">Field name</Label>
   423→   423→            <Input
   424→   424→              id="field-name"
   425→   425→              placeholder="North Field"
   426→   426→              value={name}
   427→   427→              onChange={(event) => setName(event.target.value)}
   428→   428→            />
   429→   429→          </div>
   430→   430→          <GeometryInput id="field-geometry" value={geometry} onChange={setGeometry} error={parseError} />
   431→   431→
   432→   432→          {created ? (
   433→   433→            <div className="space-y-2">
   434→   434→              <Alert className="border-emerald-300 bg-emerald-50 [&>svg]:text-emerald-700">
   435→   435→                <MapPin className="h-4 w-4" />
   436→   436→                <AlertTitle className="text-emerald-900">
   437→   437→                  Field “{created.name}” created — {formatArea(created.area_ha)} ha
   438→   438→                </AlertTitle>
   439→   439→                <AlertDescription className="text-emerald-900">
   440→   440→                  Boundary v{created.boundary_version} · passport status: {created.data_passport_status}
   441→   441→                </AlertDescription>
   442→   442→              </Alert>
   443→   443→              <GeometryWarningsAlert
   444→   444→                warnings={created.geometry_warnings}
   445→   445→                repaired={created.geometry_repaired}
   446→   446→                title="Geometry warnings from PostGIS"
   447→   447→              />
   448→   448→              <div className="flex justify-end gap-2">
   449→   449→                <Button
   450→   450→                  type="button"
   451→   451→                  variant="outline"
   452→   452→                  onClick={() => {
   453→   453→                    onFieldSelected({ id: created.id, name: created.name, farmId: farm.id })
   454→   454→                    handleOpenChange(false)
   455→   455→                  }}
   456→   456→                >
   457→   457→                  Open the new field
   458→   458→                </Button>
   459→   459→              </div>
   460→   460→            </div>
   461→   461→          ) : null}
   462→   462→
   463→   463→          {apiError ? (
   464→   464→            <Alert variant="destructive">
   465→   465→              <TriangleAlert className="h-4 w-4" />
   466→   466→              <AlertTitle>{apiError.title}</AlertTitle>
   467→   467→              <AlertDescription>{apiError.description ?? 'Adjust the geometry and try again.'}</AlertDescription>
   468→   468→            </Alert>
   469→   469→          ) : null}
   470→   470→
   471→   471→          <DialogFooter>
   472→   472→            <Button type="button" variant="outline" onClick={() => handleOpenChange(false)}>

   648→   648→                <MapPin className="h-4 w-4" />
   649→   649→                <AlertTitle className="text-emerald-900">
