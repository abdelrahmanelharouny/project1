   204→@router.post("/soil-samples/import")
   205→async def import_soil_samples(
   206→    file: UploadFile = File(...),
   207→    mapping: str | None = Form(None),
   208→    principal: Any = Depends(get_principal),
   209→    db: Session = Depends(get_db),
   210→) -> dict:
   211→    """Laboratory CSV import. `mapping` is an optional JSON mapping spec
   212→    (column → canonical field). Re-importing the same file is a no-op by
   213→    checksum (idempotency)."""
   214→    content = await file.read()
   215→    mapping_spec = None
   216→    if mapping:
   217→        try:
   218→            mapping_spec = json.loads(mapping)
   219→        except json.JSONDecodeError:
   220→            raise ValidationFailed("mapping must be valid JSON.", details=[{"field": "mapping"}]) from None
   221→    try:
   222→        parsed, report = parse_soil_lab_csv(content, mapping_spec=mapping_spec)
   223→    except Exception as exc:  # noqa: BLE001
   224→        raise ValidationFailed(str(exc), details=[{"field": "file"}]) from None
   225→
   226→    # idempotency by file checksum: skip re-import of identical content
   227→    existing = db.scalar(
   228→        select(dbm.IngestionJob).where(
   229→            dbm.IngestionJob.idempotency_key == report.idempotency_key,
   230→            dbm.IngestionJob.job_type == "soil_lab_import",
   231→        )
   232→    ) if hasattr(dbm.IngestionJob, "idempotency_key") else None
   233→    if existing is not None:
   234→        return {"status": "already_imported", "report": report.to_dict(), "import_job_id": str(existing.id)}
   235→
   236→    # resolve the organization for registries (via any accessible farm of the principal)
   237→    # note: import is org-level for labs/methods, then per-sample field resolution
   238→    created_samples = 0
   239→    created_measurements = 0
   240→    sample_cache: dict[str, dbm.SoilSample] = {}
   241→    layer_cache: dict[tuple[str, float, float], dbm.SoilSampleLayer] = {}
   242→    errors: list[str] = []
   243→
   244→    # group parsed rows by sample_code; each row needs a field context —
   245→    # the import request carries the field binding via per-sample field_id
   246→    # resolved from the sample_code prefix when provided, else via
   247→    # the caller's default_field_id in the mapping spec.
   248→    default_field_id = (mapping_spec or {}).get("default_field_id")
   249→    field_cache: dict[str, dbm.Field] = {}
   250→
   251→    def _field_for(sample_code: str) -> dbm.Field | None:
   252→        if default_field_id:
   253→            return field_cache.get(default_field_id)
   254→        return None
   255→
   256→    if not default_field_id:
   257→        raise ValidationFailed(
   258→            "the mapping spec must include 'default_field_id' (the field the samples belong to) "
   259→            "— per-sample field binding arrives with multi-field templates.",
   260→            details=[{"field": "mapping.default_field_id"}],
   261→        )
   262→    target_field = get_field_for_principal(db, principal, uuid.UUID(str(default_field_id)))
   263→    field_cache[str(default_field_id)] = target_field
   264→    organization_id = soil_service._org_id_for_field(db, target_field)
   265→
   266→    for row in parsed:
   267→        # savepoint: a rejected row must not poison the surrounding transaction
   268→        nested = db.begin_nested()
   269→        try:
   270→            sample = sample_cache.get(row["sample_code"])
   271→            if sample is None:
   272→                # create a minimal sampling event for the lab batch (composite import)
   273→                observed = date.fromisoformat(row["observed_at"])
   274→                event = dbm.SoilSamplingEvent(
   275→                    field_id=target_field.id,
   276→                    sampling_date=observed,
   277→                    purpose="laboratory analysis",
   278→                    sampling_method="import",
   279→                    composite=True,
   280→                    quality_status="valid",
   281→                    metadata={"import_template": report.template, "file_sha256": report.file_sha256},
   282→                )
   283→                db.add(event)
   284→                db.flush()
   285→                sample = dbm.SoilSample(
   286→                    event_id=event.id,
   287→                    field_id=target_field.id,
   288→                    sample_code=row["sample_code"],
   289→                    report_reference=row.get("report_reference"),
   290→                    metadata={"source": "soil_lab_import", "file_sha256": report.file_sha256},
   291→                )
   292→                if row.get("laboratory_name"):
   293→                    lab = soil_service.get_or_create_laboratory(db, organization_id, row["laboratory_name"])
   294→                    sample.laboratory_id = lab.id
   295→                db.add(sample)
   296→                db.flush()
   297→                sample_cache[row["sample_code"]] = sample
   298→                created_samples += 1
   299→            layer_key = (row["sample_code"], row["top_depth_cm"], row["bottom_depth_cm"])
   300→            layer = layer_cache.get(layer_key)
   301→            if layer is None:
   302→                layer = dbm.SoilSampleLayer(
   303→                    sample_id=sample.id,
   304→                    top_depth_cm=row["top_depth_cm"],
   305→                    bottom_depth_cm=row["bottom_depth_cm"],
   306→                    original_depth_unit=row["original_depth_unit"],
   307→                    original_top=row["original_top"],
   308→                    original_bottom=row["original_bottom"],
   309→                )
   310→                db.add(layer)
   311→                db.flush()
   312→                layer_cache[layer_key] = layer
   313→            method_params: dict[str, Any] = {}
   314→            method_row = None
   315→            if row.get("method_code"):
   316→                method_row = soil_service.get_or_create_method(db, organization_id, {
   317→                    "code": row["method_code"],
   318→                    "property_family": sc.soil_variable_definition(row["variable_code"])["family"],
   319→                })
   320→                method_params = {"code": row["method_code"]}
   321→            vdef = sc.soil_variable_definition(row["variable_code"])
   322→            measurement = dbm.SoilMeasurement(
   323→                field_id=target_field.id,
   324→                layer_id=layer.id,
   325→                variable_code=row["variable_code"],
   326→                value=row["value"],
   327→                unit=row["unit"],
   328→                top_depth_cm=row["top_depth_cm"],
   329→                bottom_depth_cm=row["bottom_depth_cm"],
   330→                evidence_class="farmer_reported_record" if (mapping_spec or {}).get("farmer_reported") else "direct_field_measurement",
   331→                spatial_support_type="lab_sample",
   332→                source_kind="lab_analysis",
   333→                method_id=method_row.id if method_row else None,
   334→                method_params=method_params,
   335→                detection_limit=row["detection_limit"],
   336→                result_qualifier=row["result_qualifier"],
   337→                is_synthetic=False,
   338→                observed_at=date.fromisoformat(row["observed_at"]),
   339→                metadata={"report_reference": row.get("report_reference"), "notes": row.get("notes")},
   340→            )
   341→            db.add(measurement)
   342→            db.flush()
   343→            created_measurements += 1
   344→            nested.commit()
   345→        except Exception as exc:  # noqa: BLE001 — per-row accounting
   346→            nested.rollback()
   347→            errors.append(f"{row.get('sample_code')}/{row['variable_code']}: {exc}")
   348→
   349→    db.commit()
   350→    return {
   351→        "status": "imported",
   352→        "report": report.to_dict(),
   353→        "created_samples": created_samples,
   354→        "created_measurements": created_measurements,
   355→        "errors": errors[:20],
   356→    }

   433→@router.post("/fields/{field_id}/soc-stock-runs", status_code=status.HTTP_201_CREATED)
   434→def create_soc_run(
   435→    field_id: uuid.UUID,
   436→    body: dict[str, Any],
   437→    principal: Any = Depends(get_principal),
   438→    db: Session = Depends(get_db),
   439→) -> dict:
   440→    field = get_field_for_principal(db, principal, field_id)
   441→    idempotency_key = body.get("idempotency_key")
   442→    if idempotency_key:
   443→        existing = db.scalar(
   444→            select(dbm.SocStockRun).where(
   445→                dbm.SocStockRun.idempotency_key == idempotency_key,
   446→                dbm.SocStockRun.field_id == field.id,
   447→            )
   448→        )
   449→        if existing is not None:
   450→            return _soc_run_payload(existing)
   451→
   452→    from app.soil.layers import make_depth_interval
   453→    from app.soil.soc import LayerStockInput
   454→
   455→    sample_ids = body.get("sample_ids") or []
   456→    layers_in = body.get("layers") or []
   457→    if not layers_in:
   458→        raise ValidationFailed(
   459→            "layers are required (per-layer OC, bulk density + basis, optional coarse "
   460→            "fragments, depth interval) — SOC stock is computed from explicit inputs.",
   461→            details=[{"field": "layers"}],
   462→        )
   463→    inputs: list[LayerStockInput] = []
   464→    warnings: list[str] = []
   465→    for spec in layers_in:
   466→        try:
   467→            interval = make_depth_interval(float(spec["top_depth"]), float(spec["bottom_depth"]),
   468→                                           str(spec.get("depth_unit") or "cm"))
   469→        except (KeyError, TypeError, sc.SoilValidationError) as exc:
   470→            raise ValidationFailed(f"invalid layer depth: {exc}", details=[{"field": "layers"}]) from None
   471→        bd_basis = spec.get("bd_basis")
   472→        if bd_basis is not None and bd_basis not in ("whole_soil", "fine_earth"):
   473→            raise ValidationFailed(
   474→                f"bd_basis must be 'whole_soil' or 'fine_earth' (got '{bd_basis}')",
   475→                details=[{"field": "layers"}],
   476→            )
   477→        vcf = spec.get("coarse_fragment_vol_fraction")
   478→        inputs.append(LayerStockInput(
   479→            interval=interval,
   480→            oc_g_kg=spec.get("oc_g_kg"),
   481→            bulk_density_g_cm3=spec.get("bulk_density_g_cm3"),
   482→            bd_basis=bd_basis,
   483→            coarse_fragment_vol_fraction=vcf,
   484→            evidence_class=str(spec.get("evidence_class") or "farmer_reported_record"),
   485→            is_prediction=bool(spec.get("is_prediction", False)),
   486→            source_refs=tuple(spec.get("source_refs") or ()),
   487→        ))
   488→        if spec.get("is_prediction"):
   489→            warnings.append(
   490→                f"layer {interval.top_cm}-{interval.bottom_cm} cm uses mapped predictions "
   491→                "— the stock is a prediction, not a field measurement"
   492→            )
   493→
   494→    mode = str(body.get("comparison_mode") or "single_profile")
   495→    run = dbm.SocStockRun(
   496→        field_id=field.id,
   497→        status="running",
   498→        label=str(body.get("label") or "SOC stock run")[:120],
   499→        input_sample_ids=[str(s) for s in sample_ids],
   500→        comparison_mode=mode,
   501→        engine_version=sc.SOC_STOCK_VERSION,
   502→        methods_version=sc.SOIL_METHODS_VERSION,
   503→        request_parameters=dict(body.get("request_parameters") or {}),
   504→        idempotency_key=idempotency_key,
   505→    )
   506→    db.add(run)
   507→    db.flush()
   508→
   509→    try:
   510→        result = soc_mod.profile_soc_stock(inputs)
   511→        comparison_block = None
   512→        if mode == "single_profile":
   513→            pass
   514→        else:
   515→            baseline_run_id = body.get("baseline_run_id")
   516→            if baseline_run_id is None:
   517→                raise ValidationFailed(
   518→                    "comparison modes require baseline_run_id.",
   519→                    details=[{"field": "baseline_run_id"}],
   520→                )
   521→            baseline = db.get(dbm.SocStockRun, uuid.UUID(str(baseline_run_id)))
   522→            if baseline is None or baseline.field_id != field.id:
   523→                raise NotFound("Baseline SOC run not found for this field.")
   524→            base_layers = [
   525→                LayerStockInput(
   526→                    interval=make_depth_interval(l["top_cm"], l["bottom_cm"], "cm"),
   527→                    oc_g_kg=(l["inputs_used"] or {}).get("oc_g_kg"),
   528→                    bulk_density_g_cm3=(l["inputs_used"] or {}).get("bulk_density_g_cm3"),
   529→                    bd_basis=(l["inputs_used"] or {}).get("bd_basis"),
   530→                    coarse_fragment_vol_fraction=(l["inputs_used"] or {}).get("coarse_fragment_vol_fraction"),
   531→                )
   532→                for l in baseline.result.get("layers", [])
   533→            ]
   534→            if mode == "fixed_depth":
   535→                comparison_block = soc_mod.fixed_depth_comparison(base_layers, inputs)
   536→            elif mode == "equivalent_soil_mass":
   537→                comparison_block = soc_mod.equivalent_soil_mass_comparison(base_layers, inputs)
   538→            else:
   539→                raise ValidationFailed(f"unknown comparison_mode '{mode}'.", details=[{"field": "comparison_mode"}])
   540→            result["comparison"] = comparison_block
   541→
   542→        run.status = "succeeded"
   543→        run.result = result
   544→        run.warnings = warnings + result.get("warnings", [])
   545→        run.not_evaluated = result.get("not_evaluated", [])
   546→        run.checksum_sha256 = soc_mod.soc_run_checksum(result)
   547→        db.flush()
   548→    except ValidationFailed:
   549→        run.status = "failed"
   550→        run.error_summary = "validation failed"
   551→        db.flush()
   552→        raise
   553→    except Exception as exc:  # noqa: BLE001 — job accounting
   554→        run.status = "failed"
   555→        run.error_summary = f"{type(exc).__name__}: {exc}"
   556→        db.flush()
   557→        raise
   558→    db.commit()
   559→    db.refresh(run)
   560→    return _soc_run_payload(run)

   646→def _nutrient_application_from_payload(db: Session, field: dbm.Field, body: dict[str, Any]) -> dbm.NutrientApplication:
   647→    product = None
   648→    if body.get("product_id"):
   649→        product = db.get(dbm.NutrientProduct, uuid.UUID(str(body["product_id"])))
   650→        if product is None:
   651→            raise NotFound("Nutrient product not found.")
   652→    n_kg = p_kg = k_kg = None
   653→    reporting_basis = None
   654→    if product is not None:
   655→        reporting_basis = product.reporting_basis
   656→        mass = float(soil_service._require(body, "applied_mass_kg"))
   657→        n_kg = (product.n_pct / 100.0 * mass) if product.n_pct is not None else None
   658→        p_kg = (product.p_pct / 100.0 * mass) if product.p_pct is not None else None
   659→        k_kg = (product.k_pct / 100.0 * mass) if product.k_pct is not None else None
   660→    else:
   661→        n_kg = body.get("n_kg")

   663→        k_kg = body.get("k_kg")

   668→            "explicit n/p/kg masses) — an application without nutrient content is "
   669→            "not a nutrient input.",
   670→            details=[{"field": "nutrients"}],
   671→        )
   672→    availability = body.get("availability_fraction")
   673→    input_kind = str(soil_service._require(body, "input_kind"))
   674→    if input_kind in ("manure", "compost", "organic_amendment") and availability is None:
   675→        raise ValidationFailed(

   690→        p_kg=None if p_kg is None else float(p_kg),
   691→        k_kg=None if k_kg is None else float(k_kg),

   695→        application_method=body.get("application_method"),

   697→        evidence_class=str(body.get("evidence_class") or "farmer_reported_record"),
   698→        is_synthetic=bool(body.get("is_synthetic", False)),

   775→@router.post("/fields/{field_id}/nutrient-budget-runs", status_code=status.HTTP_201_CREATED)
   776→def create_nutrient_budget_run(
   777→    field_id: uuid.UUID,
   778→    body: dict[str, Any],
   779→    principal: Any = Depends(get_principal),
   780→    db: Session = Depends(get_db),
   781→) -> dict:
   782→    field = get_field_for_principal(db, principal, field_id)
   783→    idempotency_key = body.get("idempotency_key")
   784→    if idempotency_key:
   785→        existing = db.scalar(
   786→            select(dbm.NutrientBudgetRun).where(
   787→                dbm.NutrientBudgetRun.idempotency_key == idempotency_key,
   788→                dbm.NutrientBudgetRun.field_id == field.id,
   789→            )
   790→        )
   791→        if existing is not None:
   792→            return _budget_run_payload(existing)
   793→
   794→    seasons = body.get("seasons") or []
   795→    if not seasons:
   796→        raise ValidationFailed("at least one season label is required.", details=[{"field": "seasons"}])
   797→
   798→    applications = db.scalars(
   799→        select(dbm.NutrientApplication).where(dbm.NutrientApplication.field_id == field.id)
   800→    ).all()
   801→    removals = db.scalars(
   802→        select(dbm.NutrientRemoval).where(dbm.NutrientRemoval.field_id == field.id)
   803→    ).all()
   804→
   805→    run = dbm.NutrientBudgetRun(
   806→        field_id=field.id,
   807→        status="running",
   808→        label=str(body.get("label") or "Nutrient budget run")[:120],
   809→        seasons=list(seasons),
   810→        engine_version=sc.NUTRIENT_BUDGET_VERSION,
   811→        methods_version=sc.SOIL_METHODS_VERSION,
   812→        request_parameters=dict(body.get("request_parameters") or {}),
   813→        idempotency_key=idempotency_key,
   814→    )
   815→    db.add(run)
   816→    db.flush()
   817→
   818→    try:
   819→        season_results = []
   820→        for season_label in seasons:
   821→            season_inputs = []
   822→            for app_row in applications:
   823→                if app_row.season_label != season_label:
   824→                    continue
   825→                if app_row.input_kind == "biological_nitrogen_fixation":
   826→                    # BNF enters only as the recorded estimate (never inferred here)
   827→                    season_inputs.append(nutrient_mod.NutrientInputRecord(
   828→                        flow_id=str(app_row.id), input_kind=app_row.input_kind,
   829→                        n_kg=app_row.n_kg, p_kg=app_row.p_kg, k_kg=app_row.k_kg,
   830→                        area_ha=app_row.area_ha, reporting_basis=app_row.reporting_basis or "elemental",
   831→                        evidence_class=app_row.evidence_class, is_estimate=True,
   832→                        citation_text=app_row.source_reference,
   833→                    ))
   834→                    continue
   835→                season_inputs.append(nutrient_mod.NutrientInputRecord(
   836→                    flow_id=str(app_row.id), input_kind=app_row.input_kind,
   837→                    n_kg=app_row.n_kg, p_kg=app_row.p_kg, k_kg=app_row.k_kg,
   838→                    area_ha=app_row.area_ha, reporting_basis=app_row.reporting_basis or "elemental",
   839→                    evidence_class=app_row.evidence_class,
   840→                    availability_fraction=app_row.availability_fraction,
   841→                    availability_basis=app_row.availability_basis,
   842→                ))
   843→            season_outputs = []
   844→            for r in removals:
   845→                if r.season_label != season_label:
   846→                    continue
   847→                # concentration g/kg × biomass kg = g nutrient → ÷1000 = kg total
   848→                area_ha = float(body.get("area_ha") or field.area_ha)
   849→                n_kg_ha = (r.n_concentration_g_kg * r.biomass_kg / 1000.0 / area_ha) if r.n_concentration_g_kg is not None else None
   850→                p_kg_ha = (r.p_concentration_g_kg * r.biomass_kg / 1000.0 / area_ha) if r.p_concentration_g_kg is not None else None
   851→                k_kg_ha = (r.k_concentration_g_kg * r.biomass_kg / 1000.0 / area_ha) if r.k_concentration_g_kg is not None else None
   852→                season_outputs.append(nutrient_mod.NutrientOutputRecord(
   853→                    flow_id=str(r.id), output_kind=r.output_kind,
   854→                    n_kg_ha=n_kg_ha, p_kg_ha=p_kg_ha, k_kg_ha=k_kg_ha,
   855→                    concentration_basis=r.concentration_basis or r.moisture_basis,
   856→                    moisture_basis_disclosed=bool(r.moisture_basis),
   857→                    evidence_class=r.evidence_class,
   858→                ))
   859→            season_results.append(nutrient_mod.nutrient_budget(
   860→                inputs=season_inputs, outputs=season_outputs, season_label=str(season_label)
   861→            ))
   862→        cumulative = nutrient_mod.cumulative_budget(season_results)
   863→        result = {
   864→            "seasons": season_results,
   865→            "cumulative": cumulative,
   866→            "engine_version": sc.NUTRIENT_BUDGET_VERSION,
   867→            "methods_version": sc.SOIL_METHODS_VERSION,
   868→            "reference_library_note": (
   869→                "the nutrient-removal reference library is intentionally EMPTY pending "
   870→                "verified sources — concentrations come from farmer/lab records only "
   871→                "(see M6B_RESEARCH_BACKLOG.md)"
   872→            ),
   873→        }
   874→        run.status = "succeeded"
   875→        run.result = result
   876→        all_warnings = [w for s in season_results for w in s["warnings"]]
   877→        run.warnings = all_warnings
   878→        run.not_evaluated = {
   879→            s["season_label"]: s["missing_flows"] for s in season_results if s["missing_flows"]
   880→        }
   881→        run.checksum_sha256 = nutrient_mod.budget_checksum(result)
   882→        db.flush()
   883→    except ValidationFailed:
   884→        run.status = "failed"
   885→        run.error_summary = "validation failed"
   886→        db.flush()
   887→        raise
   888→    except Exception as exc:  # noqa: BLE001
   889→        run.status = "failed"
   890→        run.error_summary = f"{type(exc).__name__}: {exc}"
   891→        db.flush()
   892→        raise
   893→    db.commit()
   894→    db.refresh(run)
   895→    return _budget_run_payload(run)

   991→    field_id: uuid.UUID,
   992→    body: dict[str, Any],
   993→    principal: Any = Depends(get_principal),
   994→    db: Session = Depends(get_db),
   995→) -> dict:
   996→    field = get_field_for_principal(db, principal, field_id)
   997→    idempotency_key = body.get("idempotency_key")
   998→    if idempotency_key:
   999→        existing = db.scalar(
  1000→            select(dbm.SoilAssessmentRun).where(
  1001→                dbm.SoilAssessmentRun.idempotency_key == idempotency_key,
  1002→                dbm.SoilAssessmentRun.field_id == field.id,
  1003→            )
  1004→        )
  1005→        if existing is not None:
  1006→            return _assessment_payload(existing)
  1007→    measurements = [
  1008→        soil_service.measurement_payload(m) for m in soil_service.list_measurements(db, field.id)
  1009→    ]
  1010→    run = dbm.SoilAssessmentRun(
  1011→        field_id=field.id,
  1012→        status="running",
  1013→        label=str(body.get("label") or "Soil assessment")[:120],
  1014→        engine_version=sc.SOIL_ENGINE_VERSION,
  1015→        methods_version=sc.SOIL_METHODS_VERSION,
  1016→        request_parameters=dict(body.get("request_parameters") or {}),
  1017→        idempotency_key=idempotency_key,
  1018→    )
  1019→    db.add(run)
  1020→    db.flush()
  1021→    result = assessment_mod.assess_domains(measurements, field_id=str(field.id))
  1022→    run.status = "succeeded"
  1023→    run.result = result
  1024→    run.checksum_sha256 = assessment_mod.assessment_checksum(result)
  1025→    db.flush()
  1026→    db.commit()
  1027→    db.refresh(run)
  1028→    return _assessment_payload(run)
