   337→# -------------------------------------------------------- background jobs
   338→class BackgroundJob(Base, TimestampMixin):
   339→    """Queue-level job record for work executed outside the HTTP request.
   340→
   341→    One row per submitted task. Domain tables (e.g. ``ingestion_jobs``) keep
   342→    their own domain-specific state; a BackgroundJob row references the domain
   343→    row via ``ref_kind``/``ref_id`` so clients can poll a single unified
   344→    endpoint (``GET /api/v1/jobs/background``) regardless of job kind.
   345→
   346→    Lifecycle: queued -> running -> succeeded | failed. Rows are never
   347→    hard-deleted (audit/provenance). Jobs left 'running' by a process crash
   348→    are marked failed('interrupted') on the next startup — only 'queued'
   349→    jobs are re-enqueued.
   350→    """
   351→
   352→    __tablename__ = "background_jobs"
   353→
   354→    id: Mapped[uuid.UUID] = mapped_column(UUID(as_uuid=True), primary_key=True, default=_uuid)
   355→    kind: Mapped[str] = mapped_column(String(60), nullable=False)
   356→    status: Mapped[str] = mapped_column(String(20), nullable=False, default="queued")
   357→    progress: Mapped[int] = mapped_column(Integer, nullable=False, default=0)
   358→    stage: Mapped[str | None] = mapped_column(String(200), nullable=True)
   359→    ref_kind: Mapped[str | None] = mapped_column(String(60), nullable=True)
   360→    ref_id: Mapped[uuid.UUID | None] = mapped_column(UUID(as_uuid=True), nullable=True)
   361→    label: Mapped[str | None] = mapped_column(Text, nullable=True)
   362→    result: Mapped[dict[str, Any] | None] = mapped_column(JSONB, nullable=True)
   363→    error: Mapped[str | None] = mapped_column(Text, nullable=True)
   364→    request_id: Mapped[str | None] = mapped_column(String(64), nullable=True)
   365→    created_by: Mapped[uuid.UUID | None] = mapped_column(UUID(as_uuid=True), nullable=True)
   366→    started_at: Mapped[datetime | None] = mapped_column(DateTime(timezone=True), nullable=True)
   367→    finished_at: Mapped[datetime | None] = mapped_column(DateTime(timezone=True), nullable=True)
   368→
   369→    __table_args__ = (
   370→        CheckConstraint(
   371→            "status IN ('queued','running','succeeded','failed','cancelled')",
   372→            name="ck_background_job_status",
   373→        ),
   374→        CheckConstraint("progress >= 0 AND progress <= 100", name="ck_background_job_progress"),
   375→        Index("ix_background_jobs_status", status),
   376→        Index("ix_background_jobs_ref", ref_kind, ref_id),
   377→        Index("ix_background_jobs_created_by", created_by),
   378→    )
