    60→def _validate_seasons(raw: list[dict[str, Any]]) -> list[SeasonSpec]:
    61→    if not raw:
    62→        raise ValidationFailed(
    63→            "At least one planning season is required.",
    64→            details=[{"field": "seasons"}],
    65→        )
    66→    if len(raw) > MAX_SEASONS:
    67→        raise ValidationFailed(
    68→            f"At most {MAX_SEASONS} planning seasons per run.",
    69→            details=[{"field": "seasons"}],
    70→        )
    71→    seasons: list[SeasonSpec] = []
    72→    labels: set[str] = set()
    73→    for item in raw:
    74→        label = str(item.get("label") or "").strip()
    75→        if not label or len(label) > 60:
    76→            raise ValidationFailed(
    77→                "Season labels must be 1-60 characters.",
    78→                details=[{"field": "seasons", "reason": str(item)[:80]}],
    79→            )
    80→        if label in labels:
    81→            raise ValidationFailed(
    82→                f"Duplicate season label '{label}'.",
    83→                details=[{"field": "seasons", "reason": label}],
    84→            )
    85→        labels.add(label)
    86→        try:
    87→            start = date.fromisoformat(str(item["start_date"]))
    88→            end = date.fromisoformat(str(item["end_date"]))
    89→        except (KeyError, ValueError):
    90→            raise ValidationFailed(
    91→                "Seasons need start_date and end_date (ISO dates).",
    92→                details=[{"field": "seasons", "reason": label}],
    93→            ) from None
    94→        if start > end:
    95→            raise ValidationFailed(
    96→                f"Season '{label}' start_date is after end_date.",
    97→                details=[{"field": "seasons", "reason": label}],
    98→            )
    99→        if (end - start).days > 400:
   100→            raise ValidationFailed(
   101→                f"Season '{label}' is longer than 400 days — split it.",
   102→                details=[{"field": "seasons", "reason": label}],
   103→            )
   104→        seasons.append(SeasonSpec(label=label, start_date=start, end_date=end))
   105→    return seasons

   108→def _pick_evidence_run(db: Session, field_id: Any, explicit_id: str | None) -> dbm.FeatureRun | None:
   109→    if explicit_id:
   110→        run = db.get(dbm.FeatureRun, uuid.UUID(explicit_id))
   111→        if run is None or str(run.field_id) != str(field_id):
   112→            raise NotFound("Feature run not found for this field.")
   113→        if run.status != "succeeded":
   114→            raise ValidationFailed(
   115→                "The referenced feature run has not succeeded.",
   116→                details=[{"field": "evidence_run_id", "reason": run.status}],
   117→            )
   118→        return run
   119→    return db.scalar(
   120→        select(dbm.FeatureRun)
   121→        .where(dbm.FeatureRun.field_id == field_id, dbm.FeatureRun.status == "succeeded")
   122→        .order_by(dbm.FeatureRun.completed_at.desc())
   123→        .limit(1)
   124→    )

   127→def _season_climate(
   128→    run: dbm.FeatureRun, season: SeasonSpec
   129→) -> ClimateEvidence | None:
   130→    """Aggregate the run's monthly feature values over one planning season as a
   131→    MONTH-OF-YEAR climatology (historical proxy — planning seasons are usually
   132→    in the future; no forecast is claimed).
   133→
   134→    Coverage = fraction of season days whose month-of-year has any covered
   135→    value in the run; below MIN_CLIMATE_COVERAGE the season's climate evidence
   136→    is None (the dimension is unevaluated, never approximated)."""
   137→    monthly: dict[str, dict[int, list[tuple[int, float]]]] = {}  # code -> month -> [(year, value)]
   138→    for row in run.values:
   139→        if row.aggregation_level != "monthly" or row.value is None:
   140→            continue
   141→        monthly.setdefault(row.feature_code, {}).setdefault(
   142→            row.period_start.month, []
   143→        ).append((row.period_start.year, row.value))
   144→
   145→    import calendar as cal
   146→
   147→    def months_in_season(season: SeasonSpec) -> list[tuple[int, int, int]]:
   148→        """(year, month, days of that month inside the season)."""
   149→        out = []
   150→        y, m = season.start_date.year, season.start_date.month
   151→        while (y, m) <= (season.end_date.year, season.end_date.month):
   152→            if y == season.start_date.year and m == season.start_date.month:
   153→                month_start = season.start_date
   154→            else:
   155→                month_start = date(y, m, 1)
   156→            if y == season.end_date.year and m == season.end_date.month:
   157→                month_end = season.end_date
   158→            else:
   159→                month_end = date(y, m, cal.monthrange(y, m)[1])
   160→            out.append((y, m, (month_end - month_start).days + 1))
   161→            m += 1
   162→            if m == 13:
   163→                y, m = y + 1, 1
   164→        return out
   165→
   166→    month_days = months_in_season(season)
   167→    total_days = sum(d for _, _, d in month_days)
   168→
   169→    years_used = sorted(
   170→        {
   171→            y
   172→            for table in monthly.values()
   173→            for entries in table.values()
   174→            for y, _v in entries
   175→        }
   176→    )
   177→
   178→    def collect(code: str, mode: str) -> tuple[float | None, int]:
   179→        """Season aggregate of month-of-year climatological means.
   180→
   181→        mode='total' (precip, ET0, hot-day counts): monthly TOTALS scaled by
   182→        the fraction of that month inside the season, summed.
   183→        mode='mean' (temperature): monthly means day-weighted over the season.
   184→        Returns (value, days covered by present months)."""
   185→        table = monthly.get(code, {})
   186→        acc = 0.0
   187→        covered = 0
   188→        for y, m, days in month_days:
   189→            entries = table.get(m)
   190→            if entries:
   191→                mean = sum(v for _yr, v in entries) / len(entries)
   192→                if mode == "total":
   193→                    frac = days / cal.monthrange(y, m)[1]
   194→                    acc += mean * frac
   195→                else:
   196→                    acc += mean * days
   197→                covered += days
   198→        if mode == "mean":
   199→            acc = acc / total_days if total_days else 0.0
   200→        return acc, covered
   201→
   202→    warnings: list[str] = []
   203→    precip_total, covered = collect(CLIMATE_FEATURE_CODES["precipitation"], "total")
   204→    tmean_wsum, covered_t = collect(CLIMATE_FEATURE_CODES["tmean"], "mean")
   205→    hot_days, covered_h = collect(CLIMATE_FEATURE_CODES["hot_days"], "total")
   206→    et0_total, covered_e = collect(CLIMATE_FEATURE_CODES["et0"], "total")
   207→
   208→    coverage = covered / total_days if total_days else 0.0
   209→
   210→    monthly_et0: list[tuple[int, int, float]] = []
   211→    et0_table = monthly.get(CLIMATE_FEATURE_CODES["et0"], {})
   212→    for y, m, _days in month_days:
   213→        entries = et0_table.get(m)
   214→        if entries:
   215→            monthly_et0.append((y, m, sum(v for _yr, v in entries) / len(entries)))
   216→
   217→    # M6A: month-of-year precipitation climatology for the net irrigation
   218→    # requirement (months absent stay absent — never zero-filled).
   219→    monthly_precip: list[tuple[int, int, float]] = []
   220→    precip_table = monthly.get(CLIMATE_FEATURE_CODES["precipitation"], {})
   221→    for y, m, _days in month_days:
   222→        entries = precip_table.get(m)
   223→        if entries:
   224→            monthly_precip.append((y, m, sum(v for _yr, v in entries) / len(entries)))
   225→
   226→    if coverage < MIN_CLIMATE_COVERAGE:
   227→        return None
   228→    if covered < total_days:
   229→        warnings.append(
   230→            "precipitation months incomplete — the season total includes only the "
   231→            "months present in the evidence run (an under-count, never filled)"
   232→        )
   233→
   234→    tmean = tmean_wsum if covered_t == total_days else (
   235→        tmean_wsum * total_days / covered_t if covered_t else None
   236→    )
   237→    if covered_t < total_days:
   238→        warnings.append("temperature months incomplete — mean weighted by covered days")
   239→    hot = round(hot_days, 1) if covered_h == total_days else None
   240→    if covered_h < total_days and hot is None:
   241→        warnings.append("hot-day counts incomplete — not evaluated for this season")
   242→    et0 = et0_total if covered_e == total_days else None
   243→    if covered_e < total_days and et0 is None:
   244→        warnings.append("ET0 months incomplete — water estimates not evaluated")
   245→
   246→    year_note = (
   247→        f"years {years_used[0]}-{years_used[-1]}" if len(years_used) > 1
   248→        else f"year {years_used[0]}" if years_used else "no years"
   249→    )
   250→    return ClimateEvidence(
   251→        feature_run_id=str(run.id),
   252→        is_synthetic=bool(run.is_synthetic),
   253→        season_precipitation_mm=round(precip_total, 2) if covered else None,
   254→        season_mean_temp_degC=round(tmean, 2) if tmean is not None else None,
   255→        season_mean_tmax_degC=None,
   256→        season_hot_days_35c=hot,
   257→        season_et0_mm=round(et0, 2) if et0 is not None else None,
   258→        monthly_et0=tuple(monthly_et0),
   259→        monthly_precip=tuple(monthly_precip),
   260→        coverage_ratio=round(coverage, 4),
   261→        years_used=tuple(years_used),
   262→        basis=(
   263→            f"month-of-year climatology from feature run {str(run.id)[:8]} "
   264→            f"({year_note}) — a historical proxy, NOT a forecast for the "
   265→            "planning season."
   266→        ),
   267→        warnings=tuple(warnings),
   268→    )

   271→def _soil_evidence(db: Session, field_id: Any) -> tuple[SoilEvidence | None, str | None]:
   272→    rows = db.execute(
   273→        select(dbm.Observation)
   274→        .where(
   275→            dbm.Observation.field_id == field_id,
   276→            dbm.Observation.variable_code.in_(SOIL_VARIABLES),
   277→            dbm.Observation.value.is_not(None),
   278→        )
   279→        .order_by(dbm.Observation.observed_at.desc())
   280→        .limit(50)
   281→    ).scalars().all()
   282→    latest: dict[str, dbm.Observation] = {}
   283→    for obs in rows:
   284→        latest.setdefault(obs.variable_code, obs)
   285→    if not latest:
   286→        return None, None
   287→    evidence_classes = sorted({o.evidence_class or "unknown" for o in latest.values()})
   288→
   289→    def val(code: str) -> tuple[float, str] | None:
   290→        obs = latest.get(code)
   291→        if obs is None or obs.value is None:
   292→            return None
   293→        return (float(obs.value), obs.evidence_class or "unknown")
   294→
   295→    soil = SoilEvidence(
   296→        ph=val("soil_ph"),
   297→        organic_carbon_g_kg=val("soil_organic_carbon"),
   298→        clay_g_kg=val("soil_clay_content"),
   299→        sand_g_kg=val("soil_sand_content"),
   300→        bulk_density=val("soil_bulk_density"),
   301→    )
   302→    return soil, ", ".join(evidence_classes)

   305→def build_farm_context(
   306→    db: Session,
   307→    field: dbm.Field,
   308→    seasons: list[SeasonSpec],
   309→    evidence_run_id: str | None,
   310→    hemisphere: str,
   311→) -> tuple[FarmContext, list[str], list[str], dbm.FeatureRun | None]:
   312→    warnings: list[str] = []
   313→    run = _pick_evidence_run(db, field.id, evidence_run_id)
   314→    climate_by_season: dict[str, ClimateEvidence] = {}
   315→    climate_run_refs: dict[str, str] = {}
   316→    if run is None:
   317→        warnings.append(
   318→            "No succeeded feature run exists for this field — climate dimensions are "
   319→            "unevaluated (missing evidence is never treated as zero)."
   320→        )
   321→    else:
   322→        for season in seasons:
   323→            evidence = _season_climate(run, season)
   324→            if evidence is None:
   325→                warnings.append(
   326→                    f"Season '{season.label}': climate coverage below "
   327→                    f"{MIN_CLIMATE_COVERAGE:.0%} in the evidence run — climate dimensions "
   328→                    "unevaluated for this season."
   329→                )
   330→            else:
   331→                climate_by_season[season.label] = evidence
   332→                climate_run_refs[season.label] = str(run.id)
   333→                for w in evidence.warnings:
   334→                    warnings.append(f"Season '{season.label}': {w}")
   335→        if run.is_synthetic:
   336→            warnings.append(
   337→                "The evidence feature run used SYNTHETIC data — every climate-informed "
   338→                "outcome is a demonstration, never real farm evidence."
   339→            )
   340→
   341→    soil, soil_class = _soil_evidence(db, field.id)
   342→
   343→    history_rows = db.scalars(
   344→        select(dbm.CropHistory)
   345→        .where(dbm.CropHistory.field_id == field.id)
   346→        .order_by(dbm.CropHistory.sowing_date.asc().nullslast(), dbm.CropHistory.created_at.asc())
   347→    ).all()
   348→    crop_code_by_id = {
   349→        c.id: c.crop_code
   350→        for c in db.scalars(select(dbm.Crop)).all()
   351→    }
   352→    history: list[dict[str, Any]] = []
   353→    for h in history_rows:
   354→        history.append({
   355→            "id": str(h.id),
   356→            "season_label": h.season_label,
   357→            "crop_code": crop_code_by_id.get(h.crop_id) if h.crop_id else None,
   358→            "crop_label": h.crop_label,
   359→            "sowing_date": h.sowing_date.isoformat() if h.sowing_date else None,
   360→            "harvest_date": h.harvest_date.isoformat() if h.harvest_date else None,
   361→        })
   362→
   363→    constraints_rows = db.scalars(
   364→        select(dbm.FarmConstraint).where(dbm.FarmConstraint.farm_id == field.farm_id)
   365→    ).all()
   366→    market_exclusions: list[str] = []
   367→    equipment_limits: list[str] = []
   368→    water_allocation: float | None = None
   369→    margin_index: dict[str, float] = {}
   370→    for c in constraints_rows:
   371→        if c.constraint_code == "market_exclusion" and c.value_text:
   372→            market_exclusions.extend(
   373→                [p.strip() for p in c.value_text.split(",") if p.strip()]
   374→            )
   375→        elif c.constraint_code == "equipment_limitation" and c.value_text:
   376→            equipment_limits.extend(
   377→                [p.strip() for p in c.value_text.split(",") if p.strip()]
   378→            )
   379→        elif c.constraint_code == "water_allocation_m3_per_season" and c.value_numeric is not None:
   380→            water_allocation = float(c.value_numeric)
   381→        elif c.constraint_code == "market_margin_index" and c.value_text and c.value_numeric is not None:
   382→            margin_index[c.value_text.strip()] = float(c.value_numeric)
   383→
   384→    farm = FarmContext(
   385→        field_id=str(field.id),
   386→        farm_id=str(field.farm_id),
   387→        area_ha=float(field.area_ha),
   388→        country_code=field.farm.country_code,
   389→        hemisphere=hemisphere,
   390→        seasons=tuple(seasons),
   391→        soil=soil,
   392→        climate_by_season=climate_by_season,
   393→        climate_run_refs=climate_run_refs,
   394→        history=tuple(history),
   395→        market_exclusions=tuple(sorted(set(market_exclusions))),
   396→        equipment_limits=tuple(sorted(set(equipment_limits))),
   397→        water_allocation_m3=water_allocation,
   398→        margin_index_by_crop=margin_index,
   399→        synthetic_evidence_in_use=bool(run.is_synthetic) if run else False,
   400→        soil_evidence_class=soil_class,
   401→    )
   402→    return farm, warnings, [str(h["id"]) for h in history], run

   405→def _baseline_assignment(
   406→    seasons: list[SeasonSpec], history: tuple[dict[str, Any], ...]
   407→) -> tuple[tuple[str, str | None], ...]:
   408→    assignment: list[tuple[str, str | None]] = []
   409→    for season in seasons:
   410→        crop_code = None
   411→        for h in history:
   412→            if h.get("season_label") == season.label:
   413→                crop_code = h.get("crop_code")
   414→                break
   415→        if crop_code is None:
   416→            for h in history:
   417→                sow = h.get("sowing_date")
   418→                harv = h.get("harvest_date")
   419→                if sow and harv:
   420→                    if date.fromisoformat(sow) <= season.end_date and date.fromisoformat(
   421→                        harv
   422→                    ) >= season.start_date:
   423→                        crop_code = h.get("crop_code")
   424→                        break
   425→        assignment.append((season.label, crop_code))
   426→    return tuple(assignment)

   429→def _solution_checksum(
   430→    solution: StrategySolution, problem: RotationProblem
   431→) -> str:
   432→    payload = {
   433→        "strategy": solution.strategy_key,
   434→        "assignment": [[lbl, cc] for lbl, cc in solution.assignment],
   435→        "objective": solution.objective_value,
   436→        "criteria": [
   437→            [c.criterion_code, c.raw_value, c.normalized] for c in solution.criteria
   438→        ],
   439→        "problem": problem.checksum(),
   440→        "engine": ROTATION_ENGINE_VERSION,
   441→    }
   442→    canonical = json.dumps(payload, sort_keys=True, separators=(",", ":"))
   443→    return hashlib.sha256(canonical.encode("utf-8")).hexdigest()

   446→def start_rotation_run(
   447→    db: Session,
   448→    *,
   449→    field: dbm.Field,
   450→    payload: dict[str, Any],
   451→    created_by: Any = None,
   452→) -> dbm.RotationRun:
   453→    """Validate, execute inline and persist a rotation run (202-style job)."""
   454→    idempotency_key = payload.get("idempotency_key")
   455→    if idempotency_key:
   456→        # Deepening F-3: replay lookup scoped to THIS field (cross-tenant
   457→        # replay of a known key must never serialize another tenant's run).
   458→        existing = db.scalar(
   459→            select(dbm.RotationRun).where(
   460→                dbm.RotationRun.idempotency_key == idempotency_key,
   461→                dbm.RotationRun.field_id == field.id,
   462→            )
   463→        )
   464→        if existing is not None:
   465→            return existing
   466→
   467→    seasons = _validate_seasons(payload.get("seasons") or [])
   468→    hemisphere = payload.get("hemisphere") or "north"
   469→    if hemisphere not in ("north", "south"):
   470→        raise ValidationFailed(
   471→            "hemisphere must be 'north' or 'south'.",
   472→            details=[{"field": "hemisphere", "reason": hemisphere}],
   473→        )
   474→
   475→    requested_profiles = payload.get("strategy_profiles") or list(optimizer_mod.DEFAULT_PROFILES)
   476→    if not isinstance(requested_profiles, list) or not requested_profiles:
   477→        raise ValidationFailed(
   478→            "strategy_profiles must be a non-empty list.",
   479→            details=[{"field": "strategy_profiles"}],
   480→        )
   481→    known = set(optimizer_mod.DEFAULT_PROFILES) | {"custom"}
   482→    for p in requested_profiles:
   483→        if p not in known:
   484→            raise ValidationFailed(
   485→                f"Unknown strategy profile '{p}'.",
   486→                details=[{"field": "strategy_profiles", "reason": p}],
   487→            )
   488→    custom_weights = payload.get("custom_weights") or {}
   489→    if "custom" in requested_profiles:
   490→        if not custom_weights:
   491→            raise ValidationFailed(
   492→                "The 'custom' profile needs custom_weights.",
   493→                details=[{"field": "custom_weights"}],
   494→            )
   495→        for code, w in custom_weights.items():
   496→            if code not in objectives_mod.CRITERION_FUNCS:
   497→                raise ValidationFailed(
   498→                    f"Unknown criterion '{code}' in custom_weights.",
   499→                    details=[{"field": "custom_weights", "reason": code}],
   500→                )
   501→            if not isinstance(w, (int, float)) or w < 0:
   502→                raise ValidationFailed(
   503→                    f"Custom weight for '{code}' must be a non-negative number.",
   504→                    details=[{"field": "custom_weights", "reason": code}],
   505→                )
   506→        if sum(float(w) for w in custom_weights.values()) <= 0:
   507→            raise ValidationFailed(
   508→                "custom_weights must sum to a positive value.",
   509→                details=[{"field": "custom_weights"}],
   510→            )
   511→
   512→    water_allocation = payload.get("water_allocation_m3")
   513→    if water_allocation is not None:
   514→        try:
   515→            water_allocation = float(water_allocation)
   516→        except (TypeError, ValueError):
   517→            raise ValidationFailed(
   518→                "water_allocation_m3 must be a number.",
   519→                details=[{"field": "water_allocation_m3"}],
   520→            ) from None
   521→        if water_allocation <= 0:
   522→            raise ValidationFailed(
   523→                "water_allocation_m3 must be positive.",
   524→                details=[{"field": "water_allocation_m3"}],
   525→            )
   526→
   527→    knowledge = load_knowledge(db)
   528→    if not knowledge.crops:
   529→        raise ValidationFailed(
   530→            "No active knowledge pack with crops — import a knowledge pack first.",
   531→            details=[{"field": "knowledge"}],
   532→        )
   533→
   534→    run = dbm.RotationRun(
   535→        field_id=field.id,
   536→        status="running",
   537→        seasons=[
   538→            {"label": s.label, "start_date": s.start_date.isoformat(), "end_date": s.end_date.isoformat()}
   539→            for s in seasons
   540→        ],
   541→        strategy_profiles=list(requested_profiles),
   542→        water_allocation_m3=water_allocation,
   543→        knowledge_pack_refs=[dict(r) for r in knowledge.pack_refs],
   544→        rotation_engine_version=ROTATION_ENGINE_VERSION,
   545→        solver_name=SOLVER_NAME,
   546→        random_seed=RANDOM_SEED,
   547→        request_parameters=dict(payload.get("request_parameters") or {}),
   548→        idempotency_key=idempotency_key,
   549→        created_by=getattr(created_by, "id", None),
   550→        started_at=datetime.now(UTC),
   551→    )
   552→    db.add(run)
   553→    db.flush()
   554→
   555→    try:
   556→        farm, warnings, history_ids, evidence_run = build_farm_context(
   557→            db, field, seasons, payload.get("evidence_run_id"), hemisphere
   558→        )
   559→        run.feature_run_ids = (
   560→            [str(evidence_run.id)] if evidence_run is not None else []
   561→        )
   562→        run.crop_history_ids = history_ids
   563→
   564→        assessments = candidates_mod.generate_candidates(knowledge, farm)
   565→        baseline_assignment = _baseline_assignment(seasons, farm.history)
   566→        problem = RotationProblem(
   567→            field_id=str(field.id),
   568→            seasons=tuple(seasons),
   569→            candidates=assessments,
   570→            knowledge=knowledge,
   571→            farm=farm,
   572→            water_allocation_m3=water_allocation,
   573→            baseline_assignment=baseline_assignment,
   574→            synthetic_evidence=farm.synthetic_evidence_in_use,
   575→        )
   576→        run.knowledge_pack_refs = [dict(r) for r in knowledge.pack_refs]
   577→
   578→        solutions: list[StrategySolution] = []
   579→        infeasible_profiles: list[dict[str, Any]] = []
   580→        solver_failures: list[dict[str, Any]] = []
   581→        baseline = optimizer_mod.build_baseline_solution(problem)
   582→        solutions.append(baseline)
   583→
   584→        for profile in requested_profiles:
   585→            weights = (
   586→                {k: float(v) for k, v in custom_weights.items()}
   587→                if profile == "custom"
   588→                else dict(optimizer_mod.PROFILE_WEIGHTS[profile])
   589→            )
   590→            result = optimizer_mod.optimize_strategy(problem, profile, weights)
   591→            if isinstance(result, optimizer_mod.InfeasibleResult):
   592→                # Deepening F-4: only a PROVEN infeasible model is an infeasible
   593→                # profile; solver failures (model_invalid / unknown) are kept
   594→                # separate so a solver breakdown never reads as an agronomic
   595→                # infeasibility statement.
   596→                entry = {
   597→                    "strategy_key": profile,
   598→                    "conflicts": result.conflicts,
   599→                    "notes": result.notes,
   600→                }
   601→                if result.status == "infeasible":
   602→                    infeasible_profiles.append(entry)
   603→                else:
   604→                    entry["solver_status"] = result.status
   605→                    solver_failures.append(entry)
   606→                continue
   607→            solutions.append(result)
   608→
   609→        # evaluate + normalize + objective + pareto
   610→        evaluated: list[StrategySolution] = []
   611→        for sol in solutions:
   612→            criteria = objectives_mod.evaluate_criteria(problem, dict(sol.assignment))
   613→            checks = constraints_mod.check_all(problem, dict(sol.assignment))
   614→            evaluated.append(
   615→                StrategySolution(
   616→                    strategy_key=sol.strategy_key,
   617→                    strategy_kind=sol.strategy_kind,
   618→                    is_baseline=sol.is_baseline,
   619→                    profile_weights=sol.profile_weights,
   620→                    weights_effective=sol.weights_effective,
   621→                    skipped_weights=sol.skipped_weights,
   622→                    assignment=sol.assignment,
   623→                    objective_value=sol.objective_value,
   624→                    criteria=criteria,
   625→                    constraints_checked=tuple(checks),
   626→                    evidence_coverage=None,
   627→                    solver_status=sol.solver_status,
   628→                    solver_wall_time_ms=sol.solver_wall_time_ms,
   629→                )
   630→            )
   631→        objectives_mod.normalize_within_set(evaluated)
   632→        final: list[StrategySolution] = []
   633→        for sol in evaluated:
   634→            objective, effective, skipped = objectives_mod.weighted_objective(sol)
   635→            final.append(
   636→                StrategySolution(
   637→                    strategy_key=sol.strategy_key,
   638→                    strategy_kind=sol.strategy_kind,
   639→                    is_baseline=sol.is_baseline,
   640→                    profile_weights=sol.profile_weights,
   641→                    weights_effective=effective,
   642→                    skipped_weights=skipped,
   643→                    assignment=sol.assignment,
   644→                    objective_value=objective,
   645→                    criteria=sol.criteria,
   646→                    constraints_checked=sol.constraints_checked,
   647→                    evidence_coverage=objectives_mod.evidence_coverage_of(sol),
   648→                    solver_status=sol.solver_status,
   649→                    solver_wall_time_ms=sol.solver_wall_time_ms,
   650→                )
   651→            )
   652→        objectives_mod.compute_pareto(final)
   653→        solutions = final
   654→
   655→        # persist strategies
   656→        not_evaluated: dict[str, Any] = {}
   657→        for season_label, season_assessments in assessments.items():
   658→            unevaluated = {
   659→                code: [d["code"] for d in a.unevaluated_dimensions]
   660→                for code, a in season_assessments.items()
   661→                if a.status == "unevaluated" or a.unevaluated_dimensions
   662→            }
   663→            if unevaluated:
   664→                not_evaluated[season_label] = unevaluated
   665→
   666→        run.not_evaluated = not_evaluated
   667→        run.warnings = warnings
   668→        solver_status = None
   669→        optimized_solutions = [s for s in solutions if not s.is_baseline]
   670→        statuses = [s.solver_status for s in optimized_solutions]
   671→        if infeasible_profiles and not optimized_solutions:
   672→            solver_status = "infeasible"
   673→        elif solver_failures and not optimized_solutions:
   674→            # mirror the solver failure upward (model_invalid / unknown):
   675→            # a solver breakdown is never reported as an agronomic infeasibility.
   676→            solver_status = solver_failures[0]["solver_status"]
   677→        elif infeasible_profiles or solver_failures:
   678→            # mixed outcome: some profiles solved (possibly only feasibly),
   679→            # some did not — the details block carries the specifics.
   680→            solver_status = "feasible"
   681→            warnings.append(
   682→                "Some strategy profiles did not reach a proven optimum "
   683→                f"({len(infeasible_profiles)} infeasible, {len(solver_failures)} solver "
   684→                "failure(s)); see infeasibility details."
   685→            )
   686→        elif optimized_solutions and all(st == "optimal" for st in statuses):
   687→            # 'optimal' is claimed only when EVERY optimized profile proved
   688→            # optimality (a time-limited FEASIBLE result is not optimality).
   689→            solver_status = "optimal"
   690→        elif optimized_solutions:
   691→            solver_status = "feasible"
   692→        else:
   693→            solver_status = "unknown"
   694→        run.solver_status = solver_status
   695→        run.infeasibility_details = {
   696→            "profiles": infeasible_profiles,
   697→            "solver_failures": solver_failures,
   698→            "note": (
   699→                "Infeasible profiles are reported explicitly; no hard constraint was "
   700→                "relaxed to produce the other strategies."
   701→            )
   702→            if infeasible_profiles
   703→            else {},
   704→        }
   705→
   706→        for sol in solutions:
   707→            explanations = build_explanations(problem, sol, solutions)
   708→            explanations["_weights_effective"] = sol.weights_effective
   709→            explanations["_skipped_weights"] = sol.skipped_weights
   710→            db.add(
   711→                dbm.RotationStrategy(
   712→                    run_id=run.id,
   713→                    strategy_key=sol.strategy_key,
   714→                    strategy_kind=sol.strategy_kind,
   715→                    is_baseline=sol.is_baseline,
   716→                    profile_weights=sol.profile_weights,
   717→                    assignment=[
   718→                        {"season_label": lbl, "crop_code": cc} for lbl, cc in sol.assignment
   719→                    ],
   720→                    objective_value=sol.objective_value,
   721→                    criteria=[
   722→                        {
   723→                            "criterion_code": c.criterion_code,
   724→                            "raw_value": c.raw_value,
   725→                            "unit": c.unit,
   726→                            "direction": c.direction,
   727→                            "normalized": c.normalized,
   728→                            "evaluable": c.evaluable,
   729→                            "missing_evidence": list(c.missing_evidence),
   730→                            "derivation": c.derivation,
   731→                            "source_refs": list(c.source_refs),
   732→                        }
   733→                        for c in sol.criteria
   734→                    ],
   735→                    constraints_checked=[
   736→                        {
   737→                            "constraint_code": c.constraint_code,
   738→                            "status": c.status,
   739→                            "description": c.description,
   740→                            "detail": c.detail,
   741→                            "rule_codes": list(c.rule_codes),
   742→                            "source_refs": list(c.source_refs),
   743→                        }
   744→                        for c in sol.constraints_checked
   745→                    ],
   746→                    pareto={
   747→                        "on_front": sol.on_pareto_front,
   748→                        "dominated_by": list(sol.dominated_by),
   749→                        "dominates": list(sol.dominates),
   750→                    },
   751→                    explanations=explanations,
   752→                    evidence_coverage=sol.evidence_coverage,
   753→                    checksum_sha256=_solution_checksum(sol, problem),
   754→                )
   755→            )
   756→        run.status = "succeeded"
   757→        run.completed_at = datetime.now(UTC)
   758→        db.flush()
   759→    except ValidationFailed:
   760→        run.status = "failed"
   761→        run.error_summary = "validation failed"
   762→        run.completed_at = datetime.now(UTC)
   763→        raise
   764→    except Exception as exc:  # noqa: BLE001 — job failure accounting
   765→        run.status = "failed"
   766→        run.error_summary = f"{type(exc).__name__}: {exc}"
   767→        run.completed_at = datetime.now(UTC)
   768→        db.flush()
   769→        raise
   770→    return run

   781→    data: dict[str, Any] = {

   785→        "solver_status": run.solver_status,
   786→        "seasons": run.seasons,
   787→        "strategy_profiles": run.strategy_profiles,
   788→        "water_allocation_m3": run.water_allocation_m3,
   789→        "knowledge_pack_refs": run.knowledge_pack_refs,
   790→        "feature_run_ids": run.feature_run_ids,
   791→        "crop_history_ids": run.crop_history_ids,
   792→        "rotation_engine_version": run.rotation_engine_version,
   793→        "solver_name": run.solver_name,
   794→        "solver_version": run.solver_version,
   795→        "random_seed": run.random_seed,
   796→        "infeasibility_details": run.infeasibility_details,
   797→        "warnings": run.warnings,
   798→        "not_evaluated": run.not_evaluated,
   799→        "records_written": len(run.strategies) if run.strategies else 0,
   800→        "error_summary": run.error_summary,
   801→        "started_at": run.started_at.isoformat() if run.started_at else None,
   802→        "completed_at": run.completed_at.isoformat() if run.completed_at else None,
