    62→function apiErrorText(err: unknown): string {
    63→  if (err instanceof ApiError) {
    64→    const detail = err.details[0]
    65→    const detailText = detail?.reason ? `${detail.field}: ${detail.reason}` : detail?.field
    66→    return detailText ? `${err.message} (${detailText})` : err.message
    67→  }
    68→  return err instanceof Error ? err.message : 'Unexpected error'
    69→}
    70→
    71→function fmt(value: number | null | undefined, digits = 4): string {
    72→  if (value === null || value === undefined) return '—'
    73→  if (Number.isInteger(value)) return String(value)
    74→  return value.toFixed(digits)
    75→}
    76→
    77→function shortHash(hash: string | null | undefined): string {
    78→  return hash ? `${hash.slice(0, 12)}…` : '—'
    79→}

   624→function DefinitionForm({
   625→  run,
   626→  referenceValues,
   627→  explorableDetail,
   628→  onCreated,
   629→}: {
   630→  run: ScenarioExecutionRun
   631→  referenceValues: Record<string, number>
   632→  explorableDetail: Record<string, { explorable: boolean; reference_value: number | null; reason: string | null }>
   633→  onCreated: (analysis: SensitivityAnalysis) => void
   634→}) {
   635→  const queryClient = useQueryClient()
   636→  const bothExplorable =
   637→    explorableDetail.temperature_delta_c?.explorable && explorableDetail.precipitation_multiplier?.explorable
   638→
   639→  const defaults: Record<string, BoundState> = {
   640→    temperature_delta_c: { lower: '-2', upper: '6', points: '9' },
   641→    precipitation_multiplier: { lower: '0.5', upper: '1.5', points: '9' },
   642→  }
   643→  const [bounds, setBounds] = useState<Record<string, BoundState>>({
   644→    temperature_delta_c: defaults.temperature_delta_c,
   645→    precipitation_multiplier: defaults.precipitation_multiplier,
   646→  })
   647→  const [includeGrid, setIncludeGrid] = useState(false)
   648→  const [formError, setFormError] = useState<string | null>(null)
   649→
   650→  const explorableCodes = Object.entries(explorableDetail)
   651→    .filter(([, v]) => v.explorable)
   652→    .map(([code]) => code)
   653→
   654→  const create = useMutation({
   655→    mutationFn: () => {
   656→      const parameter_design: Record<string, { lower: number; upper: number; n_points: number }> = {}
   657→      for (const code of explorableCodes) {
   658→        const b = bounds[code]
   659→        parameter_design[code] = {
   660→          lower: Number(b.lower),
   661→          upper: Number(b.upper),
   662→          n_points: Number(b.points),
   663→        }
   664→      }
   665→      return createSensitivityAnalysis(run.run_id, {
   666→        parameter_design,
   667→        grid: includeGrid && bothExplorable ? true : undefined,
   668→      })
   669→    },
   670→    onSuccess: (analysis) => {
   671→      queryClient.invalidateQueries({ queryKey: ['sensitivity-analyses', run.run_id] })
   672→      onCreated(analysis)
   673→    },
   674→  })
   675→
   676→  const refDelta = referenceValues.temperature_delta_c
   677→  const refMult = referenceValues.precipitation_multiplier
   678→
   679→  return (
   680→    <Card>
   681→      <CardHeader className="pb-3">
   682→        <CardTitle className="text-base">Define the exploration</CardTitle>
   683→        <CardDescription>
   684→          Bounds are user-supplied assumptions for a deterministic experiment — they
   685→          contain the source execution&apos;s reference values and imply no probability
   686→          distribution.
   687→        </CardDescription>
   688→      </CardHeader>
   689→      <CardContent className="space-y-4">
   690→        <div className="grid gap-4 md:grid-cols-2">
   691→          {Object.entries(explorableDetail).map(([code, detail]) => {
   692→            const ref = code === 'temperature_delta_c' ? refDelta : refMult
   693→            if (!detail.explorable) {
   694→              return (
   695→                <div key={code} className="rounded-md border border-stone-200 bg-stone-50 p-3">
   696→                  <p className="text-xs font-medium text-stone-700">{code.replace(/_/g, ' ')}</p>

   699→                  </p>
   700→                </div>
   701→              )
   702→            }
   703→            const b = bounds[code]
   704→            const set = (patch: Partial<BoundState>) =>

   706→            return (

   708→                <div className="flex items-center justify-between">
   709→                  <p className="text-xs font-medium text-stone-700">{code.replace(/_/g, ' ')}</p>

   721→                      className="h-8 text-xs"
   722→                      inputMode="decimal"
   723→                      value={b.lower}
   724→                      onChange={(e) => set({ lower: e.target.value })}

   775→            setFormError(null)
   776→            create.mutate()
   777→          }}
   778→          disabled={create.isPending}

   781→            <Loader2 className="mr-1.5 h-3.5 w-3.5 animate-spin" aria-hidden />

   785→          Run analysis

   788→    </Card>
