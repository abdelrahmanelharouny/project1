     1→     1→"""Feature-run orchestration (Milestone 4).
     2→     2→
     3→     3→A feature run is a reproducible computation: the run row records the resolved
     4→     4→source selection (ONE dataset product per variable — synthetic and real data
     5→     5→are never blended), the calendar basis, the window, the engine and catalog
     6→     6→versions, and every warning. Values are written once by the run; comparison
     7→     7→across runs is the supported way to observe change.
     8→     8→
     9→     9→Honesty rules enforced here:
    10→    10→- Source selection never mixes providers for the same variable inside a run.
    11→    11→- Runs over synthetic data are flagged `is_synthetic` end to end.
    12→    12→- Missing source variables drop the dependent features with an explicit
    13→    13→  reason — they are never silently skipped nor computed from substitutes.
    14→    14→- Crop-season windows are user-supplied; no crop calendar knowledge is
    15→    15→  invented here (the sourced knowledge base arrives in a later milestone).
    16→    16→"""
    17→    17→
    18→    18→import logging
    19→    19→import uuid
    20→    20→from datetime import UTC, date, datetime, timedelta
    21→    21→from typing import Any
    22→    22→
    23→    23→from sqlalchemy import Boolean, Integer, func, select
    24→    24→from sqlalchemy.exc import IntegrityError
    25→    25→from sqlalchemy.orm import Session
    26→    26→
    27→    27→from app.core.errors import InsufficientData, NotFound, ValidationFailed
    28→    28→from app.data_platform.contracts.variables import VARIABLE_CATALOG
    29→    29→from app.db.models import (
    30→    30→    DatasetProduct,
    31→    31→    DataSource,

    34→    34→    FeatureValue,

    36→    36→    Observation,
    37→    37→)
    38→    38→from app.features import engine as fengine
    39→    39→from app.features.calendar import resolve_timezone
    40→    40→from app.features.catalog import (
    41→    41→    FEATURE_CATALOG,

    45→    45→
    46→    46→logger = logging.getLogger("fieldshift.features")
    47→    47→
    48→    48→ALLOWED_AGGREGATION_LEVELS = ("daily", "monthly", "seasonal", "crop_season")
    49→    49→MAX_WINDOW_YEARS = 31
    50→    50→MAX_DAILY_WINDOW_DAYS = 1100  # ~3 years, protects the daily-value volume
    51→    51→MAX_CROP_SEASONS = 20

    56→    56→    from app.features.catalog import catalog_payload
    57→    57→
    58→    58→    existing = {
    59→    59→        (row.feature_code, row.definition_version)
    60→    60→        for row in db.scalars(
    61→    61→            select(FeatureDefinition).where(
    62→    62→                FeatureDefinition.definition_version == FEATURE_CATALOG_VERSION

    65→    65→    }
    66→    66→    created = False
    67→    67→    for item in catalog_payload():
    68→    68→        key = (item["feature_code"], item["definition_version"])
    69→    69→        if key in existing:
    70→    70→            continue
    71→    71→        db.add(
    72→    72→            FeatureDefinition(
    73→    73→                feature_code=item["feature_code"],
    74→    74→                name=item["name"],
    75→    75→                description=item["description"],
    76→    76→                unit=item["unit"],
    77→    77→                category=item["category"],
    78→    78→                aggregation_levels=item["aggregation_levels"],
    79→    79→                required_variables=item["required_variables"],
    80→    80→                min_coverage_ratio=item["min_coverage_ratio"],
    81→    81→                definition_version=item["definition_version"],
    82→    82→                formula_reference=item["formula_reference"],

    84→    84→            )

    91→    91→def resolve_source_selection(
    92→    92→    db: Session,
    93→    93→    *,
    94→    94→    field_id: uuid.UUID,
    95→    95→    variables: list[str],
    96→    96→    overrides: dict[str, str] | None = None,
    97→    97→    window_start: datetime | None = None,
    98→    98→    window_end: datetime | None = None,
    99→    99→) -> tuple[dict[str, dict[str, Any]], list[str], bool]:
   100→   100→    """Pick ONE dataset product per variable for the run.
   101→   101→
   102→   102→    Preference (window-aware): real (non-synthetic) products first, then the
   103→   103→    product with the most valid observations INSIDE the run window, then the
   104→   104→    best overall valid ratio, then the most recent retrieval. A real product
   105→   105→    with no data in the window is therefore not preferred over a synthetic
   106→   106→    fixture that actually covers the window — the fallback is warned about and
   107→   107→    the run is flagged `is_synthetic`. Overrides pin a specific product and
   108→   108→    are validated against the field's actual observations.
   109→   109→    """
   110→   110→    overrides = overrides or {}
   111→   111→    stmt = (
   112→   112→        select(
   113→   113→            Observation.variable_code,
   114→   114→            Observation.dataset_product_id,
   115→   115→            func.count().label("n"),
   116→   116→            func.sum(func.cast(Observation.value.is_not(None), Integer)).label("valid_n"),
   117→   117→            func.max(Observation.retrieved_at).label("last_retrieval"),
   118→   118→            func.cast(func.bool_or(Observation.is_synthetic), Boolean).label("has_synthetic"),
   119→   119→        )
   120→   120→        .where(Observation.field_id == field_id, Observation.variable_code.in_(variables))
   121→   121→        .group_by(Observation.variable_code, Observation.dataset_product_id)
   122→   122→    )
   123→   123→    if window_start is not None and window_end is not None:
   124→   124→        # in-window counts drive the preference; keep a margin of one day on
   125→   125→        # both sides for daily periods that straddle the window boundary
   126→   126→        margin = timedelta(days=1)
   127→   127→        stmt = stmt.where(
   128→   128→            Observation.observed_at >= window_start - margin,
   129→   129→            Observation.observed_at <= window_end + margin,
   130→   130→        )
   131→   131→    rows = db.execute(stmt).all()
   132→   132→
   133→   133→    if not rows:
   134→   134→        raise InsufficientData(
   135→   135→            "No observations are available for the requested variables on this field.",
   136→   136→            details=[{"field": "variables", "reason": ", ".join(sorted(variables))}],
   137→   137→        )
   138→   138→
   139→   139→    products = {
   140→   140→        p.id: p
   141→   141→        for p in db.scalars(
   142→   142→            select(DatasetProduct).where(
   143→   143→                DatasetProduct.id.in_([r.dataset_product_id for r in rows])
   144→   144→            )
   145→   145→        ).all()
   146→   146→    }
   147→   147→
   148→   148→    grouped: dict[str, list[dict[str, Any]]] = {}
   149→   149→    for r in rows:
   150→   150→        product = products.get(r.dataset_product_id)
   151→   151→        if product is None:
   152→   152→            continue
   153→   153→        meta = product.metadata or {}
   154→   154→        grouped.setdefault(r.variable_code, []).append(
   155→   155→            {
   156→   156→                "dataset_product_id": str(product.id),
   157→   157→                "product_code": product.product_code,
   158→   158→                "product_version": product.product_version,
   159→   159→                "is_synthetic": bool(meta.get("is_synthetic")) or bool(r.has_synthetic),
   160→   160→                "n": int(r.n),
   161→   161→                "valid_n": int(r.valid_n or 0),
   162→   162→                "last_retrieval": r.last_retrieval,
   163→   163→            }
   164→   164→        )
   165→   165→
   166→   166→    selection: dict[str, dict[str, Any]] = {}
   167→   167→    warnings: list[str] = []
   168→   168→    for variable in variables:
   169→   169→        candidates = grouped.get(variable) or []
   170→   170→        if not candidates:
   171→   171→            continue
   172→   172→        override_id = overrides.get(variable)
   173→   173→        if override_id:
   174→   174→            chosen = next(
   175→   175→                (c for c in candidates if c["dataset_product_id"] == str(override_id)), None
   176→   176→            )
   177→   177→            if chosen is None:
   178→   178→                raise ValidationFailed(
   179→   179→                    f"Source override for '{variable}' does not match any product with "
   180→   180→                    "observations on this field.",
   181→   181→                    details=[{"field": "source_overrides", "reason": variable}],
   182→   182→                )
   183→   183→        else:
   184→   184→            chosen = sorted(
   185→   185→                candidates,
   186→   186→                key=lambda c: (
   187→   187→                    c["is_synthetic"],
   188→   188→                    -(c["valid_n"]),
   189→   189→                    -(c["valid_n"] / max(1, c["n"])),
   190→   190→                    -(c["last_retrieval"].timestamp() if c["last_retrieval"] else 0),
   191→   191→                ),
   192→   192→            )[0]
   193→   193→        selection[variable] = {
   194→   194→            "dataset_product_id": chosen["dataset_product_id"],
   195→   195→            "product_code": chosen["product_code"],
   196→   196→            "product_version": chosen["product_version"],
   197→   197→            "is_synthetic": chosen["is_synthetic"],
   198→   198→        }
   199→   199→        skipped = [c for c in candidates if c["dataset_product_id"] != chosen["dataset_product_id"]]
   200→   200→        for c in skipped:
   201→   201→            if chosen["is_synthetic"] and not c["is_synthetic"]:
   202→   202→                warnings.append(
   203→   203→                    f"{variable}: real product '{c['product_code']}' ({c['product_version']}) has no "
   204→   204→                    "observations inside the run window; the synthetic fixture was selected instead "
   205→   205→                    "(the run is flagged synthetic). Products are never blended."
   206→   206→                )
   207→   207→            else:
   208→   208→                warnings.append(
   209→   209→                    f"{variable}: observations from product '{c['product_code']}' "
   210→   210→                    f"({c['product_version']}) were not selected for this run; one source per "
   211→   211→                    "variable is used so products are never blended."
   212→   212→                )
   213→   213→
   214→   214→    is_synthetic = any(s["is_synthetic"] for s in selection.values())
   215→   215→    return selection, warnings, is_synthetic

   218→   218→def start_feature_run(
   219→   219→    db: Session,
   220→   220→    *,
   221→   221→    field: Field,
   222→   222→    payload: dict[str, Any],
   223→   223→    created_by: Any = None,
   224→   224→) -> FeatureRun:
   225→   225→    """Validate the request, create (or return) the run, execute it inline."""
   226→   226→    ensure_feature_definitions(db)
   227→   227→
   228→   228→    idempotency_key = payload.get("idempotency_key")
   229→   229→    if idempotency_key:
   230→   230→        existing = db.scalar(select(FeatureRun).where(FeatureRun.idempotency_key == idempotency_key))
   231→   231→        if existing is not None:
   232→   232→            return existing
   233→   233→
   234→   234→    start_date: date = payload["start_date"]
   235→   235→    end_date: date = payload["end_date"]
   236→   236→    if start_date > end_date:
   237→   237→        raise ValidationFailed(
   238→   238→            "start_date must be on or before end_date.",
   239→   239→            details=[{"field": "start_date", "reason": str(start_date)}],
   240→   240→        )
   241→   241→    if (end_date - start_date).days > MAX_WINDOW_YEARS * 366:
   242→   242→        raise ValidationFailed(
   243→   243→            f"Feature run window exceeds {MAX_WINDOW_YEARS} years.",
   244→   244→            details=[{"field": "end_date", "reason": "window too long"}],
   245→   245→        )
   246→   246→
   247→   247→    levels = payload.get("aggregation_levels") or ["monthly"]
   248→   248→    unknown_levels = [l for l in levels if l not in ALLOWED_AGGREGATION_LEVELS]
   249→   249→    if unknown_levels:
   250→   250→        raise ValidationFailed(
   251→   251→            "Unknown aggregation levels.",
   252→   252→            details=[{"field": "aggregation_levels", "reason": ", ".join(unknown_levels)}],
   253→   253→        )
   254→   254→    if "daily" in levels and (end_date - start_date).days > MAX_DAILY_WINDOW_DAYS:
   255→   255→        raise ValidationFailed(
   256→   256→            f"Daily aggregation is limited to {MAX_DAILY_WINDOW_DAYS} days per run.",
   257→   257→            details=[{"field": "end_date", "reason": "window too long for daily level"}],
   258→   258→        )
   259→   259→
   260→   260→    hemisphere = payload.get("hemisphere") or "north"
   261→   261→    if hemisphere not in ("north", "south"):
   262→   262→        raise ValidationFailed(
   263→   263→            "hemisphere must be 'north' or 'south'.",
   264→   264→            details=[{"field": "hemisphere", "reason": hemisphere}],
   265→   265→        )
   266→   266→
   267→   267→    calendar_timezone = payload.get("calendar_timezone") or field.farm.timezone or "UTC"
   268→   268→    try:
   269→   269→        tz = resolve_timezone(calendar_timezone)
   270→   270→    except ValueError as exc:
   271→   271→        raise ValidationFailed(str(exc), details=[{"field": "calendar_timezone"}]) from None
   272→   272→
   273→   273→    crop_seasons_raw = payload.get("crop_seasons") or []
   274→   274→    if crop_seasons_raw and "crop_season" not in levels:
   275→   275→        raise ValidationFailed(
   276→   276→            "crop_seasons supplied but the 'crop_season' aggregation level is not requested.",
   277→   277→            details=[{"field": "aggregation_levels", "reason": "add 'crop_season'"}],
   278→   278→        )
   279→   279→    if len(crop_seasons_raw) > MAX_CROP_SEASONS:
   280→   280→        raise ValidationFailed(
   281→   281→            f"Too many crop-season windows (max {MAX_CROP_SEASONS}).",
   282→   282→            details=[{"field": "crop_seasons"}],
   283→   283→        )
   284→   284→    crop_seasons: list[dict[str, Any]] = []
   285→   285→    seen_labels: set[str] = set()
   286→   286→    for cs in crop_seasons_raw:
   287→   287→        label = str(cs.get("label") or "").strip()
   288→   288→        if not label or len(label) > 40:
   289→   289→            raise ValidationFailed(
   290→   290→                "Crop-season labels must be 1-40 characters.",
   291→   291→                details=[{"field": "crop_seasons", "reason": str(cs)[:80]}],
   292→   292→            )
   293→   293→        if label in seen_labels:
   294→   294→            raise ValidationFailed(
   295→   295→                f"Duplicate crop-season label '{label}'.",
   296→   296→                details=[{"field": "crop_seasons", "reason": label}],
   297→   297→            )
   298→   298→        seen_labels.add(label)
   299→   299→        try:
   300→   300→            cs_start = date.fromisoformat(str(cs["start_date"]))
   301→   301→            cs_end = date.fromisoformat(str(cs["end_date"]))
   302→   302→        except (KeyError, ValueError):
   303→   303→            raise ValidationFailed(
   304→   304→                "Crop seasons need start_date and end_date (ISO dates).",
   305→   305→                details=[{"field": "crop_seasons", "reason": label}],
   306→   306→            ) from None
   307→   307→        if cs_start > cs_end:
   308→   308→            raise ValidationFailed(
   309→   309→                f"Crop season '{label}' start_date is after end_date.",
   310→   310→                details=[{"field": "crop_seasons", "reason": label}],
   311→   311→            )
   312→   312→        crop_seasons.append({"label": label, "start_date": cs_start, "end_date": cs_end})
   313→   313→
   314→   314→    # Feature specs to compute.
   315→   315→    requested_codes = payload.get("feature_codes")
   316→   316→    if requested_codes:
   317→   317→        unknown = [c for c in requested_codes if c not in FEATURE_CATALOG]
   318→   318→        if unknown:
   319→   319→            raise ValidationFailed(
   320→   320→                "Unknown feature codes.",
   321→   321→                details=[{"field": "feature_codes", "reason": ", ".join(sorted(unknown))}],
   322→   322→            )
   323→   323→        specs = [FEATURE_CATALOG[c] for c in sorted(set(requested_codes))]
   324→   324→    else:
   325→   325→        specs = [
   326→   326→            s
   327→   327→            for s in FEATURE_CATALOG.values()
   328→   328→            if any(level in s.aggregation_levels for level in levels)
   329→   329→        ]
   330→   330→    if not specs:
   331→   331→        raise ValidationFailed(
   332→   332→            "No features match the requested aggregation levels.",
   333→   333→            details=[{"field": "feature_codes"}],
   334→   334→        )
   335→   335→
   336→   336→    variables = payload.get("variables")
   337→   337→    required_vars = sorted({v for spec in specs for v in spec.required_variables})
   338→   338→    if variables:
   339→   339→        unknown_vars = [v for v in variables if v not in VARIABLE_CATALOG]
   340→   340→        if unknown_vars:
   341→   341→            raise ValidationFailed(
   342→   342→                "Unknown variable codes in request.",
   343→   343→                details=[{"field": "variables", "reason": ", ".join(unknown_vars)}],
   344→   344→            )
   345→   345→        # Restrict features to what the requested variables can supply.
   346→   346→        allowed = set(variables)
   347→   347→        specs = [s for s in specs if set(s.required_variables) <= allowed]
   348→   348→        if not specs:
   349→   349→            raise ValidationFailed(
   350→   350→                "None of the catalog features can be computed from the requested variables.",
   351→   351→                details=[{"field": "variables", "reason": ", ".join(sorted(variables))}],
   352→   352→            )
   353→   353→        required_vars = sorted({v for spec in specs for v in spec.required_variables})
   354→   354→
   355→   355→    window_start = (
   356→   356→        datetime(start_date.year, start_date.month, start_date.day, tzinfo=tz).astimezone(UTC)
   357→   357→    )
   358→   358→    window_end = (
   359→   359→        datetime(end_date.year, end_date.month, end_date.day, tzinfo=tz) + timedelta(days=1)
   360→   360→    ).astimezone(UTC) - timedelta(seconds=1)
   361→   361→
   362→   362→    overrides = payload.get("source_overrides") or {}
   363→   363→    selection, selection_warnings, is_synthetic = resolve_source_selection(
   364→   364→        db,
   365→   365→        field_id=field.id,
   366→   366→        variables=required_vars,
   367→   367→        overrides=overrides,
   368→   368→        window_start=window_start,
   369→   369→        window_end=window_end,
   370→   370→    )
   371→   371→
   372→   372→    missing_vars = [v for v in required_vars if v not in selection]
   373→   373→    resolvable_specs = [s for s in specs if set(s.required_variables) <= set(selection)]
   374→   374→    not_evaluated: dict[str, str] = {
   375→   375→        f"variable:{v}": "Required input variable has no observations on this field."
   376→   376→        for v in missing_vars
   377→   377→    }
   378→   378→    dropped_codes = sorted({s.feature_code for s in specs} - {s.feature_code for s in resolvable_specs})
   379→   379→
   380→   380→    request_parameters = {
   381→   381→        "start_date": start_date.isoformat(),
   382→   382→        "end_date": end_date.isoformat(),
   383→   383→        "aggregation_levels": levels,
   384→   384→        "hemisphere": hemisphere,
   385→   385→        "calendar_timezone": calendar_timezone,
   386→   386→        "crop_seasons": [
   387→   387→            {"label": cs["label"], "start_date": cs["start_date"].isoformat(), "end_date": cs["end_date"].isoformat()}
   388→   388→            for cs in crop_seasons
   389→   389→        ],
   390→   390→        "feature_codes": sorted(s.feature_code for s in specs),
   391→   391→        "source_overrides": {k: str(v) for k, v in overrides.items()},
   392→   392→    }
   393→   393→
   394→   394→    run = FeatureRun(
   395→   395→        field_id=field.id,
   396→   396→        status="queued",
   397→   397→        calendar_timezone=calendar_timezone,
   398→   398→        hemisphere=hemisphere,
   399→   399→        period_start=window_start,
   400→   400→        period_end=window_end,
   401→   401→        aggregation_levels=levels,
   402→   402→        requested_variables=required_vars,
   403→   403→        source_selection=selection,
   404→   404→        crop_seasons=request_parameters["crop_seasons"],
   405→   405→        feature_engine_version=FEATURE_ENGINE_VERSION,
   406→   406→        feature_catalog_version=FEATURE_CATALOG_VERSION,
   407→   407→        is_synthetic=is_synthetic,
   408→   408→        request_parameters=request_parameters,
   409→   409→        warnings=list(selection_warnings),
   410→   410→        not_evaluated_reasons=not_evaluated,
   411→   411→        idempotency_key=idempotency_key,
   412→   412→        created_by=created_by,
   413→   413→    )
   414→   414→    db.add(run)
   415→   415→    try:
   416→   416→        db.commit()
   417→   417→    except IntegrityError:
   418→   418→        db.rollback()
   419→   419→        if idempotency_key:
   420→   420→            existing = db.scalar(select(FeatureRun).where(FeatureRun.idempotency_key == idempotency_key))
   421→   421→            if existing is not None:
   422→   422→                return existing
   423→   423→        raise
   424→   424→
   425→   425→    if not resolvable_specs:
   426→   426→        run.status = "failed"
   427→   427→        run.error_summary = "None of the requested features have their input variables available."
   428→   428→        run.completed_at = datetime.now(UTC)
   429→   429→        db.commit()
   430→   430→        return run
   431→   431→
   432→   432→    _execute_run(
   433→   433→        db,
   434→   434→        run,
   435→   435→        specs=resolvable_specs,
   436→   436→        tz=tz,
   437→   437→        window_start=window_start,
   438→   438→        window_end=window_end,
   439→   439→        crop_seasons=crop_seasons,
   440→   440→        not_evaluated=not_evaluated,
   441→   441→        dropped_codes=dropped_codes,
   442→   442→    )
   443→   443→    return run
   444→   444→
   445→   445→
   446→   446→def _execute_run(
   447→   447→    db: Session,
   448→   448→    run: FeatureRun,
   449→   449→    *,
   450→   450→    specs: list,
   451→   451→    tz,
   452→   452→    window_start: datetime,
   453→   453→    window_end: datetime,
   454→   454→    crop_seasons: list[dict[str, Any]],
   455→   455→    not_evaluated: dict[str, str],
   456→   456→    dropped_codes: list[str],
   457→   457→) -> None:
   458→   458→    run.status = "running"
   459→   459→    run.started_at = datetime.now(UTC)
   460→   460→    db.commit()
   461→   461→
   462→   462→    try:
   463→   463→        selected_ids = [s["dataset_product_id"] for s in run.source_selection.values()]
   464→   464→        margin = timedelta(days=2)
   465→   465→        obs_rows = db.execute(
   466→   466→            select(Observation)
   467→   467→            .where(
   468→   468→                Observation.field_id == run.field_id,
   469→   469→                Observation.dataset_product_id.in_(selected_ids),
   470→   470→                Observation.observed_at >= run.period_start - margin,
   471→   471→                Observation.observed_at <= run.period_end + margin,
   472→   472→            )
   473→   473→            .order_by(Observation.period_start, Observation.variable_code)
   474→   474→        ).scalars().all()
   475→   475→
   476→   476→        observations = [
   477→   477→            {
   478→   478→                "id": str(o.id),
   479→   479→                "variable_code": o.variable_code,
   480→   480→                "value": o.value,
   481→   481→                "unit": o.unit,
   482→   482→                "quality_status": o.quality_status,
   483→   483→                "observed_at": o.observed_at,
   484→   484→                "period_start": o.period_start,
   485→   485→                "period_end": o.period_end,
   486→   486→                "time_support": o.time_support,
   487→   487→                "is_synthetic": o.is_synthetic,
   488→   488→                "processing_version": o.processing_version,
   489→   489→                "evidence_class": o.evidence_class,
   490→   490→                "dataset_product_id": str(o.dataset_product_id),
   491→   491→            }
   492→   492→            for o in obs_rows
   493→   493→        ]
   494→   494→
   495→   495→        crop_windows = [
   496→   496→            {
   497→   497→                "label": cs["label"],
   498→   498→                "start": datetime(
   499→   499→                    cs["start_date"].year, cs["start_date"].month, cs["start_date"].day, tzinfo=tz
   500→   500→                ).astimezone(UTC),
   501→   501→                "end": (
   502→   502→                    datetime(
   503→   503→                        cs["end_date"].year, cs["end_date"].month, cs["end_date"].day, tzinfo=tz
   504→   504→                    )
   505→   505→                    + timedelta(days=1)
   506→   506→                ).astimezone(UTC)
   507→   507→                - timedelta(seconds=1),
   508→   508→            }
   509→   509→            for cs in crop_seasons
   510→   510→        ]
   511→   511→
   512→   512→        drafts, engine_not_evaluated, engine_warnings = fengine.compute_feature_drafts(
   513→   513→            observations=observations,
   514→   514→            specs=specs,
   515→   515→            tz=tz,
   516→   516→            hemisphere=run.hemisphere,
   517→   517→            window_start=window_start,
   518→   518→            window_end=window_end,
   519→   519→            aggregation_levels=list(run.aggregation_levels),
   520→   520→            crop_seasons=crop_windows,
   521→   521→            engine_version=run.feature_engine_version,
   522→   522→            definition_version=run.feature_catalog_version,
   523→   523→        )
   524→   524→
   525→   525→        all_warnings = list(run.warnings) + list(engine_warnings)
   526→   526→        if dropped_codes:
   527→   527→            all_warnings.append(
   528→   528→                "Features without their required inputs were not computed: "
   529→   529→                + ", ".join(dropped_codes)
   530→   530→                + "."
   531→   531→            )
   532→   532→        if run.is_synthetic:
   533→   533→            all_warnings.insert(
   534→   534→                0,
   535→   535→                "This feature run is computed from SYNTHETIC fixture data — values are for "
   536→   536→                "pipeline demonstration and testing only, never for real-world conclusions.",
   537→   537→            )
   538→   538→
   539→   539→        written = 0
   540→   540→        for draft in drafts:
   541→   541→            row = draft.to_row()
   542→   542→            db.add(
   543→   543→                FeatureValue(
   544→   544→                    run_id=run.id,
   545→   545→                    field_id=run.field_id,
   546→   546→                    feature_code=row["feature_code"],
   547→   547→                    aggregation_level=row["aggregation_level"],
   548→   548→                    period_label=row["period_label"],
   549→   549→                    period_start=row["period_start"],
   550→   550→                    period_end=row["period_end"],
   551→   551→                    value=row["value"],
   552→   552→                    unit=row["unit"],
   553→   553→                    coverage_ratio=row["coverage_ratio"],
   554→   554→                    n_input=row["n_input"],
   555→   555→                    n_missing=row["n_missing"],
   556→   556→                    quality_status=row["quality_status"],
   557→   557→                    uncertainty_value=row["uncertainty_value"],
   558→   558→                    uncertainty_type=row["uncertainty_type"],
   559→   559→                    baseline_reference=row["baseline_reference"],
   560→   560→                    provenance=row["provenance"],
   561→   561→                )
   562→   562→            )
   563→   563→            written += 1
   564→   564→            if written % 500 == 0:
   565→   565→                db.flush()
   566→   566→
   567→   567→        merged_reasons = dict(not_evaluated)
   568→   568→        for code, reason in engine_not_evaluated.items():
   569→   569→            merged_reasons.setdefault(code, reason)
   570→   570→
   571→   571→        run.status = "succeeded"
   572→   572→        run.records_written = written
   573→   573→        run.warnings = all_warnings[:100]
   574→   574→        run.not_evaluated_reasons = merged_reasons
   575→   575→        run.completed_at = datetime.now(UTC)
   576→   576→        db.commit()
   577→   577→    except Exception as exc:  # noqa: BLE001 — run lifecycle must capture failures
   578→   578→        db.rollback()
   579→   579→        run.status = "failed"
   580→   580→        run.error_summary = f"{type(exc).__name__}: {exc}"
   581→   581→        run.completed_at = datetime.now(UTC)
   582→   582→        db.commit()
   583→   583→        logger.exception("feature run failed", extra={"run_id": str(run.id)})

   596→   596→def run_payload(run: FeatureRun) -> dict[str, Any]:
   597→   597→    # Display the period in the run's calendar basis, not UTC: the farmer chose
   598→   598→    # local dates, so "2015-01-01" must not drift back to "2014-12-31".
   599→   599→    try:
   600→   600→        run_tz = resolve_timezone(run.calendar_timezone)
   601→   601→        period_start_date = run.period_start.astimezone(run_tz).date().isoformat()

   605→   605→        period_end_date = run.period_end.date().isoformat()

   654→   654→def compare_runs(
   655→   655→    db: Session, run: FeatureRun, against: FeatureRun
   656→   656→) -> dict[str, Any]:
   657→   657→    """Compare two runs on matching (feature_code, aggregation_level, period_label)."""
   658→   658→    left_rows = db.scalars(select(FeatureValue).where(FeatureValue.run_id == run.id)).all()
   659→   659→    right_rows = db.scalars(select(FeatureValue).where(FeatureValue.run_id == against.id)).all()
   660→   660→    right_by_key = {
   661→   661→        (v.feature_code, v.aggregation_level, v.period_label): v for v in right_rows
   662→   662→    }
   663→   663→    deltas: list[dict[str, Any]] = []
   664→   664→    matched = 0
   665→   665→    only_in_run = 0
   666→   666→    for left in left_rows:
   667→   667→        key = (left.feature_code, left.aggregation_level, left.period_label)
   668→   668→        right = right_by_key.get(key)
   669→   669→        if right is None:
   670→   670→            only_in_run += 1
   671→   671→            continue
   672→   672→        if left.value is None or right.value is None:
   673→   673→            continue
   674→   674→        matched += 1
   675→   675→        delta = round(left.value - right.value, 4)
   676→   676→        delta_pct = (
   677→   677→            round((left.value - right.value) / abs(right.value), 4) if right.value != 0 else None
   678→   678→        )
   679→   679→        deltas.append(
   680→   680→            {
   681→   681→                "feature_code": left.feature_code,
   682→   682→                "aggregation_level": left.aggregation_level,
   683→   683→                "period_label": left.period_label,
   684→   684→                "unit": left.unit,
   685→   685→                "run_value": left.value,
   686→   686→                "against_value": right.value,
   687→   687→                "delta": delta,
   688→   688→                "delta_pct": delta_pct,
   689→   689→                "quality_run": left.quality_status,
   690→   690→                "quality_against": right.quality_status,
   691→   691→            }
   692→   692→        )
   693→   693→    return {
   694→   694→        "run_id": str(run.id),
   695→   695→        "against_run_id": str(against.id),
   696→   696→        "matched_periods": matched,
   697→   697→        "only_in_run": only_in_run,
   698→   698→        "deltas": deltas,
   699→   699→        "note": (
   700→   700→            "Deltas compare two independent runs on matching periods. Runs may use different "
   701→   701→            "source selections, calendar bases or engine versions — check run metadata before "
   702→   702→            "interpreting deltas."
   703→   703→        ),
   704→   704→    }
