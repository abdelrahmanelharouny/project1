   103→   103→/** Human-readable summary of any thrown error for toasts. */
   104→   104→export function describeError(err: unknown): { title: string; description?: string } {
   105→   105→  if (err instanceof ApiError) {
   106→   106→    const firstDetail = err.details[0]
   107→   107→    const description = firstDetail
   108→   108→      ? `${firstDetail.field}: ${firstDetail.reason}${err.requestId ? ` · request ${err.requestId.slice(0, 8)}` : ''}`
   109→   109→      : err.requestId
   110→   110→        ? `request ${err.requestId.slice(0, 8)}`
   111→   111→        : undefined
   112→   112→    return { title: err.message, description }
   113→   113→  }
   114→   114→  if (err instanceof Error) {
   115→   115→    return { title: 'Network error', description: err.message }
   116→   116→  }
   117→   117→  return { title: 'Unexpected error' }
   118→   118→}

   120→   120→function withPort(path: string): string {
   121→   121→  const joiner = path.includes('?') ? '&' : '?'
   122→   122→  return `${path}${joiner}${PORT_QUERY}`
   123→   123→}

   125→   125→/**
   126→   126→ * Exchange the HttpOnly refresh cookie for a fresh access+refresh cookie pair
   127→   127→ * (rotating refresh: the presented token is revoked server-side; reusing a
   128→   128→ * superseded token revokes the whole session family).
   129→   129→ *
   130→   130→ * Single-flight: several concurrent 401s share ONE rotation call — parallel
   131→   131→ * rotations would present already-rotated tokens and revoke the family.
   132→   132→ */
   133→   133→let refreshInFlight: Promise<boolean> | null = null
   134→   134→
   135→   135→export async function refreshSession(): Promise<boolean> {
   136→   136→  if (!refreshInFlight) {
   137→   137→    refreshInFlight = authRequest('/refresh')
   138→   138→      .then(() => true)
   139→   139→      .catch(() => false)
   140→   140→      .finally(() => {
   141→   141→        refreshInFlight = null
   142→   142→      })
   143→   143→  }
   144→   144→  return refreshInFlight
   145→   145→}

   147→   147→/** POST to a Next.js /api/auth/* route handler (session cookies live here). */
   148→   148→async function authRequest<T>(path: string, body?: unknown): Promise<T> {
   149→   149→  const response = await fetch(`${AUTH_BASE}${path}`, {
   150→   150→    method: 'POST',
   151→   151→    headers: body !== undefined ? { 'Content-Type': 'application/json' } : undefined,
   152→   152→    body: body !== undefined ? JSON.stringify(body) : undefined,
   153→   153→    cache: 'no-store',
   154→   154→  })
   155→   155→  const text = await response.text()
   156→   156→  let payload: unknown = null
   157→   157→  if (text) {
   158→   158→    try {
   159→   159→      payload = JSON.parse(text)
   160→   160→    } catch {
   161→   161→      payload = null
   162→   162→    }
   163→   163→  }
   164→   164→  if (!response.ok) {
   165→   165→    const envelope = payload as ApiErrorBody | null
   166→   166→    const error =
   167→   167→      envelope && typeof envelope === 'object' && 'error' in envelope ? envelope.error : undefined
   168→   168→    throw new ApiError({
   169→   169→      status: response.status,
   170→   170→      code: error?.code ?? `HTTP_${response.status}`,
   171→   171→      message: error?.message ?? `Request failed with status ${response.status}.`,
   172→   172→      details: error?.details ?? [],
   173→   173→      requestId: error?.request_id,
   174→   174→    })
   175→   175→  }
   176→   176→  return payload as T
   177→   177→}

   179→   179→interface RequestOptions {
   180→   180→  method?: 'GET' | 'POST' | 'PATCH' | 'PUT' | 'DELETE'
   181→   181→  body?: unknown
   182→   182→  /** Multipart form body (file uploads). Content-Type is set by the browser. */
   183→   183→  formData?: FormData
   184→   184→  /** Skip the 401→refresh retry (health checks, anonymous endpoints). */
   185→   185→  anonymous?: boolean
   186→   186→}
   187→   187→
   188→   188→interface RequestInternals {
   189→   189→  /** Prevent recursive 401→refresh retries (used by refreshSession itself). */
   190→   190→  skipAuthRetry?: boolean
   191→   191→}
   192→   192→
   193→   193→async function request<T>(
   194→   194→  path: string,
   195→   195→  options: RequestOptions = {},
   196→   196→  internals: RequestInternals = {},
   197→   197→): Promise<T> {
   198→   198→  const headers: Record<string, string> = { Accept: 'application/json' }
   199→   199→  if (options.body !== undefined) headers['Content-Type'] = 'application/json'
   200→   200→
   201→   201→  let response: Response
   202→   202→  try {
   203→   203→    response = await fetch(withPort(path), {
   204→   204→      method: options.method ?? 'GET',
   205→   205→      headers,
   206→   206→      body:
   207→   207→        options.formData !== undefined
   208→   208→          ? options.formData
   209→   209→          : options.body !== undefined
   210→   210→            ? JSON.stringify(options.body)
   211→   211→            : undefined,
   212→   212→      cache: 'no-store',
   213→   213→    })
   214→   214→  } catch (err) {
   215→   215→    throw new ApiError({
   216→   216→      status: 0,
   217→   217→      code: 'NETWORK_ERROR',
   218→   218→      message: 'Could not reach the FieldShift API.',
   219→   219→      details: [],
   220→   220→    })
   221→   221→  }
   222→   222→
   223→   223→  const text = await response.text()
   224→   224→  let payload: unknown = null
   225→   225→  if (text) {
   226→   226→    try {
   227→   227→      payload = JSON.parse(text)
   228→   228→    } catch {
   229→   229→      payload = null
   230→   230→    }
   231→   231→  }
   232→   232→
   233→   233→  if (!response.ok) {
   234→   234→    const envelope = payload as ApiErrorBody | null
   235→   235→    const error = envelope && typeof envelope === 'object' && 'error' in envelope ? envelope.error : undefined
   236→   236→    // Rotating-refresh retry (P0-2/P1): an access token may have simply
   237→   237→    // expired while the session is still valid. Try ONE cookie refresh, then
   238→   238→    // retry the original request once. Never recurse from the refresh call
   239→   239→    // itself (it goes through authRequest, not here).
   240→   240→    if (response.status === 401 && !options.anonymous && !internals.skipAuthRetry) {
   241→   241→      const refreshed = await refreshSession()
   242→   242→      if (refreshed) {
   243→   243→        return request<T>(path, options, { skipAuthRetry: true })
   244→   244→      }
   245→   245→    }
   246→   246→    throw new ApiError({
   247→   247→      status: response.status,
   248→   248→      code: error?.code ?? `HTTP_${response.status}`,
   249→   249→      message: error?.message ?? `Request failed with status ${response.status}.`,
   250→   250→      details: error?.details ?? [],
   251→   251→      requestId: error?.request_id,
   252→   252→    })
   253→   253→  }
   254→   254→
   255→   255→  return payload as T
   256→   256→}

   405→   405→const JOB_TERMINAL_STATUSES = new Set(['succeeded', 'failed', 'cancelled'])
   406→   406→
   407→   407→/**
   408→   408→ * Poll an ingestion/import job until it reaches a terminal state.
   409→   409→ *
   410→   410→ * The backend executes data jobs on a background worker pool (202 + status
   411→   411→ * 'queued'); when workers are disabled it still executes inline (the response
   412→   412→ * is already terminal, returned as-is). Progress callbacks receive every
   413→   413→ * intermediate snapshot (status/progress for UI feedback).
   414→   414→ */
   415→   415→export async function pollJobUntilDone(
   416→   416→  jobId: string,
   417→   417→  opts: { onProgress?: (job: DataJob) => void; intervalMs?: number; timeoutMs?: number } = {},
   418→   418→): Promise<DataJob> {
   419→   419→  const interval = opts.intervalMs ?? 1000
   420→   420→  const deadline = Date.now() + (opts.timeoutMs ?? 180_000)
   421→   421→  let current = await getJob(jobId)
   422→   422→  while (!JOB_TERMINAL_STATUSES.has(current.status)) {
   423→   423→    opts.onProgress?.(current)
   424→   424→    if (Date.now() > deadline) {
   425→   425→      throw new Error('Timed out waiting for the job to finish — check its status later.')
   426→   426→    }
   427→   427→    await new Promise((resolve) => setTimeout(resolve, interval))
   428→   428→    current = await getJob(jobId)
   429→   429→  }
   430→   430→  return current
   431→   431→}

   462→   462→export type GeometryParseResult =
   463→   463→  | { ok: true; geometry: import('./types').GeoJSONGeometry }
   464→   464→  | { ok: false; error: string }
   465→   465→
   466→   466→/** Parse + shallow-validate a GeoJSON Polygon / MultiPolygon from user input. */
   467→   467→export function parseGeometryInput(text: string): GeometryParseResult {
   468→   468→  let parsed: unknown
   469→   469→  try {
   470→   470→    parsed = JSON.parse(text)
   471→   471→  } catch {
   472→   472→    return { ok: false, error: 'The text is not valid JSON. Check for missing quotes, brackets or trailing commas.' }
   473→   473→  }
   474→   474→  if (typeof parsed !== 'object' || parsed === null || Array.isArray(parsed)) {
   475→   475→    return { ok: false, error: 'Expected a GeoJSON object like {"type":"Polygon","coordinates":[...]}' }
   476→   476→  }
   477→   477→  const obj = parsed as Record<string, unknown>
   478→   478→  if (obj.type === 'Polygon') {
   479→   479→    if (!Array.isArray(obj.coordinates) || obj.coordinates.length === 0) {
   480→   480→      return { ok: false, error: 'Polygon coordinates must be a non-empty array of rings.' }
   481→   481→    }
   482→   482→    return { ok: true, geometry: parsed as import('./types').GeoJSONPolygon }
   483→   483→  }
   484→   484→  if (obj.type === 'MultiPolygon') {
   485→   485→    if (!Array.isArray(obj.coordinates) || obj.coordinates.length === 0) {
   486→   486→      return { ok: false, error: 'MultiPolygon coordinates must be a non-empty array of polygons.' }
   487→   487→    }
   488→   488→    return { ok: true, geometry: parsed as import('./types').GeoJSONMultiPolygon }
   489→   489→  }

   510→   510→/** Client-side guard mirroring backend job validation. Returns an error message or null. */
   511→   511→export function validateJobWindow(
   512→   512→  start: string,
   513→   513→  end: string,
   514→   514→  bounds?: { min?: string; max?: string; maxYears?: number },
   515→   515→): string | null {
   516→   516→  const dateRe = /^\d{4}-\d{2}-\d{2}$/
   517→   517→  if (!dateRe.test(start) || !dateRe.test(end)) return 'Enter both dates as YYYY-MM-DD.'
   518→   518→  const min = bounds?.min ?? COVERAGE_START
   519→   519→  const max = bounds?.max ?? COVERAGE_END
   520→   520→  if (start < min || end > max) {
   521→   521→    return `Dates must fall inside the dataset coverage ${min} → ${max}.`
   522→   522→  }
   523→   523→  if (end < start) return 'The end date must be on or after the start date.'
   524→   524→  const years = (Date.parse(end) - Date.parse(start)) / (365.25 * 24 * 3600 * 1000)
   525→   525→  const maxYears = bounds?.maxYears ?? MAX_JOB_YEARS
   526→   526→  if (years > maxYears) return `A single job covers at most ${maxYears} years. Split the range.`
   527→   527→  return null
   528→   528→}

   966→   966→// there, so the component's (fieldId, file) signature is preserved.
   967→   967→
   968→   968→export async function importSoilLabCsv(fieldId: string, file: File): Promise<SoilLabImportResponse> {
   969→   969→  const formData = new FormData()
   970→   970→  formData.append('file', file)
   971→   971→  formData.append('mapping', JSON.stringify({ default_field_id: fieldId }))
   972→   972→  const data = await request<SoilLabImportResponse>('/api/v1/soil-samples/import', {
   973→   973→    method: 'POST',
   974→   974→    formData,
   975→   975→  })
   976→   976→  // The backend's idempotent "already_imported" reply omits the counters;
   977→   977→  // normalize so callers always see numbers (re-import of identical content
   978→   978→  // creates nothing — 0 is the truthful value there).
   979→   979→  return {
   980→   980→    ...data,
   981→   981→    created_samples: data.created_samples ?? 0,
   982→   982→    created_measurements: data.created_measurements ?? 0,
   983→   983→    errors: data.errors ?? [],
   984→   984→  }
   985→   985→}
