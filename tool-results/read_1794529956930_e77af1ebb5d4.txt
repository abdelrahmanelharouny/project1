   154→function CreateFarmDialog({ open, onOpenChange }: { open: boolean; onOpenChange: (open: boolean) => void }) {
   155→  const queryClient = useQueryClient()
   156→  const t = useTranslations('farms')
   157→  const tCommon = useTranslations('common')
   158→  const [name, setName] = useState('')
   159→  const [country, setCountry] = useState('')
   160→  const [adminRegion, setAdminRegion] = useState('')
   161→  const [timezone, setTimezone] = useState('UTC')
   162→  const [currency, setCurrency] = useState('')
   163→  const [unitSystem, setUnitSystem] = useState<UnitSystem>('metric')
   164→  const [language, setLanguage] = useState('en')
   165→  const [formError, setFormError] = useState<string | null>(null)
   166→
   167→  // Reset form state when the dialog closes (handler-based, no effect needed).
   168→  const handleOpenChange = (next: boolean) => {
   169→    if (!next) {
   170→      setName('')
   171→      setCountry('')
   172→      setAdminRegion('')
   173→      setTimezone('UTC')
   174→      setCurrency('')
   175→      setUnitSystem('metric')
   176→      setLanguage('en')
   177→      setFormError(null)
   178→    }
   179→    onOpenChange(next)
   180→  }
   181→
   182→  const mutation = useMutation({
   183→    mutationFn: (input: FarmCreateInput) => createFarm(input),
   184→    onSuccess: (farm) => {
   185→      queryClient.invalidateQueries({ queryKey: ['farms'] })
   186→      toast.success(t('farmCreated', { name: farm.name }), {
   187→        description: `${farm.timezone} · ${farm.currency_code} · ${farm.unit_system}`,

   196→  })

   199→    event.preventDefault()

   201→    const currencyClean = currency.trim().toUpperCase()
   202→    const languageClean = language.trim().toLowerCase()

   208→    mutation.mutate({
   209→      name: name.trim(),
   210→      country_code: countryClean,
   211→      admin_region: adminRegion.trim() || undefined,
   212→      timezone,
   213→      currency_code: currencyClean,
   214→      unit_system: unitSystem,
   215→      preferred_language: languageClean,

   217→  }
   218→

   221→      <DialogContent className="max-h-[85vh] overflow-y-auto scrollbar-thin sm:max-w-lg">
   222→        <DialogHeader>
   223→          <DialogTitle>{t('createTitle')}</DialogTitle>
   224→          <DialogDescription>
   225→            {t('createDesc')}

   227→        </DialogHeader>
   228→        <form onSubmit={submit} className="space-y-4" aria-label={t('createFormAria')}>
   229→          <div className="space-y-1.5">
   230→            <Label htmlFor="farm-name">{t('nameLabel')}</Label>
   231→            <Input
   232→              id="farm-name"
   233→              placeholder={t('namePlaceholder')}

   235→              onChange={(event) => setName(event.target.value)}

   238→          <div className="grid gap-4 sm:grid-cols-2">
   239→            <div className="space-y-1.5">
   240→              <Label htmlFor="farm-country">{t('countryLabel')}</Label>
   241→              <Input
   242→                id="farm-country"
   243→                maxLength={2}
   244→                placeholder="EG"
   245→                value={country}
   246→                onChange={(event) => setCountry(event.target.value.toUpperCase())}
   247→                className="uppercase"
   248→              />
   249→              <p className="text-xs text-stone-500">{t('countryHint')}</p>

   251→            <div className="space-y-1.5">

   339→function AddFieldDialog({
   340→  farm,
   341→  open,
   342→  onOpenChange,
   343→  onFieldSelected,
   344→}: {
   345→  farm: Farm
   346→  open: boolean
   347→  onOpenChange: (open: boolean) => void
   348→  onFieldSelected: (field: { id: string; name: string; farmId: string }) => void
   349→}) {
   350→  const queryClient = useQueryClient()
   351→  const t = useTranslations('farms')
   352→  const tCommon = useTranslations('common')
   353→  const [name, setName] = useState('')
   354→  const [geometry, setGeometry] = useState('')
   355→  const [parseError, setParseError] = useState<string | null>(null)
   356→  const [apiError, setApiError] = useState<{ title: string; description?: string } | null>(null)
   357→  const [created, setCreated] = useState<FieldCreateResponse | null>(null)
   358→
   359→  // Reset form state when the dialog closes.
   360→  const handleOpenChange = (next: boolean) => {
   361→    if (!next) {
   362→      setName('')
   363→      setGeometry('')
   364→      setParseError(null)
   365→      setApiError(null)
   366→      setCreated(null)
   367→    }
   368→    onOpenChange(next)
   369→  }
   370→
   371→  const mutation = useMutation({
   372→    mutationFn: (payload: { name: string; geometry: GeoJSONGeometry }) =>
   373→      createField(farm.id, { name: payload.name, geometry: payload.geometry }),
   374→    onSuccess: (result) => {
   375→      queryClient.invalidateQueries({ queryKey: ['fields', farm.id] })
   376→      queryClient.invalidateQueries({ queryKey: ['farms'] })
   377→      if (result.geometry_warnings.length > 0) {
   378→        setCreated(result)
   379→        toast.warning(t('fieldCreatedWarnings'), {
   380→          description: t('warningsFromPostgis', { count: result.geometry_warnings.length }),
   381→        })
   382→      } else {
   383→        toast.success(t('fieldCreated', { name: result.name }), {
   384→          description: t('fieldCreatedDesc', {
   385→            area: formatArea(result.area_ha),
   386→            unit: tCommon('ha'),
   387→            version: result.boundary_version,
   388→          }),
   389→        })
   390→        onFieldSelected({ id: result.id, name: result.name, farmId: farm.id })
   391→        handleOpenChange(false)
   392→      }
   393→    },
   394→    onError: (err) => {
   395→      const { title, description } = describeError(err)
   396→      setApiError({ title, description })
   397→      toast.error(title, { description })
   398→    },
   399→  })
   400→
   401→  const submit = (event: FormEvent) => {
   402→    event.preventDefault()
   403→    setApiError(null)
   404→    setCreated(null)
   405→    if (!name.trim()) {
   406→      setParseError(t('fieldNameRequired'))
   407→      return
   408→    }
   409→    const parsed = parseGeometryInput(geometry)
   410→    if (!parsed.ok) {
   411→      setParseError(parsed.error)
   412→      return
   413→    }
   414→    setParseError(null)
   415→    mutation.mutate({ name: name.trim(), geometry: parsed.geometry })
   416→  }
   417→
   418→  return (
   419→    <Dialog open={open} onOpenChange={handleOpenChange}>
   420→      <DialogContent className="max-h-[85vh] overflow-y-auto scrollbar-thin sm:max-w-xl">
   421→        <DialogHeader>
   422→          <DialogTitle>{t('addFieldTitle', { farm: farm.name })}</DialogTitle>
   423→          <DialogDescription>
   424→            {t('addFieldDesc')}
   425→          </DialogDescription>
   426→        </DialogHeader>
   427→        <form onSubmit={submit} className="space-y-4" aria-label={t('addFieldFormAria')}>
   428→          <div className="space-y-1.5">
   429→            <Label htmlFor="field-name">{t('fieldLabel')}</Label>
   430→            <Input
   431→              id="field-name"
   432→              placeholder={t('fieldPlaceholder')}
   433→              value={name}
   434→              onChange={(event) => setName(event.target.value)}
   435→            />
   436→          </div>
   437→          <GeometryInput id="field-geometry" value={geometry} onChange={setGeometry} error={parseError} />
   438→
   439→          {created ? (
   440→            <div className="space-y-2">
   441→              <Alert className="border-emerald-300 bg-emerald-50 [&>svg]:text-emerald-700">
   442→                <MapPin className="h-4 w-4" />
   443→                <AlertTitle className="text-emerald-900">
   444→                  {t('fieldCreatedArea', {
   445→                    name: created.name,
   446→                    area: formatArea(created.area_ha),
   447→                    unit: tCommon('ha'),
   448→                  })}
   449→                </AlertTitle>
   450→                <AlertDescription className="text-emerald-900">
   451→                  {t('boundaryPassport', {
   452→                    version: created.boundary_version,
   453→                    status: created.data_passport_status,
   454→                  })}
   455→                </AlertDescription>
   456→              </Alert>
   457→              <GeometryWarningsAlert
   458→                warnings={created.geometry_warnings}
   459→                repaired={created.geometry_repaired}
   460→                title={t('postgisWarnings')}
   461→              />
   462→              <div className="flex justify-end gap-2">
   463→                <Button
   464→                  type="button"
   465→                  variant="outline"
   466→                  onClick={() => {
   467→                    onFieldSelected({ id: created.id, name: created.name, farmId: farm.id })
   468→                    handleOpenChange(false)
   469→                  }}
   470→                >
   471→                  {t('openNewField')}
   472→                </Button>
   473→              </div>
   474→            </div>
   475→          ) : null}
   476→
   477→          {apiError ? (
   478→            <Alert variant="destructive">
   479→              <TriangleAlert className="h-4 w-4" />
   480→              <AlertTitle>{apiError.title}</AlertTitle>
   481→              <AlertDescription>{apiError.description ?? t('adjustGeometryRetry')}</AlertDescription>
   482→            </Alert>
   483→          ) : null}
   484→
   485→          <DialogFooter>
   486→            <Button type="button" variant="outline" onClick={() => handleOpenChange(false)}>
   487→              {created ? tCommon('close') : tCommon('cancel')}
   488→            </Button>
   489→            <Button type="submit" className="bg-emerald-700 hover:bg-emerald-800" disabled={mutation.isPending}>
   490→              {mutation.isPending ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : <Plus className="h-4 w-4" aria-hidden />}
   491→              {t('createFieldSubmit')}
   492→            </Button>
   493→          </DialogFooter>
   494→        </form>
   495→      </DialogContent>
   496→    </Dialog>
   497→  )
   498→}

   504→function RenameFieldDialog({
   505→  field,
   506→  open,
   507→  onOpenChange,
   508→}: {
   509→  field: Field
   510→  open: boolean
   511→  onOpenChange: (open: boolean) => void
   512→}) {
   513→  const queryClient = useQueryClient()
   514→  const t = useTranslations('farms')
   515→  const tCommon = useTranslations('common')
   516→  const renameSelectedField = useCropTwistUi((s) => s.renameSelectedField)
   517→  const [name, setName] = useState(field.name)
   518→
   519→  // Sync the input with the current field name each time the dialog opens.
   520→  const handleOpenChange = (next: boolean) => {
   521→    if (next) setName(field.name)
   522→    onOpenChange(next)
   523→  }
   524→
   525→  const mutation = useMutation({
   526→    mutationFn: () => patchField(field.id, { name: name.trim() }),
   527→    onSuccess: (updated) => {
   528→      queryClient.invalidateQueries({ queryKey: ['fields'] })
   529→      renameSelectedField(updated.id, updated.name)
   530→      toast.success(t('fieldRenamed', { name: updated.name }))
   531→      onOpenChange(false)
   532→    },
   533→    onError: (err) => {
   534→      const { title, description } = describeError(err)
   535→      toast.error(title, { description })
   536→    },
   537→  })
   538→
   539→  const submit = (event: FormEvent) => {
   540→    event.preventDefault()
   541→    if (!name.trim() || name.trim() === field.name) {
   542→      onOpenChange(false)
   543→      return
   544→    }
   545→    mutation.mutate()
   546→  }
   547→
   548→  return (
   549→    <Dialog open={open} onOpenChange={handleOpenChange}>
   550→      <DialogContent className="sm:max-w-sm">
   551→        <DialogHeader>
   552→          <DialogTitle>{t('renameTitle')}</DialogTitle>
   553→          <DialogDescription>{t('renameDesc')}</DialogDescription>
   554→        </DialogHeader>
   555→        <form onSubmit={submit} className="space-y-4" aria-label={t('renameFormAria')}>
   556→          <div className="space-y-1.5">
   557→            <Label htmlFor="rename-field">{t('fieldLabel')}</Label>
   558→            <Input id="rename-field" value={name} onChange={(event) => setName(event.target.value)} />
   559→          </div>
   560→          <DialogFooter>
   561→            <Button type="button" variant="outline" onClick={() => handleOpenChange(false)}>
   562→              {tCommon('cancel')}
   563→            </Button>
   564→            <Button type="submit" className="bg-emerald-700 hover:bg-emerald-800" disabled={mutation.isPending}>
   565→              {mutation.isPending ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : null}
   566→              {t('saveName')}
   567→            </Button>
   568→          </DialogFooter>
   569→        </form>
   570→      </DialogContent>
   571→    </Dialog>
   572→  )
   573→}

   577→  open,
   578→  onOpenChange,
   579→}: {

   591→
   592→  // Reset form state when the dialog closes.
   593→  const handleOpenChange = (next: boolean) => {

   595→      setGeometry('')
   596→      setReason('')
   597→      setParseError(null)
   598→      setResult(null)
   599→    }

   601→  }

   609→      queryClient.invalidateQueries({ queryKey: ['boundary-versions', field.id] })

   639→      return
   640→    }

   643→  }

   682→                warnings={result.geometry_warnings}
   683→                repaired={result.geometry_repaired}
