    44→
    45→def _trait(trait_code, *, value=None, text=None, low=None, high=None, unit=None,
    46→           source=_FAO56, evidence=EVIDENCE_PUBLISHED, confidence="moderate",
    47→           conditions=None, notes=None):
    48→    row = {"trait_code": trait_code}
    49→    if value is not None:
    50→        row["value_numeric"] = value
    51→    if text is not None:
    52→        row["value_text"] = text
    53→    if low is not None:
    54→        row["value_low"] = low
    55→    if high is not None:
    56→        row["value_high"] = high
    57→    if unit is not None:
    58→        row["unit"] = unit
    59→    row["source_key"] = source
    60→    row["evidence_class"] = evidence
    61→    row["confidence"] = confidence
    62→    if conditions:
    63→        row["conditions"] = conditions
    64→    if notes:
    65→        row["notes"] = notes
    66→    return row
    67→
    68→
    69→def _window(planting_start, planting_end, harvest_start=None, harvest_end=None, *,
    70→            hemisphere="north", region=None, season_code=None, notes=None):
    71→    row = {
    72→        "planting_start": planting_start,
    73→        "planting_end": planting_end,
    74→        "hemisphere": hemisphere,
    75→        "source_key": _FAO56,  # station-context reference; synthesis declared below
    76→        "evidence_class": EVIDENCE_SYNTHESIZED,
    77→        "validation_status": "reference",
    78→    }
    79→    if harvest_start:
    80→        row["harvest_start"] = harvest_start
    81→    if harvest_end:
    82→        row["harvest_end"] = harvest_end
    83→    if region:
    84→        row["region_code"] = region
    85→    if season_code:
    86→        row["season_code"] = season_code
    87→    row["notes"] = notes or (
    88→        "Coarse reference window consistent with FAO-56 Table 11 station contexts; "
    89→        "requires regional calibration before operational use."
    90→    )
    91→    return row
    92→
    93→
    94→def _p(value):
    95→    """FAO-56 Table 22 soil water depletion fraction p (for ETc ≈ 5 mm/day).
    96→    The ETc adjustment p = p_table + 0.04·(5 − ETc) is applied by the water
    97→    engine when a season ETc mean exists (Table 22 footnote 2, verified)."""
    98→    return _trait(
    99→        "stress_depletion_fraction_p", value=value, unit="fraction",
   100→        notes="FAO-56 Table 22 depletion fraction p for ETc ≈ 5 mm/day; "
   101→              "adjusted by the water engine with p = p + 0.04·(5 − ETc).",
   102→    )

   105→def _kc(stage, value, stage_days, *, notes=None):
   106→    return {
   107→        "growth_stage": stage,
   108→        "kc_value": value,
   109→        "kc_method": "single",
   110→        "stage_length_days": stage_days,
   111→        "source_key": _FAO56,
   112→        "evidence_class": EVIDENCE_PUBLISHED,
   113→        "confidence": "moderate",
   114→        "notes": notes or "FAO-56 single Kc (Table 11/12 typical values).",
   115→    }

   118→def _eco(temp_min_abs, opt_min, opt_max, precip_min, precip_max, ph_min, ph_max):
   119→    return [
   120→        _trait("temp_min_abs_degC", value=temp_min_abs, unit="degC", source=_ECOCROP,
   121→               confidence="moderate"),
   122→        _trait("temp_opt_min_degC", value=opt_min, unit="degC", source=_ECOCROP,
   123→               confidence="moderate"),
   124→        _trait("temp_opt_max_degC", value=opt_max, unit="degC", source=_ECOCROP,
   125→               confidence="moderate"),
   126→        _trait("precip_min_annual_mm", value=precip_min, unit="mm", source=_ECOCROP,
   127→               confidence="moderate"),
   128→        _trait("precip_max_annual_mm", value=precip_max, unit="mm", source=_ECOCROP,
   129→               confidence="moderate"),
   130→        _trait("ph_opt_min", value=ph_min, unit="pH", source=_ECOCROP, confidence="low"),
   131→        _trait("ph_opt_max", value=ph_max, unit="pH", source=_ECOCROP, confidence="low"),
   132→    ]

   134→
   135→def _stages(init, dev, mid, late):
   136→    return [
   137→        _trait("stage_init_days", value=init, unit="d"),
   138→        _trait("stage_dev_days", value=dev, unit="d"),
   139→        _trait("stage_mid_days", value=mid, unit="d"),
   140→        _trait("stage_late_days", value=late, unit="d"),
   141→        _trait("duration_days_total", value=init + dev + mid + late, unit="d"),
   142→    ]
   143→
   144→
   145→def _kcs(ini, mid, end, notes=None):
   146→    """Stage Kc anchor rows; the FAO-56 single-Kc curve (linear ramps across
   147→    development/late stages) is constructed by the water estimator from these
   148→    anchors + the stage_* trait lengths. Stage lengths live on crop traits."""
   149→    base_notes = notes or "FAO-56 single Kc (Table 11/12 typical values)."
   150→    return [
   151→        _kc("initial", ini, None, notes=base_notes),
   152→        _kc("mid", mid, None, notes=base_notes),
   153→        _kc("late", end, None, notes=base_notes),
   154→    ]
