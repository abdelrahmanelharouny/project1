   124→def _has_arabic(text: str) -> bool:
   125→    return any(
   126→        lo <= ord(ch) <= hi
   127→        for ch in text
   128→        for lo, hi in _ARABIC_RANGES
   129→    )

   132→# Compact Arabic→Latin transliteration (applied AFTER NFKC, which folds the
   133→# presentation forms produced by the shaper back to base letters).

   135→    "ا": "a", "ب": "b", "ت": "t", "ث": "th", "ج": "j", "ح": "h", "خ": "kh",

   138→    "ك": "k", "ل": "l", "م": "m", "ن": "n", "ه": "h", "و": "w", "ي": "y",
   139→    "آ": "aa", "أ": "a", "إ": "i", "ٱ": "a", " ئ": "y", "ؤ": "w", "ى": "a",
   140→    "ة": "h", "ء": "'",

   149→def _transliterate_arabic(text: str) -> str:
   150→    """Last-resort Latin fallback when no Arabic-capable font is embedded."""
   151→    folded = unicodedata.normalize("NFKC", text)
   152→    out: list[str] = []
   153→    for ch in folded:
   154→        out.append(_AR_LATIN.get(ch, ch if ord(ch) < 0x0600 else ""))
   155→    return "".join(out)

   158→def _find_arabic_font() -> tuple[str, str | None] | None:
   159→    """Locate a Unicode TTF with real Arabic coverage on this host.
   160→
   161→    Returns (regular_path, bold_path|None) or None. Coverage is VERIFIED
   162→    (base Arabic + presentation forms B) — a font that would render tofu is
   163→    treated as absent so the caller can degrade honestly instead.
   164→    """
   165→    candidates = (
   166→        "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
   167→        "/usr/share/fonts/truetype/noto/NotoNaskhArabic-Regular.ttf",
   168→        "/usr/share/fonts/noto/NotoNaskhArabic-Regular.ttf",
   169→        "/usr/share/fonts/truetype/freefont/FreeSans.ttf",
   170→        "/usr/share/fonts/TTF/DejaVuSans.ttf",
   171→        "/usr/share/fonts/dejavu/DejaVuSans.ttf",
   172→    )
   173→    bold_candidates = (
   174→        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf",
   175→        "/usr/share/fonts/truetype/noto/NotoNaskhArabic-Bold.ttf",
   176→        "/usr/share/fonts/truetype/freefont/FreeSansBold.ttf",
   177→    )
   178→    probe = (0x0627, 0x0644, 0x0639, 0xFEDF, 0xFEE0, 0xFE8D, 0xFEF4)
   179→    for regular in candidates:
   180→        try:
   181→            with open(regular, "rb") as fh:
   182→                if not fh.read(2):
   183→                    continue
   184→        except OSError:
   185→            continue
   186→        if not _font_covers(regular, probe):
   187→            continue
   188→        bold: str | None = None
   189→        for cand in bold_candidates:
   190→            try:
   191→                with open(cand, "rb") as fh:
   192→                    fh.read(2)
   193→                if _font_covers(cand, probe):
   194→                    bold = cand
   195→                    break
   196→            except OSError:
   197→                continue
   198→        return regular, bold
   199→    return None

   202→def _font_covers(path: str, codepoints: tuple[int, ...]) -> bool:
   203→    """True when the TTF at `path` maps every codepoint in `codepoints`."""
   204→    try:
   205→        from reportlab.pdfbase.ttfonts import TTFontFile
   206→
   207→        font = TTFontFile(path)
   208→        return all(cp in font.charToGlyph for cp in codepoints)
   209→    except Exception:  # unreadable/unsupported font → do not trust it
   210→        return False

   216→def _ensure_arabic_font_registered() -> tuple[str, str] | None:
   217→    """Register the verified Arabic-capable TTF with reportlab (once per
   218→    process). Returns (base_font, heading_font) or None when no usable font
   219→    exists — the caller then degrades honestly."""
   220→    from reportlab.pdfbase import pdfmetrics
   221→    from reportlab.pdfbase.ttfonts import TTFont
   222→
   223→    if _FONT_STATE:
   224→        return (_FONT_STATE["base"], _FONT_STATE["heading"]) if "base" in _FONT_STATE else None
   225→    found = _find_arabic_font()
   226→    if found is None:
   227→        _FONT_STATE["unavailable"] = "1"
   228→        return None
   229→    regular_path, bold_path = found
   230→    try:
   231→        pdfmetrics.registerFont(TTFont("CropTwistUnicode", regular_path))
   232→        base = "CropTwistUnicode"
   233→        if bold_path:
   234→            pdfmetrics.registerFont(TTFont("CropTwistUnicode-Bold", bold_path))
   235→            heading = "CropTwistUnicode-Bold"
   236→        else:
   237→            heading = base
   238→    except Exception:
   239→        _FONT_STATE["unavailable"] = "1"
   240→        return None
   241→    _FONT_STATE["base"] = base
   242→    _FONT_STATE["heading"] = heading
   243→    return base, heading

   255→def _as_utc_text(value: Any, locale: str) -> str:
   256→    """Render a datetime/ISO string in locale format, always labelled UTC."""
   257→    dt: datetime | None = None
   258→    if isinstance(value, datetime):
   259→        dt = value
   260→    elif isinstance(value, str):
   261→        try:
   262→            dt = datetime.fromisoformat(value.replace("Z", "+00:00"))
   263→        except ValueError:
   264→            return value
   265→    if dt is None:
   266→        return str(value)
   267→    if dt.tzinfo is not None:
   268→        dt = dt.astimezone(UTC)
   269→    if locale == "ar":
   270→        return dt.strftime("%Y/%m/%d %H:%M") + " UTC"
   271→    return dt.strftime("%d %b %Y %H:%M") + " UTC"
   272→
   273→
   274→def _fmt_date(value: Any) -> str:
   275→    if isinstance(value, datetime):
   276→        return value.date().isoformat()
   277→    return str(value)

   280→def _normalize(payload: dict[str, Any], locale: str) -> dict[str, Any]:
   281→    """Fill defaults and return a shallow-normalized copy of the payload."""
   282→    data = dict(payload or {})
   283→    data.setdefault("title", "Report")
   284→    data.setdefault("report_type", "custom_range")
   285→    data.setdefault("primary_section", None)
   286→    data.setdefault("org_name", "")
   287→    data.setdefault("farm_name", None)
   288→    data["generated_at"] = _as_utc_text(data.get("generated_at") or datetime.now(UTC), locale)
   289→    dr = dict(data.get("date_range") or {})
   290→    data["date_range"] = {
   291→        "from": _fmt_date(dr.get("from") or ""),
   292→        "to": _fmt_date(dr.get("to") or ""),
   293→    }
   294→    sections = []
   295→    for section in data.get("sections") or []:
   296→        sec = dict(section or {})
   297→        sec.setdefault("heading", "")
   298→        sec.setdefault("prose", None)
   299→        sec.setdefault("table", None)
   300→        sec.setdefault("kpis", [])
   301→        sec.setdefault("provenance", "missing")
   302→        sec.setdefault("note", None)
   303→        sections.append(sec)
   304→    data["sections"] = sections
   305→    data.setdefault("provenance_legend", DEFAULT_LEGEND)
   306→    for key in ("assumptions", "limitations", "sources", "rendering_notes"):
   307→        data.setdefault(key, [])
   308→    return data

   325→def _headers_with_units(table: dict[str, Any]) -> list[str]:
   326→    columns = [str(c) for c in table.get("columns") or []]
   327→    units = table.get("units")
   328→    if not units:
   329→        return columns
   330→    out = []
   331→    for i, col in enumerate(columns):
   332→        unit = units[i] if i < len(units) else None
   333→        out.append(f"{col} ({unit})" if unit else col)
   334→    return out

   337→def _primary_table(payload: dict[str, Any]) -> tuple[dict[str, Any] | None, str | None]:
   338→    """Choose the table flattened into the CSV export.
   339→
   340→    Preference: the section whose key matches payload["primary_section"]
   341→    (when it has a table) → the first section with a table → None.
   342→    """
   343→    wanted = payload.get("primary_section")
   344→    sections = payload.get("sections") or []
   345→    if wanted:
   346→        for sec in sections:
   347→            if sec.get("key") == wanted and sec.get("table"):
   348→                return sec["table"], sec.get("heading")
   349→    for sec in sections:
   350→        if sec.get("table"):
   351→            return sec["table"], sec.get("heading")
   352→    return None, None

   355→def _kpi_table(payload: dict[str, Any]) -> dict[str, Any] | None:
   356→    rows: list[list[Any]] = []
   357→    for sec in payload.get("sections") or []:
   358→        for kpi in sec.get("kpis") or []:
   359→            rows.append([
   360→                kpi.get("label"),
   361→                "" if kpi.get("value") is None else kpi.get("value"),
   362→                kpi.get("unit") or "",
   363→                kpi.get("provenance") or "missing",
   364→            ])
   365→    if not rows:
   366→        return None
   367→    return {
   368→        "columns": ["Label", "Value", "Unit", "Provenance"],
   369→        "rows": rows,
   370→        "units": None,
   371→    }

   396→# ------------------------------------------------------------------- PDF
   397→def render_pdf(payload: dict[str, Any], locale: str = "en") -> bytes:
   398→    """Render the payload as a paginated A4 PDF (platypus layout).
   399→
   400→    Arabic (locale=ar): text is shaped via arabic_reshaper + python-bidi and
   401→    drawn with a verified Unicode TTF when one exists on the host; without
   402→    one the document falls back to transliterated Latin and records
   403→    ``"arabic font not embedded"`` in payload["rendering_notes"] — never
   404→    silent tofu.
   405→    """
   406→    from reportlab.lib import colors
   407→    from reportlab.lib.enums import TA_LEFT, TA_RIGHT
   408→    from reportlab.lib.pagesizes import A4
   409→    from reportlab.lib.styles import ParagraphStyle
   410→    from reportlab.pdfbase import pdfmetrics
   411→    from reportlab.pdfbase.ttfonts import TTFont
   412→    from reportlab.platypus import (
   413→        HRFlowable,
   414→        Paragraph,
   415→        SimpleDocTemplate,
   416→        Spacer,
   417→        Table,
   418→        TableStyle,
   419→    )
   420→
   421→    data = _normalize(payload, locale)
   422→    t = _chrome(locale)
   423→    rtl = locale == "ar"
   424→
   425→    base_font = "Helvetica"
   426→    heading_font = "Helvetica-Bold"
   427→    if rtl and _has_arabic(_flatten_text(data)):
   428→        registered = _ensure_arabic_font_registered()
   429→        if registered is None:
   430→            data["rendering_notes"].append("arabic font not embedded")
   431→        else:
   432→            base_font, heading_font = registered
   433→
   434→    def shape(text: str) -> str:
   435→        if rtl and _has_arabic(text) and base_font != "Helvetica":
   436→            try:
   437→                return _shape_arabic(text)
   438→            except Exception:
   439→                return _transliterate_arabic(text)
   440→        if rtl and _has_arabic(text) and base_font == "Helvetica":
   441→            return _transliterate_arabic(text)
   442→        return text
   443→
   444→    alignment = TA_RIGHT if rtl else TA_LEFT
   445→    wrap = "RTL" if rtl else None
   446→
   447→    def style(name: str, size: int, *, bold: bool = False, grey: bool = False) -> ParagraphStyle:
   448→        return ParagraphStyle(
   449→            name,
   450→            fontName=heading_font if bold else base_font,
   451→            fontSize=size,
   452→            leading=size * 1.35,
   453→            alignment=alignment,
   454→            wordWrap=wrap,
   455→            textColor=colors.HexColor("#555555") if grey else colors.HexColor("#111111"),
   456→            spaceAfter=4,
   457→        )
   458→
   459→    st_title = style("title", 18, bold=True)
   460→    st_h2 = style("h2", 13, bold=True)
   461→    st_body = style("body", 9.5)
   462→    st_small = style("small", 8, grey=True)
   463→    st_cell = ParagraphStyle(
   464→        "cell", fontName=base_font, fontSize=7.5, leading=9.5,
   465→        alignment=alignment, wordWrap=wrap,
   466→    )
   467→    st_cell_head = ParagraphStyle(
   468→        "cellhead", fontName=heading_font, fontSize=7.5, leading=9.5,
   469→        alignment=alignment, wordWrap=wrap, textColor=colors.white,
   470→    )
   471→
   472→    def para(text: str, st: ParagraphStyle) -> Paragraph:
   473→        return Paragraph(saxutils.escape(shape(str(text))), st)
   474→
   475→    buf = io.BytesIO()
   476→    doc = SimpleDocTemplate(
   477→        buf, pagesize=A4,
   478→        leftMargin=42, rightMargin=42, topMargin=42, bottomMargin=48,
   479→        title=str(data.get("title") or "Report"),
   480→        author=t["identifier"],
   481→    )
   482→
   483→    def footer(canvas, _doc) -> None:  # noqa: ANN001 — reportlab signature
   484→        canvas.saveState()
   485→        canvas.setFont(base_font, 7.5)
   486→        canvas.setFillColor(colors.HexColor("#777777"))
   487→        label = shape(f"{t['identifier']} — {t['page']} {canvas.getPageNumber()}")
   488→        if rtl:
   489→            canvas.drawRightString(A4[0] - 42, 30, label)
   490→        else:
   491→            canvas.drawString(42, 30, label)
   492→        canvas.restoreState()
   493→
   494→    story: list[Any] = []
   495→    story.append(para(data.get("title") or "Report", st_title))
   496→    meta_lines = [
   497→        f"{t['org']}: {data.get('org_name') or t['missing']}",
   498→    ]
   499→    if data.get("farm_name"):
   500→        meta_lines.append(f"{t['farm']}: {data['farm_name']}")
   501→    meta_lines.append(f"{t['report_type']}: {data.get('report_type')}")
   502→    meta_lines.append(f"{t['generated']}: {data['generated_at']}")
   503→    meta_lines.append(
   504→        f"{t['date_range']}: {data['date_range']['from']} {t['to']} {data['date_range']['to']}"
   505→    )
   506→    for line in meta_lines:
   507→        story.append(para(line, st_body))
   508→    story.append(Spacer(1, 4))
   509→    story.append(HRFlowable(width="100%", thickness=0.7, color=colors.HexColor("#BBBBBB")))
   510→    story.append(Spacer(1, 8))
   511→
   512→    for sec in data["sections"]:
   513→        story.append(para(sec["heading"] or "", st_h2))
   514→        story.append(para(f"{t['provenance']}: {sec['provenance']}", st_small))
   515→        if sec.get("prose"):
   516→            story.append(para(sec["prose"], st_body))
   517→        if sec.get("note"):
   518→            story.append(para(f"{t['note']}: {sec['note']}", st_small))
   519→        kpis = sec.get("kpis") or []
   520→        if kpis:
   521→            story.append(Spacer(1, 3))
   522→            story.append(para(t["kpis"], st_body))
   523→            kpi_rows = [[t["label"], t["value"], t["unit"], t["provenance"]]]
   524→            for kpi in kpis:
   525→                value = kpi.get("value")
   526→                kpi_rows.append([
   527→                    kpi.get("label") or "",
   528→                    _cell_text(value, locale),
   529→                    kpi.get("unit") or "",
   530→                    kpi.get("provenance") or "missing",
   531→                ])
   532→            story.append(_styled_table(kpi_rows, st_cell, st_cell_head, colors, rtl))
   533→            story.append(Spacer(1, 4))
   534→        table = sec.get("table")
   535→        if table and (table.get("rows") or table.get("columns")):
   536→            story.append(Spacer(1, 3))
   537→            headers = [shape(h) for h in _headers_with_units(table)]
   538→            rows = [[_cell_text(v, locale) for v in row] for row in (table.get("rows") or [])]
   539→            omitted = 0
   540→            if len(rows) > PDF_ROWS_CAP:
   541→                omitted = len(rows) - PDF_ROWS_CAP
   542→                rows = rows[:PDF_ROWS_CAP]
   543→            if not rows:
   544→                rows = [[t["no_data"]] + [""] * max(0, len(headers) - 1)]
   545→            story.append(_styled_table([headers, *rows], st_cell, st_cell_head, colors, rtl))
   546→            if omitted:
   547→                story.append(para(t["rows_omitted"].format(n=omitted), st_small))
   548→        elif not kpis:
   549→            story.append(para(t["no_data"], st_small))
   550→        story.append(Spacer(1, 10))
   551→
   552→    story.append(HRFlowable(width="100%", thickness=0.7, color=colors.HexColor("#BBBBBB")))
   553→    story.append(para(t["provenance_legend"], st_h2))
   554→    for entry in data.get("provenance_legend") or []:
   555→        story.append(para(f"{entry.get('code')}: {entry.get('meaning')}", st_small))
   556→    for key, title_key in (
   557→        ("assumptions", "assumptions"),
   558→        ("limitations", "limitations"),
   559→        ("sources", "sources"),
   560→    ):
   561→        items = data.get(key) or []
   562→        story.append(Spacer(1, 6))
   563→        story.append(para(t[title_key], st_h2))
   564→        if not items:
   565→            story.append(para(t["missing"], st_small))
   566→        for item in items:
   567→            story.append(para(f"• {item}", st_body))
   568→
   569→    doc.build(story, onFirstPage=footer, onLaterPages=footer)
   570→    return buf.getvalue()

   573→def _flatten_text(data: dict[str, Any]) -> str:
   574→    """Every display string in the payload, for the Arabic-detection probe."""
   575→    parts: list[str] = [
   576→        str(data.get("title") or ""), str(data.get("org_name") or ""),
   577→        str(data.get("farm_name") or ""),
   578→    ]
   579→    for sec in data.get("sections") or []:
   580→        parts.append(str(sec.get("heading") or ""))
   581→        parts.append(str(sec.get("prose") or ""))
   582→        parts.append(str(sec.get("note") or ""))
   583→        table = sec.get("table") or {}
   584→        parts.extend(str(c) for c in table.get("columns") or [])
   585→        for row in table.get("rows") or []:
   586→            parts.extend(str(v) for v in row if v is not None)
   587→        for kpi in sec.get("kpis") or []:
   588→            parts.append(str(kpi.get("label") or ""))
   589→    for key in ("assumptions", "limitations", "sources"):
   590→        parts.extend(str(x) for x in data.get(key) or [])
   591→    return " ".join(parts)

   594→def _styled_table(rows: list[list[Any]], st_cell, st_cell_head, colors, rtl: bool):  # noqa: ANN001
   595→    """Build a wrapped reportlab Table with header styling."""
   596→    from reportlab.lib.pagesizes import A4
   597→    from reportlab.platypus import Paragraph, Table, TableStyle
   598→
   599→    ncols = max(len(r) for r in rows)
   600→    norm = [list(r) + [""] * (ncols - len(r)) for r in rows]
   601→    wrapped: list[list[Any]] = []
   602→    for i, row in enumerate(norm):
   603→        st = st_cell_head if i == 0 else st_cell
   604→        wrapped.append([Paragraph(str(v), st) for v in row])
   605→    page_w = A4[0] - 84  # usable width between the 42pt margins
   606→    widths = [page_w / ncols] * ncols
   607→    tbl = Table(wrapped, colWidths=widths, repeatRows=1)
   608→    tbl.setStyle(TableStyle([
   609→        ("BACKGROUND", (0, 0), (-1, 0), colors.HexColor("#33526B")),
   610→        ("GRID", (0, 0), (-1, -1), 0.4, colors.HexColor("#CCCCCC")),
   611→        ("VALIGN", (0, 0), (-1, -1), "TOP"),
   612→        ("ROWBACKGROUNDS", (0, 1), (-1, -1), [colors.white, colors.HexColor("#F4F6F8")]),
   613→        ("LEFTPADDING", (0, 0), (-1, -1), 4),
   614→        ("RIGHTPADDING", (0, 0), (-1, -1), 4),
   615→        ("TOPPADDING", (0, 0), (-1, -1), 3),
   616→        ("BOTTOMPADDING", (0, 0), (-1, -1), 3),
   617→    ]))
   618→    return tbl

   624→def render_xlsx(payload: dict[str, Any], locale: str = "en") -> bytes:
   625→    """Render the payload as an XLSX workbook.
   626→
   627→    One sheet per section (heading as sheet name, sanitized to Excel's 31-char
   628→    limit), styled header row, units embedded in column headers, provenance
   629→    column on KPI blocks, plus a "Metadata" sheet (org, farm, generated_at,
   630→    period, assumptions, limitations, sources, rendering notes). Excel shapes
   631→    Arabic text natively — cell values are stored unshaped (logical order).
   632→    """
   633→    from openpyxl import Workbook
   634→    from openpyxl.styles import Alignment, Font, PatternFill
   635→    from openpyxl.utils import get_column_letter
   636→
   637→    data = _normalize(payload, locale)
   638→    t = _chrome(locale)
   639→
   640→    wb = Workbook()
   641→    wb.remove(wb.active)
   642→
   643→    header_fill = PatternFill("solid", fgColor="33526B")
   644→    header_font = Font(bold=True, color="FFFFFF")
   645→    title_font = Font(bold=True, size=12)
   646→    small_font = Font(size=9, color="666666")
   647→
   648→    used_names: set[str] = set()
   649→
   650→    def sheet_name(heading: str, index: int) -> str:
   651→        cleaned = "".join(c for c in str(heading or t["metadata"]) if c not in '[]:*?/\\')
   652→        cleaned = cleaned.strip() or t["metadata"]
   653→        name = cleaned[:28]
   654→        candidate = name
   655→        n = 1
   656→        while candidate.lower() in used_names:
   657→            suffix = f" ({n})"
   658→            candidate = name[: 28 - len(suffix)] + suffix
   659→            n += 1
   660→        used_names.add(candidate.lower())
   661→        return candidate
   662→
   663→    for index, sec in enumerate(data["sections"], start=1):
   664→        ws = wb.create_sheet(title=sheet_name(sec.get("heading", ""), index))
   665→        ws.sheet_view.rightToLeft = locale == "ar"
   666→        row = 1
   667→        ws.cell(row=row, column=1, value=sec.get("heading") or "").font = title_font
   668→        row += 1
   669→        prov_line = f"{t['provenance']}: {sec.get('provenance')}"
   670→        if sec.get("note"):
   671→            prov_line += f" | {t['note']}: {sec['note']}"
   672→        ws.cell(row=row, column=1, value=prov_line).font = small_font
   673→        row += 2
   674→        kpis = sec.get("kpis") or []
   675→        if kpis:
   676→            ws.cell(row=row, column=1, value=t["kpis"]).font = title_font
   677→            row += 1
   678→            headers = [t["label"], t["value"], t["unit"], t["provenance"]]
   679→            for col, header in enumerate(headers, start=1):
   680→                cell = ws.cell(row=row, column=col, value=header)
   681→                cell.font = header_font
   682→                cell.fill = header_fill
   683→            row += 1
   684→            for kpi in kpis:
   685→                ws.cell(row=row, column=1, value=kpi.get("label"))
   686→                value = kpi.get("value")
   687→                ws.cell(row=row, column=2, value=value if value is not None else t["missing"])
   688→                ws.cell(row=row, column=3, value=kpi.get("unit") or "")
   689→                ws.cell(row=row, column=4, value=kpi.get("provenance") or "missing")
   690→                row += 1
   691→            row += 1
   692→        table = sec.get("table")
   693→        if table and (table.get("columns") or table.get("rows")):
   694→            headers = _headers_with_units(table)
   695→            for col, header in enumerate(headers, start=1):
   696→                cell = ws.cell(row=row, column=col, value=header)
   697→                cell.font = header_font
   698→                cell.fill = header_fill
   699→                cell.alignment = Alignment(wrap_text=True, vertical="top")
   700→            row += 1
   701→            for record in table.get("rows") or []:
   702→                for col, value in enumerate(record, start=1):
   703→                    ws.cell(
   704→                        row=row, column=col,
   705→                        value=value if value is not None else t["missing"],
   706→                    )
   707→                row += 1
   708→        elif not kpis:
   709→            ws.cell(row=row, column=1, value=t["no_data"]).font = small_font
   710→            row += 1
   711→        if sec.get("prose"):
   712→            row += 1
   713→            ws.cell(row=row, column=1, value=str(sec["prose"])).font = small_font
   714→        for col_idx in range(1, 9):
   715→            ws.column_dimensions[get_column_letter(col_idx)].width = 22
   716→
   717→    # Metadata sheet (always last).
   718→    meta = _meta_table(data)
   719→    ws = wb.create_sheet(title=t["metadata"][:31])
   720→    ws.sheet_view.rightToLeft = locale == "ar"
   721→    for col, header in enumerate(meta["columns"], start=1):
   722→        cell = ws.cell(row=1, column=col, value=header)
   723→        cell.font = header_font
   724→        cell.fill = header_fill
   725→    for r, record in enumerate(meta["rows"], start=2):
   726→        ws.cell(row=r, column=1, value=record[0])
   727→        ws.cell(row=r, column=2, value=record[1])
   728→    ws.column_dimensions["A"].width = 28
   729→    ws.column_dimensions["B"].width = 80
   730→
   731→    buf = io.BytesIO()
   732→    wb.save(buf)
   733→    return buf.getvalue()

   736→# -------------------------------------------------------------------- CSV
   737→def render_csv(payload: dict[str, Any], locale: str = "en") -> bytes:
   738→    """Render the payload as a single UTF-8 **with BOM** CSV (Excel-friendly).
   739→
   740→    The flattened table is the section selected by payload["primary_section"]
   741→    (falling back to the first section that carries a table, then to KPIs,
   742→    then to a metadata listing). Header comment lines carry the report
   743→    identity and provenance.
   744→    """
   745→    data = _normalize(payload, locale)
   746→    t = _chrome(locale)
   747→
   748→    buf = io.StringIO()
   749→    writer = csv.writer(buf)
   750→
   751→    writer.writerow([f"# {data.get('title') or 'Report'}"])
   752→    writer.writerow([f"# {t['org']}: {data.get('org_name') or ''}"])
   753→    if data.get("farm_name"):
   754→        writer.writerow([f"# {t['farm']}: {data['farm_name']}"])
   755→    writer.writerow([f"# {t['report_type']}: {data.get('report_type')}"])
   756→    writer.writerow([f"# {t['generated']}: {data['generated_at']}"])
   757→    writer.writerow([
   758→        f"# {t['date_range']}: {data['date_range']['from']} {t['to']} {data['date_range']['to']}",
   759→    ])
   760→    for note in data.get("rendering_notes") or []:
   761→        writer.writerow([f"# rendering_note: {note}"])
   762→
   763→    table, section_heading = _primary_table(data)
   764→    if table is not None:
   765→        writer.writerow([])
   766→        writer.writerow([f"# section: {section_heading or ''}"])
   767→        headers = _headers_with_units(table)
   768→        writer.writerow(headers)
   769→        for record in table.get("rows") or []:
   770→            writer.writerow(["" if v is None else v for v in record])
   771→    else:
   772→        kpis = _kpi_table(data)
   773→        if kpis is not None:
   774→            writer.writerow([])
   775→            writer.writerow([f"# {t['kpis']}"])
   776→            writer.writerow(kpis["columns"])
   777→            writer.writerows(kpis["rows"])
   778→        else:
   779→            meta = _meta_table(data)
   780→            writer.writerow([])
   781→            writer.writerow([f"# {t['metadata']}"])
   782→            writer.writerow(meta["columns"])
   783→            writer.writerows(meta["rows"])
   784→
   785→    # utf-8-sig prepends the BOM so Excel detects the encoding.
   786→    return buf.getvalue().encode("utf-8-sig")
