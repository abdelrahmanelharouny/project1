    53→const AGGREGATION_LEVELS = [
    54→  { value: 'monthly', label: 'Monthly', hint: 'Totals, means, indices, anomalies per calendar month' },
    55→  { value: 'seasonal', label: 'Seasonal', hint: 'Meteorological seasons of the chosen hemisphere' },

    68→function fmtNum(value: number | null | undefined, digits = 2): string {
    69→  if (value === null || value === undefined) return '—'
    70→  return value.toLocaleString(undefined, { maximumFractionDigits: digits })
    71→}
    72→
    73→function coveragePct(ratio: number | null): string {
    74→  if (ratio === null || ratio === undefined) return '—'
    75→  return `${Math.round(ratio * 100)}%`
    76→}

    78→function featureQualityTone(status: FeatureValue['quality_status']): string {
    79→  switch (status) {
    80→    case 'valid':
    81→      return 'border-emerald-200 bg-emerald-50 text-emerald-800'
    82→    case 'partial':
    83→      return 'border-amber-200 bg-amber-50 text-amber-900'
    84→    case 'insufficient_coverage':
    85→      return 'border-orange-200 bg-orange-50 text-orange-900'
    86→    case 'missing':
    87→      return 'border-stone-200 bg-stone-100 text-stone-600'
    88→    default:
    89→      return 'border-stone-300 bg-white text-stone-600'
    90→  }
    91→}

    93→function CreateRunForm({
    94→  selectedField,
    95→  onCreated,
    96→}: {
    97→  selectedField: SelectedField
    98→  onCreated: (run: FeatureRun) => void
    99→}) {
   100→  const [start, setStart] = useState('2023-01-01')
   101→  const [end, setEnd] = useState('2023-12-31')
   102→  const [levels, setLevels] = useState<string[]>(['monthly'])
   103→  const [hemisphere, setHemisphere] = useState<'north' | 'south'>('north')
   104→  const [seasons, setSeasons] = useState<SeasonWindow[]>([])
   105→  const queryClient = useQueryClient()
   106→
   107→  const toggleLevel = (level: string) => {
   108→    setLevels((prev) => {
   109→      const next = prev.includes(level) ? prev.filter((l) => l !== level) : [...prev, level]
   110→      return next.length ? next : prev // never empty
   111→    })
   112→  }
   113→
   114→  const validate = (): string | null => {
   115→    const dateRe = /^\d{4}-\d{2}-\d{2}$/
   116→    if (!dateRe.test(start) || !dateRe.test(end)) return 'Enter start and end as YYYY-MM-DD.'
   117→    if (end < start) return 'The end date must be on or after the start date.'
   118→    const years = (Date.parse(end) - Date.parse(start)) / (365.25 * 24 * 3600 * 1000)
   119→    if (years > 31) return 'A run covers at most 31 years.'
   120→    if (levels.includes('daily') && Date.parse(end) - Date.parse(start) > MAX_DAILY_DAYS * 86_400_000) {
   121→      return 'Daily aggregation is limited to ~3 years per run — uncheck Daily or shorten the window.'
   122→    }
   123→    if (levels.includes('crop_season') && seasons.length === 0) {
   124→      return 'Add at least one crop-season window or uncheck Crop seasons.'
   125→    }
   126→    for (const s of seasons) {
   127→      if (!s.label.trim()) return 'Every crop season needs a label.'
   128→      if (!dateRe.test(s.start) || !dateRe.test(s.end)) return `Season "${s.label}" needs YYYY-MM-DD dates.`
   129→      if (s.end < s.start) return `Season "${s.label}" ends before it starts.`
   130→    }
   131→    return null
   132→  }
   133→
   134→  const mutation = useMutation({
   135→    mutationFn: () =>
   136→      createFeatureRun(selectedField.id, {
   137→        start_date: start,
   138→        end_date: end,
   139→        aggregation_levels: levels,
   140→        hemisphere,
   141→        crop_seasons:
   142→          levels.includes('crop_season') && seasons.length
   143→            ? seasons.map((s) => ({ label: s.label.trim(), start_date: s.start, end_date: s.end }))
   144→            : [],
   145→      }),
   146→    onSuccess: (run) => {
   147→      queryClient.invalidateQueries({ queryKey: ['feature-runs', selectedField.id] })
   148→      if (run.status === 'succeeded') {
   149→        toast.success('Feature run completed', {
   150→          description: `${run.records_written} values written · engine ${run.feature_engine_version}`,
   151→        })
   152→        onCreated(run)
   153→      } else if (run.status === 'failed') {
   154→        toast.error('Feature run failed', { description: run.error_summary ?? undefined })
   155→      }
   156→    },
   157→    onError: (err) => {
   158→      const { title, description } = describeError(err)
   159→      toast.error(title, { description })
   160→    },
   161→  })
   162→
   163→  return (
   164→    <Card>
   165→      <CardHeader>
   166→        <CardTitle className="flex items-center gap-2 text-base">
   167→          <CalendarRange className="h-4 w-4 text-emerald-700" aria-hidden />
   168→          Create feature run
   169→        </CardTitle>
   170→        <CardDescription>
   171→          Derived features are computed from ONE source per variable (products are never blended).
   172→          Coverage below a definition&apos;s minimum leaves the value explicitly empty — missing data
   173→          is never treated as zero.
   174→        </CardDescription>
   175→      </CardHeader>
   176→      <CardContent className="space-y-4">
   177→        <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-4">
   178→          <div className="space-y-1.5">
   179→            <Label htmlFor="feature-start">Period start</Label>
   180→            <Input id="feature-start" type="date" value={start} onChange={(e) => setStart(e.target.value)} />
   181→          </div>
   182→          <div className="space-y-1.5">
   183→            <Label htmlFor="feature-end">Period end</Label>
   184→            <Input id="feature-end" type="date" value={end} onChange={(e) => setEnd(e.target.value)} />
   185→          </div>
   186→          <div className="space-y-1.5">
   187→            <Label htmlFor="feature-hemisphere">Season hemisphere</Label>
   188→            <Select value={hemisphere} onValueChange={(v) => setHemisphere(v as 'north' | 'south')}>
   189→              <SelectTrigger id="feature-hemisphere" aria-label="Season hemisphere">
   190→                <SelectValue />
   191→              </SelectTrigger>
   192→              <SelectContent>
   193→                <SelectItem value="north">North (DJF · MAM · JJA · SON)</SelectItem>
   194→                <SelectItem value="south">South (JJA · SON · DJF · MAM)</SelectItem>
   195→              </SelectContent>
   196→            </Select>
   197→          </div>
   198→          <div className="space-y-1.5">
   199→            <Label>Aggregation levels</Label>
   200→            <div className="flex flex-wrap gap-x-4 gap-y-1.5 pt-1">
   201→              {AGGREGATION_LEVELS.map((level) => (
   202→                <label key={level.value} className="flex min-w-0 items-center gap-1.5 text-sm text-stone-700">
   203→                  <Checkbox
   204→                    checked={levels.includes(level.value)}
   205→                    onCheckedChange={() => toggleLevel(level.value)}
   206→                    aria-label={`Aggregation level ${level.label}`}
   207→                  />
   208→                  {level.label}
   209→                </label>
   210→              ))}
   211→            </div>
   212→          </div>
   213→        </div>
   214→
   215→        {levels.includes('crop_season') && (
   216→          <div className="space-y-2 rounded-lg border border-stone-200 bg-stone-50 p-3">
   217→            <div className="flex items-center justify-between">
   218→              <p className="text-sm font-medium text-stone-700">Crop-season windows</p>
   219→              <Button
   220→                type="button"
   221→                variant="outline"
   222→                size="sm"
   223→                onClick={() => setSeasons((prev) => [...prev, { label: '', start: '', end: '' }])}
   224→              >
   225→                Add window
   226→              </Button>
   227→            </div>
   228→            <p className="text-xs text-stone-500">
   229→              Windows are recorded verbatim in run provenance. Crop calendars from the sourced
   230→              knowledge base arrive in a later milestone — nothing is invented here.
   231→            </p>
   232→            {seasons.map((season, index) => (
   233→              <div key={index} className="grid gap-2 sm:grid-cols-[1fr_auto_auto_auto]">
   234→                <Input
   235→                  placeholder="Label, e.g. maize_2024"
   236→                  value={season.label}
   237→                  onChange={(e) =>
   238→                    setSeasons((prev) => prev.map((s, i) => (i === index ? { ...s, label: e.target.value } : s)))
   239→                  }
   240→                  aria-label={`Season ${index + 1} label`}
   241→                />
   242→                <Input
   243→                  type="date"
   244→                  value={season.start}
   245→                  onChange={(e) =>
   246→                    setSeasons((prev) => prev.map((s, i) => (i === index ? { ...s, start: e.target.value } : s)))
   247→                  }
   248→                  aria-label={`Season ${index + 1} start`}
   249→                />
   250→                <Input
   251→                  type="date"
   252→                  value={season.end}
   253→                  onChange={(e) =>
   254→                    setSeasons((prev) => prev.map((s, i) => (i === index ? { ...s, end: e.target.value } : s)))
   255→                  }
   256→                  aria-label={`Season ${index + 1} end`}
   257→                />
   258→                <Button
   259→                  type="button"
   260→                  variant="ghost"
   261→                  size="sm"
   262→                  onClick={() => setSeasons((prev) => prev.filter((_, i) => i !== index))}
   263→                  aria-label={`Remove season ${index + 1}`}
   264→                >
   265→                  <XCircle className="h-4 w-4" aria-hidden />
   266→                </Button>
   267→              </div>
   268→            ))}
   269→          </div>
   270→        )}
   271→
   272→        {levels.includes('daily') && (
   273→          <Alert>
   274→            <Info className="h-4 w-4" aria-hidden />
   275→            <AlertTitle>Daily values are voluminous</AlertTitle>
   276→            <AlertDescription>
   277→              Each requested day becomes one stored value per feature. Keep daily windows to about
   278→              three years per run.
   279→            </AlertDescription>
   280→          </Alert>
   281→        )}
   282→
   283→        <div className="flex flex-wrap items-center justify-between gap-3">
   284→          <p className="text-xs text-stone-500">
   285→            Indices follow cited meteorological conventions (WMO dry day, ETCCDI temperature
   286→            indices). Crop-specific stress thresholds remain unevaluated until the sourced crop
   287→            knowledge base exists.
   288→          </p>
   289→          <Button
   290→            onClick={() => {
   291→              const problem = validate()
   292→              if (problem) {
   293→                toast.error('Check the run settings', { description: problem })
   294→                return
   295→              }
   296→              mutation.mutate()
   297→            }}
   298→            disabled={mutation.isPending}
   299→            className="bg-emerald-700 text-white hover:bg-emerald-800"
   300→          >
   301→            {mutation.isPending ? (
   302→              <Loader2 className="h-4 w-4 animate-spin" aria-hidden />
   303→            ) : (
   304→              <FlaskConical className="h-4 w-4" aria-hidden />
   305→            )}
   306→            Run features
   307→          </Button>
   308→        </div>
   309→      </CardContent>
   310→    </Card>
   311→  )
   312→}
   313→
   314→function RunDetail({ fieldId, run }: { fieldId: string; run: FeatureRun }) {
   315→  const [level, setLevel] = useState<string>(run.aggregation_levels[0] ?? 'monthly')
   316→  const runsQuery = useFeatureRuns(fieldId)
   317→  const valuesQuery = useFeatureValues(run.run_id, { aggregationLevel: level })
   318→  const [againstId, setAgainstId] = useState<string>('')
   319→  const comparisonQuery = useFeatureComparison(run.run_id, againstId || null, Boolean(againstId))
   320→
   321→  const otherRuns = useMemo(
   322→    () => (runsQuery.data?.items ?? []).filter((r) => r.run_id !== run.run_id),
   323→    [runsQuery.data, run.run_id],
   324→  )
   325→
   326→  return (
   327→    <Card>
   328→      <CardHeader>
   329→        <div className="flex flex-wrap items-start justify-between gap-2">
   330→          <div className="min-w-0">
   331→            <CardTitle className="flex flex-wrap items-center gap-2 text-base">
   332→              Run {run.run_id.slice(0, 8)}
   333→              <JobStatusBadge status={run.status} />
   334→              {run.is_synthetic && <SyntheticBadge />}
   335→            </CardTitle>
   336→            <CardDescription className="mt-1">
   337→              {run.period.start} → {run.period.end} · {run.calendar_timezone} ({run.hemisphere})
   338→              · engine {run.feature_engine_version} · catalog {run.feature_catalog_version}
   339→            </CardDescription>
   340→          </div>
   341→          <div className="flex flex-wrap items-center gap-2">
   342→            {run.aggregation_levels.map((l) => (
   343→              <button
   344→                key={l}
   345→                type="button"
   346→                onClick={() => setLevel(l)}
   347→                className={cn(
   348→                  'rounded-md border px-2.5 py-1 text-xs font-medium transition',
   349→                  level === l
   350→                    ? 'border-emerald-700 bg-emerald-700 text-white'
   351→                    : 'border-stone-300 bg-white text-stone-600 hover:bg-stone-50',
   352→                )}
   353→              >
   354→                {l}
   355→              </button>
   356→            ))}
   357→          </div>
   358→        </div>
   359→      </CardHeader>
   360→      <CardContent className="space-y-4">
   361→        {run.is_synthetic && (
   362→          <Alert className="border-amber-300 bg-amber-50">
   363→            <AlertTriangle className="h-4 w-4 text-amber-700" aria-hidden />
   364→            <AlertTitle className="text-amber-900">Synthetic fixture data</AlertTitle>
   365→            <AlertDescription className="text-amber-900">
   366→              This run is computed from SYNTHETIC fixture data — values demonstrate the pipeline
   367→              and are never valid for real-world conclusions.
   368→            </AlertDescription>
   369→          </Alert>
   370→        )}
   371→
   372→        {run.warnings.length > 0 && (
   373→          <div className="space-y-1.5 rounded-lg border border-stone-200 bg-stone-50 p-3">
   374→            <p className="text-xs font-semibold uppercase tracking-wide text-stone-500">
   375→              Run warnings ({run.warnings.length})
   376→            </p>
   377→            <ul className="max-h-32 space-y-1 overflow-y-auto text-xs text-stone-600">
   378→              {run.warnings.map((w, i) => (
   379→                <li key={i} className="flex gap-1.5">
   380→                  <span aria-hidden>•</span>
   381→                  <span className="min-w-0 break-words">{w}</span>
   382→                </li>
   383→              ))}
   384→            </ul>
   385→          </div>
   386→        )}
   387→
   388→        {Object.keys(run.not_evaluated_reasons).length > 0 && (
   389→          <div className="space-y-1.5 rounded-lg border border-stone-200 bg-stone-50 p-3">
   390→            <p className="text-xs font-semibold uppercase tracking-wide text-stone-500">
   391→              Not evaluated — with reasons
   392→            </p>
   393→            <ul className="max-h-32 space-y-1 overflow-y-auto text-xs text-stone-600">
   394→              {Object.entries(run.not_evaluated_reasons).map(([code, reason]) => (
   395→                <li key={code}>
   396→                  <span className="font-medium">{code.replace('variable:', '')}</span> — {reason}
   397→                </li>
   398→              ))}
   399→            </ul>
   400→          </div>
   401→        )}
   402→
   403→        {run.error_summary && (
   404→          <Alert variant="destructive">
   405→            <XCircle className="h-4 w-4" aria-hidden />
   406→            <AlertTitle>Run failed</AlertTitle>
   407→            <AlertDescription className="break-words">{run.error_summary}</AlertDescription>
   408→          </Alert>
   409→        )}
   410→
   411→        {/* values */}
   412→        {valuesQuery.isLoading ? (
   413→          <div className="flex h-24 items-center justify-center text-sm text-stone-500">
   414→            <Loader2 className="mr-2 h-4 w-4 animate-spin" aria-hidden /> Loading values…
   415→          </div>
   416→        ) : valuesQuery.data && valuesQuery.data.items.length > 0 ? (
   417→          <div className="space-y-1">
   418→            {valuesQuery.data.total > valuesQuery.data.items.length ? (
   419→              <p className="text-[11px] text-stone-500">
   420→                Showing the first {valuesQuery.data.items.length} of {valuesQuery.data.total}{' '}
   421→                values — narrow the level or window to see everything.
   422→              </p>
   423→            ) : null}
   424→            <div className="max-h-96 overflow-y-auto rounded-lg border border-stone-200">
   425→            <Table>
   426→              <TableHeader className="sticky top-0 bg-white">
   427→                <TableRow>
   428→                  <TableHead>Period</TableHead>
   429→                  <TableHead>Feature</TableHead>
   430→                  <TableHead className="text-right">Value</TableHead>
   431→                  <TableHead className="text-right">Unit</TableHead>
   432→                  <TableHead className="text-right">Coverage</TableHead>
   433→                  <TableHead>Quality</TableHead>
   434→                  <TableHead>Provenance</TableHead>
   435→                </TableRow>
   436→              </TableHeader>
   437→              <TableBody>
   438→                {valuesQuery.data.items.map((v) => (
   439→                  <TableRow key={v.id}>
   440→                    <TableCell className="whitespace-nowrap font-mono text-xs">{v.period_label}</TableCell>
   441→                    <TableCell className="max-w-56 truncate text-xs" title={v.feature_code}>
   442→                      {v.feature_code}
   443→                    </TableCell>
   444→                    <TableCell className="text-right font-mono text-xs">{fmtNum(v.value, 3)}</TableCell>
   445→                    <TableCell className="text-right text-xs text-stone-500">{v.unit}</TableCell>
   446→                    <TableCell className="text-right text-xs text-stone-500">
   447→                      {coveragePct(v.coverage_ratio)}
   448→                    </TableCell>
   449→                    <TableCell>
   450→                      <Badge variant="outline" className={cn('px-1.5 py-0 text-[10px]', featureQualityTone(v.quality_status))}>
   451→                        {v.quality_status.replace('_', ' ')}
   452→                      </Badge>
   453→                    </TableCell>
   454→                    <TableCell className="text-xs text-stone-500">
   455→                      {v.provenance.observation_count ?? v.provenance.inputs?.observation_count ?? '—'} obs
   456→                      {v.provenance.baseline
   457→                        ? ` · baseline ${v.provenance.baseline.n_years}y`
   458→                        : ''}
   459→                    </TableCell>
   460→                  </TableRow>
   461→                ))}
   462→              </TableBody>
   463→            </Table>
   464→            </div>
   465→          </div>
   466→        ) : (
   467→          <p className="rounded-lg border border-dashed border-stone-300 p-6 text-center text-sm text-stone-500">
   468→            No stored values for level “{level}” in this run.
   469→          </p>
   470→        )}
   471→
   472→        {/* comparison */}
   473→        {otherRuns.length > 0 && (
   474→          <div className="space-y-3 rounded-lg border border-stone-200 p-3">
   475→            <div className="flex flex-wrap items-center gap-2">
   476→              <GitCompareArrows className="h-4 w-4 text-stone-500" aria-hidden />
   477→              <p className="text-sm font-medium text-stone-700">Compare with another run</p>
   478→              <Select value={againstId || undefined} onValueChange={setAgainstId}>
   479→                <SelectTrigger className="h-8 w-56" aria-label="Select run to compare against">
   480→                  <SelectValue placeholder="Choose a run…" />
   481→                </SelectTrigger>
   482→                <SelectContent>
   483→                  {otherRuns.map((r) => (
   484→                    <SelectItem key={r.run_id} value={r.run_id}>
   485→                      {r.run_id.slice(0, 8)} · {r.period.start} → {r.period.end}
   486→                    </SelectItem>
   487→                  ))}
   488→                </SelectContent>
   489→              </Select>
   490→            </div>
   491→            {comparisonQuery.data && comparisonQuery.data.deltas.length > 0 && (
   492→              <div className="max-h-72 overflow-y-auto rounded-md border border-stone-200">
   493→                <Table>
   494→                  <TableHeader className="sticky top-0 bg-white">
   495→                    <TableRow>
   496→                      <TableHead>Period</TableHead>
   497→                      <TableHead>Feature</TableHead>
   498→                      <TableHead className="text-right">This run</TableHead>
   499→                      <TableHead className="text-right">Against</TableHead>
   500→                      <TableHead className="text-right">Δ</TableHead>
   501→                    </TableRow>
   502→                  </TableHeader>
   503→                  <TableBody>
   504→                    {comparisonQuery.data.deltas.map((d) => (
   505→                      <TableRow key={`${d.feature_code}-${d.period_label}`}>
   506→                        <TableCell className="whitespace-nowrap font-mono text-xs">{d.period_label}</TableCell>
   507→                        <TableCell className="max-w-52 truncate text-xs" title={d.feature_code}>
   508→                          {d.feature_code}
   509→                        </TableCell>
   510→                        <TableCell className="text-right font-mono text-xs">{fmtNum(d.run_value, 3)}</TableCell>
   511→                        <TableCell className="text-right font-mono text-xs">{fmtNum(d.against_value, 3)}</TableCell>
   512→                        <TableCell
   513→                          className={cn(

   519→                          {fmtNum(d.delta, 3)}

   522→                    ))}
   523→                  </TableBody>
   524→                </Table>
   525→              </div>

   527→            {comparisonQuery.data && comparisonQuery.data.deltas.length === 0 && (
   528→              <p className="text-xs text-stone-500">

   530→              </p>
   531→            )}
   532→            {againstId && (

   534→            )}
   535→          </div>
   536→        )}
   537→      </CardContent>
