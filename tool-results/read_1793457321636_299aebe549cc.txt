    60→def _validate_seasons(raw: list[dict[str, Any]]) -> list[SeasonSpec]:
    61→    if not raw:
    62→        raise ValidationFailed(
    63→            "At least one planning season is required.",
    64→            details=[{"field": "seasons"}],
    65→        )
    66→    if len(raw) > MAX_SEASONS:
    67→        raise ValidationFailed(
    68→            f"At most {MAX_SEASONS} planning seasons per run.",
    69→            details=[{"field": "seasons"}],
    70→        )
    71→    seasons: list[SeasonSpec] = []
    72→    labels: set[str] = set()
    73→    for item in raw:
    74→        label = str(item.get("label") or "").strip()
    75→        if not label or len(label) > 60:
    76→            raise ValidationFailed(
    77→                "Season labels must be 1-60 characters.",
    78→                details=[{"field": "seasons", "reason": str(item)[:80]}],
    79→            )
    80→        if label in labels:
    81→            raise ValidationFailed(
    82→                f"Duplicate season label '{label}'.",
    83→                details=[{"field": "seasons", "reason": label}],
    84→            )
    85→        labels.add(label)
    86→        try:
    87→            start = date.fromisoformat(str(item["start_date"]))
    88→            end = date.fromisoformat(str(item["end_date"]))
    89→        except (KeyError, ValueError):
    90→            raise ValidationFailed(
    91→                "Seasons need start_date and end_date (ISO dates).",
    92→                details=[{"field": "seasons", "reason": label}],
    93→            ) from None
    94→        if start > end:
    95→            raise ValidationFailed(
    96→                f"Season '{label}' start_date is after end_date.",
    97→                details=[{"field": "seasons", "reason": label}],
    98→            )
    99→        if (end - start).days > 400:
   100→            raise ValidationFailed(
   101→                f"Season '{label}' is longer than 400 days — split it.",
   102→                details=[{"field": "seasons", "reason": label}],
   103→            )
   104→        seasons.append(SeasonSpec(label=label, start_date=start, end_date=end))
   105→    return seasons

   108→def _pick_evidence_run(db: Session, field_id: Any, explicit_id: str | None) -> dbm.FeatureRun | None:
   109→    if explicit_id:
   110→        run = db.get(dbm.FeatureRun, uuid.UUID(explicit_id))
   111→        if run is None or str(run.field_id) != str(field_id):
   112→            raise NotFound("Feature run not found for this field.")
   113→        if run.status != "succeeded":
   114→            raise ValidationFailed(
   115→                "The referenced feature run has not succeeded.",
   116→                details=[{"field": "evidence_run_id", "reason": run.status}],
   117→            )
   118→        return run
   119→    return db.scalar(
   120→        select(dbm.FeatureRun)
   121→        .where(dbm.FeatureRun.field_id == field_id, dbm.FeatureRun.status == "succeeded")
   122→        .order_by(dbm.FeatureRun.completed_at.desc())
   123→        .limit(1)

   127→def _season_climate(
   128→    run: dbm.FeatureRun, season: SeasonSpec
   129→) -> ClimateEvidence | None:
   130→    """Aggregate the run's monthly feature values over one planning season as a
   131→    MONTH-OF-YEAR climatology (historical proxy — planning seasons are usually
   132→    in the future; no forecast is claimed).
   133→
   134→    Coverage = fraction of season days whose month-of-year has any covered
   135→    value in the run; below MIN_CLIMATE_COVERAGE the season's climate evidence
   136→    is None (the dimension is unevaluated, never approximated)."""
   137→    monthly: dict[str, dict[int, list[tuple[int, float]]]] = {}  # code -> month -> [(year, value)]
   138→    for row in run.values:
   139→        if row.aggregation_level != "monthly" or row.value is None:
   140→            continue
   141→        monthly.setdefault(row.feature_code, {}).setdefault(
   142→            row.period_start.month, []
   143→        ).append((row.period_start.year, row.value))
   144→
   145→    import calendar as cal
   146→
   147→    def months_in_season(season: SeasonSpec) -> list[tuple[int, int, int]]:
   148→        """(year, month, days of that month inside the season)."""
   149→        out = []
   150→        y, m = season.start_date.year, season.start_date.month
   151→        while (y, m) <= (season.end_date.year, season.end_date.month):
   152→            if y == season.start_date.year and m == season.start_date.month:
   153→                month_start = season.start_date
   154→            else:
   155→                month_start = date(y, m, 1)
   156→            if y == season.end_date.year and m == season.end_date.month:
   157→                month_end = season.end_date
   158→            else:
   159→                month_end = date(y, m, cal.monthrange(y, m)[1])
   160→            out.append((y, m, (month_end - month_start).days + 1))
   161→            m += 1
   162→            if m == 13:
   163→                y, m = y + 1, 1
   164→        return out
   165→
   166→    month_days = months_in_season(season)
   167→    total_days = sum(d for _, _, d in month_days)
   168→
   169→    years_used = sorted(
   170→        {
   171→            y
   172→            for table in monthly.values()
   173→            for entries in table.values()
   174→            for y, _v in entries
   175→        }
   176→    )
   177→
   178→    def collect(code: str, mode: str) -> tuple[float | None, int]:
   179→        """Season aggregate of month-of-year climatological means.
   180→
   181→        mode='total' (precip, ET0, hot-day counts): monthly TOTALS scaled by
   182→        the fraction of that month inside the season, summed.
   183→        mode='mean' (temperature): monthly means day-weighted over the season.
   184→        Returns (value, days covered by present months)."""
   185→        table = monthly.get(code, {})
   186→        acc = 0.0
   187→        covered = 0
   188→        for y, m, days in month_days:
   189→            entries = table.get(m)
   190→            if entries:
   191→                mean = sum(v for _yr, v in entries) / len(entries)
   192→                if mode == "total":
   193→                    frac = days / cal.monthrange(y, m)[1]
   194→                    acc += mean * frac
   195→                else:
   196→                    acc += mean * days
   197→                covered += days
   198→        if mode == "mean":
   199→            acc = acc / total_days if total_days else 0.0
   200→        return acc, covered
   201→
   202→    warnings: list[str] = []
   203→    precip_total, covered = collect(CLIMATE_FEATURE_CODES["precipitation"], "total")
   204→    tmean_wsum, covered_t = collect(CLIMATE_FEATURE_CODES["tmean"], "mean")
   205→    hot_days, covered_h = collect(CLIMATE_FEATURE_CODES["hot_days"], "total")
   206→    et0_total, covered_e = collect(CLIMATE_FEATURE_CODES["et0"], "total")
   207→
   208→    coverage = covered / total_days if total_days else 0.0
   209→
   210→    monthly_et0: list[tuple[int, int, float]] = []
   211→    et0_table = monthly.get(CLIMATE_FEATURE_CODES["et0"], {})
   212→    for y, m, _days in month_days:
   213→        entries = et0_table.get(m)
   214→        if entries:
   215→            monthly_et0.append((y, m, sum(v for _yr, v in entries) / len(entries)))
   216→
   217→    if coverage < MIN_CLIMATE_COVERAGE:
   218→        return None
   219→
   220→    tmean = tmean_wsum if covered_t == total_days else (
   221→        tmean_wsum * total_days / covered_t if covered_t else None
   222→    )
   223→    if covered_t < total_days:
   224→        warnings.append("temperature months incomplete — mean weighted by covered days")
   225→    hot = round(hot_days, 1) if covered_h == total_days else None
   226→    if covered_h < total_days and hot is None:
   227→        warnings.append("hot-day counts incomplete — not evaluated for this season")
   228→    et0 = et0_total if covered_e == total_days else None
   229→    if covered_e < total_days and et0 is None:
   230→        warnings.append("ET0 months incomplete — water estimates not evaluated")
   231→
   232→    year_note = (
   233→        f"years {years_used[0]}-{years_used[-1]}" if len(years_used) > 1
   234→        else f"year {years_used[0]}" if years_used else "no years"
   235→    )
   236→    return ClimateEvidence(
   237→        feature_run_id=str(run.id),
   238→        is_synthetic=bool(run.is_synthetic),
   239→        season_precipitation_mm=round(precip_total, 2) if covered else None,
   240→        season_mean_temp_degC=round(tmean, 2) if tmean is not None else None,
   241→        season_mean_tmax_degC=None,
   242→        season_hot_days_35c=hot,
   243→        season_et0_mm=round(et0, 2) if et0 is not None else None,
   244→        monthly_et0=tuple(monthly_et0),
   245→        coverage_ratio=round(coverage, 4),
   246→        years_used=tuple(years_used),
   247→        basis=(
   248→            f"month-of-year climatology from feature run {str(run.id)[:8]} "
   249→            f"({year_note}) — a historical proxy, NOT a forecast for the "
   250→            "planning season."
   251→        ),
   252→        warnings=tuple(warnings),
   253→    )

   256→def _soil_evidence(db: Session, field_id: Any) -> tuple[SoilEvidence | None, str | None]:
   257→    rows = db.execute(
   258→        select(dbm.Observation)
   259→        .where(
   260→            dbm.Observation.field_id == field_id,
   261→            dbm.Observation.variable_code.in_(SOIL_VARIABLES),
   262→            dbm.Observation.value.is_not(None),
   263→        )
   264→        .order_by(dbm.Observation.observed_at.desc())
   265→        .limit(50)
   266→    ).scalars().all()
   267→    latest: dict[str, dbm.Observation] = {}
   268→    for obs in rows:
   269→        latest.setdefault(obs.variable_code, obs)
   270→    if not latest:
   271→        return None, None
   272→    evidence_classes = sorted({o.evidence_class or "unknown" for o in latest.values()})
   273→
   274→    def val(code: str) -> tuple[float, str] | None:
   275→        obs = latest.get(code)
   276→        if obs is None or obs.value is None:
   277→            return None
   278→        return (float(obs.value), obs.evidence_class or "unknown")
   279→
   280→    soil = SoilEvidence(
   281→        ph=val("soil_ph"),
   282→        organic_carbon_g_kg=val("soil_organic_carbon"),
   283→        clay_g_kg=val("soil_clay_content"),
   284→        sand_g_kg=val("soil_sand_content"),
   285→        bulk_density=val("soil_bulk_density"),
   286→    )
   287→    return soil, ", ".join(evidence_classes)

   290→def build_farm_context(
   291→    db: Session,
   292→    field: dbm.Field,
   293→    seasons: list[SeasonSpec],
   294→    evidence_run_id: str | None,
   295→    hemisphere: str,
   296→) -> tuple[FarmContext, list[str], list[str], dbm.FeatureRun | None]:
   297→    warnings: list[str] = []
   298→    run = _pick_evidence_run(db, field.id, evidence_run_id)
   299→    climate_by_season: dict[str, ClimateEvidence] = {}
   300→    climate_run_refs: dict[str, str] = {}
   301→    if run is None:
   302→        warnings.append(
   303→            "No succeeded feature run exists for this field — climate dimensions are "
   304→            "unevaluated (missing evidence is never treated as zero)."
   305→        )
   306→    else:
   307→        for season in seasons:
   308→            evidence = _season_climate(run, season)
   309→            if evidence is None:
   310→                warnings.append(
   311→                    f"Season '{season.label}': climate coverage below "
   312→                    f"{MIN_CLIMATE_COVERAGE:.0%} in the evidence run — climate dimensions "
   313→                    "unevaluated for this season."
   314→                )
   315→            else:
   316→                climate_by_season[season.label] = evidence
   317→                climate_run_refs[season.label] = str(run.id)
   318→                for w in evidence.warnings:
   319→                    warnings.append(f"Season '{season.label}': {w}")
   320→        if run.is_synthetic:
   321→            warnings.append(
   322→                "The evidence feature run used SYNTHETIC data — every climate-informed "
   323→                "outcome is a demonstration, never real farm evidence."
   324→            )
   325→
   326→    soil, soil_class = _soil_evidence(db, field.id)
   327→
   328→    history_rows = db.scalars(
   329→        select(dbm.CropHistory)
   330→        .where(dbm.CropHistory.field_id == field.id)
   331→        .order_by(dbm.CropHistory.sowing_date.asc().nullslast(), dbm.CropHistory.created_at.asc())
   332→    ).all()
   333→    crop_code_by_id = {
   334→        c.id: c.crop_code
   335→        for c in db.scalars(select(dbm.Crop)).all()
   336→    }
   337→    history: list[dict[str, Any]] = []
   338→    for h in history_rows:
   339→        history.append({
   340→            "id": str(h.id),
   341→            "season_label": h.season_label,
   342→            "crop_code": crop_code_by_id.get(h.crop_id) if h.crop_id else None,
   343→            "crop_label": h.crop_label,
   344→            "sowing_date": h.sowing_date.isoformat() if h.sowing_date else None,
   345→            "harvest_date": h.harvest_date.isoformat() if h.harvest_date else None,
   346→        })
   347→
   348→    constraints_rows = db.scalars(
   349→        select(dbm.FarmConstraint).where(dbm.FarmConstraint.farm_id == field.farm_id)
   350→    ).all()
   351→    market_exclusions: list[str] = []
   352→    equipment_limits: list[str] = []
   353→    water_allocation: float | None = None
   354→    margin_index: dict[str, float] = {}
   355→    for c in constraints_rows:
   356→        if c.constraint_code == "market_exclusion" and c.value_text:
   357→            market_exclusions.extend(
   358→                [p.strip() for p in c.value_text.split(",") if p.strip()]
   359→            )
   360→        elif c.constraint_code == "equipment_limitation" and c.value_text:
   361→            equipment_limits.extend(
   362→                [p.strip() for p in c.value_text.split(",") if p.strip()]
   363→            )
   364→        elif c.constraint_code == "water_allocation_m3_per_season" and c.value_numeric is not None:
   365→            water_allocation = float(c.value_numeric)
   366→        elif c.constraint_code == "market_margin_index" and c.value_text and c.value_numeric is not None:
   367→            margin_index[c.value_text.strip()] = float(c.value_numeric)
   368→
   369→    farm = FarmContext(
   370→        field_id=str(field.id),
   371→        farm_id=str(field.farm_id),
   372→        area_ha=float(field.area_ha),
   373→        country_code=field.farm.country_code,
   374→        hemisphere=hemisphere,
   375→        seasons=tuple(seasons),
   376→        soil=soil,
   377→        climate_by_season=climate_by_season,
   378→        climate_run_refs=climate_run_refs,
   379→        history=tuple(history),
   380→        market_exclusions=tuple(sorted(set(market_exclusions))),
   381→        equipment_limits=tuple(sorted(set(equipment_limits))),
   382→        water_allocation_m3=water_allocation,
   383→        margin_index_by_crop=margin_index,
   384→        synthetic_evidence_in_use=bool(run.is_synthetic) if run else False,
   385→        soil_evidence_class=soil_class,
   386→    )
   387→    return farm, warnings, [str(h["id"]) for h in history], run

   390→def _baseline_assignment(
   391→    seasons: list[SeasonSpec], history: tuple[dict[str, Any], ...]
   392→) -> tuple[tuple[str, str | None], ...]:
   393→    assignment: list[tuple[str, str | None]] = []
   394→    for season in seasons:
   395→        crop_code = None
   396→        for h in history:
   397→            if h.get("season_label") == season.label:
   398→                crop_code = h.get("crop_code")
   399→                break
   400→        if crop_code is None:
   401→            for h in history:
   402→                sow = h.get("sowing_date")
   403→                harv = h.get("harvest_date")
   404→                if sow and harv:
   405→                    if date.fromisoformat(sow) <= season.end_date and date.fromisoformat(
   406→                        harv
   407→                    ) >= season.start_date:
   408→                        crop_code = h.get("crop_code")
   409→                        break
   410→        assignment.append((season.label, crop_code))
   411→    return tuple(assignment)

   414→def _solution_checksum(
   415→    solution: StrategySolution, problem: RotationProblem
   416→) -> str:
   417→    payload = {
   418→        "strategy": solution.strategy_key,
   419→        "assignment": [[lbl, cc] for lbl, cc in solution.assignment],
   420→        "objective": solution.objective_value,
   421→        "criteria": [
   422→            [c.criterion_code, c.raw_value, c.normalized] for c in solution.criteria
   423→        ],
   424→        "problem": problem.checksum(),
   425→        "engine": ROTATION_ENGINE_VERSION,
   426→    }
   427→    canonical = json.dumps(payload, sort_keys=True, separators=(",", ":"))
   428→    return hashlib.sha256(canonical.encode("utf-8")).hexdigest()

   431→def start_rotation_run(
   432→    db: Session,
   433→    *,
   434→    field: dbm.Field,
   435→    payload: dict[str, Any],
   436→    created_by: Any = None,
   437→) -> dbm.RotationRun:
   438→    """Validate, execute inline and persist a rotation run (202-style job)."""
   439→    idempotency_key = payload.get("idempotency_key")
   440→    if idempotency_key:
   441→        existing = db.scalar(
   442→            select(dbm.RotationRun).where(dbm.RotationRun.idempotency_key == idempotency_key)
   443→        )
   444→        if existing is not None:
   445→            return existing
   446→
   447→    seasons = _validate_seasons(payload.get("seasons") or [])
   448→    hemisphere = payload.get("hemisphere") or "north"
   449→    if hemisphere not in ("north", "south"):
   450→        raise ValidationFailed(
   451→            "hemisphere must be 'north' or 'south'.",
   452→            details=[{"field": "hemisphere", "reason": hemisphere}],
   453→        )
   454→
   455→    requested_profiles = payload.get("strategy_profiles") or list(optimizer_mod.DEFAULT_PROFILES)
   456→    if not isinstance(requested_profiles, list) or not requested_profiles:
   457→        raise ValidationFailed(
   458→            "strategy_profiles must be a non-empty list.",
   459→            details=[{"field": "strategy_profiles"}],
   460→        )
   461→    known = set(optimizer_mod.DEFAULT_PROFILES) | {"custom"}
   462→    for p in requested_profiles:
   463→        if p not in known:
   464→            raise ValidationFailed(
   465→                f"Unknown strategy profile '{p}'.",
   466→                details=[{"field": "strategy_profiles", "reason": p}],
   467→            )
   468→    custom_weights = payload.get("custom_weights") or {}
   469→    if "custom" in requested_profiles:
   470→        if not custom_weights:
   471→            raise ValidationFailed(
   472→                "The 'custom' profile needs custom_weights.",
   473→                details=[{"field": "custom_weights"}],
   474→            )
   475→        for code, w in custom_weights.items():
   476→            if code not in objectives_mod.CRITERION_FUNCS:
   477→                raise ValidationFailed(
   478→                    f"Unknown criterion '{code}' in custom_weights.",
   479→                    details=[{"field": "custom_weights", "reason": code}],
   480→                )
   481→            if not isinstance(w, (int, float)) or w < 0:
   482→                raise ValidationFailed(
   483→                    f"Custom weight for '{code}' must be a non-negative number.",
   484→                    details=[{"field": "custom_weights", "reason": code}],
   485→                )
   486→        if sum(float(w) for w in custom_weights.values()) <= 0:
   487→            raise ValidationFailed(
   488→                "custom_weights must sum to a positive value.",
   489→                details=[{"field": "custom_weights"}],
   490→            )
   491→
   492→    water_allocation = payload.get("water_allocation_m3")
   493→    if water_allocation is not None:
   494→        try:
   495→            water_allocation = float(water_allocation)
   496→        except (TypeError, ValueError):
   497→            raise ValidationFailed(
   498→                "water_allocation_m3 must be a number.",
   499→                details=[{"field": "water_allocation_m3"}],
   500→            ) from None
   501→        if water_allocation <= 0:
   502→            raise ValidationFailed(
   503→                "water_allocation_m3 must be positive.",
   504→                details=[{"field": "water_allocation_m3"}],
   505→            )
   506→
   507→    knowledge = load_knowledge(db)
   508→    if not knowledge.crops:
   509→        raise ValidationFailed(
   510→            "No active knowledge pack with crops — import a knowledge pack first.",
   511→            details=[{"field": "knowledge"}],
   512→        )
   513→
   514→    run = dbm.RotationRun(
   515→        field_id=field.id,
   516→        status="running",
   517→        seasons=[
   518→            {"label": s.label, "start_date": s.start_date.isoformat(), "end_date": s.end_date.isoformat()}
   519→            for s in seasons
   520→        ],
   521→        strategy_profiles=list(requested_profiles),
   522→        water_allocation_m3=water_allocation,
   523→        knowledge_pack_refs=[dict(r) for r in knowledge.pack_refs],
   524→        rotation_engine_version=ROTATION_ENGINE_VERSION,
   525→        solver_name=SOLVER_NAME,
   526→        random_seed=RANDOM_SEED,
   527→        request_parameters=dict(payload.get("request_parameters") or {}),
   528→        idempotency_key=idempotency_key,
   529→        created_by=getattr(created_by, "id", None),
   530→        started_at=datetime.now(UTC),
   531→    )
   532→    db.add(run)
   533→    db.flush()
   534→
   535→    try:
   536→        farm, warnings, history_ids, evidence_run = build_farm_context(
   537→            db, field, seasons, payload.get("evidence_run_id"), hemisphere
   538→        )
   539→        run.feature_run_ids = (
   540→            [str(evidence_run.id)] if evidence_run is not None else []
   541→        )
   542→        run.crop_history_ids = history_ids
   543→
   544→        assessments = candidates_mod.generate_candidates(knowledge, farm)
   545→        baseline_assignment = _baseline_assignment(seasons, farm.history)
   546→        problem = RotationProblem(
   547→            field_id=str(field.id),
   548→            seasons=tuple(seasons),
   549→            candidates=assessments,
   550→            knowledge=knowledge,
   551→            farm=farm,
   552→            water_allocation_m3=water_allocation,
   553→            baseline_assignment=baseline_assignment,
   554→            synthetic_evidence=farm.synthetic_evidence_in_use,
   555→        )
   556→        run.knowledge_pack_refs = [dict(r) for r in knowledge.pack_refs]
   557→
   558→        solutions: list[StrategySolution] = []
   559→        infeasible_profiles: list[dict[str, Any]] = []
   560→        baseline = optimizer_mod.build_baseline_solution(problem)
   561→        solutions.append(baseline)
   562→
   563→        for profile in requested_profiles:
   564→            weights = (
   565→                {k: float(v) for k, v in custom_weights.items()}
   566→                if profile == "custom"
   567→                else dict(optimizer_mod.PROFILE_WEIGHTS[profile])
   568→            )
   569→            result = optimizer_mod.optimize_strategy(problem, profile, weights)
   570→            if isinstance(result, optimizer_mod.InfeasibleResult):
   571→                infeasible_profiles.append({
   572→                    "strategy_key": profile,
   573→                    "conflicts": result.conflicts,
   574→                    "notes": result.notes,
   575→                })
   576→                continue
   577→            solutions.append(result)
   578→
   579→        # evaluate + normalize + objective + pareto
   580→        evaluated: list[StrategySolution] = []
   581→        for sol in solutions:
   582→            criteria = objectives_mod.evaluate_criteria(problem, dict(sol.assignment))
   583→            checks = constraints_mod.check_all(problem, dict(sol.assignment))
   584→            evaluated.append(
   585→                StrategySolution(
   586→                    strategy_key=sol.strategy_key,
   587→                    strategy_kind=sol.strategy_kind,
   588→                    is_baseline=sol.is_baseline,
   589→                    profile_weights=sol.profile_weights,
   590→                    weights_effective=sol.weights_effective,
   591→                    skipped_weights=sol.skipped_weights,
   592→                    assignment=sol.assignment,
   593→                    objective_value=sol.objective_value,
   594→                    criteria=criteria,
   595→                    constraints_checked=tuple(checks),
   596→                    evidence_coverage=None,
   597→                    solver_status=sol.solver_status,
   598→                    solver_wall_time_ms=sol.solver_wall_time_ms,
   599→                )
   600→            )
   601→        objectives_mod.normalize_within_set(evaluated)
   602→        final: list[StrategySolution] = []
   603→        for sol in evaluated:
   604→            objective, effective, skipped = objectives_mod.weighted_objective(sol)
   605→            final.append(
   606→                StrategySolution(
   607→                    strategy_key=sol.strategy_key,
   608→                    strategy_kind=sol.strategy_kind,
   609→                    is_baseline=sol.is_baseline,
   610→                    profile_weights=sol.profile_weights,
   611→                    weights_effective=effective,
   612→                    skipped_weights=skipped,
   613→                    assignment=sol.assignment,
   614→                    objective_value=objective,
   615→                    criteria=sol.criteria,
   616→                    constraints_checked=sol.constraints_checked,
   617→                    evidence_coverage=objectives_mod.evidence_coverage_of(sol),
   618→                    solver_status=sol.solver_status,
   619→                    solver_wall_time_ms=sol.solver_wall_time_ms,
   620→                )
   621→            )
   622→        objectives_mod.compute_pareto(final)
   623→        solutions = final
   624→
   625→        # persist strategies
   626→        not_evaluated: dict[str, Any] = {}
   627→        for season_label, season_assessments in assessments.items():
   628→            unevaluated = {
   629→                code: [d["code"] for d in a.unevaluated_dimensions]
   630→                for code, a in season_assessments.items()
   631→                if a.status == "unevaluated" or a.unevaluated_dimensions
   632→            }
   633→            if unevaluated:
   634→                not_evaluated[season_label] = unevaluated
   635→
   636→        run.not_evaluated = not_evaluated
   637→        run.warnings = warnings
   638→        solver_status = None
   639→        optimized_solutions = [s for s in solutions if not s.is_baseline]
   640→        if infeasible_profiles and not optimized_solutions:
   641→            solver_status = "infeasible"
   642→        elif infeasible_profiles:
   643→            solver_status = "feasible"
   644→        elif optimized_solutions:
   645→            solver_status = "optimal"
   646→        else:
   647→            solver_status = "unknown"
   648→        run.solver_status = solver_status
   649→        run.infeasibility_details = {
   650→            "profiles": infeasible_profiles,
   651→            "note": (
   652→                "Infeasible profiles are reported explicitly; no hard constraint was "
   653→                "relaxed to produce the other strategies."
   654→            )
   655→            if infeasible_profiles
   656→            else {},
   657→        }
   658→
   659→        for sol in solutions:
   660→            explanations = build_explanations(problem, sol, solutions)
   661→            explanations["_weights_effective"] = sol.weights_effective
   662→            explanations["_skipped_weights"] = sol.skipped_weights
   663→            db.add(
   664→                dbm.RotationStrategy(
   665→                    run_id=run.id,
   666→                    strategy_key=sol.strategy_key,
   667→                    strategy_kind=sol.strategy_kind,
   668→                    is_baseline=sol.is_baseline,
   669→                    profile_weights=sol.profile_weights,
   670→                    assignment=[
   671→                        {"season_label": lbl, "crop_code": cc} for lbl, cc in sol.assignment
   672→                    ],
   673→                    objective_value=sol.objective_value,
   674→                    criteria=[
   675→                        {
   676→                            "criterion_code": c.criterion_code,
   677→                            "raw_value": c.raw_value,
   678→                            "unit": c.unit,
   679→                            "direction": c.direction,
   680→                            "normalized": c.normalized,
   681→                            "evaluable": c.evaluable,
   682→                            "missing_evidence": list(c.missing_evidence),
   683→                            "derivation": c.derivation,
   684→                        }
   685→                        for c in sol.criteria
   686→                    ],
   687→                    constraints_checked=[
   688→                        {
   689→                            "constraint_code": c.constraint_code,
   690→                            "status": c.status,
   691→                            "description": c.description,
   692→                            "detail": c.detail,
   693→                            "rule_codes": list(c.rule_codes),
   694→                            "source_refs": list(c.source_refs),
   695→                        }
   696→                        for c in sol.constraints_checked
   697→                    ],
   698→                    pareto={
   699→                        "on_front": sol.on_pareto_front,
   700→                        "dominated_by": list(sol.dominated_by),
   701→                        "dominates": list(sol.dominates),
   702→                    },
   703→                    explanations=explanations,
   704→                    evidence_coverage=sol.evidence_coverage,
   705→                    checksum_sha256=_solution_checksum(sol, problem),
   706→                )
   707→            )
   708→        run.status = "succeeded"
   709→        run.completed_at = datetime.now(UTC)
   710→        db.flush()
   711→    except ValidationFailed:
   712→        run.status = "failed"
   713→        run.error_summary = "validation failed"
   714→        run.completed_at = datetime.now(UTC)
   715→        raise
   716→    except Exception as exc:  # noqa: BLE001 — job failure accounting
   717→        run.status = "failed"
   718→        run.error_summary = f"{type(exc).__name__}: {exc}"
   719→        run.completed_at = datetime.now(UTC)
   720→        db.flush()
   721→        raise
   722→    return run
