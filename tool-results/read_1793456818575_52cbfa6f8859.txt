   125→function ProviderStatusDot({ provider }: { provider: ProviderHealth }) {
   126→  const probe = provider.probe?.status
   127→  let dotClass = 'bg-emerald-600'
   128→  let label = 'available'
   129→  if (provider.status === 'optional' || provider.status === 'degraded') {
   130→    dotClass = 'bg-amber-500'
   131→    label = provider.status
   132→  }
   133→  if (probe === 'unreachable') {
   134→    dotClass = 'bg-red-500'
   135→    label = 'unreachable right now'
   136→  } else if (probe === 'degraded') {
   137→    dotClass = 'bg-amber-500'
   138→    label = 'slow / degraded'
   139→  } else if (probe === 'ok') {
   140→    dotClass = 'bg-emerald-600'
   141→    label = `live (${provider.probe?.latency_ms ?? '?'} ms)`
   142→  }
   143→  return (
   144→    <span className="inline-flex items-center gap-1.5 text-[11px] text-stone-500">
   145→      <span className={`inline-block h-2 w-2 rounded-full ${dotClass}`} aria-hidden />
   146→      {label}
   147→    </span>
   148→  )
   149→}

   151→/** Provenance label for non-synthetic rows, derived from observation metadata. */
   152→function ExternalProvenanceBadge({ observation }: { observation: Observation }) {
   153→  const meta = observation.metadata as Record<string, unknown> | null
   154→  let label = 'external'
   155→  if (meta && typeof meta === 'object') {
   156→    if (meta.import_kind === 'observations_csv') label = 'CSV import'
   157→    else if (meta.import_kind === 'chirps_geotiff_extract') label = 'CHIRPS raster'
   158→    else if (typeof meta.model === 'string') label = `reanalysis · ${meta.model}`
   159→  }
   160→  return (
   161→    <Badge variant="outline" className="border-stone-300 text-[10px] text-stone-500">
   162→      {label}
   163→    </Badge>
   164→  )
   165→}

   209→export function Observations() {
   210→  const queryClient = useQueryClient()
   211→  const selectedField = useFieldShiftUi((s) => s.selectedField)
   212→  const setSelectedField = useFieldShiftUi((s) => s.setSelectedField)
   213→
   214→  const catalog = useCatalog()
   215→  const providersHealth = useProvidersHealth()
   216→
   217→  // ---- provider / ingestion form state (job + errors remember their field) ----
   218→  const [provider, setProvider] = useState<JobProviderCode>('synthetic')
   219→  const [model, setModel] = useState('era5_land')
   220→  const [selectedVariables, setSelectedVariables] = useState<string[]>(['precipitation'])
   221→  const [profile, setProfile] = useState<SyntheticProfile>('mediterranean_dry_summer')
   222→  const [seed, setSeed] = useState(42)
   223→  const [jobStart, setJobStart] = useState(DEFAULT_START)
   224→  const [jobEnd, setJobEnd] = useState(DEFAULT_END)
   225→  const [formError, setFormError] = useState<{ fieldId: string; message: string } | null>(null)
   226→  const [job, setJob] = useState<{ fieldId: string; data: DataJob } | null>(null)
   227→
   228→  // ---- file import state ----
   229→  const csvInputRef = useRef<HTMLInputElement>(null)
   230→  const [csvFile, setCsvFile] = useState<File | null>(null)
   231→  const [csvJob, setCsvJob] = useState<{ fieldId: string; data: DataJob } | null>(null)
   232→  const chirpsInputRef = useRef<HTMLInputElement>(null)
   233→  const [chirpsFile, setChirpsFile] = useState<File | null>(null)
   234→  const [chirpsDate, setChirpsDate] = useState('2024-01-31')
   235→  const [chirpsKind, setChirpsKind] = useState<'daily' | 'pentad' | 'monthly'>('daily')
   236→  const [chirpsStatus, setChirpsStatus] = useState<'preliminary' | 'final'>('final')
   237→  const [chirpsJob, setChirpsJob] = useState<{ fieldId: string; data: DataJob } | null>(null)
   238→
   239→  // ---- loaded observation window ----
   240→  const [windowStart, setWindowStart] = useState(DEFAULT_START)
   241→  const [windowEnd, setWindowEnd] = useState(DEFAULT_END)
   242→  const [appliedWindow, setAppliedWindow] = useState({ start: DEFAULT_START, end: DEFAULT_END })
   243→  const [tableVariable, setTableVariable] = useState<string>('all')
   244→
   245→  const observationsQuery = useObservations(
   246→    selectedField?.id ?? null,
   247→    appliedWindow.start,
   248→    appliedWindow.end,
   249→    Boolean(selectedField),
   250→  )
   251→  const coverageQuery = useDataCoverage(selectedField?.id ?? null, Boolean(selectedField))
   252→  const items = useMemo(() => observationsQuery.data?.items ?? [], [observationsQuery.data])
   253→
   254→  const visibleJob = job && selectedField && job.fieldId === selectedField.id ? job.data : null
   255→  const visibleFormError =
   256→    formError && selectedField && formError.fieldId === selectedField.id ? formError.message : null
   257→  const visibleCsvJob = csvJob && selectedField && csvJob.fieldId === selectedField.id ? csvJob.data : null
   258→  const visibleChirpsJob =
   259→    chirpsJob && selectedField && chirpsJob.fieldId === selectedField.id ? chirpsJob.data : null
   260→
   261→  const syntheticProvider = providersHealth.data?.providers.find((p) => p.provider_code === 'synthetic')
   262→  const openMeteoProvider = providersHealth.data?.providers.find((p) => p.provider_code === 'open_meteo')
   263→  const chirpsProvider = providersHealth.data?.providers.find((p) => p.provider_code === 'chirps_import')
   264→
   265→  const activeProvider = provider === 'synthetic' ? syntheticProvider : openMeteoProvider
   266→  const models = openMeteoProvider?.details.models ?? []
   267→  const activeModel = models.find((m) => m.code === model)
   268→
   269→  /** Variables available for the current provider (+ model for open_meteo). */
   270→  const selectableVariables = useMemo(() => {
   271→    const all = catalog.data?.variables ?? []
   272→    if (provider === 'open_meteo') {
   273→      const modelVars = new Set(activeModel?.variables ?? [])
   274→      return all.filter((v) => modelVars.has(v.code))
   275→    }
   276→    const syntheticVars = new Set(syntheticProvider?.details.supported_variables ?? [])
   277→    if (syntheticVars.size === 0) return all
   278→    return all.filter((v) => syntheticVars.has(v.code))
   279→  }, [catalog.data, provider, activeModel, syntheticProvider])
   280→
   281→  // Keep the effective selection inside the currently selectable set; default to
   282→  // precipitation, else the first selectable variable. Derived (not synced) so
   283→  // provider/model switches never wedge.
   284→  const effectiveSelectedVariables = useMemo(() => {
   285→    const codes = new Set(selectableVariables.map((v) => v.code))
   286→    const kept = selectedVariables.filter((code) => codes.has(code))
   287→    if (kept.length > 0) return kept
   288→    if (codes.has('precipitation')) return ['precipitation']
   289→    return selectableVariables.length > 0 ? [selectableVariables[0].code] : []
   290→  }, [selectableVariables, selectedVariables])
   291→
   292→  const invalidateFieldData = (fieldId: string) => {
   293→    queryClient.invalidateQueries({ queryKey: ['observations', fieldId] })
   294→    queryClient.invalidateQueries({ queryKey: ['data-passport', fieldId] })
   295→    queryClient.invalidateQueries({ queryKey: ['data-coverage', fieldId] })
   296→    queryClient.invalidateQueries({ queryKey: ['fields'] })
   297→  }
   298→
   299→  const variableName = (code: string) =>
   300→    catalog.data?.variables.find((v) => v.code === code)?.name ?? humanizeCode(code)
   301→
   302→  const ingestMutation = useMutation({
   303→    mutationFn: (fieldId: string) =>
   304→      createDataJob(fieldId, {
   305→        provider_code: provider,
   306→        variables: effectiveSelectedVariables,
   307→        start_date: jobStart,
   308→        end_date: jobEnd,
   309→        ...(provider === 'synthetic' ? { profile, seed } : { model }),
   310→      }),
   311→    onSuccess: (jobResult) => {
   312→      if (!selectedField) return
   313→      setJob({ fieldId: selectedField.id, data: jobResult })
   314→      if (jobResult.status === 'succeeded') {
   315→        toast.success('Ingestion job succeeded', {
   316→          description: `${jobResult.records_received} records stored · ${jobResult.records_rejected} rejected.`,
   317→        })
   318→        const params = jobResult.request_parameters as { start_date?: string; end_date?: string }
   319→        const start = params.start_date ?? jobStart
   320→        const end = params.end_date ?? jobEnd
   321→        setWindowStart(start)
   322→        setWindowEnd(end)
   323→        setAppliedWindow({ start, end })
   324→        if (selectedField) {
   325→          invalidateFieldData(selectedField.id)
   326→        }
   327→      } else {
   328→        toast.error('Ingestion job failed', { description: jobResult.error_summary ?? 'No error summary returned.' })
   329→      }
   330→    },
   331→    onError: (err) => {
   332→      const { title, description } = describeError(err)
   333→      if (selectedField) setFormError({ fieldId: selectedField.id, message: title })
   334→      toast.error(title, { description })
   335→    },
   336→  })
   337→
   338→  const csvImportMutation = useMutation({
   339→    mutationFn: (fieldId: string) => importObservationsCsv(fieldId, csvFile as File),
   340→    onSuccess: (jobResult) => {
   341→      if (!selectedField) return
   342→      setCsvJob({ fieldId: selectedField.id, data: jobResult })
   343→      if (jobResult.status === 'succeeded') {
   344→        toast.success('CSV import finished', {
   345→          description: `${jobResult.records_received} rows read · ${jobResult.records_rejected} rejected.`,
   346→        })
   347→        invalidateFieldData(selectedField.id)
   348→        setCsvFile(null)
   349→        if (csvInputRef.current) csvInputRef.current.value = ''
   350→      } else {
   351→        toast.error('CSV import failed', { description: jobResult.error_summary ?? undefined })
   352→      }
   353→    },
   354→    onError: (err) => {
   355→      const { title, description } = describeError(err)
   356→      toast.error(title, { description })
   357→    },
   358→  })
   359→
   360→  const chirpsMutation = useMutation({
   361→    mutationFn: (fieldId: string) =>
   362→      extractChirps(fieldId, chirpsFile as File, {
   363→        observedAt: chirpsDate,
   364→        periodKind: chirpsKind,
   365→        productStatus: chirpsStatus,
   366→      }),
   367→    onSuccess: (jobResult) => {
   368→      if (!selectedField) return
   369→      setChirpsJob({ fieldId: selectedField.id, data: jobResult })
   370→      if (jobResult.status === 'succeeded') {
   371→        toast.success('CHIRPS extraction finished', {
   372→          description: `${jobResult.records_received} observation(s) extracted from the raster.`,
   373→        })
   374→        invalidateFieldData(selectedField.id)
   375→        setChirpsFile(null)
   376→        if (chirpsInputRef.current) chirpsInputRef.current.value = ''
   377→      } else {
   378→        toast.error('CHIRPS extraction failed', { description: jobResult.error_summary ?? undefined })
   379→      }
   380→    },
   381→    onError: (err) => {
   382→      const { title, description } = describeError(err)
   383→      toast.error(title, { description })
   384→    },
   385→  })
   386→
   387→  const submitJob = () => {
   388→    setFormError(null)
   389→    if (!selectedField) return
   390→    if (effectiveSelectedVariables.length === 0) {
   391→      setFormError({ fieldId: selectedField.id, message: 'Pick at least one variable to ingest.' })
   392→      return
   393→    }
   394→    const windowProblem =
   395→      provider === 'open_meteo'
   396→        ? validateJobWindow(jobStart, jobEnd, {
   397→            min: '1940-01-01',
   398→            max: new Date(Date.now() - 6 * 86_400_000).toISOString().slice(0, 10),
   399→            maxYears: 10,
   400→          })
   401→        : validateJobWindow(jobStart, jobEnd)
   402→    if (windowProblem) {
   403→      setFormError({ fieldId: selectedField.id, message: windowProblem })
   404→      return
   405→    }
   406→    ingestMutation.mutate(selectedField.id)
   407→  }
   408→
   409→  const loadWindow = () => {
   410→    const problem = validateJobWindow(windowStart, windowEnd)
   411→    if (problem) {
   412→      toast.error(problem)
   413→      return
   414→    }
   415→    setAppliedWindow({ start: windowStart, end: windowEnd })
   416→  }
   417→
   418→  // ---- derived table data ----
   419→  const presentVariables = useMemo(
   420→    () => Array.from(new Set(items.map((item) => item.variable_code))).sort(),
   421→    [items],
   422→  )
   423→  // If the selected filter variable is not present in the loaded window, fall back to "all".
   424→  const effectiveTableVariable =
   425→    tableVariable !== 'all' && !presentVariables.includes(tableVariable) ? 'all' : tableVariable
   426→  const filteredItems = useMemo(
   427→    () =>
   428→      (effectiveTableVariable === 'all'
   429→        ? items
   430→        : items.filter((item) => item.variable_code === effectiveTableVariable)
   431→      )
   432→        .slice()
   433→        .sort((a, b) => (a.observed_at < b.observed_at ? 1 : a.observed_at > b.observed_at ? -1 : 0)),
   434→    [items, effectiveTableVariable],
   435→  )
   436→
   437→  const tempSeries = useMemo(
   438→    () =>
   439→      items
   440→        .filter((o) => o.variable_code === 'air_temperature_mean' && o.value !== null)
   441→        .slice()
   442→        .sort((a, b) => (a.observed_at < b.observed_at ? -1 : 1))
   443→        .map((o) => ({ date: o.observed_at.slice(0, 10), value: o.value })),
   444→    [items],
   445→  )
   446→  const precipSeries = useMemo(
   447→    () =>
   448→      items
   449→        .filter((o) => o.variable_code === 'precipitation' && o.value !== null)
   450→        .slice()
   451→        .sort((a, b) => (a.observed_at < b.observed_at ? -1 : 1))
   452→        .map((o) => ({ date: o.observed_at.slice(0, 10), value: o.value })),
   453→    [items],
   454→  )
   455→
   456→  // ------------------------------------------------------------------ render
   457→
   458→  if (!selectedField) {
   459→    return (
   460→      <div className="space-y-4" aria-label="Observations">
   461→        <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   462→          <div>
   463→            <h2 className="text-lg font-semibold text-stone-900">Observations</h2>
   464→            <p className="text-sm text-stone-500">
   465→              Run ingestion jobs (synthetic fixtures or the real Open-Meteo archive), import files, and inspect what
   466→              landed.
   467→            </p>
   468→          </div>
   469→          <FieldPicker
   470→            value={null}
   471→            onValueChange={setSelectedField}
   472→            emptyHint={
   473→              <div className="flex items-center gap-2 text-sm text-stone-500">
   474→                <span>Create a farm and field first.</span>
   475→                <GoToFarmsButton label="Go to Farms & Fields" />
   476→              </div>
   477→            }
   478→          />
   479→        </div>
   480→        <Card className="border-dashed border-stone-300 bg-white">
   481→          <CardContent className="flex flex-col items-center gap-3 p-8 text-center">
   482→            <Database className="h-8 w-8 text-stone-400" aria-hidden />
   483→            <p className="text-sm text-stone-500">Pick a field above to ingest and inspect its observation history.</p>
   484→          </CardContent>
   485→        </Card>
   486→      </div>
   487→    )
   488→  }
   489→
   490→  return (
   491→    <div className="space-y-6" aria-label="Observations">
   492→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   493→        <div>
   494→          <h2 className="text-lg font-semibold text-stone-900">Observations</h2>
   495→          <p className="text-sm text-stone-500">
   496→            Run ingestion jobs (synthetic fixtures or the real Open-Meteo archive), import files, and inspect what
   497→            landed.
   498→          </p>
   499→        </div>
   500→        <FieldPicker value={selectedField.id} onValueChange={setSelectedField} />
   501→      </div>
   502→
   503→      {catalog.isLoading ? <Skeleton className="h-4 w-64" /> : null}
   504→
   505→      <div className="grid gap-6 lg:grid-cols-2">
   506→        {/* ---------------- ingestion form ---------------- */}
   507→        <Card className="border-stone-200 bg-white shadow-sm">
   508→          <CardHeader className="pb-3">
   509→            <CardTitle className="flex items-center gap-2 text-base">
   510→              {provider === 'synthetic' ? (
   511→                <FlaskConical className="h-4 w-4 text-emerald-700" aria-hidden />
   512→              ) : (
   513→                <Globe2 className="h-4 w-4 text-emerald-700" aria-hidden />
   514→              )}
   515→              Ingest data
   516→            </CardTitle>
   517→            <CardDescription>
   518→              {provider === 'synthetic'
   519→                ? 'Software fixtures for testing — always labelled SYNTHETIC downstream.'
   520→                : 'Real historical reanalysis (gridded model estimates, never field sensors). Open-Meteo data is CC BY 4.0.'}
