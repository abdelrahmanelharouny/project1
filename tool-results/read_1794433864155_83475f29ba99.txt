    62→_BIN_OPS = {
    63→    sa_ops.eq: operator.eq,
    64→    sa_ops.ne: operator.ne,
    65→    sa_ops.lt: operator.lt,
    66→    sa_ops.le: operator.le,
    67→    sa_ops.gt: operator.gt,
    68→    sa_ops.ge: operator.ge,

    72→def _clause_matches(obj, clause, table_name=None) -> bool:
    73→    """Tiny WHERE evaluator: simple comparisons/IN/IS on the primary table.
    74→    Conditions referencing a JOINed table (e.g. endpoint status in the retry
    75→    sweep) are skipped — they cannot be evaluated against the row."""
    76→    if clause is None:
    77→        return True
    78→    if isinstance(clause, BooleanClauseList):
    79→        return all(_clause_matches(obj, c, table_name) for c in clause.clauses)
    80→    if isinstance(clause, BinaryExpression):
    81→        name = getattr(clause.left, "key", None)
    82→        if name is None:
    83→            return True
    84→        if table_name is not None:
    85→            left_table = getattr(getattr(clause.left, "table", None), "name", None)
    86→            if left_table is not None and left_table != table_name:
    87→                return True
    88→        actual = getattr(obj, name, None)
    89→        value = getattr(clause.right, "value", None)
    90→        op = clause.operator
    91→        if op is sa_ops.in_op:
    92→            values = value if isinstance(value, (list, tuple, set)) else []
    93→            return actual in values
    94→        if op is sa_ops.is_:
    95→            return actual is value
    96→        if op is sa_ops.isnot:
    97→            return actual is not value
    98→        fn = _BIN_OPS.get(op)
    99→        if fn is None:
   100→            return True
   101→        try:
   102→            return bool(fn(actual, value))
   103→        except TypeError:
   104→            return True
   105→    return True

   122→class FakeSession:
   123→    def __init__(self):
   124→        self.objects: list = []
   125→        self.commits = 0
   126→
   127→    def add(self, obj):
   128→        if getattr(obj, "id", None) is None and hasattr(type(obj), "id"):
   129→            obj.id = uuid.uuid4()
   130→        self.objects.append(obj)
   131→
   132→    def flush(self):
   133→        seen = set()
   134→        for obj in self.objects:
   135→            if isinstance(obj, WebhookDelivery):
   136→                k = (obj.endpoint_id, obj.event_id)
   137→                if k in seen:
   138→                    raise IntegrityError("uq_webhook_delivery_event", {}, Exception("dup"))
   139→                seen.add(k)
   140→
   141→    def begin_nested(self):
   142→        return _FakeNested(self)
   143→
   144→    def commit(self):
   145→        self.commits += 1
   146→
   147→    def delete(self, obj):
   148→        if obj in self.objects:
   149→            self.objects.remove(obj)
   150→
   151→    def get(self, entity, pk):
   152→        for obj in self.objects:
   153→            if isinstance(obj, entity) and getattr(obj, "id", None) == pk:
   154→                return obj
   155→        return None
   156→
   157→    def _matching(self, stmt):
   158→        entity = _stmt_entity(stmt)
   159→        if entity is None:
   160→            return []
   161→        table_name = getattr(getattr(entity, "__table__", None), "name", None)
   162→        return [
   163→            o for o in self.objects
   164→            if isinstance(o, entity) and _clause_matches(o, stmt.whereclause, table_name)
   165→        ]
   166→
   167→    def scalar(self, stmt):
   168→        if _stmt_entity(stmt) is None:  # e.g. select(func.count()) → zero
   169→            return 0
   170→        rows = self._matching(stmt)
   171→        return rows[0] if rows else None
   172→
   173→    def scalars(self, stmt):
   174→        return FakeResult(self._matching(stmt))
   175→
   176→    def execute(self, stmt):
   177→        return FakeExecResult(self._matching(stmt))
   178→

   183→
   184→    def scalars(self):

   222→def make_delivery(session, endpoint, *, event="farm.updated", event_id="evt-1",
   223→                  status="pending", attempt=0) -> WebhookDelivery:
   224→    delivery = WebhookDelivery(
   225→        endpoint_id=endpoint.id,

   230→            "id": str(uuid.uuid4()),

   237→        status=status,

   254→def test_sign_and_verify_roundtrip():
   255→    body = b'{"event":"farm.updated"}'
   256→    header = webhooks.sign_payload(SECRET, body, 1_700_000_000)
   257→    assert header.startswith("t=1700000000,v1=")
   258→    mac = header.split("v1=", 1)[1]
   259→    expected = hmac_mod.new(
   260→        SECRET.encode(), b"1700000000." + body, hashlib.sha256
   261→    ).hexdigest()
   262→    assert mac == expected
   263→    assert webhooks.verify_signature(
   264→        SECRET, body, header, tolerance_seconds=300,
   265→        now=datetime.fromtimestamp(1_700_000_000, tz=UTC),
   266→    )
