     1→"""Feature-run orchestration (Milestone 4).
     2→
     3→A feature run is a reproducible computation: the run row records the resolved
     4→source selection (ONE dataset product per variable — synthetic and real data
     5→are never blended), the calendar basis, the window, the engine and catalog
     6→versions, and every warning. Values are written once by the run; comparison
     7→across runs is the supported way to observe change.
     8→
     9→Honesty rules enforced here:
    10→- Source selection never mixes providers for the same variable inside a run.
    11→- Runs over synthetic data are flagged `is_synthetic` end to end.
    12→- Missing source variables drop the dependent features with an explicit
    13→  reason — they are never silently skipped nor computed from substitutes.
    14→- Crop-season windows are user-supplied; no crop calendar knowledge is
    15→  invented here (the sourced knowledge base arrives in a later milestone).
    16→"""
    17→
    18→import logging
    19→import uuid
    20→from datetime import UTC, date, datetime, timedelta
    21→from typing import Any
    22→
    23→from sqlalchemy import Boolean, Integer, func, select
    24→from sqlalchemy.exc import IntegrityError
    25→from sqlalchemy.orm import Session
    26→
    27→from app.core.errors import InsufficientData, NotFound, ValidationFailed
    28→from app.data_platform.contracts.variables import VARIABLE_CATALOG
    29→from app.db.models import (
    30→    DatasetProduct,
    31→    DataSource,
    32→    FeatureDefinition,
    33→    FeatureRun,
    34→    FeatureValue,
    35→    Field,
    36→    Observation,
    37→)
    38→from app.features import engine as fengine
    39→from app.features.calendar import resolve_timezone
    40→from app.features.catalog import (
    41→    FEATURE_CATALOG,
    42→    FEATURE_CATALOG_VERSION,
    43→    FEATURE_ENGINE_VERSION,
    44→)
    45→
    46→logger = logging.getLogger("fieldshift.features")
    47→
    48→ALLOWED_AGGREGATION_LEVELS = ("daily", "monthly", "seasonal", "crop_season")
    49→MAX_WINDOW_YEARS = 31
    50→MAX_DAILY_WINDOW_DAYS = 1100  # ~3 years, protects the daily-value volume
    51→MAX_CROP_SEASONS = 20

    54→def ensure_feature_definitions(db: Session) -> None:
    55→    """Seed the versioned feature definitions table from the code catalog."""
    56→    from app.features.catalog import catalog_payload
    57→
    58→    existing = {
    59→        (row.feature_code, row.definition_version)
    60→        for row in db.scalars(
    61→            select(FeatureDefinition).where(
    62→                FeatureDefinition.definition_version == FEATURE_CATALOG_VERSION
    63→            )
    64→        ).all()
    65→    }
    66→    created = False
    67→    for item in catalog_payload():
    68→        key = (item["feature_code"], item["definition_version"])
    69→        if key in existing:
    70→            continue
    71→        db.add(
    72→            FeatureDefinition(
    73→                feature_code=item["feature_code"],
    74→                name=item["name"],
    75→                description=item["description"],
    76→                unit=item["unit"],
    77→                category=item["category"],
    78→                aggregation_levels=item["aggregation_levels"],
    79→                required_variables=item["required_variables"],
    80→                min_coverage_ratio=item["min_coverage_ratio"],
    81→                definition_version=item["definition_version"],
    82→                formula_reference=item["formula_reference"],
    83→                is_active=True,
    84→            )
    85→        )
    86→        created = True
    87→    if created:
    88→        db.commit()

    91→def resolve_source_selection(
    92→    db: Session,
    93→    *,
    94→    field_id: uuid.UUID,
    95→    variables: list[str],
    96→    overrides: dict[str, str] | None = None,
    97→    window_start: datetime | None = None,
    98→    window_end: datetime | None = None,
    99→) -> tuple[dict[str, dict[str, Any]], list[str], bool]:
   100→    """Pick ONE dataset product per variable for the run.
   101→
   102→    Preference (window-aware): real (non-synthetic) products first, then the
   103→    product with the most valid observations INSIDE the run window, then the
   104→    best overall valid ratio, then the most recent retrieval. A real product
   105→    with no data in the window is therefore not preferred over a synthetic
   106→    fixture that actually covers the window — the fallback is warned about and
   107→    the run is flagged `is_synthetic`. Overrides pin a specific product and
   108→    are validated against the field's actual observations.
   109→    """
   110→    overrides = overrides or {}
   111→    stmt = (
   112→        select(
   113→            Observation.variable_code,
   114→            Observation.dataset_product_id,
   115→            func.count().label("n"),
   116→            func.sum(func.cast(Observation.value.is_not(None), Integer)).label("valid_n"),
   117→            func.max(Observation.retrieved_at).label("last_retrieval"),
   118→            func.cast(func.bool_or(Observation.is_synthetic), Boolean).label("has_synthetic"),
   119→        )
   120→        .where(Observation.field_id == field_id, Observation.variable_code.in_(variables))
   121→        .group_by(Observation.variable_code, Observation.dataset_product_id)
   122→    )
   123→    if window_start is not None and window_end is not None:
   124→        # in-window counts drive the preference; keep a margin of one day on
   125→        # both sides for daily periods that straddle the window boundary
   126→        margin = timedelta(days=1)
   127→        stmt = stmt.where(
   128→            Observation.observed_at >= window_start - margin,
   129→            Observation.observed_at <= window_end + margin,
   130→        )
   131→    rows = db.execute(stmt).all()
   132→
   133→    if not rows:
   134→        raise InsufficientData(
   135→            "No observations are available for the requested variables on this field.",
   136→            details=[{"field": "variables", "reason": ", ".join(sorted(variables))}],
   137→        )
   138→
   139→    products = {
   140→        p.id: p
   141→        for p in db.scalars(
   142→            select(DatasetProduct).where(
   143→                DatasetProduct.id.in_([r.dataset_product_id for r in rows])
   144→            )
   145→        ).all()
   146→    }
   147→
   148→    grouped: dict[str, list[dict[str, Any]]] = {}
   149→    for r in rows:
   150→        product = products.get(r.dataset_product_id)
   151→        if product is None:
   152→            continue
   153→        meta = product.metadata or {}
   154→        grouped.setdefault(r.variable_code, []).append(
   155→            {
   156→                "dataset_product_id": str(product.id),
   157→                "product_code": product.product_code,
   158→                "product_version": product.product_version,
   159→                "is_synthetic": bool(meta.get("is_synthetic")) or bool(r.has_synthetic),
   160→                "n": int(r.n),
   161→                "valid_n": int(r.valid_n or 0),
   162→                "last_retrieval": r.last_retrieval,
   163→            }
   164→        )
   165→
   166→    selection: dict[str, dict[str, Any]] = {}
   167→    warnings: list[str] = []
   168→    for variable in variables:
   169→        candidates = grouped.get(variable) or []
   170→        if not candidates:
   171→            continue
   172→        override_id = overrides.get(variable)
   173→        if override_id:
   174→            chosen = next(
   175→                (c for c in candidates if c["dataset_product_id"] == str(override_id)), None
   176→            )
   177→            if chosen is None:
   178→                raise ValidationFailed(
   179→                    f"Source override for '{variable}' does not match any product with "
   180→                    "observations on this field.",
   181→                    details=[{"field": "source_overrides", "reason": variable}],
   182→                )
   183→        else:
   184→            chosen = sorted(
   185→                candidates,
   186→                key=lambda c: (
   187→                    c["is_synthetic"],
   188→                    -(c["valid_n"]),
   189→                    -(c["valid_n"] / max(1, c["n"])),
   190→                    -(c["last_retrieval"].timestamp() if c["last_retrieval"] else 0),
   191→                ),
   192→            )[0]
   193→        selection[variable] = {
   194→            "dataset_product_id": chosen["dataset_product_id"],
   195→            "product_code": chosen["product_code"],
   196→            "product_version": chosen["product_version"],
   197→            "is_synthetic": chosen["is_synthetic"],
   198→        }
   199→        skipped = [c for c in candidates if c["dataset_product_id"] != chosen["dataset_product_id"]]
   200→        for c in skipped:
   201→            if chosen["is_synthetic"] and not c["is_synthetic"]:
   202→                warnings.append(
   203→                    f"{variable}: real product '{c['product_code']}' ({c['product_version']}) has no "
   204→                    "observations inside the run window; the synthetic fixture was selected instead "
   205→                    "(the run is flagged synthetic). Products are never blended."
   206→                )
   207→            else:
   208→                warnings.append(
   209→                    f"{variable}: observations from product '{c['product_code']}' "
   210→                    f"({c['product_version']}) were not selected for this run; one source per "
   211→                    "variable is used so products are never blended."
   212→                )
   213→
   214→    is_synthetic = any(s["is_synthetic"] for s in selection.values())
   215→    return selection, warnings, is_synthetic

   218→def start_feature_run(
   219→    db: Session,
   220→    *,
   221→    field: Field,
   222→    payload: dict[str, Any],
   223→    created_by: Any = None,
   224→) -> FeatureRun:
   225→    """Validate the request, create (or return) the run, execute it inline."""
   226→    ensure_feature_definitions(db)
   227→
   228→    idempotency_key = payload.get("idempotency_key")
   229→    if idempotency_key:
   230→        # Deepening F-3: the replay lookup is scoped to THIS field — a global
   231→        # key lookup could return another tenant's run if a key collided or
   232→        # was guessed (cross-tenant information disclosure).
   233→        existing = db.scalar(
   234→            select(FeatureRun).where(
   235→                FeatureRun.idempotency_key == idempotency_key,
   236→                FeatureRun.field_id == field.id,
   237→            )
   238→        )
   239→        if existing is not None:
   240→            return existing
   241→
   242→    start_date: date = payload["start_date"]
   243→    end_date: date = payload["end_date"]
   244→    if start_date > end_date:
   245→        raise ValidationFailed(
   246→            "start_date must be on or before end_date.",
   247→            details=[{"field": "start_date", "reason": str(start_date)}],
   248→        )
   249→    if (end_date - start_date).days > MAX_WINDOW_YEARS * 366:
   250→        raise ValidationFailed(
   251→            f"Feature run window exceeds {MAX_WINDOW_YEARS} years.",
   252→            details=[{"field": "end_date", "reason": "window too long"}],
   253→        )
   254→
   255→    levels = payload.get("aggregation_levels") or ["monthly"]
   256→    unknown_levels = [l for l in levels if l not in ALLOWED_AGGREGATION_LEVELS]
   257→    if unknown_levels:
   258→        raise ValidationFailed(
   259→            "Unknown aggregation levels.",
   260→            details=[{"field": "aggregation_levels", "reason": ", ".join(unknown_levels)}],
   261→        )
   262→    if "daily" in levels and (end_date - start_date).days > MAX_DAILY_WINDOW_DAYS:
   263→        raise ValidationFailed(
   264→            f"Daily aggregation is limited to {MAX_DAILY_WINDOW_DAYS} days per run.",
   265→            details=[{"field": "end_date", "reason": "window too long for daily level"}],
   266→        )
   267→
   268→    hemisphere = payload.get("hemisphere") or "north"
   269→    if hemisphere not in ("north", "south"):
   270→        raise ValidationFailed(
   271→            "hemisphere must be 'north' or 'south'.",
   272→            details=[{"field": "hemisphere", "reason": hemisphere}],
   273→        )
   274→
   275→    calendar_timezone = payload.get("calendar_timezone") or field.farm.timezone or "UTC"
   276→    try:
   277→        tz = resolve_timezone(calendar_timezone)
   278→    except ValueError as exc:
   279→        raise ValidationFailed(str(exc), details=[{"field": "calendar_timezone"}]) from None
   280→
   281→    crop_seasons_raw = payload.get("crop_seasons") or []
   282→    if crop_seasons_raw and "crop_season" not in levels:
   283→        raise ValidationFailed(
   284→            "crop_seasons supplied but the 'crop_season' aggregation level is not requested.",
   285→            details=[{"field": "aggregation_levels", "reason": "add 'crop_season'"}],
   286→        )
   287→    if len(crop_seasons_raw) > MAX_CROP_SEASONS:
   288→        raise ValidationFailed(
   289→            f"Too many crop-season windows (max {MAX_CROP_SEASONS}).",
   290→            details=[{"field": "crop_seasons"}],
   291→        )
   292→    crop_seasons: list[dict[str, Any]] = []
   293→    seen_labels: set[str] = set()
   294→    for cs in crop_seasons_raw:
   295→        label = str(cs.get("label") or "").strip()
   296→        if not label or len(label) > 40:
   297→            raise ValidationFailed(
   298→                "Crop-season labels must be 1-40 characters.",
   299→                details=[{"field": "crop_seasons", "reason": str(cs)[:80]}],
   300→            )
   301→        if label in seen_labels:
   302→            raise ValidationFailed(
   303→                f"Duplicate crop-season label '{label}'.",
   304→                details=[{"field": "crop_seasons", "reason": label}],
   305→            )
   306→        seen_labels.add(label)
   307→        try:
   308→            cs_start = date.fromisoformat(str(cs["start_date"]))
   309→            cs_end = date.fromisoformat(str(cs["end_date"]))
   310→        except (KeyError, ValueError):
   311→            raise ValidationFailed(
   312→                "Crop seasons need start_date and end_date (ISO dates).",
   313→                details=[{"field": "crop_seasons", "reason": label}],
   314→            ) from None
   315→        if cs_start > cs_end:
   316→            raise ValidationFailed(
   317→                f"Crop season '{label}' start_date is after end_date.",
   318→                details=[{"field": "crop_seasons", "reason": label}],
   319→            )
   320→        crop_seasons.append({"label": label, "start_date": cs_start, "end_date": cs_end})
   321→
   322→    # Feature specs to compute.
   323→    requested_codes = payload.get("feature_codes")
   324→    if requested_codes:
   325→        unknown = [c for c in requested_codes if c not in FEATURE_CATALOG]
   326→        if unknown:
   327→            raise ValidationFailed(
   328→                "Unknown feature codes.",
   329→                details=[{"field": "feature_codes", "reason": ", ".join(sorted(unknown))}],
   330→            )
   331→        specs = [FEATURE_CATALOG[c] for c in sorted(set(requested_codes))]
   332→    else:
   333→        specs = [
   334→            s
   335→            for s in FEATURE_CATALOG.values()
   336→            if any(level in s.aggregation_levels for level in levels)
   337→        ]
   338→    if not specs:
   339→        raise ValidationFailed(
   340→            "No features match the requested aggregation levels.",
   341→            details=[{"field": "feature_codes"}],
   342→        )
   343→
   344→    variables = payload.get("variables")
   345→    required_vars = sorted({v for spec in specs for v in spec.required_variables})
   346→    if variables:
   347→        unknown_vars = [v for v in variables if v not in VARIABLE_CATALOG]
   348→        if unknown_vars:
   349→            raise ValidationFailed(
   350→                "Unknown variable codes in request.",
   351→                details=[{"field": "variables", "reason": ", ".join(unknown_vars)}],
   352→            )
   353→        # Restrict features to what the requested variables can supply.
   354→        allowed = set(variables)
   355→        specs = [s for s in specs if set(s.required_variables) <= allowed]
   356→        if not specs:
   357→            raise ValidationFailed(
   358→                "None of the catalog features can be computed from the requested variables.",
   359→                details=[{"field": "variables", "reason": ", ".join(sorted(variables))}],
   360→            )
   361→        required_vars = sorted({v for spec in specs for v in spec.required_variables})
   362→
   363→    window_start = (
   364→        datetime(start_date.year, start_date.month, start_date.day, tzinfo=tz).astimezone(UTC)
   365→    )
   366→    window_end = (
   367→        datetime(end_date.year, end_date.month, end_date.day, tzinfo=tz) + timedelta(days=1)
   368→    ).astimezone(UTC) - timedelta(seconds=1)
   369→
   370→    overrides = payload.get("source_overrides") or {}
   371→    selection, selection_warnings, is_synthetic = resolve_source_selection(
   372→        db,
   373→        field_id=field.id,
   374→        variables=required_vars,
   375→        overrides=overrides,
   376→        window_start=window_start,
   377→        window_end=window_end,
   378→    )
   379→
   380→    missing_vars = [v for v in required_vars if v not in selection]
   381→    resolvable_specs = [s for s in specs if set(s.required_variables) <= set(selection)]
   382→    not_evaluated: dict[str, str] = {
   383→        f"variable:{v}": "Required input variable has no observations on this field."
   384→        for v in missing_vars
   385→    }
   386→    dropped_codes = sorted({s.feature_code for s in specs} - {s.feature_code for s in resolvable_specs})
   387→
   388→    request_parameters = {
   389→        "start_date": start_date.isoformat(),
   390→        "end_date": end_date.isoformat(),
   391→        "aggregation_levels": levels,
   392→        "hemisphere": hemisphere,
   393→        "calendar_timezone": calendar_timezone,
   394→        "crop_seasons": [
   395→            {"label": cs["label"], "start_date": cs["start_date"].isoformat(), "end_date": cs["end_date"].isoformat()}
   396→            for cs in crop_seasons
   397→        ],
   398→        "feature_codes": sorted(s.feature_code for s in specs),
   399→        "source_overrides": {k: str(v) for k, v in overrides.items()},
   400→    }
   401→
   402→    run = FeatureRun(
   403→        field_id=field.id,
   404→        status="queued",
   405→        calendar_timezone=calendar_timezone,
   406→        hemisphere=hemisphere,
   407→        period_start=window_start,
   408→        period_end=window_end,
   409→        aggregation_levels=levels,
   410→        requested_variables=required_vars,
   411→        source_selection=selection,
   412→        crop_seasons=request_parameters["crop_seasons"],
   413→        feature_engine_version=FEATURE_ENGINE_VERSION,
   414→        feature_catalog_version=FEATURE_CATALOG_VERSION,
   415→        is_synthetic=is_synthetic,
   416→        request_parameters=request_parameters,
   417→        warnings=list(selection_warnings),
   418→        not_evaluated_reasons=not_evaluated,
   419→        idempotency_key=idempotency_key,
   420→        created_by=created_by,
   421→    )
   422→    db.add(run)
   423→    try:
   424→        db.commit()
   425→    except IntegrityError:
   426→        db.rollback()
   427→        # Deepening F-3: race recovery is ALSO field-scoped — a duplicate-key
   428→        # collision with another field's run must never serialize that run.
   429→        if idempotency_key:
   430→            existing = db.scalar(
   431→                select(FeatureRun).where(
   432→                    FeatureRun.idempotency_key == idempotency_key,
   433→                    FeatureRun.field_id == field.id,
   434→                )
   435→            )
   436→            if existing is not None:
   437→                return existing
   438→        raise
   439→
   440→    if not resolvable_specs:
   441→        run.status = "failed"
   442→        run.error_summary = "None of the requested features have their input variables available."
   443→        run.completed_at = datetime.now(UTC)
   444→        db.commit()
   445→        return run
   446→
   447→    _execute_run(
   448→        db,
   449→        run,
   450→        specs=resolvable_specs,
   451→        tz=tz,
   452→        window_start=window_start,
   453→        window_end=window_end,
   454→        crop_seasons=crop_seasons,
   455→        not_evaluated=not_evaluated,
   456→        dropped_codes=dropped_codes,
   457→    )
   458→    return run
   459→
   460→
   461→def _execute_run(
   462→    db: Session,
   463→    run: FeatureRun,
   464→    *,
   465→    specs: list,
   466→    tz,
   467→    window_start: datetime,
   468→    window_end: datetime,
   469→    crop_seasons: list[dict[str, Any]],
   470→    not_evaluated: dict[str, str],
   471→    dropped_codes: list[str],
   472→) -> None:
   473→    run.status = "running"
   474→    run.started_at = datetime.now(UTC)
   475→    db.commit()
   476→
   477→    try:
   478→        selected_ids = [s["dataset_product_id"] for s in run.source_selection.values()]
   479→        margin = timedelta(days=2)
   480→        obs_rows = db.execute(
   481→            select(Observation)
   482→            .where(
   483→                Observation.field_id == run.field_id,
   484→                Observation.dataset_product_id.in_(selected_ids),
   485→                Observation.observed_at >= run.period_start - margin,
   486→                Observation.observed_at <= run.period_end + margin,
   487→            )
   488→            .order_by(Observation.period_start, Observation.variable_code)
   489→        ).scalars().all()
   490→
   491→        observations = [
   492→            {
   493→                "id": str(o.id),
   494→                "variable_code": o.variable_code,
   495→                "value": o.value,
   496→                "unit": o.unit,
   497→                "quality_status": o.quality_status,
   498→                "observed_at": o.observed_at,
   499→                "period_start": o.period_start,
   500→                "period_end": o.period_end,
   501→                "time_support": o.time_support,
   502→                "is_synthetic": o.is_synthetic,
   503→                "processing_version": o.processing_version,
   504→                "evidence_class": o.evidence_class,
   505→                "dataset_product_id": str(o.dataset_product_id),
   506→            }
   507→            for o in obs_rows
   508→        ]
   509→
   510→        crop_windows = [
   511→            {
   512→                "label": cs["label"],
   513→                "start": datetime(
   514→                    cs["start_date"].year, cs["start_date"].month, cs["start_date"].day, tzinfo=tz
   515→                ).astimezone(UTC),
   516→                "end": (
   517→                    datetime(
   518→                        cs["end_date"].year, cs["end_date"].month, cs["end_date"].day, tzinfo=tz
   519→                    )
   520→                    + timedelta(days=1)
   521→                ).astimezone(UTC)
   522→                - timedelta(seconds=1),
   523→            }
   524→            for cs in crop_seasons
   525→        ]
   526→
   527→        drafts, engine_not_evaluated, engine_warnings = fengine.compute_feature_drafts(
   528→            observations=observations,
   529→            specs=specs,
   530→            tz=tz,
   531→            hemisphere=run.hemisphere,
   532→            window_start=window_start,
   533→            window_end=window_end,
   534→            aggregation_levels=list(run.aggregation_levels),
   535→            crop_seasons=crop_windows,
   536→            engine_version=run.feature_engine_version,
   537→            definition_version=run.feature_catalog_version,
   538→        )
   539→
   540→        all_warnings = list(run.warnings) + list(engine_warnings)
   541→        if dropped_codes:
   542→            all_warnings.append(
   543→                "Features without their required inputs were not computed: "
   544→                + ", ".join(dropped_codes)
   545→                + "."
   546→            )
   547→        if run.is_synthetic:
   548→            all_warnings.insert(
   549→                0,
   550→                "This feature run is computed from SYNTHETIC fixture data — values are for "
   551→                "pipeline demonstration and testing only, never for real-world conclusions.",
   552→            )
   553→
   554→        written = 0
   555→        for draft in drafts:
   556→            row = draft.to_row()
   557→            db.add(
   558→                FeatureValue(
   559→                    run_id=run.id,
   560→                    field_id=run.field_id,
   561→                    feature_code=row["feature_code"],
   562→                    aggregation_level=row["aggregation_level"],
   563→                    period_label=row["period_label"],
   564→                    period_start=row["period_start"],
   565→                    period_end=row["period_end"],
   566→                    value=row["value"],
   567→                    unit=row["unit"],
   568→                    coverage_ratio=row["coverage_ratio"],
   569→                    n_input=row["n_input"],
   570→                    n_missing=row["n_missing"],
   571→                    quality_status=row["quality_status"],
   572→                    uncertainty_value=row["uncertainty_value"],
   573→                    uncertainty_type=row["uncertainty_type"],
   574→                    baseline_reference=row["baseline_reference"],
   575→                    provenance=row["provenance"],
   576→                )
   577→            )
   578→            written += 1
   579→            if written % 500 == 0:
   580→                db.flush()
   581→
   582→        merged_reasons = dict(not_evaluated)
   583→        for code, reason in engine_not_evaluated.items():
   584→            merged_reasons.setdefault(code, reason)
   585→
   586→        run.status = "succeeded"
   587→        run.records_written = written
   588→        run.warnings = all_warnings[:100]
   589→        run.not_evaluated_reasons = merged_reasons
   590→        run.completed_at = datetime.now(UTC)
   591→        db.commit()
   592→    except Exception as exc:  # noqa: BLE001 — run lifecycle must capture failures
   593→        db.rollback()
   594→        run.status = "failed"
   595→        run.error_summary = f"{type(exc).__name__}: {exc}"
   596→        run.completed_at = datetime.now(UTC)
   597→        db.commit()
   598→        logger.exception("feature run failed", extra={"run_id": str(run.id)})

   669→def compare_runs(
   670→    db: Session, run: FeatureRun, against: FeatureRun
   671→) -> dict[str, Any]:
   672→    """Compare two runs on matching (feature_code, aggregation_level, period_label)."""
   673→    left_rows = db.scalars(select(FeatureValue).where(FeatureValue.run_id == run.id)).all()
   674→    right_rows = db.scalars(select(FeatureValue).where(FeatureValue.run_id == against.id)).all()
   675→    right_by_key = {
   676→        (v.feature_code, v.aggregation_level, v.period_label): v for v in right_rows
   677→    }
   678→    deltas: list[dict[str, Any]] = []
   679→    matched = 0
   680→    only_in_run = 0
   681→    for left in left_rows:
   682→        key = (left.feature_code, left.aggregation_level, left.period_label)
   683→        right = right_by_key.get(key)
   684→        if right is None:
   685→            only_in_run += 1
   686→            continue
   687→        if left.value is None or right.value is None:
   688→            continue
   689→        matched += 1
   690→        delta = round(left.value - right.value, 4)
   691→        delta_pct = (
   692→            round((left.value - right.value) / abs(right.value), 4) if right.value != 0 else None
   693→        )
   694→        deltas.append(
   695→            {
   696→                "feature_code": left.feature_code,
   697→                "aggregation_level": left.aggregation_level,
   698→                "period_label": left.period_label,
   699→                "unit": left.unit,
   700→                "run_value": left.value,
   701→                "against_value": right.value,
   702→                "delta": delta,
   703→                "delta_pct": delta_pct,
   704→                "quality_run": left.quality_status,
   705→                "quality_against": right.quality_status,
   706→            }
   707→        )
   708→    return {
   709→        "run_id": str(run.id),
   710→        "against_run_id": str(against.id),
   711→        "matched_periods": matched,
   712→        "only_in_run": only_in_run,
   713→        "deltas": deltas,
   714→        "note": (
   715→            "Deltas compare two independent runs on matching periods. Runs may use different "
   716→            "source selections, calendar bases or engine versions — check run metadata before "
   717→            "interpreting deltas."
   718→        ),
   719→    }
