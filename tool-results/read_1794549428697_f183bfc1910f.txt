    88→    88→function statusLabel(t: T, status: string): string {
    89→    89→  switch (status) {
    90→    90→    case 'queued':
    91→    91→      return t('reports.statusQueued')
    92→    92→    case 'running':
    93→    93→      return t('reports.statusRunning')
    94→    94→    case 'succeeded':
    95→    95→      return t('reports.statusSucceeded')
    96→    96→    case 'failed':
    97→    97→      return t('reports.statusFailed')
    98→    98→    case 'expired':
    99→    99→      return t('reports.statusExpired')
   100→   100→    default:
   101→   101→      return status
   102→   102→  }
   103→   103→}

   142→   142→function formatSize(bytes: number | null): string {
   143→   143→  if (bytes === null || bytes === undefined) return '—'
   144→   144→  if (bytes < 1024) return `${bytes} B`
   145→   145→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
   146→   146→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`
   147→   147→}

   149→   149→function toastMutationError(t: T, err: unknown): void {
   150→   150→  if (err instanceof ApiError && err.status === 403) {

   154→   154→  const d = describeError(err)
   155→   155→  toast.error(d.title, { description: d.description })

   175→   175→async function saveBlob(blob: Blob, filename: string): Promise<void> {
   176→   176→  const url = URL.createObjectURL(blob)
   177→   177→  try {
   178→   178→    const a = document.createElement('a')
   179→   179→    a.href = url
   180→   180→    a.download = filename
   181→   181→    document.body.appendChild(a)
   182→   182→    a.click()
   183→   183→    a.remove()
   184→   184→  } finally {
   185→   185→    URL.revokeObjectURL(url)
   186→   186→  }
   187→   187→}
