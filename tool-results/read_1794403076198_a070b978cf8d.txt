    88→/** Thrown for any non-2xx API response; carries the parsed error envelope. */
    89→export class ApiError extends Error {
    90→  readonly status: number
    91→  readonly code: string
    92→  readonly details: ApiErrorDetail[]
    93→  readonly requestId: string | undefined
    94→
    95→  constructor(params: {
    96→    status: number
    97→    code: string
    98→    message: string
    99→    details?: ApiErrorDetail[]
   100→    requestId?: string
   101→  }) {
   102→    super(params.message)
   103→    this.name = 'ApiError'
   104→    this.status = params.status
   105→    this.code = params.code
   106→    this.details = params.details ?? []
   107→    this.requestId = params.requestId
   108→  }
   109→}

   111→/** Human-readable summary of any thrown error for toasts. */
   112→export function describeError(err: unknown): { title: string; description?: string } {
   113→  if (err instanceof ApiError) {
   114→    const firstDetail = err.details[0]
   115→    const description = firstDetail
   116→      ? `${firstDetail.field}: ${firstDetail.reason}${err.requestId ? ` · request ${err.requestId.slice(0, 8)}` : ''}`
   117→      : err.requestId
   118→        ? `request ${err.requestId.slice(0, 8)}`
   119→        : undefined
   120→    return { title: err.message, description }
   121→  }
   122→  if (err instanceof Error) {
   123→    return { title: 'Network error', description: err.message }
   124→  }
   125→  return { title: 'Unexpected error' }
   126→}

   128→function withPort(path: string): string {
   129→  const joiner = path.includes('?') ? '&' : '?'
   130→  return `${path}${joiner}${PORT_QUERY}`
   131→}

   133→/**
   134→ * Exchange the HttpOnly refresh cookie for a fresh access+refresh cookie pair
   135→ * (rotating refresh: the presented token is revoked server-side; reusing a
   136→ * superseded token revokes the whole session family).
   137→ *
   138→ * Single-flight: several concurrent 401s share ONE rotation call — parallel
   139→ * rotations would present already-rotated tokens and revoke the family.
   140→ */
   141→let refreshInFlight: Promise<boolean> | null = null
   142→
   143→export async function refreshSession(): Promise<boolean> {
   144→  if (!refreshInFlight) {
   145→    refreshInFlight = authRequest('/refresh')
   146→      .then(() => true)
   147→      .catch(() => false)
   148→      .finally(() => {
   149→        refreshInFlight = null
   150→      })
   151→  }
   152→  return refreshInFlight
   153→}

   155→/** POST to a Next.js /api/auth/* route handler (session cookies live here). */
   156→async function authRequest<T>(path: string, body?: unknown): Promise<T> {
   157→  const response = await fetch(`${AUTH_BASE}${path}`, {
   158→    method: 'POST',
   159→    headers: body !== undefined ? { 'Content-Type': 'application/json' } : undefined,
   160→    body: body !== undefined ? JSON.stringify(body) : undefined,
   161→    cache: 'no-store',
   162→  })
   163→  const text = await response.text()
   164→  let payload: unknown = null
   165→  if (text) {
   166→    try {
   167→      payload = JSON.parse(text)
   168→    } catch {
   169→      payload = null
   170→    }
   171→  }
   172→  if (!response.ok) {
   173→    const envelope = payload as ApiErrorBody | null
   174→    const error =
   175→      envelope && typeof envelope === 'object' && 'error' in envelope ? envelope.error : undefined
   176→    throw new ApiError({
   177→      status: response.status,
   178→      code: error?.code ?? `HTTP_${response.status}`,
   179→      message: error?.message ?? `Request failed with status ${response.status}.`,
   180→      details: error?.details ?? [],
   181→      requestId: error?.request_id,
   182→    })
   183→  }
   184→  return payload as T
   185→}
   186→
   187→interface RequestOptions {
   188→  method?: 'GET' | 'POST' | 'PATCH' | 'PUT' | 'DELETE'
   189→  body?: unknown
   190→  /** Multipart form body (file uploads). Content-Type is set by the browser. */
   191→  formData?: FormData
   192→  /** Skip the 401→refresh retry (health checks, anonymous endpoints). */
   193→  anonymous?: boolean
   194→}
   195→
   196→interface RequestInternals {
   197→  /** Prevent recursive 401→refresh retries (used by refreshSession itself). */
   198→  skipAuthRetry?: boolean
   199→}
   200→
   201→async function request<T>(
   202→  path: string,
   203→  options: RequestOptions = {},
   204→  internals: RequestInternals = {},
   205→): Promise<T> {
   206→  const headers: Record<string, string> = { Accept: 'application/json' }
   207→  if (options.body !== undefined) headers['Content-Type'] = 'application/json'
   208→
   209→  let response: Response
   210→  try {
   211→    response = await fetch(withPort(path), {
   212→      method: options.method ?? 'GET',
   213→      headers,
   214→      body:
   215→        options.formData !== undefined
   216→          ? options.formData
   217→          : options.body !== undefined
   218→            ? JSON.stringify(options.body)
   219→            : undefined,
   220→      cache: 'no-store',
   221→    })
   222→  } catch (err) {
   223→    throw new ApiError({
   224→      status: 0,
   225→      code: 'NETWORK_ERROR',
   226→      message: 'Could not reach the CropTwist API.',
   227→      details: [],
   228→    })
   229→  }
   230→
   231→  const text = await response.text()
   232→  let payload: unknown = null
   233→  if (text) {
   234→    try {
   235→      payload = JSON.parse(text)
   236→    } catch {
   237→      payload = null
   238→    }
   239→  }
   240→
   241→  if (!response.ok) {
   242→    const envelope = payload as ApiErrorBody | null
   243→    const error = envelope && typeof envelope === 'object' && 'error' in envelope ? envelope.error : undefined
   244→    // Rotating-refresh retry (P0-2/P1): an access token may have simply
   245→    // expired while the session is still valid. Try ONE cookie refresh, then
   246→    // retry the original request once. Never recurse from the refresh call
   247→    // itself (it goes through authRequest, not here).
   248→    if (response.status === 401 && !options.anonymous && !internals.skipAuthRetry) {
   249→      const refreshed = await refreshSession()
   250→      if (refreshed) {
   251→        return request<T>(path, options, { skipAuthRetry: true })
   252→      }
   253→    }
   254→    throw new ApiError({
   255→      status: response.status,
   256→      code: error?.code ?? `HTTP_${response.status}`,
   257→      message: error?.message ?? `Request failed with status ${response.status}.`,
   258→      details: error?.details ?? [],
   259→      requestId: error?.request_id,
   260→    })
   261→  }
   262→
   263→  return payload as T
   264→}

   413→const JOB_TERMINAL_STATUSES = new Set(['succeeded', 'failed', 'cancelled'])
   414→
   415→/**
   416→ * Poll an ingestion/import job until it reaches a terminal state.
   417→ *
   418→ * The backend executes data jobs on a background worker pool (202 + status
   419→ * 'queued'); when workers are disabled it still executes inline (the response
   420→ * is already terminal, returned as-is). Progress callbacks receive every
   421→ * intermediate snapshot (status/progress for UI feedback).
   422→ */
   423→export async function pollJobUntilDone(
   424→  jobId: string,
   425→  opts: { onProgress?: (job: DataJob) => void; intervalMs?: number; timeoutMs?: number } = {},
   426→): Promise<DataJob> {
   427→  const interval = opts.intervalMs ?? 1000
   428→  const deadline = Date.now() + (opts.timeoutMs ?? 180_000)
   429→  let current = await getJob(jobId)
   430→  while (!JOB_TERMINAL_STATUSES.has(current.status)) {
   431→    opts.onProgress?.(current)
   432→    if (Date.now() > deadline) {
   433→      throw new Error('Timed out waiting for the job to finish — check its status later.')
   434→    }
   435→    await new Promise((resolve) => setTimeout(resolve, interval))
   436→    current = await getJob(jobId)
   437→  }
   438→  return current
   439→}

   470→export type GeometryParseResult =
   471→  | { ok: true; geometry: import('./types').GeoJSONGeometry }
   472→  | { ok: false; error: string }
   473→
   474→/** Parse + shallow-validate a GeoJSON Polygon / MultiPolygon from user input. */
   475→export function parseGeometryInput(text: string): GeometryParseResult {
   476→  let parsed: unknown
   477→  try {
   478→    parsed = JSON.parse(text)
   479→  } catch {
   480→    return { ok: false, error: 'The text is not valid JSON. Check for missing quotes, brackets or trailing commas.' }
   481→  }
   482→  if (typeof parsed !== 'object' || parsed === null || Array.isArray(parsed)) {
   483→    return { ok: false, error: 'Expected a GeoJSON object like {"type":"Polygon","coordinates":[...]}' }
   484→  }
   485→  const obj = parsed as Record<string, unknown>
   486→  if (obj.type === 'Polygon') {
   487→    if (!Array.isArray(obj.coordinates) || obj.coordinates.length === 0) {
   488→      return { ok: false, error: 'Polygon coordinates must be a non-empty array of rings.' }
   489→    }
   490→    return { ok: true, geometry: parsed as import('./types').GeoJSONPolygon }
   491→  }
   492→  if (obj.type === 'MultiPolygon') {
   493→    if (!Array.isArray(obj.coordinates) || obj.coordinates.length === 0) {
   494→      return { ok: false, error: 'MultiPolygon coordinates must be a non-empty array of polygons.' }
   495→    }
   496→    return { ok: true, geometry: parsed as import('./types').GeoJSONMultiPolygon }
   497→  }
   498→  return { ok: false, error: `Unsupported geometry type "${String(obj.type)}". Use Polygon or MultiPolygon.` }
   499→}

   518→/** Client-side guard mirroring backend job validation. Returns an error message or null. */
   519→export function validateJobWindow(
   520→  start: string,
   521→  end: string,
   522→  bounds?: { min?: string; max?: string; maxYears?: number },
   523→): string | null {
   524→  const dateRe = /^\d{4}-\d{2}-\d{2}$/
   525→  if (!dateRe.test(start) || !dateRe.test(end)) return 'Enter both dates as YYYY-MM-DD.'
   526→  const min = bounds?.min ?? COVERAGE_START
   527→  const max = bounds?.max ?? COVERAGE_END
   528→  if (start < min || end > max) {
   529→    return `Dates must fall inside the dataset coverage ${min} → ${max}.`
   530→  }
   531→  if (end < start) return 'The end date must be on or after the start date.'
   532→  const years = (Date.parse(end) - Date.parse(start)) / (365.25 * 24 * 3600 * 1000)
   533→  const maxYears = bounds?.maxYears ?? MAX_JOB_YEARS
   534→  if (years > maxYears) return `A single job covers at most ${maxYears} years. Split the range.`
   535→  return null
   536→}

   971→// --- laboratory CSV import ------------------------------------------------------
   972→// Backend contract: POST /soil-samples/import (org-level) requires a `mapping`
   973→// form field whose JSON carries default_field_id — the field binding arrives
   974→// there, so the component's (fieldId, file) signature is preserved.
   975→
   976→export async function importSoilLabCsv(fieldId: string, file: File): Promise<SoilLabImportResponse> {
   977→  const formData = new FormData()
   978→  formData.append('file', file)
   979→  formData.append('mapping', JSON.stringify({ default_field_id: fieldId }))
   980→  const data = await request<SoilLabImportResponse>('/api/v1/soil-samples/import', {
   981→    method: 'POST',
   982→    formData,
   983→  })
   984→  // The backend's idempotent "already_imported" reply omits the counters;
   985→  // normalize so callers always see numbers (re-import of identical content
   986→  // creates nothing — 0 is the truthful value there).
   987→  return {
   988→    ...data,
   989→    created_samples: data.created_samples ?? 0,
   990→    created_measurements: data.created_measurements ?? 0,
   991→    errors: data.errors ?? [],
   992→  }
   993→}
