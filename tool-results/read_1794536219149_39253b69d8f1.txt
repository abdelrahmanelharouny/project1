   101→function StatusBadge({ status }: { status: string }) {
   102→  const tone =
   103→    status === 'eligible' || status === 'evaluated' || status === 'succeeded' || status === 'scheduled'
   104→      ? 'border-emerald-200 bg-emerald-50 text-emerald-800'
   105→      : status === 'conditional'
   106→        ? 'border-amber-200 bg-amber-50 text-amber-900'
   107→        : status === 'excluded' || status === 'failed' || status === 'over_allocated'
   108→          ? 'border-red-200 bg-red-50 text-red-800'
   109→          : status === 'not_evaluated' || status === 'unknown' || status === 'unscheduled'
   110→            ? 'border-stone-300 bg-stone-100 text-stone-600'
   111→            : 'border-stone-200 bg-stone-50 text-stone-700'
   112→  return (
   113→    <Badge variant="outline" className={`px-1.5 py-0 text-[10px] ${tone}`}>
   114→      {status.replace(/_/g, ' ')}
   115→    </Badge>
   116→  )
   117→}

   136→function ErrorNote({ error }: { error: unknown }) {
   137→  if (!error) return null
   138→  const described = describeError(error)
   139→  return (
   140→    <p className="text-xs font-medium text-red-700" role="alert">
   141→      {described.title}
   142→      {described.description ? ` — ${described.description}` : ''}
   143→    </p>
   144→  )

   245→function PlanView({ fieldId, expert }: { fieldId: string; expert: boolean }) {
   246→  const t = useTranslations('management')
   247→  const qc = useQueryClient()
   248→  const [created, setCreated] = useState<ManagementPlan | null>(null)
   249→  const [error, setError] = useState<unknown>(null)
   250→
   251→  const createPlan = useMutation({
   252→    mutationFn: () =>
   253→      createManagementPlan(fieldId, {
   254→        window_start: daysAgo(180),
   255→        window_end: today(),
   256→        season_label: undefined,
   257→        label: 'Management plan',
   258→      }),
   259→    onSuccess: (plan) => {
   260→      setCreated(plan)
   261→      setError(null)
   262→      qc.invalidateQueries({ queryKey: ['management-plan', fieldId] })
   263→    },
   264→    onError: setError,
   265→  })
   266→
   267→  const plan = created
   268→  if (!plan) {
   269→    return (
   270→      <div className="space-y-3">
   271→        <p className="text-sm text-stone-600">
   272→          {t('planIntro')}

   277→        <ErrorNote error={error} />
   278→      </div>
   279→    )
   280→  }
   281→

   283→  const m = r.soil_cover.metrics

   415→function CoverStudio({ fieldId, expert }: { fieldId: string; expert: boolean }) {
   416→  const t = useTranslations('management')
   417→  const qc = useQueryClient()
   418→  const [windowStart, setWindowStart] = useState(today())
   419→  const [windowEnd, setWindowEnd] = useState(daysAgo(-120))
   420→  const [nextCrop, setNextCrop] = useState('')
   421→  const [plan, setPlan] = useState<CoverPlan | null>(null)
   422→  const [adopted, setAdopted] = useState<string | null>(null)
   423→  const [error, setError] = useState<unknown>(null)
   424→
   425→  const items = useQuery({
   426→    queryKey: ['cover-items', fieldId],
   427→    queryFn: () => import('@/lib/api').then((m) => m.listCoverCropPlanItems(fieldId)),
   428→    enabled: Boolean(fieldId),
   429→  })
   430→
   431→  const run = useMutation({
   432→    mutationFn: () => createCoverCropPlan(fieldId, { window_start: windowStart, window_end: windowEnd, next_crop_label: nextCrop || undefined }),
   433→    onSuccess: (p) => {
   434→      setPlan(p)
   435→      setError(null)
   436→      qc.invalidateQueries({ queryKey: ['cover-items', fieldId] })
   437→    },
   438→    onError: setError,
   439→  })
   440→
   441→  const adopt = useMutation({
   442→    mutationFn: (candidate: CoverCandidate) =>
   443→      createCoverCropPlanItem(fieldId, {
   444→        crop_label: candidate.crop_code,
   445→        plan_run_id: plan?.id,
   446→        intended_sowing_start: candidate.sowing_window?.[0],
   447→        intended_termination_window_start: undefined,
   448→      }),
   449→    onSuccess: (item) => {
   450→      setAdopted(item.id)
   451→      qc.invalidateQueries({ queryKey: ['cover-items', fieldId] })
   452→    },
   453→    onError: setError,
   454→  })
   455→
   456→  const candidates = plan?.result.candidates ?? []
   457→
   458→  return (
   459→    <div className="space-y-4">
   460→      <p className="text-sm text-stone-600">
   461→        {t.rich('coverIntro', { cond: (chunks) => <span className="font-medium">{chunks}</span> })}
   462→      </p>
   463→      <div className="grid gap-2 sm:grid-cols-4">
   464→        <div className="space-y-1">
   465→          <Label htmlFor="cover-start" className="text-xs">{t('windowStart')}</Label>
   466→          <Input id="cover-start" type="date" value={windowStart} onChange={(e) => setWindowStart(e.target.value)} className="h-9 text-sm" />
   467→        </div>
   468→        <div className="space-y-1">
   469→          <Label htmlFor="cover-end" className="text-xs">{t('windowEnd')}</Label>
   470→          <Input id="cover-end" type="date" value={windowEnd} onChange={(e) => setWindowEnd(e.target.value)} className="h-9 text-sm" />
   471→        </div>
   472→        <div className="space-y-1">
   473→          <Label htmlFor="cover-next" className="text-xs">{t('nextCropLabel')}</Label>
   474→          <Input id="cover-next" value={nextCrop} onChange={(e) => setNextCrop(e.target.value)} placeholder={t('cropPlaceholder')} className="h-9 text-sm" />
   475→        </div>
   476→        <div className="flex items-end">
   477→          <Button onClick={() => run.mutate()} disabled={run.isPending} className="w-full bg-emerald-700 hover:bg-emerald-800">
   478→            {run.isPending ? t('evaluating') : t('evaluateCandidates')}
   479→          </Button>
   480→        </div>
   481→      </div>
   482→      <ErrorNote error={error} />
   483→
   484→      {plan ? (
   485→        <div className="space-y-3" data-testid="cover-candidates">
   486→          {plan.warnings.map((w, i) => (
   487→            <p key={i} className="rounded-md border border-amber-200 bg-amber-50 px-3 py-2 text-xs text-amber-900">{w}</p>
   488→          ))}
   489→          {candidates.length === 0 ? (
   490→            <HonestEmpty>
   491→              {t('noCandidates')}
   492→            </HonestEmpty>
   493→          ) : (

   930→              <SelectItem value="steep">{t('slopeSteep')}</SelectItem>
   931→            </SelectContent>
   932→          </Select>
   933→        </div>
   934→        <div className="flex items-center gap-2 pb-2 text-xs text-stone-500">
