    33→    33→def _crop(code, family=None, *, legume=False, cover=False, kc=None, duration=120,
    34→    34→          temp_opt=(15, 28), temp_min_abs=5, ph=None, windows=None, traits_extra=None):
    35→    35→    traits: dict[str, TraitValue] = {}
    36→    36→
    37→    37→    def t(code_, value=None, low=None, high=None, unit=None):
    38→    38→        traits[code_] = TraitValue(
    39→    39→            trait_code=code_, value_numeric=value, value_text=None, value_low=low,
    40→    40→            value_high=high, unit=unit, region_code=None,
    41→    41→            evidence_class="published_reference", confidence="moderate",
    42→    42→            validation_status="reference", source_key="s", citation_text="citation",
    43→    43→            pack_code="kp", pack_version=PACK_VERSION,
    44→    44→        )
    45→    45→
    46→    46→    if duration is not None:
    47→    47→        t("duration_days_total", value=duration, unit="d")
    48→    48→    if kc is not None:
    49→    49→        t("stage_init_days", value=20, unit="d")
    50→    50→        t("stage_dev_days", value=30, unit="d")
    51→    51→        t("stage_mid_days", value=40, unit="d")
    52→    52→        t("stage_late_days", value=30, unit="d")
    53→    53→    if temp_opt:
    54→    54→        t("temp_opt_min_degC", value=temp_opt[0], unit="degC")
    55→    55→        t("temp_opt_max_degC", value=temp_opt[1], unit="degC")
    56→    56→    if temp_min_abs is not None:
    57→    57→        t("temp_min_abs_degC", value=temp_min_abs, unit="degC")
    58→    58→    if ph:
    59→    59→        t("ph_opt_min", value=ph[0], unit="pH")
    60→    60→        t("ph_opt_max", value=ph[1], unit="pH")
    61→    61→    for code_, value in (traits_extra or {}).items():
    62→    62→        t(code_, value=value)
    63→    63→    return KnowledgeCrop(
    64→    64→        crop_code=code, pack_code="kp", pack_version=PACK_VERSION, family_key=family,
    65→    65→        scientific_name=None, common_name=code, crop_category=None,
    66→    66→        life_cycle_type="annual", is_legume=legume, is_cover_crop_capable=cover,
    67→    67→        traits=traits,
    68→    68→        calendar_windows=tuple(windows or ()),
    69→    69→        water_params=WaterParams(
    70→    70→            kc_initial=kc[0], kc_mid=kc[1], kc_late=kc[2],
    71→    71→            evidence_class="published_reference", source_key="s",
    72→    72→            citation_text="citation", pack_code="kp", pack_version=PACK_VERSION,
    73→    73→        ) if kc else None,
    74→    74→    )
