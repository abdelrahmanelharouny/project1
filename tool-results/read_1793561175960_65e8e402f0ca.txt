   152→function CreateFarmDialog({ open, onOpenChange }: { open: boolean; onOpenChange: (open: boolean) => void }) {
   153→  const queryClient = useQueryClient()
   154→  const [name, setName] = useState('')
   155→  const [country, setCountry] = useState('')
   156→  const [adminRegion, setAdminRegion] = useState('')
   157→  const [timezone, setTimezone] = useState('UTC')
   158→  const [currency, setCurrency] = useState('')
   159→  const [unitSystem, setUnitSystem] = useState<UnitSystem>('metric')
   160→  const [language, setLanguage] = useState('en')
   161→  const [formError, setFormError] = useState<string | null>(null)
   162→
   163→  // Reset form state when the dialog closes (handler-based, no effect needed).
   164→  const handleOpenChange = (next: boolean) => {
   165→    if (!next) {
   166→      setName('')
   167→      setCountry('')
   168→      setAdminRegion('')
   169→      setTimezone('UTC')
   170→      setCurrency('')
   171→      setUnitSystem('metric')
   172→      setLanguage('en')
   173→      setFormError(null)
   174→    }
   175→    onOpenChange(next)
   176→  }
   177→
   178→  const mutation = useMutation({
   179→    mutationFn: (input: FarmCreateInput) => createFarm(input),
   180→    onSuccess: (farm) => {
   181→      queryClient.invalidateQueries({ queryKey: ['farms'] })
   182→      toast.success(`Farm “${farm.name}” created`, {
   183→        description: `${farm.timezone} · ${farm.currency_code} · ${farm.unit_system}`,

   204→    mutation.mutate({
   205→      name: name.trim(),
   206→      country_code: countryClean,
   207→      admin_region: adminRegion.trim() || undefined,
   208→      timezone,
   209→      currency_code: currencyClean,
   210→      unit_system: unitSystem,
   211→      preferred_language: languageClean,
   212→    })
   213→  }

   221→            A farm groups fields and sets the geographic + currency context for decisions.
   222→          </DialogDescription>
   223→        </DialogHeader>
   224→        <form onSubmit={submit} className="space-y-4" aria-label="Create farm form">

   237→              <Input
   238→                id="farm-country"
   239→                maxLength={2}
   240→                placeholder="EG"
   241→                value={country}

   243→                className="uppercase"

   245→              <p className="text-xs text-stone-500">2 letters, e.g. EG, IT, US, TH.</p>

   335→function AddFieldDialog({
   336→  farm,
   337→  open,
   338→  onOpenChange,
   339→  onFieldSelected,
   340→}: {
   341→  farm: Farm
   342→  open: boolean
   343→  onOpenChange: (open: boolean) => void
   344→  onFieldSelected: (field: { id: string; name: string; farmId: string }) => void
   345→}) {
   346→  const queryClient = useQueryClient()
   347→  const [name, setName] = useState('')
   348→  const [geometry, setGeometry] = useState('')
   349→  const [parseError, setParseError] = useState<string | null>(null)
   350→  const [apiError, setApiError] = useState<{ title: string; description?: string } | null>(null)
   351→  const [created, setCreated] = useState<FieldCreateResponse | null>(null)
   352→
   353→  // Reset form state when the dialog closes.
   354→  const handleOpenChange = (next: boolean) => {
   355→    if (!next) {
   356→      setName('')
   357→      setGeometry('')
   358→      setParseError(null)
   359→      setApiError(null)
   360→      setCreated(null)
   361→    }
   362→    onOpenChange(next)
   363→  }
   364→
   365→  const mutation = useMutation({
   366→    mutationFn: (payload: { name: string; geometry: GeoJSONGeometry }) =>
   367→      createField(farm.id, { name: payload.name, geometry: payload.geometry }),
   368→    onSuccess: (result) => {
   369→      queryClient.invalidateQueries({ queryKey: ['fields', farm.id] })
   370→      queryClient.invalidateQueries({ queryKey: ['farms'] })
   371→      if (result.geometry_warnings.length > 0) {
   372→        setCreated(result)
   373→        toast.warning('Field created — but review the geometry warnings', {
   374→          description: `${result.geometry_warnings.length} warning(s) from PostGIS validation.`,
   375→        })
   376→      } else {
   377→        toast.success(`Field “${result.name}” created`, {
   378→          description: `${formatArea(result.area_ha)} ha · boundary v${result.boundary_version}`,
   379→        })
   380→        onFieldSelected({ id: result.id, name: result.name, farmId: farm.id })
   381→        handleOpenChange(false)
   382→      }
   383→    },
   384→    onError: (err) => {
   385→      const { title, description } = describeError(err)
   386→      setApiError({ title, description })
   387→      toast.error(title, { description })
   388→    },
   389→  })
   390→
   391→  const submit = (event: FormEvent) => {
   392→    event.preventDefault()
   393→    setApiError(null)
   394→    setCreated(null)
   395→    if (!name.trim()) {
   396→      setParseError('Give the field a name first.')
   397→      return
   398→    }
   399→    const parsed = parseGeometryInput(geometry)
   400→    if (!parsed.ok) {
   401→      setParseError(parsed.error)
   402→      return
   403→    }
   404→    setParseError(null)
   405→    mutation.mutate({ name: name.trim(), geometry: parsed.geometry })
   406→  }
   407→
   408→  return (
   409→    <Dialog open={open} onOpenChange={handleOpenChange}>
   410→      <DialogContent className="max-h-[85vh] overflow-y-auto scrollbar-thin sm:max-w-xl">
   411→        <DialogHeader>
   412→          <DialogTitle>Add a field to “{farm.name}”</DialogTitle>
   413→          <DialogDescription>
   414→            The boundary is validated by PostGIS. If something is off (self-intersections, unclosed rings), the
   415→            response tells you exactly what was fixed.
   416→          </DialogDescription>
   417→        </DialogHeader>
   418→        <form onSubmit={submit} className="space-y-4" aria-label="Add field form">
   419→          <div className="space-y-1.5">
   420→            <Label htmlFor="field-name">Field name</Label>
   421→            <Input
   422→              id="field-name"
   423→              placeholder="North Field"
   424→              value={name}
   425→              onChange={(event) => setName(event.target.value)}
   426→            />
   427→          </div>
   428→          <GeometryInput id="field-geometry" value={geometry} onChange={setGeometry} error={parseError} />
   429→
   430→          {created ? (
   431→            <div className="space-y-2">
   432→              <Alert className="border-emerald-300 bg-emerald-50 [&>svg]:text-emerald-700">
   433→                <MapPin className="h-4 w-4" />
   434→                <AlertTitle className="text-emerald-900">
   435→                  Field “{created.name}” created — {formatArea(created.area_ha)} ha
   436→                </AlertTitle>
   437→                <AlertDescription className="text-emerald-900">
   438→                  Boundary v{created.boundary_version} · passport status: {created.data_passport_status}
   439→                </AlertDescription>
   440→              </Alert>
   441→              <GeometryWarningsAlert
   442→                warnings={created.geometry_warnings}
   443→                repaired={created.geometry_repaired}
   444→                title="Geometry warnings from PostGIS"
   445→              />
   446→              <div className="flex justify-end gap-2">
   447→                <Button
   448→                  type="button"
   449→                  variant="outline"
   450→                  onClick={() => {
   451→                    onFieldSelected({ id: created.id, name: created.name, farmId: farm.id })
   452→                    handleOpenChange(false)
   453→                  }}
   454→                >
   455→                  Open the new field
   456→                </Button>
   457→              </div>
   458→            </div>
   459→          ) : null}
   460→
   461→          {apiError ? (
   462→            <Alert variant="destructive">
   463→              <TriangleAlert className="h-4 w-4" />
   464→              <AlertTitle>{apiError.title}</AlertTitle>
   465→              <AlertDescription>{apiError.description ?? 'Adjust the geometry and try again.'}</AlertDescription>
   466→            </Alert>
   467→          ) : null}
   468→
   469→          <DialogFooter>
   470→            <Button type="button" variant="outline" onClick={() => handleOpenChange(false)}>
   471→              {created ? 'Close' : 'Cancel'}
   472→            </Button>
   473→            <Button type="submit" className="bg-emerald-700 hover:bg-emerald-800" disabled={mutation.isPending}>
   474→              {mutation.isPending ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : <Plus className="h-4 w-4" aria-hidden />}
   475→              Create field
   476→            </Button>
   477→          </DialogFooter>
   478→        </form>
   479→      </DialogContent>
   480→    </Dialog>
   481→  )
   482→}

   488→function RenameFieldDialog({
   489→  field,
   490→  open,
   491→  onOpenChange,
   492→}: {
   493→  field: Field
   494→  open: boolean
   495→  onOpenChange: (open: boolean) => void
   496→}) {
   497→  const queryClient = useQueryClient()
   498→  const renameSelectedField = useFieldShiftUi((s) => s.renameSelectedField)
   499→  const [name, setName] = useState(field.name)
   500→
   501→  // Sync the input with the current field name each time the dialog opens.
   502→  const handleOpenChange = (next: boolean) => {
   503→    if (next) setName(field.name)
   504→    onOpenChange(next)
   505→  }
   506→
   507→  const mutation = useMutation({
   508→    mutationFn: () => patchField(field.id, { name: name.trim() }),
   509→    onSuccess: (updated) => {
   510→      queryClient.invalidateQueries({ queryKey: ['fields'] })
   511→      renameSelectedField(updated.id, updated.name)
   512→      toast.success(`Field renamed to “${updated.name}”`)
   513→      onOpenChange(false)
   514→    },
   515→    onError: (err) => {
   516→      const { title, description } = describeError(err)
   517→      toast.error(title, { description })
   518→    },
   519→  })
   520→
   521→  const submit = (event: FormEvent) => {
   522→    event.preventDefault()
   523→    if (!name.trim() || name.trim() === field.name) {
   524→      onOpenChange(false)
   525→      return
   526→    }
   527→    mutation.mutate()
   528→  }
   529→
   530→  return (
   531→    <Dialog open={open} onOpenChange={handleOpenChange}>
   532→      <DialogContent className="sm:max-w-sm">
   533→        <DialogHeader>
   534→          <DialogTitle>Rename field</DialogTitle>
   535→          <DialogDescription>Boundaries and observations keep their history — only the label changes.</DialogDescription>
   536→        </DialogHeader>
   537→        <form onSubmit={submit} className="space-y-4" aria-label="Rename field form">
   538→          <div className="space-y-1.5">
   539→            <Label htmlFor="rename-field">Field name</Label>
   540→            <Input id="rename-field" value={name} onChange={(event) => setName(event.target.value)} />
   541→          </div>
   542→          <DialogFooter>
   543→            <Button type="button" variant="outline" onClick={() => handleOpenChange(false)}>
   544→              Cancel
   545→            </Button>
   546→            <Button type="submit" className="bg-emerald-700 hover:bg-emerald-800" disabled={mutation.isPending}>
   547→              {mutation.isPending ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : null}
   548→              Save name
   549→            </Button>
   550→          </DialogFooter>
   551→        </form>
   552→      </DialogContent>
   553→    </Dialog>
   554→  )
   555→}

   583→  const mutation = useMutation({
   584→    mutationFn: (payload: { geometry: GeoJSONGeometry; change_reason?: string }) =>
   585→      createBoundaryVersion(field.id, payload),
   586→    onSuccess: (response) => {
   587→      queryClient.invalidateQueries({ queryKey: ['fields'] })
   588→      queryClient.invalidateQueries({ queryKey: ['field-map', field.id] })
   589→      queryClient.invalidateQueries({ queryKey: ['boundary-versions', field.id] })
   590→      queryClient.invalidateQueries({ queryKey: ['data-passport', field.id] })
   591→      if (response.geometry_warnings.length > 0) {
   592→        setResult(response)
   593→        toast.warning(`Boundary v${response.boundary_version} stored with warnings`, {
   594→          description: 'Review the PostGIS warnings below.',
   595→        })
   596→      } else {
   597→        toast.success(`Boundary updated to v${response.boundary_version}`, {
   598→          description: `${formatArea(response.area_ha)} ha · geodesic WGS84`,
   599→        })
   600→        handleOpenChange(false)
   601→      }
   602→    },
   603→    onError: (err) => {
   604→      const { title, description } = describeError(err)
   605→      setParseError(title)
   606→      toast.error(title, { description })
   607→    },
   608→  })
   609→
   610→  const submit = (event: FormEvent) => {
   611→    event.preventDefault()
   612→    setResult(null)
   613→    const parsed = parseGeometryInput(geometry)
   614→    if (!parsed.ok) {
   615→      setParseError(parsed.error)
   616→      return
   617→    }
   618→    setParseError(null)
   619→    mutation.mutate({ geometry: parsed.geometry, change_reason: reason.trim() || undefined })
   620→  }
   621→
   622→  return (
   623→    <Dialog open={open} onOpenChange={handleOpenChange}>
   624→      <DialogContent className="max-h-[85vh] overflow-y-auto scrollbar-thin sm:max-w-xl">
   625→        <DialogHeader>
   626→          <DialogTitle>Update boundary — creates version {field.boundary_version + 1}</DialogTitle>
   627→          <DialogDescription>
   628→            Boundary history is never overwritten: every change is stored as a new version with its recomputed area.
   629→          </DialogDescription>
   630→        </DialogHeader>
   631→        <form onSubmit={submit} className="space-y-4" aria-label="Update boundary form">
   632→          <GeometryInput id="boundary-geometry" value={geometry} onChange={setGeometry} error={parseError} />
   633→          <div className="space-y-1.5">
   634→            <Label htmlFor="boundary-reason">Change reason (optional)</Label>
   635→            <Input
   636→              id="boundary-reason"
   637→              placeholder="Re-measured after land swap"
   638→              value={reason}
   639→              onChange={(event) => setReason(event.target.value)}
   640→            />
   641→          </div>
   642→
   643→          {result ? (
   644→            <div className="space-y-2">
   645→              <Alert className="border-emerald-300 bg-emerald-50 [&>svg]:text-emerald-700">
   646→                <MapPin className="h-4 w-4" />
   647→                <AlertTitle className="text-emerald-900">
   648→                  Boundary v{result.boundary_version} stored — {formatArea(result.area_ha)} ha
   649→                </AlertTitle>
   650→                <AlertDescription className="text-emerald-900">
   651→                  The map and version list will refresh automatically.
   652→                </AlertDescription>
   653→              </Alert>
   654→              <GeometryWarningsAlert
   655→                warnings={result.geometry_warnings}
   656→                repaired={result.geometry_repaired}
   657→                title="Geometry warnings from PostGIS"
   658→              />
   659→            </div>
   660→          ) : null}
   661→
   662→          <DialogFooter>
   663→            <Button type="button" variant="outline" onClick={() => handleOpenChange(false)}>
   664→              {result ? 'Close' : 'Cancel'}
   665→            </Button>
   666→            <Button type="submit" className="bg-emerald-700 hover:bg-emerald-800" disabled={mutation.isPending}>
   667→              {mutation.isPending ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : null}
   668→              Store new version
   669→            </Button>
   670→          </DialogFooter>
   671→        </form>
   672→      </DialogContent>
   673→    </Dialog>
