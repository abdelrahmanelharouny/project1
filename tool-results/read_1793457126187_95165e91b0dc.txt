     1→"""Deterministic feature computation engine (Milestone 4).
     2→
     3→Pure functions over normalized observations. The engine:
     4→
     5→- assigns daily observations to LOCAL calendar days via the explicit midpoint
     6→  rule (app.features.calendar);
     7→- NEVER mixes temporal resolutions: monthly/pentadal observations do not
     8→  participate in daily-bucket aggregation (they are counted and reported);
     9→- NEVER treats missing observations as zero: accumulations sum valid days
    10→  only and stay NULL below the definition's minimum coverage; a data gap
    11→  interrupts dry spells instead of assuming dryness;
    12→- propagates quality (valid/partial/insufficient_coverage/missing) and, for
    13→  means, the standard error of the mean as uncertainty;
    14→- computes baselines and anomalies only when the baseline window holds enough
    15→  distinct years (>= 5), otherwise the feature is explicitly `not_evaluated`
    16→  with the reason embedded in provenance;
    17→- stamps every draft with full provenance (contributing observation ids,
    18→  dataset products, processing versions, evidence classes, calendar basis)
    19→  and a reproducibility checksum.
    20→"""
    21→
    22→import hashlib
    23→import json
    24→import statistics
    25→from dataclasses import dataclass, field
    26→from datetime import date, datetime, timedelta
    27→from typing import Any

    40→@dataclass

    51→@dataclass
    52→class DailySeries:
    53→    variable: str
    54→    cells: dict[date, DailyCell] = field(default_factory=dict)
    55→    duplicate_assignments: int = 0
    56→    excluded_time_supports: dict[str, int] = field(default_factory=dict)

    58→
    59→def build_daily_series(observations: list[dict[str, Any]], variable: str, tz: ZoneInfo) -> DailySeries:
    60→    """Bucket one variable's DAILY observations into local calendar days.
    61→
    62→    Observations whose time_support is not 'daily' are excluded (counted) —
    63→    mixing e.g. monthly product totals into daily buckets would double-count
    64→    or misrepresent periods. Only observations with a usable value are
    65→    quality-valid; missing observations create explicit empty slots? No —
    66→    missing observations are simply absent from `cells`, and coverage math
    67→    treats absent days as missing (never as zero).
    68→    """
    69→    series = DailySeries(variable=variable)
    70→    obs_list = [o for o in observations if o["variable_code"] == variable]
    71→    ordered = sorted(obs_list, key=lambda o: (o["period_start"], o["observed_at"]))
    72→    for obs in ordered:
    73→        ts = obs.get("time_support") or "daily"
    74→        if ts != "daily":
    75→            series.excluded_time_supports[ts] = series.excluded_time_supports.get(ts, 0) + 1
    76→            continue
    77→        local_day = fscal.assign_local_date(obs["period_start"], obs["period_end"], tz)
    78→        if local_day in series.cells:
    79→            series.duplicate_assignments += 1
    80→            continue  # first (earliest) wins; deterministic
    81→        value = obs.get("value")
    82→        if obs.get("quality_status") == "missing" or value is None:
    83→            continue  # missing stays missing; the day counts as uncovered
    84→        series.cells[local_day] = DailyCell(
    85→            value=float(value),
    86→            obs_id=str(obs["id"]),
    87→            dataset_product_id=str(obs["dataset_product_id"]),
    88→            processing_version=obs.get("processing_version") or "unknown",
    89→            evidence_class=obs.get("evidence_class") or "unknown",
    90→            is_synthetic=bool(obs.get("is_synthetic")),
    91→            quality_status=obs.get("quality_status") or "valid",
    92→        )
    93→    return series

   135→def _periods_overlap_days(p_start: datetime, p_end: datetime, w_start: datetime, w_end: datetime) -> tuple[datetime, datetime]:
   136→    s = max(p_start, w_start)
   137→    e = min(p_end, w_end)
   138→    return s, e
   139→
   140→
   141→def _expected_days(p_start_utc: datetime, p_end_utc: datetime, tz: ZoneInfo) -> int:
   142→    """Number of local days whose [start, next-start) lies within the period."""
   143→    local_start = p_start_utc.astimezone(tz).date()
   144→    local_end = p_end_utc.astimezone(tz).date()
   145→    days = 0
   146→    d = local_start
   147→    while d <= local_end:
   148→        days += 1
   149→        d += timedelta(days=1)
   150→    return max(1, days)

   153→def _checksum(draft: FeatureDraft, obs_ids: list[str]) -> str:
   154→    payload = json.dumps(
   155→        {
   156→            "feature_code": draft.feature_code,
   157→            "aggregation_level": draft.aggregation_level,
   158→            "period_label": draft.period_label,
   159→            "value": None if draft.value is None else round(draft.value, 6),
   160→            "n_input": draft.n_input,
   161→            "n_missing": draft.n_missing,
   162→            "coverage_ratio": draft.coverage_ratio,
   163→            "quality_status": draft.quality_status,
   164→            "observation_ids": obs_ids,
   165→        },
   166→        sort_keys=True,
   167→        separators=(",", ":"),
   168→    )
   169→    return hashlib.sha256(payload.encode("utf-8")).hexdigest()

   172→def _provenance(
   173→    *,
   174→    spec: FeatureSpec,
   175→    draft: FeatureDraft,
   176→    cells_used: list[DailyCell],
   177→    calendar_meta: dict[str, Any],
   178→    extra: dict[str, Any] | None = None,
   179→) -> dict[str, Any]:
   180→    obs_ids = [c.obs_id for c in cells_used]
   181→    variables: dict[str, Any] = {}
   182→    for c in cells_used:
   183→        entry = variables.setdefault(
   184→            "inputs",
   185→            {
   186→                "observation_count": 0,
   187→                "dataset_products": [],
   188→                "processing_versions": [],
   189→                "evidence_classes": [],
   190→                "is_synthetic": False,
   191→            },
   192→        )
   193→        entry["observation_count"] += 1
   194→        if c.dataset_product_id not in entry["dataset_products"]:
   195→            entry["dataset_products"].append(c.dataset_product_id)
   196→        if c.processing_version not in entry["processing_versions"]:
   197→            entry["processing_versions"].append(c.processing_version)
   198→        if c.evidence_class not in entry["evidence_classes"]:
   199→            entry["evidence_classes"].append(c.evidence_class)
   200→        entry["is_synthetic"] = entry["is_synthetic"] or c.is_synthetic
   201→    prov = {
   202→        "engine_version": calendar_meta["engine_version"],
   203→        "definition_version": calendar_meta["definition_version"],
   204→        "calendar": {
   205→            "timezone": calendar_meta["timezone"],
   206→            "hemisphere": calendar_meta["hemisphere"],
   207→            "assignment_rule": fscal.CALENDAR_ASSIGNMENT_RULE,
   208→        },
   209→        "required_variables": list(spec.required_variables),
   210→        "inputs": variables.get("inputs", {
   211→            "observation_count": 0,
   212→            "dataset_products": [],
   213→            "processing_versions": [],
   214→            "evidence_classes": [],
   215→            "is_synthetic": False,
   216→        }),
   217→        "observation_ids": obs_ids[:PROVENANCE_OBS_CAP],
   218→        "observation_count": len(obs_ids),
   219→        "observation_ids_capped": len(obs_ids) > PROVENANCE_OBS_CAP,
   220→        "checksum": _checksum(draft, obs_ids),
   221→    }
   222→    if extra:
   223→        prov.update(extra)
   224→    return prov

   227→# ------------------------------------------------------------ core engine
   228→def _iter_period_days(
   229→    p_start_utc: datetime, p_end_utc: datetime, tz: ZoneInfo
   230→) -> list[date]:
   231→    d = p_start_utc.astimezone(tz).date()
   232→    end = p_end_utc.astimezone(tz).date()
   233→    out = []
   234→    while d <= end:
   235→        out.append(d)
   236→        d += timedelta(days=1)
   237→    return out

   245→def _aggregate_period(
   246→    spec: FeatureSpec,
   247→    strategy: str,
   248→    series: DailySeries,
   249→    p_start: datetime,
   250→    p_end: datetime,
   251→    tz: ZoneInfo,
   252→    calendar_meta: dict[str, Any],
   253→    level: str,
   254→    label: str,
   255→    *,
   256→    window_start: datetime,
   257→    window_end: datetime,
   258→) -> FeatureDraft:
   259→    """One period aggregation for accumulation/mean/count/dry_spell strategies."""
   260→    s, e = _periods_overlap_days(p_start, p_end, window_start, window_end)
   261→    days = _iter_period_days(s, e, tz)
   262→    expected = len(days)
   263→    if expected == 0:
   264→        return FeatureDraft(
   265→            feature_code=spec.feature_code, aggregation_level=level, period_label=label,
   266→            period_start=s, period_end=e, value=None, unit=spec.unit,
   267→            coverage_ratio=0.0, n_input=0, n_missing=0,
   268→            quality_status="not_evaluated",
   269→            provenance={"reason": "Period does not overlap the run window."},
   270→        )
   271→
   272→    cells = [series.cells.get(d) for d in days]
   273→    valid_cells = [c for c in cells if c is not None and c.value is not None]
   274→    valid_days = len(valid_cells)
   275→    missing_days = expected - valid_days
   276→    coverage = valid_days / expected
   277→
   278→    base_kwargs = dict(
   279→        feature_code=spec.feature_code, aggregation_level=level, period_label=label,
   280→        period_start=s, period_end=e, unit=spec.unit,
   281→        coverage_ratio=round(coverage, 4), n_input=valid_days, n_missing=missing_days,
   282→    )
   283→
   284→    if valid_days == 0:
   285→        return FeatureDraft(
   286→            value=None, quality_status="missing", **base_kwargs,
   287→            provenance=_provenance(
   288→                spec=spec,
   289→                draft=FeatureDraft(
   290→                    feature_code=spec.feature_code, aggregation_level=level, period_label=label,
   291→                    period_start=s, period_end=e, value=None, unit=spec.unit,
   292→                    coverage_ratio=0.0, n_input=0, n_missing=missing_days, quality_status="missing",
   293→                ),
   294→                cells_used=[],
   295→                calendar_meta=calendar_meta,
   296→                extra={"reason": "No valid observations for this period."},
   297→            ),
   298→        )
   299→
   300→    if coverage < spec.min_coverage_ratio:
   301→        return FeatureDraft(
   302→            value=None, quality_status="insufficient_coverage", **base_kwargs,
   303→            provenance=_provenance(
   304→                spec=spec,
   305→                draft=FeatureDraft(
   306→                    feature_code=spec.feature_code, aggregation_level=level, period_label=label,
   307→                    period_start=s, period_end=e, value=None, unit=spec.unit,
   308→                    coverage_ratio=coverage, n_input=valid_days, n_missing=missing_days,
   309→                    quality_status="insufficient_coverage",
   310→                ),
   311→                cells_used=valid_cells,
   312→                calendar_meta=calendar_meta,
   313→                extra={
   314→                    "reason": (
   315→                        f"Coverage {coverage:.2f} is below the definition minimum "
   316→                        f"{spec.min_coverage_ratio:.2f}; value not computed (missing data is "
   317→                        "never treated as zero and totals are never scaled up)."
   318→                    )
   319→                },
   320→            ),
   321→        )
   322→
   323→    value: float | None = None
   324→    uncertainty: float | None = None
   325→    uncertainty_type: str | None = None
   326→    if strategy == "accumulation":
   327→        value = sum(c.value for c in valid_cells)  # type: ignore[arg-type]
   328→    elif strategy == "mean":
   329→        values = [c.value for c in valid_cells]  # type: ignore[arg-type]
   330→        value = sum(values) / len(values)  # type: ignore[arg-type]
   331→        if len(values) >= 2:
   332→            uncertainty = statistics.stdev(values) / (len(values) ** 0.5)
   333→            uncertainty_type = "standard_error_of_mean"
   334→    elif strategy == "count_days":
   335→        op = spec.strategy_params["op"]
   336→        thr = float(spec.strategy_params["threshold"])
   337→        preds = {
   338→            ">=": lambda v: v >= thr,
   339→            ">": lambda v: v > thr,
   340→            "<": lambda v: v < thr,
   341→            "<=": lambda v: v <= thr,
   342→        }
   343→        pred = preds[op]
   344→        value = float(sum(1 for c in valid_cells if pred(float(c.value))))  # type: ignore[arg-type]
   345→    elif strategy == "dry_spell_max":
   346→        dry_below = float(spec.strategy_params["dry_below_mm"])
   347→        current = best = 0
   348→        for d in days:  # ordered; missing cells interrupt the spell
   349→            cell = series.cells.get(d)
   350→            if cell is None or cell.value is None:
   351→                current = 0
   352→                continue
   353→            if float(cell.value) < dry_below:
   354→                current += 1
   355→                best = max(best, current)
   356→            else:
   357→                current = 0
   358→        value = float(best)
   359→    else:  # pragma: no cover — catalog guards strategies
   360→        raise ValueError(f"Unknown aggregation strategy '{strategy}'.")
   361→
   362→    quality = "valid" if missing_days == 0 else "partial"
   363→    draft = FeatureDraft(
   364→        value=round(float(value), 4), quality_status=quality,
   365→        uncertainty_value=uncertainty, uncertainty_type=uncertainty_type, **base_kwargs,
   366→    )
   367→    draft.provenance = _provenance(spec=spec, draft=draft, cells_used=valid_cells, calendar_meta=calendar_meta)
   368→    return draft
   369→

   371→def _daily_passthrough(
   372→    spec: FeatureSpec, series: DailySeries, window_days: list[date],
   373→    window_start: datetime, window_end: datetime, tz: ZoneInfo,
   374→    calendar_meta: dict[str, Any],
   375→) -> list[FeatureDraft]:
   376→    drafts = []
   377→    for d in window_days:
   378→        cell = series.cells.get(d)
   379→        p_start, _ = fscal.day_bounds_utc(d, tz)
   380→        label = d.isoformat()
   381→        if cell is None or cell.value is None:
   382→            drafts.append(FeatureDraft(
   383→                feature_code=spec.feature_code, aggregation_level="daily", period_label=label,
   384→                period_start=p_start, period_end=p_start + timedelta(days=1) - timedelta(seconds=1),
   385→                value=None, unit=spec.unit, coverage_ratio=0.0, n_input=0, n_missing=1,
   386→                quality_status="missing",
   387→                provenance={
   388→                    "reason": "No valid observation for this local day (missing is never zero).",
   389→                    "engine_version": calendar_meta["engine_version"],
   390→                    "definition_version": calendar_meta["definition_version"],
   391→                    "calendar": {
   392→                        "timezone": calendar_meta["timezone"],
   393→                        "hemisphere": calendar_meta["hemisphere"],
   394→                        "assignment_rule": fscal.CALENDAR_ASSIGNMENT_RULE,
   395→                    },
   396→                    "required_variables": list(spec.required_variables),
   397→                    "inputs": {"observation_count": 0, "dataset_products": [], "processing_versions": [], "evidence_classes": [], "is_synthetic": False},
   398→                    "observation_ids": [], "observation_count": 0, "observation_ids_capped": False,
   399→                },
   400→            ))
   401→            continue
   402→        draft = FeatureDraft(
   403→            feature_code=spec.feature_code, aggregation_level="daily", period_label=label,
   404→            period_start=p_start, period_end=p_start + timedelta(days=1) - timedelta(seconds=1),
   405→            value=round(cell.value, 4), unit=spec.unit, coverage_ratio=1.0,
   406→            n_input=1, n_missing=0, quality_status="valid",
   407→        )
   408→        draft.provenance = _provenance(spec=spec, draft=draft, cells_used=[cell], calendar_meta=calendar_meta)
   409→        drafts.append(draft)
   410→    return drafts

   413→def _baseline_by_calendar_month(
   414→    monthly_drafts: list[FeatureDraft], label: str
   415→) -> dict[str, Any] | None:
   416→    """Climatology of the same calendar month across the run window."""
   417→    month = int(label.split("-")[1])
   418→    values = []
   419→    years = set()
   420→    for d in monthly_drafts:
   421→        if d.value is None or d.quality_status in ("missing", "not_evaluated"):
   422→            continue
   423→        d_year, d_month = d.period_label.split("-")[:2]
   424→        if int(d_month) != month:
   425→            continue
   426→        values.append(float(d.value))
   427→        years.add(int(d_year))
   428→    if len(years) < BASELINE_MIN_YEARS or len(values) < BASELINE_MIN_YEARS:
   429→        return None
   430→    mean = sum(values) / len(values)
   431→    sd = statistics.stdev(values) if len(values) >= 2 else 0.0
   432→    return {
   433→        "calendar_month": month,
   434→        "window_years": sorted(years),
   435→        "n_years": len(years),
   436→        "mean": round(mean, 4),
   437→        "sd": round(sd, 4),
   438→        "min_years_required": BASELINE_MIN_YEARS,
   439→    }

   442→def _derived_draft(
   443→    spec: FeatureSpec,
   444→    level: str,
   445→    label: str,
   446→    period_start: datetime,
   447→    period_end: datetime,
   448→    base_value: float | None,
   449→    baseline: dict[str, Any] | None,
   450→    calendar_meta: dict[str, Any],
   451→    mode: str,
   452→    reason: str | None = None,
   453→) -> FeatureDraft:
   454→    draft = FeatureDraft(
   455→        feature_code=spec.feature_code, aggregation_level=level, period_label=label,
   456→        period_start=period_start, period_end=period_end, value=None, unit=spec.unit,
   457→        coverage_ratio=None, n_input=0, n_missing=0, quality_status="not_evaluated",
   458→        baseline_reference=baseline,
   459→    )
   460→    if reason:
   461→        draft.provenance = {"reason": reason, "engine_version": calendar_meta["engine_version"]}
   462→        return draft
   463→    if base_value is None:
   464→        draft.provenance = {
   465→            "reason": "Base monthly feature for this period is not available (missing or below coverage).",
   466→            "engine_version": calendar_meta["engine_version"],
   467→        }
   468→        return draft
   469→    if baseline is None:
   470→        draft.provenance = {
   471→            "reason": (
   472→                f"Baseline climatology requires >= {BASELINE_MIN_YEARS} distinct years of this "
   473→                "calendar month within the run window."
   474→            ),
   475→            "engine_version": calendar_meta["engine_version"],
   476→        }
   477→        return draft
   478→    mean = baseline["mean"]
   479→    sd = baseline["sd"]
   480→    if mode == "anomaly":
   481→        draft.value = round(base_value - mean, 4)
   482→        draft.quality_status = "valid"
   483→        draft.coverage_ratio = None
   484→        draft.n_input = baseline["n_years"]
   485→    elif mode == "anomaly_z":
   486→        if sd <= 0:
   487→            draft.provenance = {
   488→                "reason": "Baseline standard deviation is zero; standardized anomaly undefined.",
   489→                "engine_version": calendar_meta["engine_version"],
   490→            }
   491→            draft.baseline_reference = baseline
   492→            return draft
   493→        draft.value = round((base_value - mean) / sd, 4)
   494→        draft.quality_status = "valid"
   495→        draft.n_input = baseline["n_years"]
   496→    draft.provenance = {
   497→        "engine_version": calendar_meta["engine_version"],
   498→        "definition_version": calendar_meta["definition_version"],
   499→        "calendar": {
   500→            "timezone": calendar_meta["timezone"],
   501→            "hemisphere": calendar_meta["hemisphere"],
   502→            "assignment_rule": fscal.CALENDAR_ASSIGNMENT_RULE,
   503→        },
   504→        "required_variables": list(spec.required_variables),
   505→        "inputs": {"observation_count": baseline["n_years"]},
   506→        "baseline": baseline,
   507→        "formula": spec.formula_reference,
   508→    }
   509→    return draft

   512→def compute_feature_drafts(
   513→    *,
   514→    observations: list[dict[str, Any]],
   515→    specs: list[FeatureSpec],
   516→    tz: ZoneInfo,
   517→    hemisphere: str,
   518→    window_start: datetime,
   519→    window_end: datetime,
   520→    aggregation_levels: list[str],
   521→    crop_seasons: list[dict[str, Any]],
   522→    engine_version: str,
   523→    definition_version: str,
   524→) -> tuple[list[FeatureDraft], dict[str, str], list[str]]:
   525→    """Compute all requested feature drafts.
   526→
   527→    Returns (drafts ordered by (level, label, code), not_evaluated_reasons by
   528→    feature_code (representative reason), warnings).
   529→    """
   530→    calendar_meta = {
   531→        "timezone": str(tz),
   532→        "hemisphere": hemisphere,
   533→        "engine_version": engine_version,
   534→        "definition_version": definition_version,
   535→    }
   536→    warnings: list[str] = []
   537→    not_evaluated: dict[str, str] = {}
   538→
   539→    series_by_var: dict[str, DailySeries] = {}
   540→    for spec in specs:
   541→        variable = spec.strategy_params.get("variable")
   542→        if variable and variable not in series_by_var:
   543→            series_by_var[variable] = build_daily_series(observations, variable, tz)
   544→    for variable, series in series_by_var.items():
   545→        if series.duplicate_assignments:
   546→            warnings.append(
   547→                f"{variable}: {series.duplicate_assignments} duplicate local-day assignments "
   548→                "were collapsed (first observation wins)."
   549→            )
   550→        for ts, count in series.excluded_time_supports.items():
   551→            warnings.append(
   552→                f"{variable}: {count} observations with time_support '{ts}' were excluded from "
   553→                "daily-bucket aggregation (temporal resolutions are never mixed)."
   554→            )
   555→
   556→    drafts: list[FeatureDraft] = []
   557→
   558→    # ---- daily passthrough ----
   559→    if "daily" in aggregation_levels:
   560→        w_days = _iter_period_days(window_start, window_end, tz)
   561→        for spec in specs:
   562→            if "daily" not in spec.aggregation_levels:
   563→                continue
   564→            series = _series_for(spec, series_by_var)
   565→            drafts.extend(
   566→                _daily_passthrough(spec, series, w_days, window_start, window_end, tz, calendar_meta)
   567→            )
   568→
   569→    # ---- monthly ----
   570→    monthly_drafts_by_code: dict[str, list[FeatureDraft]] = {}
   571→    if "monthly" in aggregation_levels:
   572→        months = fscal.iter_months(window_start.astimezone(tz).date(), window_end.astimezone(tz).date())
   573→        for year, month in months:
   574→            m_start, m_end_excl = fscal.month_bounds_utc(year, month, tz)
   575→            m_end = m_end_excl - timedelta(seconds=1)
   576→            label = f"{year:04d}-{month:02d}"
   577→            for spec in specs:
   578→                if "monthly" not in spec.aggregation_levels:
   579→                    continue
   580→                if spec.strategy in ("accumulation", "mean", "count_days", "dry_spell_max"):
   581→                    series = _series_for(spec, series_by_var)
   582→                    draft = _aggregate_period(
   583→                        spec, spec.strategy, series, m_start, m_end, tz, calendar_meta,
   584→                        "monthly", label, window_start=window_start, window_end=window_end,
   585→                    )
   586→                    drafts.append(draft)
   587→                    monthly_drafts_by_code.setdefault(spec.feature_code, []).append(draft)
   588→
   589→    # ---- derived monthly (anomaly, z, difference) ----
   590→    if "monthly" in aggregation_levels:
   591→        for spec in specs:
   592→            if "monthly" not in spec.aggregation_levels:
   593→                continue
   594→            if spec.strategy in ("anomaly", "anomaly_z"):
   595→                base_code = spec.strategy_params["base_feature"]
   596→                base_drafts = monthly_drafts_by_code.get(base_code, [])
   597→                for base in sorted(base_drafts, key=lambda d: d.period_label):
   598→                    baseline = _baseline_by_calendar_month(base_drafts, base.period_label)
   599→                    draft = _derived_draft(
   600→                        spec, "monthly", base.period_label, base.period_start, base.period_end,
   601→                        base.value, baseline, calendar_meta, spec.strategy,
   602→                    )
   603→                    if draft.quality_status == "not_evaluated":
   604→                        not_evaluated.setdefault(spec.feature_code, draft.provenance.get("reason", ""))
   605→                    drafts.append(draft)
   606→            elif spec.strategy == "difference":
   607→                left_code = spec.strategy_params["left"]
   608→                right_code = spec.strategy_params["right"]
   609→                left_by_label = {d.period_label: d for d in monthly_drafts_by_code.get(left_code, [])}
   610→                right_by_label = {d.period_label: d for d in monthly_drafts_by_code.get(right_code, [])}
   611→                for label in sorted(set(left_by_label) | set(right_by_label)):
   612→                    left = left_by_label.get(label)
   613→                    right = right_by_label.get(label)
   614→                    if left is None or right is None or right.value is None or left.value is None:
   615→                        draft = _derived_draft(
   616→                            spec, "monthly", label,
   617→                            (left or right).period_start, (left or right).period_end,
   618→                            None, None, calendar_meta, "difference",
   619→                            reason="Requires both monthly totals (precipitation and ET0) for this month.",
   620→                        )
   621→                        not_evaluated.setdefault(spec.feature_code, draft.provenance.get("reason", ""))
   622→                        drafts.append(draft)
   623→                        continue
   624→                    draft = FeatureDraft(
   625→                        feature_code=spec.feature_code, aggregation_level="monthly",
   626→                        period_label=label, period_start=left.period_start,
   627→                        period_end=left.period_end,
   628→                        value=round(left.value - right.value, 4), unit=spec.unit,
   629→                        coverage_ratio=round(min(left.coverage_ratio or 0, right.coverage_ratio or 0), 4),
   630→                        n_input=left.n_input + right.n_input,
   631→                        n_missing=left.n_missing + right.n_missing,
   632→                        quality_status="valid" if left.quality_status == "valid" and right.quality_status == "valid" else "partial",
   633→                    )
   634→                    draft.provenance = {
   635→                        "engine_version": calendar_meta["engine_version"],
   636→                        "definition_version": calendar_meta["definition_version"],
   637→                        "calendar": {
   638→                            "timezone": calendar_meta["timezone"],
   639→                            "hemisphere": calendar_meta["hemisphere"],
   640→                            "assignment_rule": fscal.CALENDAR_ASSIGNMENT_RULE,
   641→                        },
   642→                        "required_variables": list(spec.required_variables),
   643→                        "derived_from": {
   644→                            "left": {"feature_code": left_code, "value": left.value},
   645→                            "right": {"feature_code": right_code, "value": right.value},
   646→                        },
   647→                    }
   648→                    drafts.append(draft)
   649→
   650→    # ---- seasonal ----
   651→    if "seasonal" in aggregation_levels:
   652→        season_labels = _season_labels_overlapping(window_start, window_end, tz, hemisphere)
   653→        seasonal_drafts_by_code: dict[str, list[FeatureDraft]] = {}
   654→        for label in season_labels:
   655→            s_start, s_end_excl = fscal.season_bounds_utc(label, tz)
   656→            s_end = s_end_excl - timedelta(seconds=1)
   657→            for spec in specs:
   658→                if "seasonal" not in spec.aggregation_levels:
   659→                    continue
   660→                if spec.strategy in ("accumulation", "mean", "count_days", "dry_spell_max"):
   661→                    series = _series_for(spec, series_by_var)
   662→                    draft = _aggregate_period(
   663→                        spec, spec.strategy, series, s_start, s_end, tz, calendar_meta,
   664→                        "seasonal", label, window_start=window_start, window_end=window_end,
   665→                    )
   666→                    drafts.append(draft)
   667→                    seasonal_drafts_by_code.setdefault(spec.feature_code, []).append(draft)
   668→        # CV across years per season
   669→        for spec in specs:
   670→            if "seasonal" not in spec.aggregation_levels or spec.strategy != "cv_across_years":
   671→                continue
   672→            base_code = spec.strategy_params["base_feature"]
   673→            base_drafts = seasonal_drafts_by_code.get(base_code, [])
   674→            by_season: dict[str, list[FeatureDraft]] = {}
   675→            for d in base_drafts:
   676→                if d.value is not None and d.quality_status not in ("missing", "not_evaluated"):
   677→                    by_season.setdefault(d.period_label.split("-")[1], []).append(d)
   678→            for season, items in sorted(by_season.items()):
   679→                years = {int(d.period_label.split("-")[0]) for d in items}
   680→                if len(years) < BASELINE_MIN_YEARS:
   681→                    reason = (
   682→                        f"Variability requires >= {BASELINE_MIN_YEARS} distinct years of the "
   683→                        f"{season} season within the run window; found {len(years)}."
   684→                    )
   685→                    not_evaluated.setdefault(spec.feature_code, reason)
   686→                    for d in items:
   687→                        draft = _derived_draft(
   688→                            spec, "seasonal", d.period_label, d.period_start, d.period_end,
   689→                            d.value, None, calendar_meta, "cv", reason=reason,
   690→                        )
   691→                        drafts.append(draft)
   692→                    continue
   693→                values = [float(d.value) for d in items]
   694→                mean = sum(values) / len(values)
   695→                sd = statistics.stdev(values) if len(values) >= 2 else 0.0
   696→                for d in sorted(items, key=lambda x: x.period_label):
   697→                    draft = FeatureDraft(
   698→                        feature_code=spec.feature_code, aggregation_level="seasonal",
   699→                        period_label=d.period_label, period_start=d.period_start,
   700→                        period_end=d.period_end, value=None, unit=spec.unit,
   701→                        coverage_ratio=None, n_input=len(years), n_missing=0,
   702→                        quality_status="not_evaluated",
   703→                    )
   704→                    if mean == 0:
   705→                        reason = "Mean seasonal total is zero; CV undefined."
   706→                        not_evaluated.setdefault(spec.feature_code, reason)
   707→                        draft.provenance = {"reason": reason, "engine_version": calendar_meta["engine_version"]}
   708→                        drafts.append(draft)
   709→                        continue
   710→                    draft.value = round(sd / mean, 4)
   711→                    draft.quality_status = "valid"
   712→                    draft.provenance = {
   713→                        "engine_version": calendar_meta["engine_version"],
   714→                        "definition_version": calendar_meta["definition_version"],
   715→                        "calendar": {
   716→                            "timezone": calendar_meta["timezone"],
   717→                            "hemisphere": calendar_meta["hemisphere"],
   718→                            "assignment_rule": fscal.CALENDAR_ASSIGNMENT_RULE,
   719→                        },
   720→                        "required_variables": list(spec.required_variables),
   721→                        "inputs": {"observation_count": len(items)},
   722→                        "baseline": {
   723→                            "season": season,
   724→                            "window_years": sorted(years),
   725→                            "n_years": len(years),
   726→                            "mean": round(mean, 4),
   727→                            "sd": round(sd, 4),
   728→                        },
   729→                        "formula": spec.formula_reference,
   730→                    }
   731→                    drafts.append(draft)
   732→
   733→    # ---- crop seasons (explicit user windows) ----
   734→    if "crop_season" in aggregation_levels and crop_seasons:
   735→        for season in crop_seasons:
   736→            label = str(season["label"])[:40]
   737→            cs_start = season["start"]
   738→            cs_end = season["end"]
   739→            for spec in specs:
   740→                if "crop_season" not in spec.aggregation_levels:
   741→                    continue
   742→                series = _series_for(spec, series_by_var)
   743→                draft = _aggregate_period(
   744→                    spec, spec.strategy, series, cs_start, cs_end, tz, calendar_meta,
   745→                    "crop_season", label, window_start=window_start, window_end=window_end,
   746→                )
   747→                draft.provenance = dict(draft.provenance)
   748→                draft.provenance["crop_season_window"] = {
   749→                    "label": label,
   750→                    "start": cs_start.date().isoformat(),
   751→                    "end": cs_end.date().isoformat(),
   752→                    "source": "user_supplied_window (crop knowledge base arrives in a later milestone)",
   753→                }
   754→                drafts.append(draft)
   755→
   756→    drafts.sort(key=lambda d: (
   757→        {"daily": 0, "monthly": 1, "seasonal": 2, "crop_season": 3}[d.aggregation_level],
   758→        d.period_label, d.feature_code,
   759→    ))
   760→    return drafts, not_evaluated, warnings

   762→
   763→def _season_labels_overlapping(
   764→    window_start: datetime, window_end: datetime, tz: ZoneInfo, hemisphere: str
   765→) -> list[str]:
   766→    labels: list[str] = []
   767→    start_d = window_start.astimezone(tz).date()
   768→    end_d = window_end.astimezone(tz).date()
   769→    for year in range(start_d.year - 1, end_d.year + 2):
   770→        for season in ("DJF", "MAM", "JJA", "SON"):
   771→            label = f"{year:04d}-{season}"
   772→            s_start, s_end_excl = fscal.season_bounds_utc(label, tz)
   773→            if s_start <= window_end and s_end_excl - timedelta(seconds=1) >= window_start:
   774→                # keep only seasons with at least one local day inside the window
   775→                d = s_start.astimezone(tz).date()
   776→                last = (s_end_excl - timedelta(seconds=1)).astimezone(tz).date()
   777→                while d <= last:
   778→                    if start_d <= d <= end_d:
   779→                        labels.append(label)
   780→                        break
   781→                    d += timedelta(days=1)
   782→    return labels
