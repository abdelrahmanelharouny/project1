   100→function StatusBadge({ status }: { status: string }) {
   101→  const tone =
   102→    status === 'eligible' || status === 'evaluated' || status === 'succeeded' || status === 'scheduled'
   103→      ? 'border-emerald-200 bg-emerald-50 text-emerald-800'
   104→      : status === 'conditional'
   105→        ? 'border-amber-200 bg-amber-50 text-amber-900'
   106→        : status === 'excluded' || status === 'failed' || status === 'over_allocated'
   107→          ? 'border-red-200 bg-red-50 text-red-800'
   108→          : status === 'not_evaluated' || status === 'unknown' || status === 'unscheduled'
   109→            ? 'border-stone-300 bg-stone-100 text-stone-600'
   110→            : 'border-stone-200 bg-stone-50 text-stone-700'
   111→  return (
   112→    <Badge variant="outline" className={`px-1.5 py-0 text-[10px] ${tone}`}>
   113→      {status.replace(/_/g, ' ')}
   114→    </Badge>
   115→  )
   116→}

   135→function ErrorNote({ error }: { error: unknown }) {
   136→  if (!error) return null
   137→  const described = describeError(error)
   138→  return (
   139→    <p className="text-xs font-medium text-red-700" role="alert">
   140→      {described.title}
   141→      {described.description ? ` — ${described.description}` : ''}
   142→    </p>
   143→  )

   251→function PlanView({ fieldId, expert }: { fieldId: string; expert: boolean }) {
   252→  const qc = useQueryClient()
   253→  const [created, setCreated] = useState<ManagementPlan | null>(null)
   254→  const [error, setError] = useState<unknown>(null)
   255→
   256→  const createPlan = useMutation({
   257→    mutationFn: () =>
   258→      createManagementPlan(fieldId, {
   259→        window_start: daysAgo(180),
   260→        window_end: today(),
   261→        season_label: undefined,
   262→        label: 'Management plan',
   263→      }),
   264→    onSuccess: (plan) => {
   265→      setCreated(plan)
   266→      setError(null)
   267→      qc.invalidateQueries({ queryKey: ['management-plan', fieldId] })
   268→    },
   269→    onError: setError,
   270→  })
   271→
   272→  const plan = created
   273→  if (!plan) {
   274→    return (
   275→      <div className="space-y-3">
   276→        <p className="text-sm text-stone-600">
   277→          Build a seasonal management overview for this field: what grew, where
   278→          the soil is exposed, what is planned, and what needs your attention.
   279→        </p>
   280→        <Button onClick={() => createPlan.mutate()} disabled={createPlan.isPending} className="bg-emerald-700 hover:bg-emerald-800">
   281→          {createPlan.isPending ? 'Building…' : 'Build management plan'}
   282→        </Button>
   283→        <ErrorNote error={error} />
   284→      </div>
   285→    )
   286→  }
   287→
   288→  const r = plan.result
   289→  const m = r.soil_cover.metrics

   292→      {plan.warnings.map((w, i) => (

   421→function CoverStudio({ fieldId, expert }: { fieldId: string; expert: boolean }) {
   422→  const qc = useQueryClient()
   423→  const [windowStart, setWindowStart] = useState(today())
   424→  const [windowEnd, setWindowEnd] = useState(daysAgo(-120))
   425→  const [nextCrop, setNextCrop] = useState('')
   426→  const [plan, setPlan] = useState<CoverPlan | null>(null)
   427→  const [adopted, setAdopted] = useState<string | null>(null)
   428→  const [error, setError] = useState<unknown>(null)
   429→
   430→  const items = useQuery({
   431→    queryKey: ['cover-items', fieldId],
   432→    queryFn: () => import('@/lib/api').then((m) => m.listCoverCropPlanItems(fieldId)),
   433→    enabled: Boolean(fieldId),
   434→  })
   435→
   436→  const run = useMutation({
   437→    mutationFn: () => createCoverCropPlan(fieldId, { window_start: windowStart, window_end: windowEnd, next_crop_label: nextCrop || undefined }),
   438→    onSuccess: (p) => {
   439→      setPlan(p)
   440→      setError(null)
   441→      qc.invalidateQueries({ queryKey: ['cover-items', fieldId] })
   442→    },
   443→    onError: setError,
   444→  })
   445→
   446→  const adopt = useMutation({
   447→    mutationFn: (candidate: CoverCandidate) =>
   448→      createCoverCropPlanItem(fieldId, {
   449→        crop_label: candidate.crop_code,
   450→        plan_run_id: plan?.id,
   451→        intended_sowing_start: candidate.sowing_window?.[0],
   452→        intended_termination_window_start: undefined,
   453→      }),
   454→    onSuccess: (item) => {
   455→      setAdopted(item.id)
   456→      qc.invalidateQueries({ queryKey: ['cover-items', fieldId] })
   457→    },
   458→    onError: setError,
   459→  })
   460→
   461→  const candidates = plan?.result.candidates ?? []
   462→
   463→  return (
   464→    <div className="space-y-4">
   465→      <p className="text-sm text-stone-600">
   466→        Evaluate cover-crop candidates for a window. Missing knowledge (no
   467→        regional calendar, no water curve, no termination guidance) keeps a
   468→        candidate <span className="font-medium">conditional</span> — it is
   469→        never presented as a proven recommendation.
   470→      </p>
   471→      <div className="grid gap-2 sm:grid-cols-4">
   472→        <div className="space-y-1">
   473→          <Label htmlFor="cover-start" className="text-xs">Window start</Label>
   474→          <Input id="cover-start" type="date" value={windowStart} onChange={(e) => setWindowStart(e.target.value)} className="h-9 text-sm" />
   475→        </div>
   476→        <div className="space-y-1">
   477→          <Label htmlFor="cover-end" className="text-xs">Window end</Label>
   478→          <Input id="cover-end" type="date" value={windowEnd} onChange={(e) => setWindowEnd(e.target.value)} className="h-9 text-sm" />
   479→        </div>
   480→        <div className="space-y-1">
   481→          <Label htmlFor="cover-next" className="text-xs">Next crop (optional)</Label>
   482→          <Input id="cover-next" value={nextCrop} onChange={(e) => setNextCrop(e.target.value)} placeholder="e.g. maize" className="h-9 text-sm" />
   483→        </div>
   484→        <div className="flex items-end">
   485→          <Button onClick={() => run.mutate()} disabled={run.isPending} className="w-full bg-emerald-700 hover:bg-emerald-800">
   486→            {run.isPending ? 'Evaluating…' : 'Evaluate candidates'}
   487→          </Button>
   488→        </div>
   489→      </div>
   490→      <ErrorNote error={error} />
   491→
   492→      {plan ? (
   493→        <div className="space-y-3" data-testid="cover-candidates">
   494→          {plan.warnings.map((w, i) => (
   495→            <p key={i} className="rounded-md border border-amber-200 bg-amber-50 px-3 py-2 text-xs text-amber-900">{w}</p>
   496→          ))}
   497→          {candidates.length === 0 ? (
   498→            <HonestEmpty>
   499→              No cover-crop-capable species in the active knowledge base could be
   500→              evaluated — an honest empty result.
   501→            </HonestEmpty>
   502→          ) : (
