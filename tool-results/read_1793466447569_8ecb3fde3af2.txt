     1→"""Candidate generation (Milestone 5.3, ADR-0016).
     2→
     3→Pure evaluation of knowledge-base crops for the planning seasons of one field.
     4→Outcomes are first-class: `eligible`, `conditionally_eligible`, `excluded`,
     5→`unevaluated` — each with the reasons and the evidence that produced them.
     6→Missing evidence never becomes zero and never becomes an exclusion.
     7→
     8→Water requirement is a FAO-56 single-Kc screening estimate (ADR-0017) computed
     9→only when BOTH monthly ET0 evidence and sourced Kc anchors exist; its
    10→assumptions are returned with the estimate.
    11→"""
    12→
    13→from __future__ import annotations
    14→
    15→from datetime import date
    16→from typing import Any
    17→
    18→from app.rotation.problem import (
    19→    CandidateAssessment,
    20→    ClimateEvidence,
    21→    FarmContext,
    22→    KnowledgeBaseData,
    23→    KnowledgeCrop,
    24→    SeasonSpec,
    25→    TraitValue,
    26→)
    27→
    28→DURATION_SPILL_TOLERANCE_DAYS = 14  # documented harvest-spill tolerance
    29→
    30→
    31→def _reason(
    32→    kind: str, code: str, message: str, *, severity: str = "info",
    33→    source_refs: list[dict[str, Any]] | None = None,
    34→) -> dict[str, Any]:
    35→    reason = {"kind": kind, "code": code, "message": message, "severity": severity}
    36→    if source_refs:
    37→        reason["source_refs"] = source_refs
    38→    return reason
    39→
    40→
    41→def _missing_dimension(code: str, message: str) -> dict[str, Any]:
    42→    return {"code": code, "message": message}
    43→
    44→
    45→def _trait_ref(trait: TraitValue) -> dict[str, Any]:
    46→    return {
    47→        "type": "knowledge_trait",
    48→        "trait_code": trait.trait_code,
    49→        "pack_code": trait.pack_code,
    50→        "pack_version": trait.pack_version,
    51→        "source_key": trait.source_key,
    52→        "citation_text": trait.citation_text,
    53→        "evidence_class": trait.evidence_class,
    54→        "confidence": trait.confidence,
    55→    }
    56→
    57→
    58→def _mmdd_to_date(year: int, mmdd: str) -> date:
    59→    month, day = mmdd.split("-")
    60→    return date(year, int(month), int(day))
    61→
    62→
    63→def _window_planting_overlap(
    64→    season: SeasonSpec, planting_start: str, planting_end: str
    65→) -> tuple[date, date] | None:
    66→    """Overlap of the recurring (MM-DD) planting window with the season,
    67→    handling year-wrapping windows (start > end)."""
    68→    y0, y1 = season.start_date.year, season.end_date.year
    69→    for base_year in (y0, y0 - 1, y0 + 1):
    70→        try:
    71→            ws = _mmdd_to_date(base_year, planting_start)
    72→        except ValueError:
    73→            continue
    74→        we_year = base_year + 1 if planting_end <= planting_start else base_year
    75→        try:
    76→            we = _mmdd_to_date(we_year, planting_end)
    77→        except ValueError:
    78→            continue
    79→        lo = max(ws, season.start_date)
    80→        hi = min(we, season.end_date)
    81→        if lo <= hi:
    82→            return lo, hi
    83→    return None
    84→
    85→
    86→def _windows_for_field(crop: KnowledgeCrop, hemisphere: str, country_code: str):
    87→    return [
    88→        w
    89→        for w in crop.calendar_windows
    90→        if w.hemisphere == hemisphere and (w.region_code is None or w.region_code == country_code)
    91→    ]
    92→

    94→def estimate_water_requirement_mm(
    95→    crop: KnowledgeCrop, evidence: ClimateEvidence, season: SeasonSpec
    96→) -> tuple[float | None, list[str]]:
    97→    """FAO-56 single-Kc screening estimate (ADR-0017).
    98→
    99→    Returns (mm, assumptions) or (None, [why not evaluable])."""
   100→    missing: list[str] = []
   101→    if crop.water_params is None:
   102→        missing.append("no sourced Kc parameters in the knowledge base — water "
   103→                       "requirement intentionally not estimated (never guessed)")
   104→        return None, missing
   105→    if evidence.season_et0_mm is None or not evidence.monthly_et0:
   106→        missing.append("no reference ET0 evidence for the season")
   107→        return None, missing
   108→
   109→    stage_lengths = {
   110→        "init": crop.numeric_trait("stage_init_days"),
   111→        "dev": crop.numeric_trait("stage_dev_days"),
   112→        "mid": crop.numeric_trait("stage_mid_days"),
   113→        "late": crop.numeric_trait("stage_late_days"),
   114→    }
   115→    if any(v is None for v in stage_lengths.values()):
   116→        missing.append("crop stage lengths missing from the knowledge base")
   117→        return None, missing
   118→
   119→    wp = crop.water_params
   120→    init_len = int(stage_lengths["init"])
   121→    dev_len = int(stage_lengths["dev"])
   122→    mid_len = int(stage_lengths["mid"])
   123→    late_len = int(stage_lengths["late"])
   124→
   125→    def kc_on_day(day_index: int) -> float:
   126→        if day_index < init_len:
   127→            return wp.kc_initial
   128→        if day_index < init_len + dev_len:
   129→            t = (day_index - init_len) / max(dev_len, 1)
   130→            return wp.kc_initial + t * (wp.kc_mid - wp.kc_initial)
   131→        if day_index < init_len + dev_len + mid_len:
   132→            return wp.kc_mid
   133→        t = (day_index - init_len - dev_len - mid_len) / max(late_len, 1)
   134→        t = min(max(t, 0.0), 1.0)
   135→        return wp.kc_mid + t * (wp.kc_late - wp.kc_mid)
   136→
   137→    # Monthly ET0 totals; each month assumed uniform per day (documented).
   138→    # Deepening F-07: a month of the season without ET0 evidence makes the
   139→    # whole estimate unevaluable — dropping the month would UNDER-report the
   140→    # requirement (a silent partial estimate), and partial compliance must
   141→    # never be labelled as a full estimate.
   142→    et0_by_month = {(y, m): v for y, m, v in evidence.monthly_et0}
   143→    season_months: list[tuple[int, int]] = []
   144→    y, m = season.start_date.year, season.start_date.month
   145→    while (y, m) <= (season.end_date.year, season.end_date.month):
   146→        season_months.append((y, m))
   147→        if m == 12:
   148→            y, m = y + 1, 1
   149→        else:
   150→            m += 1
   151→    absent = [(y, m) for y, m in season_months if (y, m) not in et0_by_month]
   152→    if absent:
   153→        missing.extend(
   154→            f"ET0 missing for {y:04d}-{m:02d} — the screening estimate is refused "
   155→            "rather than under-reported (no month is silently dropped)"
   156→            for y, m in absent
   157→        )
   158→        return None, missing
   159→    total = 0.0
   160→    for y, m in season_months:
   161→        et0_month = et0_by_month[(y, m)]
   162→        if y == season.start_date.year and m == season.start_date.month:
   163→            month_start = season.start_date
   164→        else:
   165→            month_start = date(y, m, 1)
   166→        if y == season.end_date.year and m == season.end_date.month:
   167→            month_end = season.end_date
   168→        else:
   169→            # last day of month
   170→            if m == 12:
   171→                month_end = date(y, 12, 31)
   172→            else:
   173→                month_end = date(y, m + 1, 1).toordinal() - 1
   174→                month_end = date.fromordinal(month_end)
   175→        import calendar as _cal
   176→
   177→        days_in_month = _cal.monthrange(y, m)[1]
   178→        month_len = (month_end - month_start).days + 1
   179→        et0_daily = et0_month / days_in_month
   180→        d = month_start
   181→        while d <= month_end:
   182→            day_index = (d - season.start_date).days
   183→            total += kc_on_day(day_index) * et0_daily
   184→            d = date.fromordinal(d.toordinal() + 1)
   185→
   186→    assumptions = [
   187→        "Sowing assumed at the planning-season start (no farmer sowing date supplied).",
   188→        "FAO-56 single-Kc curve reconstructed from sourced stage anchors "
   189→        "(linear ramp across development and late stages).",
   190→        "Monthly ET0 totals distributed uniformly within each calendar month "
   191→        "(screening estimate — the daily water balance arrives in Milestone 7).",
   192→    ]
   193→    return round(total, 2), assumptions
   194→
   195→
   196→def assess_crop(
   197→    crop: KnowledgeCrop,
   198→    season: SeasonSpec,
   199→    knowledge: KnowledgeBaseData,
   200→    farm: FarmContext,
   201→    evidence: ClimateEvidence | None,
   202→    previous_crop_code: str | None,
   203→) -> CandidateAssessment:
   204→    """Evaluate one crop for one planning season."""
   205→    country = farm.country_code
   206→    reasons: list[dict[str, Any]] = []
   207→    unevaluated: list[dict[str, Any]] = []
   208→
   209→    # 0) synthetic-evidence honesty label
   210→    if evidence is not None and evidence.is_synthetic:
   211→        reasons.append(
   212→            _reason(
   213→                "evidence",
   214→                "synthetic_climate_evidence",
   215→                "Climate evidence comes from the SYNTHETIC test fixture — "
   216→                "never real farm evidence; outcomes are demonstration only.",
   217→                severity="warning",
   218→            )
   219→        )
   220→
   221→    # 1) farmer market exclusions (hard, farmer authority)
   222→    if crop.crop_code in farm.market_exclusions:
   223→        return CandidateAssessment(
   224→            crop_code=crop.crop_code,
   225→            status="excluded",
   226→            reasons=(
   227→                _reason(
   228→                    "farmer_constraint",
   229→                    "market_exclusion",
   230→                    "Excluded by an explicit farmer market exclusion.",
   231→                    severity="hard",
   232→                ),
   233→            ),
   234→        )
   235→
   236→    # 2) equipment limits (hard): only when the knowledge base lists a
   237→    #    requirement trait; absent trait ⇒ no equipment restriction claimed.
   238→    equipment_trait = crop.trait("equipment_required")
   239→    if equipment_trait is not None and equipment_trait.value_text:
   240→        required = equipment_trait.value_text
   241→        if required in farm.equipment_limits:
   242→            return CandidateAssessment(
   243→                crop_code=crop.crop_code,
   244→                status="excluded",
   245→                reasons=(
   246→                    _reason(
   247→                        "farmer_constraint",
   248→                        "equipment_restriction",
   249→                        f"Crop requires '{required}', which the farm flags as unavailable.",
   250→                        severity="hard",
   251→                        source_refs=[_trait_ref(equipment_trait)],
   252→                    ),
   253→                ),
   254→            )
   255→
   256→    # 3) calendar compatibility
   257→    windows = _windows_for_field(crop, farm.hemisphere, country)
   258→    if not windows:
   259→        unevaluated.append(
   260→            _missing_dimension(
   261→                "crop_calendar",
   262→                f"No calendar window for hemisphere '{farm.hemisphere}' in the active "
   263→                "knowledge packs — calendar compatibility not evaluated.",
   264→            )
   265→        )
   266→        window_matched = None
   267→        overlap = None
   268→    else:
   269→        window_matched = None
   270→        overlap = None
   271→        for w in windows:
   272→            ov = _window_planting_overlap(season, w.planting_start, w.planting_end)
   273→            if ov is not None:
   274→                window_matched = w
   275→                overlap = ov
   276→                break
   277→        if window_matched is None:
   278→            return CandidateAssessment(
   279→                crop_code=crop.crop_code,
   280→                status="excluded",
   281→                reasons=(
   282→                    _reason(
   283→                        "crop_calendar",
   284→                        "planting_window_outside_season",
   285→                        "No planting window overlaps this planning season "
   286→                        f"(known windows: {', '.join(w.planting_start + '..' + w.planting_end for w in windows)}).",
   287→                        severity="hard",
   288→                        source_refs=[
   289→                            {
   290→                                "type": "crop_calendar_window",
   291→                                "pack_code": w.pack_code,
   292→                                "pack_version": w.pack_version,
   293→                                "source_key": w.source_key,
   294→                                "citation_text": w.citation_text,
   295→                                "evidence_class": w.evidence_class,
   296→                            }
   297→                            for w in windows
   298→                        ],
   299→                    ),
   300→                ),
   301→            )
   302→
   303→    # 4) duration fit
   304→    duration = crop.numeric_trait("duration_days_total")
   305→    duration_trait = crop.resolve_trait("duration_days_total", country)
   306→    if duration is not None and window_matched is not None:
   307→        overflow = int(duration) - season.length_days
   308→        if overflow <= 0:
   309→            reasons.append(
   310→                _reason(
   311→                    "crop_calendar",
   312→                    "duration_fits",
   313→                    f"Crop duration {int(duration)} d fits the {season.length_days}-d season.",
   314→                    source_refs=[_trait_ref(duration_trait)] if duration_trait else None,
   315→                )
   316→            )
   317→        elif overflow <= DURATION_SPILL_TOLERANCE_DAYS:
   318→            reasons.append(
   319→                _reason(
   320→                    "crop_calendar",
   321→                    "duration_spills",
   322→                    f"Crop duration {int(duration)} d exceeds the season by {overflow} d "
   323→                    f"(within the {DURATION_SPILL_TOLERANCE_DAYS}-d harvest-spill tolerance) "
   324→                    "— conditionally eligible.",
   325→                    severity="conditional",
   326→                    source_refs=[_trait_ref(duration_trait)] if duration_trait else None,
   327→                )
   328→            )
   329→        else:
   330→            return CandidateAssessment(
   331→                crop_code=crop.crop_code,
   332→                status="excluded",
   333→                reasons=(
   334→                    _reason(
   335→                        "crop_calendar",
   336→                        "duration_exceeds_season",
   337→                        f"Crop duration {int(duration)} d exceeds the {season.length_days}-d "
   338→                        f"season by {overflow} d (beyond tolerance).",
   339→                        severity="hard",
   340→                        source_refs=[_trait_ref(duration_trait)] if duration_trait else None,
   341→                    ),
   342→                ),
   343→            )
   344→    elif duration is None and window_matched is not None:
   345→        unevaluated.append(
   346→            _missing_dimension(
   347→                "crop_duration",
   348→                "Crop duration absent from the knowledge base — season fit not verified.",
   349→            )
   350→        )
   351→        reasons.append(
   352→            _reason(
   353→                "crop_calendar",
   354→                "window_matched_duration_unknown",
   355→                "Planting window overlaps the season; duration unknown.",
   356→                severity="conditional",
   357→            )
   358→        )
   359→
   360→    # 5) climate suitability (EcoCrop-style ranges vs season aggregates)
   361→    if evidence is None:
   362→        unevaluated.append(
   363→            _missing_dimension(
   364→                "climate_evidence",
   365→                "No feature run available for this field — climate suitability not "
   366→                "evaluated (missing evidence is never treated as zero).",
   367→            )
   368→        )
   369→    else:
   370→        temp_opt_min = crop.numeric_trait("temp_opt_min_degC")
   371→        temp_min_abs = crop.numeric_trait("temp_min_abs_degC")
   372→        temp_opt_max = crop.numeric_trait("temp_opt_max_degC")
   373→        mean_t = evidence.season_mean_temp_degC
   374→        if mean_t is not None and temp_min_abs is not None:
   375→            t_trait_min = crop.resolve_trait("temp_min_abs_degC", country)
   376→            if mean_t < temp_min_abs:
   377→                return CandidateAssessment(
   378→                    crop_code=crop.crop_code,
   379→                    status="excluded",
   380→                    reasons=(
   381→                        _reason(
   382→                            "climate",
   383→                            "temperature_below_absolute_minimum",
   384→                            f"Season mean temperature {mean_t} degC is below the crop's "
   385→                            f"absolute minimum {temp_min_abs} degC.",
   386→                            severity="hard",
   387→                            source_refs=[
   388→                                _trait_ref(t_trait_min) if t_trait_min else {"type": "knowledge_trait", "trait_code": "temp_min_abs_degC"}
   389→                            ],
   390→                        ),
   391→                    ),
   392→                    unevaluated_dimensions=tuple(unevaluated),
   393→                )
   394→        if mean_t is not None and temp_opt_min is not None and temp_opt_max is not None:
   395→            if temp_opt_min <= mean_t <= temp_opt_max:
   396→                reasons.append(
   397→                    _reason(
   398→                        "climate",
   399→                        "temperature_within_optimum",
   400→                        f"Season mean temperature {mean_t} degC within the optimum "
   401→                        f"{temp_opt_min}..{temp_opt_max} degC.",
   402→                        source_refs=[
   403→                            _trait_ref(crop.resolve_trait("temp_opt_min_degC", country))
   404→                        ],
   405→                    )
   406→                )
   407→            else:
   408→                marginal = mean_t >= temp_min_abs if temp_min_abs is not None else True
   409→                reasons.append(
   410→                    _reason(
   411→                        "climate",
   412→                        "temperature_outside_optimum",
   413→                        f"Season mean temperature {mean_t} degC outside the optimum "
   414→                        f"{temp_opt_min}..{temp_opt_max} degC — marginal, needs local "
   415→                        "confirmation.",
   416→                        severity="conditional" if marginal else "warning",
   417→                        source_refs=[
   418→                            _trait_ref(crop.resolve_trait("temp_opt_min_degC", country))
   419→                        ],
   420→                    )
   421→                )
   422→
   423→        precip_min = crop.numeric_trait("precip_min_annual_mm")
   424→        precip_max = crop.numeric_trait("precip_max_annual_mm")
   425→        precip = evidence.season_precipitation_mm
   426→        if precip is not None and precip_min is not None and precip_max is not None:
   427→            if precip < precip_min:
   428→                reasons.append(
   429→                    _reason(
   430→                        "climate",
   431→                        "precipitation_below_reference_range",
   432→                        f"Season precipitation {precip} mm is below the crop's annual "
   433→                        f"reference minimum {precip_min} mm — rainfed deficit likely; "
   434→                        "irrigation-dependent. Compared against an ANNUAL range — a "
   435→                        "coarse screening signal, not a water budget.",
   436→                        severity="conditional",
   437→                        source_refs=[_trait_ref(crop.resolve_trait("precip_min_annual_mm", country))],
   438→                    )
   439→                )
   440→            elif precip > precip_max:
   441→                reasons.append(
   442→                    _reason(
   443→                        "climate",
   444→                        "precipitation_above_reference_range",
   445→                        f"Season precipitation {precip} mm exceeds the crop's annual "
   446→                        f"reference maximum {precip_max} mm — excess-moisture risk.",
   447→                        severity="conditional",
   448→                        source_refs=[_trait_ref(crop.resolve_trait("precip_max_annual_mm", country))],
   449→                    )
   450→                )
   451→        # hot days (generic index, NOT crop-specific heat stress)
   452→        if evidence.season_hot_days_35c:
   453→            reasons.append(
   454→                _reason(
   455→                    "climate",
   456→                    "hot_days_observed",
   457→                    f"{evidence.season_hot_days_35c} days >=35 degC in the evidence window "
   458→                    "(generic meteorological index; crop-specific heat stress needs the "
   459→                    "sourced crop thresholds — not applied here).",
   460→                    severity="info",
   461→                )
   462→            )
   463→
   464→    # 6) soil pH (evidence class surfaced — SoilGrids is a prediction)
   465→    soil = farm.soil
   466→    ph_opt_min = crop.numeric_trait("ph_opt_min")
   467→    ph_opt_max = crop.numeric_trait("ph_opt_max")
   468→    if soil is not None and soil.ph is not None and ph_opt_min is not None and ph_opt_max is not None:
   469→        ph_value, ph_evidence_class = soil.ph
   470→        ph_trait = crop.resolve_trait("ph_opt_min", country)
   471→        if ph_opt_min <= ph_value <= ph_opt_max:
   472→            reasons.append(
   473→                _reason(
   474→                    "soil",
   475→                    "ph_within_optimum",
   476→                    f"Soil pH {ph_value} within crop optimum {ph_opt_min}..{ph_opt_max} "
   477→                    f"(soil evidence class: {ph_evidence_class}).",
   478→                    source_refs=[_trait_ref(ph_trait)] if ph_trait else None,
   479→                )
   480→            )
   481→        else:
   482→            reasons.append(
   483→                _reason(
   484→                    "soil",
   485→                    "ph_outside_optimum",
   486→                    f"Soil pH {ph_value} outside crop optimum {ph_opt_min}..{ph_opt_max} "
   487→                    f"— marginal (soil evidence class: {ph_evidence_class}; a 0-5 cm "
   488→                    "model prediction is not a lab measurement).",
   489→                    severity="conditional",
   490→                    source_refs=[_trait_ref(ph_trait)] if ph_trait else None,
   491→                )
   492→            )
   493→    elif soil is None or soil.ph is None:
   494→        unevaluated.append(
   495→            _missing_dimension(
   496→                "soil_ph",
   497→                "No soil pH evidence for the field — soil suitability not evaluated.",
   498→            )
   499→        )
   500→
   501→    # 7) water requirement estimate (screening, ADR-0017)
   502→    water_mm: float | None = None
   503→    water_assumptions: tuple[str, ...] = ()
   504→    if evidence is not None:
   505→        water_mm, why_not = estimate_water_requirement_mm(crop, evidence, season)
   506→        if water_mm is None:
   507→            unevaluated.append(
   508→                _missing_dimension("water_requirement", " ".join(why_not))
   509→            )
   510→        else:
   511→            water_assumptions = tuple(why_not)  # assumptions when estimated
   512→
   513→    # 8) rotation history (rule-gated, region-scoped)
   514→    if previous_crop_code is not None:
   515→        from app.knowledge.service import rules_between
   516→
   517→        rules = rules_between(knowledge, previous_crop_code, crop.crop_code, country)
   518→        for rule in rules:
   519→            if rule.min_break_seasons < 1:
   520→                reasons.append(
   521→                    _reason(
   522→                        "rotation_knowledge",
   523→                        f"rule_{rule.rule_code}",
   524→                        f"Favourable succession after {previous_crop_code}: {rule.mechanism}",
   525→                        source_refs=[{
   526→                            "type": "rotation_rule",
   527→                            "rule_code": rule.rule_code,
   528→                            "validation_status": rule.validation_status,
   529→                            "source_key": rule.source_key,
   530→                            "citation_text": rule.citation_text,
   531→                        }],
   532→                    )
   533→                )
   534→                continue
   535→            if rule.is_hard_enforceable:
   536→                return CandidateAssessment(
   537→                    crop_code=crop.crop_code,
   538→                    status="excluded",
   539→                    reasons=(
   540→                        _reason(
   541→                            "rotation_knowledge",
   542→                            f"rule_{rule.rule_code}",
   543→                            f"Regionally CONFIRMED rotation rule forbids {previous_crop_code} "
   544→                            f"→ {crop.crop_code} within {rule.min_break_seasons} season(s): "
   545→                            f"{rule.mechanism}",
   546→                            severity="hard",
   547→                            source_refs=[{
   548→                                "type": "rotation_rule",
   549→                                "rule_code": rule.rule_code,
   550→                                "validation_status": rule.validation_status,
   551→                                "source_key": rule.source_key,
   552→                                "citation_text": rule.citation_text,
   553→                            }],
   554→                        ),
   555→                    ),
   556→                )
   557→            reasons.append(
   558→                _reason(
   559→                    "rotation_knowledge",
   560→                    f"rule_{rule.rule_code}",
   561→                    f"REFERENCE rule suggests a break after {previous_crop_code} "
   562→                    f"({rule.mechanism}) — not regionally confirmed; candidate is "
   563→                    "conditionally eligible and the rule is NOT enforced.",
   564→                    severity="conditional",
   565→                    source_refs=[{
   566→                        "type": "rotation_rule",
   567→                        "rule_code": rule.rule_code,
   568→                        "validation_status": rule.validation_status,
   569→                        "source_key": rule.source_key,
   570→                        "citation_text": rule.citation_text,
   571→                    }],
   572→                )
   573→            )
   574→
   575→    # ---- resolve status
   576→    hard_excluded = any(r["severity"] == "hard" for r in reasons)
   577→    if hard_excluded:
   578→        status = "excluded"
   579→    elif unevaluated and any(
   580→        d["code"] == "crop_calendar" for d in unevaluated
   581→    ):
   582→        status = "unevaluated"
   583→    elif any(r["severity"] == "conditional" for r in reasons):
   584→        status = "conditionally_eligible"
   585→    else:
   586→        status = "eligible"
   587→
   588→    matched_labels = tuple(
   589→        f"{w.planting_start}..{w.planting_end}" for w in _windows_for_field(crop, farm.hemisphere, country)
   590→    )
   591→    return CandidateAssessment(
   592→        crop_code=crop.crop_code,
   593→        status=status,
   594→        reasons=tuple(r for r in reasons if r is not None),
   595→        unevaluated_dimensions=tuple(unevaluated),
   596→        water_requirement_mm=water_mm,
   597→        water_estimate_assumptions=water_assumptions,
   598→        calendar_windows_matched=matched_labels,
   599→    )

   602→def generate_candidates(
   603→    knowledge: KnowledgeBaseData,
   604→    farm: FarmContext,
   605→) -> dict[str, dict[str, CandidateAssessment]]:
   606→    """Evaluate every knowledge-base crop for EVERY planning season.
   607→
   608→    Season 0 checks rotation rules against the last history crop; later
   609→    seasons have no known predecessor at candidate time — pairwise rotation
   610→    rules across planned seasons are enforced by the optimizer constraints
   611→    (confirmed rules only) and surfaced as reference evidence in explanations.
   612→    """
   613→    result: dict[str, dict[str, CandidateAssessment]] = {}
   614→    history_codes = [h.get("crop_code") for h in farm.history if h.get("crop_code")]
   615→    for idx, season in enumerate(farm.seasons):
   616→        evidence = farm.climate_by_season.get(season.label)
   617→        previous_crop = history_codes[-1] if (history_codes and idx == 0) else None
   618→        assessments: dict[str, CandidateAssessment] = {}
   619→        for crop_code in sorted(knowledge.crops):
   620→            crop = knowledge.crops[crop_code]
   621→            assessments[crop_code] = assess_crop(
   622→                crop, season, knowledge, farm, evidence, previous_crop
   623→            )
   624→        result[season.label] = assessments
   625→    return result
