    66→    66→function apiErrorText(err: unknown): string {
    67→    67→  if (err instanceof ApiError) {
    68→    68→    const detail = err.details[0]
    69→    69→    const detailText = detail?.reason ? `${detail.field}: ${detail.reason}` : detail?.field
    70→    70→    return detailText ? `${err.message} (${detailText})` : err.message
    71→    71→  }
    72→    72→  return err instanceof Error ? err.message : 'Unexpected error'
    73→    73→}

    75→    75→function fmt(value: number | null | undefined, digits = 4): string {
    76→    76→  if (value === null || value === undefined) return '—'
    77→    77→  if (Number.isInteger(value)) return String(value)
    78→    78→  return value.toFixed(digits)
    79→    79→}

    81→    81→function shortHash(hash: string | null | undefined): string {
    82→    82→  return hash ? `${hash.slice(0, 12)}…` : '—'
    83→    83→}

   180→   180→  const { data, isLoading, isError, error } = useQuery({
   181→   181→    queryKey: ['sensitivity-responses', analysisId, parameterCode, classificationFilter],
   182→   182→    queryFn: () =>
   183→   183→      listSensitivityResponses(analysisId, {
   184→   184→        parameter_code: parameterCode ?? undefined,
   185→   185→        response_classification: classificationFilter || undefined,
   186→   186→      }),
   187→   187→    enabled: Boolean(analysisId),
   188→   188→  })
   189→   189→  // Unbounded feature sets would explode the DOM; render a capped window
   190→   190→  // behind the classification/parameter filters. Evaluated responses surface
   191→   191→  // first — the alphabetically-first rows are often honest not_evaluated ones.
   192→   192→  const MAX_ROWS = 100
   193→   193→  const rows = useMemo(() => {
   194→   194→    const items = data?.items ?? []
   195→   195→    const rank: Record<string, number> = {
   196→   196→      evaluated: 0,
   197→   197→      partially_evaluated: 1,
   198→   198→      not_evaluated: 2,
   199→   199→    }
   200→   200→    return [...items].sort((a, b) => {
   201→   201→      const r = (rank[a.response_status] ?? 3) - (rank[b.response_status] ?? 3)
   202→   202→      if (r !== 0) return r
   203→   203→      return `${a.feature_code}|${a.aggregation_level}|${a.period_label}`.localeCompare(
   204→   204→        `${b.feature_code}|${b.aggregation_level}|${b.period_label}`,
   205→   205→      )
   206→   206→    })
   207→   207→  }, [data])
   208→   208→  const visible = rows.slice(0, MAX_ROWS)
   209→   209→
   210→   210→  if (isLoading)
   211→   211→    return (
   212→   212→      <div className="space-y-2">
   213→   213→        <Skeleton className="h-6 w-64" />
   214→   214→        <Skeleton className="h-40 w-full" />
   215→   215→      </div>
   216→   216→    )
   217→   217→  if (isError) return <p className="text-sm text-red-700">{apiErrorText(error)}</p>
   218→   218→  if (!data) return null
   219→   219→
   220→   220→  return (
   221→   221→    <Card>
   222→   222→      <CardHeader className="pb-3">
   223→   223→        <CardTitle className="text-base">One-factor-at-a-time responses</CardTitle>

   225→   225→          Each parameter varies over the explored bounds while every other configured

   227→   227→          compared across different units.
