   153→function CreateFarmDialog({ open, onOpenChange }: { open: boolean; onOpenChange: (open: boolean) => void }) {
   154→  const queryClient = useQueryClient()
   155→  const t = useTranslations('farms')
   156→  const tCommon = useTranslations('common')
   157→  const [name, setName] = useState('')
   158→  const [country, setCountry] = useState('')
   159→  const [adminRegion, setAdminRegion] = useState('')
   160→  const [timezone, setTimezone] = useState('UTC')
   161→  const [currency, setCurrency] = useState('')
   162→  const [unitSystem, setUnitSystem] = useState<UnitSystem>('metric')
   163→  const [language, setLanguage] = useState('en')
   164→  const [formError, setFormError] = useState<string | null>(null)
   165→
   166→  // Reset form state when the dialog closes (handler-based, no effect needed).
   167→  const handleOpenChange = (next: boolean) => {
   168→    if (!next) {
   169→      setName('')
   170→      setCountry('')
   171→      setAdminRegion('')
   172→      setTimezone('UTC')
   173→      setCurrency('')
   174→      setUnitSystem('metric')
   175→      setLanguage('en')
   176→      setFormError(null)
   177→    }
   178→    onOpenChange(next)
   179→  }
   180→
   181→  const mutation = useMutation({
   182→    mutationFn: (input: FarmCreateInput) => createFarm(input),
   183→    onSuccess: (farm) => {
   184→      queryClient.invalidateQueries({ queryKey: ['farms'] })
   185→      toast.success(t('farmCreated', { name: farm.name }), {
   186→        description: `${farm.timezone} · ${farm.currency_code} · ${farm.unit_system}`,

   191→      const { title, description } = describeError(err)
   192→      setFormError(title)
   193→      toast.error(title, { description })
   194→    },
   195→  })

   197→  const submit = (event: FormEvent) => {
   198→    event.preventDefault()

   200→    const currencyClean = currency.trim().toUpperCase()
   201→    const languageClean = language.trim().toLowerCase()
   202→    if (!name.trim()) return setFormError(t('nameRequired'))
   203→    if (!/^[A-Z]{2}$/.test(countryClean)) return setFormError(t('countryRequired'))
   204→    if (!/^[A-Z]{3}$/.test(currencyClean)) return setFormError(t('currencyRequired'))
   205→    if (!/^[a-z]{2}$/.test(languageClean)) return setFormError(t('languageRequired'))

   207→    mutation.mutate({
   208→      name: name.trim(),
   209→      country_code: countryClean,
   210→      admin_region: adminRegion.trim() || undefined,
   211→      timezone,
   212→      currency_code: currencyClean,
   213→      unit_system: unitSystem,
   214→      preferred_language: languageClean,
   215→    })
   216→  }
   217→
   218→  return (
   219→    <Dialog open={open} onOpenChange={handleOpenChange}>
   220→      <DialogContent className="max-h-[85vh] overflow-y-auto scrollbar-thin sm:max-w-lg">
   221→        <DialogHeader>
   222→          <DialogTitle>{t('createTitle')}</DialogTitle>
   223→          <DialogDescription>
   224→            {t('createDesc')}
   225→          </DialogDescription>
   226→        </DialogHeader>
   227→        <form onSubmit={submit} className="space-y-4" aria-label={t('createFormAria')}>
   228→          <div className="space-y-1.5">
   229→            <Label htmlFor="farm-name">{t('nameLabel')}</Label>
   230→            <Input
   231→              id="farm-name"
   232→              placeholder={t('namePlaceholder')}

   234→              onChange={(event) => setName(event.target.value)}
   235→            />
   236→          </div>
   237→          <div className="grid gap-4 sm:grid-cols-2">
   238→            <div className="space-y-1.5">
   239→              <Label htmlFor="farm-country">{t('countryLabel')}</Label>
   240→              <Input
   241→                id="farm-country"
   242→                maxLength={2}
   243→                placeholder="EG"
   244→                value={country}
   245→                onChange={(event) => setCountry(event.target.value.toUpperCase())}
   246→                className="uppercase"
   247→              />
   248→              <p className="text-xs text-stone-500">{t('countryHint')}</p>

   250→            <div className="space-y-1.5">

   252→              <Input
   253→                id="farm-region"

   338→function AddFieldDialog({
   339→  farm,
   340→  open,
   341→  onOpenChange,
   342→  onFieldSelected,
   343→}: {
   344→  farm: Farm
   345→  open: boolean
   346→  onOpenChange: (open: boolean) => void
   347→  onFieldSelected: (field: { id: string; name: string; farmId: string }) => void
   348→}) {
   349→  const queryClient = useQueryClient()
   350→  const t = useTranslations('farms')
   351→  const tCommon = useTranslations('common')
   352→  const [name, setName] = useState('')
   353→  const [geometry, setGeometry] = useState('')
   354→  const [parseError, setParseError] = useState<string | null>(null)
   355→  const [apiError, setApiError] = useState<{ title: string; description?: string } | null>(null)
   356→  const [created, setCreated] = useState<FieldCreateResponse | null>(null)
   357→
   358→  // Reset form state when the dialog closes.
   359→  const handleOpenChange = (next: boolean) => {
   360→    if (!next) {
   361→      setName('')
   362→      setGeometry('')
   363→      setParseError(null)
   364→      setApiError(null)
   365→      setCreated(null)
   366→    }
   367→    onOpenChange(next)
   368→  }
   369→
   370→  const mutation = useMutation({
   371→    mutationFn: (payload: { name: string; geometry: GeoJSONGeometry }) =>
   372→      createField(farm.id, { name: payload.name, geometry: payload.geometry }),
   373→    onSuccess: (result) => {
   374→      queryClient.invalidateQueries({ queryKey: ['fields', farm.id] })
   375→      queryClient.invalidateQueries({ queryKey: ['farms'] })
   376→      if (result.geometry_warnings.length > 0) {
   377→        setCreated(result)
   378→        toast.warning(t('fieldCreatedWarnings'), {
   379→          description: t('warningsFromPostgis', { count: result.geometry_warnings.length }),
   380→        })
   381→      } else {
   382→        toast.success(t('fieldCreated', { name: result.name }), {
   383→          description: t('fieldCreatedDesc', {
   384→            area: formatArea(result.area_ha),
   385→            unit: tCommon('ha'),
   386→            version: result.boundary_version,
   387→          }),
   388→        })
   389→        onFieldSelected({ id: result.id, name: result.name, farmId: farm.id })
   390→        handleOpenChange(false)
   391→      }
   392→    },
   393→    onError: (err) => {
   394→      const { title, description } = describeError(err)
   395→      setApiError({ title, description })
   396→      toast.error(title, { description })
   397→    },
   398→  })
   399→
   400→  const submit = (event: FormEvent) => {
   401→    event.preventDefault()
   402→    setApiError(null)
   403→    setCreated(null)
   404→    if (!name.trim()) {
   405→      setParseError(t('fieldNameRequired'))
   406→      return
   407→    }
   408→    const parsed = parseGeometryInput(geometry)
   409→    if (!parsed.ok) {
   410→      setParseError(parsed.error)
   411→      return
   412→    }
   413→    setParseError(null)
   414→    mutation.mutate({ name: name.trim(), geometry: parsed.geometry })
   415→  }
   416→
   417→  return (
   418→    <Dialog open={open} onOpenChange={handleOpenChange}>
   419→      <DialogContent className="max-h-[85vh] overflow-y-auto scrollbar-thin sm:max-w-xl">
   420→        <DialogHeader>
   421→          <DialogTitle>{t('addFieldTitle', { farm: farm.name })}</DialogTitle>
   422→          <DialogDescription>
   423→            {t('addFieldDesc')}
   424→          </DialogDescription>
   425→        </DialogHeader>
   426→        <form onSubmit={submit} className="space-y-4" aria-label={t('addFieldFormAria')}>
   427→          <div className="space-y-1.5">
   428→            <Label htmlFor="field-name">{t('fieldLabel')}</Label>
   429→            <Input
   430→              id="field-name"
   431→              placeholder={t('fieldPlaceholder')}
   432→              value={name}
   433→              onChange={(event) => setName(event.target.value)}
   434→            />
   435→          </div>
   436→          <GeometryInput id="field-geometry" value={geometry} onChange={setGeometry} error={parseError} />
   437→
   438→          {created ? (
   439→            <div className="space-y-2">
   440→              <Alert className="border-emerald-300 bg-emerald-50 [&>svg]:text-emerald-700">
   441→                <MapPin className="h-4 w-4" />
   442→                <AlertTitle className="text-emerald-900">
   443→                  {t('fieldCreatedArea', {
   444→                    name: created.name,
   445→                    area: formatArea(created.area_ha),
   446→                    unit: tCommon('ha'),
   447→                  })}
   448→                </AlertTitle>
   449→                <AlertDescription className="text-emerald-900">
   450→                  {t('boundaryPassport', {
   451→                    version: created.boundary_version,
   452→                    status: created.data_passport_status,
   453→                  })}
   454→                </AlertDescription>
   455→              </Alert>
   456→              <GeometryWarningsAlert
   457→                warnings={created.geometry_warnings}
   458→                repaired={created.geometry_repaired}
   459→                title={t('postgisWarnings')}
   460→              />
   461→              <div className="flex justify-end gap-2">
   462→                <Button
   463→                  type="button"
   464→                  variant="outline"
   465→                  onClick={() => {
   466→                    onFieldSelected({ id: created.id, name: created.name, farmId: farm.id })
   467→                    handleOpenChange(false)
   468→                  }}
   469→                >
   470→                  {t('openNewField')}
   471→                </Button>
   472→              </div>
   473→            </div>
   474→          ) : null}
   475→
   476→          {apiError ? (
   477→            <Alert variant="destructive">
   478→              <TriangleAlert className="h-4 w-4" />
   479→              <AlertTitle>{apiError.title}</AlertTitle>
   480→              <AlertDescription>{apiError.description ?? t('adjustGeometryRetry')}</AlertDescription>
   481→            </Alert>
   482→          ) : null}
   483→
   484→          <DialogFooter>
   485→            <Button type="button" variant="outline" onClick={() => handleOpenChange(false)}>
   486→              {created ? tCommon('close') : tCommon('cancel')}
   487→            </Button>
   488→            <Button type="submit" className="bg-emerald-700 hover:bg-emerald-800" disabled={mutation.isPending}>
   489→              {mutation.isPending ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : <Plus className="h-4 w-4" aria-hidden />}
   490→              {t('createFieldSubmit')}
   491→            </Button>
   492→          </DialogFooter>
   493→        </form>
   494→      </DialogContent>
   495→    </Dialog>
   496→  )
   497→}

   503→function RenameFieldDialog({
   504→  field,
   505→  open,
   506→  onOpenChange,
   507→}: {
   508→  field: Field
   509→  open: boolean
   510→  onOpenChange: (open: boolean) => void
   511→}) {
   512→  const queryClient = useQueryClient()
   513→  const t = useTranslations('farms')
   514→  const tCommon = useTranslations('common')
   515→  const renameSelectedField = useFieldShiftUi((s) => s.renameSelectedField)
   516→  const [name, setName] = useState(field.name)
   517→
   518→  // Sync the input with the current field name each time the dialog opens.
   519→  const handleOpenChange = (next: boolean) => {
   520→    if (next) setName(field.name)
   521→    onOpenChange(next)
   522→  }
   523→
   524→  const mutation = useMutation({
   525→    mutationFn: () => patchField(field.id, { name: name.trim() }),
   526→    onSuccess: (updated) => {
   527→      queryClient.invalidateQueries({ queryKey: ['fields'] })
   528→      renameSelectedField(updated.id, updated.name)
   529→      toast.success(t('fieldRenamed', { name: updated.name }))
   530→      onOpenChange(false)
   531→    },
   532→    onError: (err) => {
   533→      const { title, description } = describeError(err)
   534→      toast.error(title, { description })
   535→    },
   536→  })
   537→
   538→  const submit = (event: FormEvent) => {
   539→    event.preventDefault()
   540→    if (!name.trim() || name.trim() === field.name) {
   541→      onOpenChange(false)
   542→      return
   543→    }
   544→    mutation.mutate()
   545→  }
   546→
   547→  return (
   548→    <Dialog open={open} onOpenChange={handleOpenChange}>
   549→      <DialogContent className="sm:max-w-sm">
   550→        <DialogHeader>
   551→          <DialogTitle>{t('renameTitle')}</DialogTitle>
   552→          <DialogDescription>{t('renameDesc')}</DialogDescription>
   553→        </DialogHeader>
   554→        <form onSubmit={submit} className="space-y-4" aria-label={t('renameFormAria')}>
   555→          <div className="space-y-1.5">
   556→            <Label htmlFor="rename-field">{t('fieldLabel')}</Label>
   557→            <Input id="rename-field" value={name} onChange={(event) => setName(event.target.value)} />
   558→          </div>
   559→          <DialogFooter>
   560→            <Button type="button" variant="outline" onClick={() => handleOpenChange(false)}>
   561→              {tCommon('cancel')}
   562→            </Button>
   563→            <Button type="submit" className="bg-emerald-700 hover:bg-emerald-800" disabled={mutation.isPending}>
   564→              {mutation.isPending ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : null}
   565→              {t('saveName')}
   566→            </Button>
   567→          </DialogFooter>
   568→        </form>
   569→      </DialogContent>
   570→    </Dialog>
   571→  )
   572→}

   639→    }

   642→  }
