   158→   158→  const [country, setCountry] = useState('')

   167→   167→  const handleOpenChange = (next: boolean) => {
   168→   168→    if (!next) {

   171→   171→      setAdminRegion('')
   172→   172→      setTimezone('UTC')

   174→   174→      setUnitSystem('metric')

   208→   208→      name: name.trim(),

   212→   212→      currency_code: currencyClean,
   213→   213→      unit_system: unitSystem,

   220→   220→      <DialogContent className="max-h-[85vh] overflow-y-auto scrollbar-thin sm:max-w-lg">

   231→   231→              id="farm-name"
   232→   232→              placeholder={t('namePlaceholder')}
   233→   233→              value={name}
   234→   234→              onChange={(event) => setName(event.target.value)}
   235→   235→            />
   236→   236→          </div>
   237→   237→          <div className="grid gap-4 sm:grid-cols-2">
   238→   238→            <div className="space-y-1.5">
   239→   239→              <Label htmlFor="farm-country">{t('countryLabel')}</Label>
   240→   240→              <Input
   241→   241→                id="farm-country"
   242→   242→                maxLength={2}
   243→   243→                placeholder="EG"
   244→   244→                value={country}
   245→   245→                onChange={(event) => setCountry(event.target.value.toUpperCase())}
   246→   246→                className="uppercase"
   247→   247→              />
   248→   248→              <p className="text-xs text-stone-500">{t('countryHint')}</p>
   249→   249→            </div>

   254→   254→                placeholder={t('regionPlaceholder')}

   256→   256→                onChange={(event) => setAdminRegion(event.target.value)}
   257→   257→              />
   258→   258→            </div>
   259→   259→          </div>
   260→   260→          <div className="grid gap-4 sm:grid-cols-2">
   261→   261→            <div className="space-y-1.5">
   262→   262→              <Label htmlFor="farm-timezone">{t('timezoneLabel')}</Label>
   263→   263→              <Select value={timezone} onValueChange={setTimezone}>
   264→   264→                <SelectTrigger id="farm-timezone" className="w-full">
   265→   265→                  <SelectValue />
   266→   266→                </SelectTrigger>
   267→   267→                <SelectContent>
   268→   268→                  {TIMEZONES.map((tz) => (

   274→   274→              </Select>

   277→   277→              <Label htmlFor="farm-currency">{t('currencyLabel')}</Label>
   278→   278→              <Input
   279→   279→                id="farm-currency"
   280→   280→                maxLength={3}

   338→   338→function AddFieldDialog({
   339→   339→  farm,
   340→   340→  open,
   341→   341→  onOpenChange,
   342→   342→  onFieldSelected,
   343→   343→}: {
   344→   344→  farm: Farm
   345→   345→  open: boolean
   346→   346→  onOpenChange: (open: boolean) => void
   347→   347→  onFieldSelected: (field: { id: string; name: string; farmId: string }) => void
   348→   348→}) {
   349→   349→  const queryClient = useQueryClient()
   350→   350→  const t = useTranslations('farms')
   351→   351→  const tCommon = useTranslations('common')
   352→   352→  const [name, setName] = useState('')
   353→   353→  const [geometry, setGeometry] = useState('')
   354→   354→  const [parseError, setParseError] = useState<string | null>(null)
   355→   355→  const [apiError, setApiError] = useState<{ title: string; description?: string } | null>(null)
   356→   356→  const [created, setCreated] = useState<FieldCreateResponse | null>(null)
   357→   357→
   358→   358→  // Reset form state when the dialog closes.
   359→   359→  const handleOpenChange = (next: boolean) => {
   360→   360→    if (!next) {
   361→   361→      setName('')
   362→   362→      setGeometry('')
   363→   363→      setParseError(null)
   364→   364→      setApiError(null)
   365→   365→      setCreated(null)
   366→   366→    }
   367→   367→    onOpenChange(next)
   368→   368→  }
   369→   369→
   370→   370→  const mutation = useMutation({
   371→   371→    mutationFn: (payload: { name: string; geometry: GeoJSONGeometry }) =>
   372→   372→      createField(farm.id, { name: payload.name, geometry: payload.geometry }),
   373→   373→    onSuccess: (result) => {
   374→   374→      queryClient.invalidateQueries({ queryKey: ['fields', farm.id] })
   375→   375→      queryClient.invalidateQueries({ queryKey: ['farms'] })
   376→   376→      if (result.geometry_warnings.length > 0) {
   377→   377→        setCreated(result)
   378→   378→        toast.warning(t('fieldCreatedWarnings'), {
   379→   379→          description: t('warningsFromPostgis', { count: result.geometry_warnings.length }),
   380→   380→        })
   381→   381→      } else {
   382→   382→        toast.success(t('fieldCreated', { name: result.name }), {
   383→   383→          description: t('fieldCreatedDesc', {
   384→   384→            area: formatArea(result.area_ha),
   385→   385→            unit: tCommon('ha'),
   386→   386→            version: result.boundary_version,
   387→   387→          }),
   388→   388→        })
   389→   389→        onFieldSelected({ id: result.id, name: result.name, farmId: farm.id })
   390→   390→        handleOpenChange(false)
   391→   391→      }
   392→   392→    },
   393→   393→    onError: (err) => {
   394→   394→      const { title, description } = describeError(err)
   395→   395→      setApiError({ title, description })
   396→   396→      toast.error(title, { description })
   397→   397→    },
   398→   398→  })
   399→   399→
   400→   400→  const submit = (event: FormEvent) => {
   401→   401→    event.preventDefault()
   402→   402→    setApiError(null)
   403→   403→    setCreated(null)
   404→   404→    if (!name.trim()) {
   405→   405→      setParseError(t('fieldNameRequired'))
   406→   406→      return
   407→   407→    }
   408→   408→    const parsed = parseGeometryInput(geometry)
   409→   409→    if (!parsed.ok) {
   410→   410→      setParseError(parsed.error)
   411→   411→      return
   412→   412→    }
   413→   413→    setParseError(null)
   414→   414→    mutation.mutate({ name: name.trim(), geometry: parsed.geometry })
   415→   415→  }
   416→   416→
   417→   417→  return (
   418→   418→    <Dialog open={open} onOpenChange={handleOpenChange}>
   419→   419→      <DialogContent className="max-h-[85vh] overflow-y-auto scrollbar-thin sm:max-w-xl">
   420→   420→        <DialogHeader>
   421→   421→          <DialogTitle>{t('addFieldTitle', { farm: farm.name })}</DialogTitle>
   422→   422→          <DialogDescription>
   423→   423→            {t('addFieldDesc')}
   424→   424→          </DialogDescription>
   425→   425→        </DialogHeader>
   426→   426→        <form onSubmit={submit} className="space-y-4" aria-label={t('addFieldFormAria')}>
   427→   427→          <div className="space-y-1.5">
   428→   428→            <Label htmlFor="field-name">{t('fieldLabel')}</Label>
   429→   429→            <Input
   430→   430→              id="field-name"
   431→   431→              placeholder={t('fieldPlaceholder')}
   432→   432→              value={name}
   433→   433→              onChange={(event) => setName(event.target.value)}
   434→   434→            />
   435→   435→          </div>
   436→   436→          <GeometryInput id="field-geometry" value={geometry} onChange={setGeometry} error={parseError} />
   437→   437→
   438→   438→          {created ? (
   439→   439→            <div className="space-y-2">
   440→   440→              <Alert className="border-emerald-300 bg-emerald-50 [&>svg]:text-emerald-700">
   441→   441→                <MapPin className="h-4 w-4" />
   442→   442→                <AlertTitle className="text-emerald-900">
   443→   443→                  {t('fieldCreatedArea', {
   444→   444→                    name: created.name,
   445→   445→                    area: formatArea(created.area_ha),
   446→   446→                    unit: tCommon('ha'),
   447→   447→                  })}
   448→   448→                </AlertTitle>
   449→   449→                <AlertDescription className="text-emerald-900">
   450→   450→                  {t('boundaryPassport', {
   451→   451→                    version: created.boundary_version,
   452→   452→                    status: created.data_passport_status,
   453→   453→                  })}
   454→   454→                </AlertDescription>
   455→   455→              </Alert>
   456→   456→              <GeometryWarningsAlert
   457→   457→                warnings={created.geometry_warnings}
   458→   458→                repaired={created.geometry_repaired}
   459→   459→                title={t('postgisWarnings')}
   460→   460→              />
   461→   461→              <div className="flex justify-end gap-2">
   462→   462→                <Button
   463→   463→                  type="button"
   464→   464→                  variant="outline"
   465→   465→                  onClick={() => {
   466→   466→                    onFieldSelected({ id: created.id, name: created.name, farmId: farm.id })
   467→   467→                    handleOpenChange(false)
   468→   468→                  }}
   469→   469→                >
   470→   470→                  {t('openNewField')}
   471→   471→                </Button>
   472→   472→              </div>
   473→   473→            </div>
   474→   474→          ) : null}
   475→   475→
   476→   476→          {apiError ? (
