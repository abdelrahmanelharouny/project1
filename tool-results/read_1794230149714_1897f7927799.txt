     3→/**
     4→ * Sensitivity & Uncertainty Workspace (M7.4).
     5→ *
     6→ * Analyses ONE immutable M7.3 execution run: varies the run's configured
     7→ * transformation parameters over user-supplied bounds, shows OAT responses,
     8→ * the explored parameter-range envelope, the optional joint grid, and the
     9→ * uncertainty registry — with complete provenance and honesty disclosures.
    10→ *
    11→ * Honesty rules rendered here (M7.4 spec §2/§4):
    12→ * - The envelope is always labelled "explored parameter-range envelope" — a
    13→ *   deterministic sensitivity result, NOT a confidence or probability interval.
    14→ * - Parameters not configured in the source execution appear as not
    15→ *   applicable with the API's reason — never hidden, never editable.
    16→ * - Elasticity is only shown when the API reports it available; otherwise the
    17→ *   structured unavailability reason is printed.
    18→ * - The interaction contrast is always labelled "finite-grid interaction
    19→ *   contrast" — never a Sobol index.
    20→ * - No probabilities, forecasts, yield impact or climate projections anywhere.
    21→ */

    66→function apiErrorText(err: unknown): string {
    67→  if (err instanceof ApiError) {
    68→    const detail = err.details[0]
    69→    const detailText = detail?.reason ? `${detail.field}: ${detail.reason}` : detail?.field
    70→    return detailText ? `${err.message} (${detailText})` : err.message
    71→  }
    72→  return err instanceof Error ? err.message : 'Unexpected error'
    73→}
    74→
    75→function fmt(value: number | null | undefined, digits = 4): string {
    76→  if (value === null || value === undefined) return '—'
    77→  if (Number.isInteger(value)) return String(value)
    78→  return value.toFixed(digits)
    79→}
    80→
    81→function shortHash(hash: string | null | undefined): string {
    82→  return hash ? `${hash.slice(0, 12)}…` : '—'
    83→}

   178→function ResponsesCard({ analysisId, parameterCode }: { analysisId: string; parameterCode: string | null }) {
   179→  const [classificationFilter, setClassificationFilter] = useState<string>('')
   180→  const { data, isLoading, isError, error } = useQuery({
   181→    queryKey: ['sensitivity-responses', analysisId, parameterCode, classificationFilter],
   182→    queryFn: () =>
   183→      listSensitivityResponses(analysisId, {
   184→        parameter_code: parameterCode ?? undefined,
   185→        response_classification: classificationFilter || undefined,
   186→      }),
   187→    enabled: Boolean(analysisId),
   188→  })
   189→  // Unbounded feature sets would explode the DOM; render a capped window
   190→  // behind the classification/parameter filters. Evaluated responses surface
   191→  // first — the alphabetically-first rows are often honest not_evaluated ones.
   192→  const MAX_ROWS = 100
   193→  const rows = useMemo(() => {
   194→    const items = data?.items ?? []
   195→    const rank: Record<string, number> = {
   196→      evaluated: 0,
   197→      partially_evaluated: 1,
   198→      not_evaluated: 2,
   199→    }
   200→    return [...items].sort((a, b) => {
   201→      const r = (rank[a.response_status] ?? 3) - (rank[b.response_status] ?? 3)
   202→      if (r !== 0) return r
   203→      return `${a.feature_code}|${a.aggregation_level}|${a.period_label}`.localeCompare(
   204→        `${b.feature_code}|${b.aggregation_level}|${b.period_label}`,
   205→      )
   206→    })
   207→  }, [data])
   208→  const visible = rows.slice(0, MAX_ROWS)
   209→
   210→  if (isLoading)
   211→    return (
   212→      <div className="space-y-2">
   213→        <Skeleton className="h-6 w-64" />
   214→        <Skeleton className="h-40 w-full" />
   215→      </div>
   216→    )
   217→  if (isError) return <p className="text-sm text-red-700">{apiErrorText(error)}</p>
   218→  if (!data) return null
   219→
   220→  return (
   221→    <Card>
   222→      <CardHeader className="pb-3">
   223→        <CardTitle className="text-base">One-factor-at-a-time responses</CardTitle>

   225→          Each parameter varies over the explored bounds while every other configured
   226→          parameter is held at its reference value. Slopes keep their units and are never
   227→          compared across different units.

   230→          {['', 'monotonic_increasing', 'monotonic_decreasing', 'threshold_suspected', 'non_monotonic', 'constant'].map(
   231→            (c) => (
   232→              <button
   233→                key={c || 'all'}
   234→                onClick={() => setClassificationFilter(c)}
   235→                className={`rounded-full border px-2.5 py-0.5 text-[11px] transition-colors ${
   236→                  classificationFilter === c
   237→                    ? 'border-stone-800 bg-stone-800 text-white'
   238→                    : 'border-stone-300 bg-white text-stone-600 hover:bg-stone-50'
   239→                }`}
   240→              >
   241→                {c === '' ? 'all' : c.replace(/_/g, ' ')}
   242→              </button>
   243→            ),
   244→          )}
   245→        </div>
   246→      </CardHeader>
   247→      <CardContent>
   248→        {rows.length === 0 ? (
   249→          <p className="py-6 text-center text-sm text-stone-500">No responses for this filter.</p>

   430→function VerifyCard({ analysisId }: { analysisId: string }) {
   431→  const queryClient = useQueryClient()
   432→  const verify = useMutation({
   433→    mutationFn: () => verifySensitivityAnalysis(analysisId),
   434→    onSuccess: () => queryClient.invalidateQueries({ queryKey: ['sensitivity-analysis', analysisId] }),
   435→  })
   436→  const result = verify.data
   437→  const flags: Array<readonly [keyof SensitivityVerifyResponse, string]> = result
   438→    ? [
   439→        ['output_checksum_valid', 'output checksum'],
   440→        ['source_run_intact', 'source run intact'],
   441→        ['parent_snapshot_intact', 'parent snapshot intact'],
   442→        ['manifest_checksum_matches', 'manifest checksum matches'],
   443→        ['reference_reproduction_consistent', 'reference reproduction consistent'],
   444→      ]
   445→    : []
   446→  return (
   447→    <Card>
   448→      <CardHeader className="pb-3">
   449→        <CardTitle className="text-base">Integrity verification</CardTitle>
   450→        <CardDescription>
   451→          Recomputes the analysis checksum from persisted artifacts and re-checks the
   452→          source-run pins. Confirms the integrity of a hypothetical analysis — never that
   453→          transformed values are observations.
   454→        </CardDescription>
   455→      </CardHeader>
   456→      <CardContent>
   457→        <Button size="sm" variant="outline" onClick={() => verify.mutate()} disabled={verify.isPending}>
   458→          {verify.isPending ? <Loader2 className="mr-1.5 h-3.5 w-3.5 animate-spin" aria-hidden /> : <ShieldCheck className="mr-1.5 h-3.5 w-3.5" aria-hidden />}
   459→          Verify analysis
   460→        </Button>
   461→        {verify.isError && (
   462→          <p className="mt-2 text-sm text-red-700">{apiErrorText(verify.error)}</p>
   463→        )}
   464→        {result && (
   465→          <div className="mt-3 space-y-1.5">
   466→            {flags.map(([flag, label]) => (
   467→              <div key={flag} className="flex items-center gap-2 text-xs">
   468→                {result[flag] ? (
   469→                  <ShieldCheck className="h-3.5 w-3.5 text-emerald-700" aria-hidden />
   470→                ) : (
   471→                  <ShieldAlert className="h-3.5 w-3.5 text-red-600" aria-hidden />
   472→                )}
   473→                <span className="text-stone-600">{label}</span>
   474→                <span className="ml-auto font-mono text-[10px] text-stone-400">{String(result[flag])}</span>
   475→              </div>
   476→            ))}
   477→          </div>
   478→        )}
   479→      </CardContent>
   480→    </Card>
   481→  )
   482→}

   484→function AnalysisResults({ analysis }: { analysis: SensitivityAnalysis }) {
   485→  const parameters = useMemo(
   486→    () => Object.keys(analysis.parameter_design ?? {}),
   487→    [analysis.parameter_design],
   488→  )
   489→  const [selectedParameter, setSelectedParameter] = useState<string | null>(null)
   490→  // Derived (no sync-effect): fall back to the first explored parameter when
   491→  // the selection is absent or no longer part of the design.
   492→  const activeParameter =
   493→    selectedParameter && parameters.includes(selectedParameter) ? selectedParameter : (parameters[0] ?? null)
   494→
   495→  const reproduction = analysis.reference_reproduction
   496→
   497→  return (
   498→    <div className="space-y-5">
   499→      <div className="flex flex-wrap items-center gap-2">
   500→        <AnalysisStatusBadge status={analysis.analysis_status} />
   501→        <Badge variant="outline" className="border-stone-200 font-normal text-stone-500">
   502→          {analysis.evaluation_count} evaluations · {analysis.evaluated_feature_count} features
   503→        </Badge>
   504→        <Badge variant="outline" className="border-stone-200 font-normal text-stone-500">
   505→          checksum {shortHash(analysis.output_checksum)}
   506→        </Badge>

   624→interface BoundState {
   625→  lower: string
   626→  upper: string
   627→  points: string
   628→}
   629→
   630→function DefinitionForm({
   631→  run,
   632→  referenceValues,
   633→  explorableDetail,
   634→  onCreated,
   635→}: {
   636→  run: ScenarioExecutionRun
   637→  referenceValues: Record<string, number>
   638→  explorableDetail: Record<string, { explorable: boolean; reference_value: number | null; reason: string | null }>
   639→  onCreated: (analysis: SensitivityAnalysis) => void
   640→}) {
   641→  const queryClient = useQueryClient()
   642→  const bothExplorable =
   643→    explorableDetail.temperature_delta_c?.explorable && explorableDetail.precipitation_multiplier?.explorable
   644→
   645→  const defaults: Record<string, BoundState> = {
   646→    temperature_delta_c: { lower: '-2', upper: '6', points: '9' },
   647→    precipitation_multiplier: { lower: '0.5', upper: '1.5', points: '9' },
   648→  }
   649→  const [bounds, setBounds] = useState<Record<string, BoundState>>({
   650→    temperature_delta_c: defaults.temperature_delta_c,
   651→    precipitation_multiplier: defaults.precipitation_multiplier,
   652→  })
   653→  const [includeGrid, setIncludeGrid] = useState(false)
   654→  const [formError, setFormError] = useState<string | null>(null)
   655→
   656→  const explorableCodes = Object.entries(explorableDetail)
   657→    .filter(([, v]) => v.explorable)
   658→    .map(([code]) => code)
   659→
   660→  const create = useMutation({
   661→    mutationFn: () => {
   662→      const parameter_design: Record<string, { lower: number; upper: number; n_points: number }> = {}
   663→      for (const code of explorableCodes) {
   664→        const b = bounds[code]
   665→        parameter_design[code] = {
   666→          lower: Number(b.lower),
   667→          upper: Number(b.upper),
   668→          n_points: Number(b.points),
   669→        }
   670→      }
   671→      return createSensitivityAnalysis(run.run_id, {
   672→        parameter_design,
   673→        grid: includeGrid && bothExplorable ? true : undefined,
   674→      })
   675→    },
   676→    onSuccess: (analysis) => {
   677→      queryClient.invalidateQueries({ queryKey: ['sensitivity-analyses', run.run_id] })
   678→      onCreated(analysis)
   679→    },
   680→  })
   681→
   682→  const refDelta = referenceValues.temperature_delta_c
   683→  const refMult = referenceValues.precipitation_multiplier
   684→
   685→  return (
   686→    <Card>
   687→      <CardHeader className="pb-3">
   688→        <CardTitle className="text-base">Define the exploration</CardTitle>
   689→        <CardDescription>
   690→          Bounds are user-supplied assumptions for a deterministic experiment — they
   691→          contain the source execution&apos;s reference values and imply no probability
   692→          distribution.
   693→        </CardDescription>
   694→      </CardHeader>
   695→      <CardContent className="space-y-4">
   696→        <div className="grid gap-4 md:grid-cols-2">
   697→          {Object.entries(explorableDetail).map(([code, detail]) => {
   698→            const ref = code === 'temperature_delta_c' ? refDelta : refMult
   699→            if (!detail.explorable) {
   700→              return (
   701→                <div key={code} className="rounded-md border border-stone-200 bg-stone-50 p-3">
   702→                  <p className="text-xs font-medium text-stone-700">{code.replace(/_/g, ' ')}</p>
   703→                  <p className="mt-1 text-xs text-stone-500">
   704→                    Not applicable — {detail.reason}
   705→                  </p>
   706→                </div>
   707→              )
   708→            }
   709→            const b = bounds[code]
   710→            const set = (patch: Partial<BoundState>) =>
   711→              setBounds((prev) => ({ ...prev, [code]: { ...prev[code], ...patch } }))
   712→            return (
   713→              <div key={code} className="rounded-md border border-stone-200 p-3">
   714→                <div className="flex items-center justify-between">
   715→                  <p className="text-xs font-medium text-stone-700">{code.replace(/_/g, ' ')}</p>
   716→                  <Badge variant="outline" className="border-stone-200 font-normal text-stone-500">
   717→                    reference {fmt(ref)}
   718→                  </Badge>
   719→                </div>
   720→                <div className="mt-2 grid grid-cols-3 gap-2">
   721→                  <div>
   722→                    <Label className="text-[11px] text-stone-500" htmlFor={`${code}-lower`}>
   723→                      Lower
   724→                    </Label>
   725→                    <Input
   726→                      id={`${code}-lower`}
   727→                      className="h-8 text-xs"
   728→                      inputMode="decimal"
   729→                      value={b.lower}
   730→                      onChange={(e) => set({ lower: e.target.value })}
   731→                    />
   732→                  </div>
   733→                  <div>
   734→                    <Label className="text-[11px] text-stone-500" htmlFor={`${code}-upper`}>
   735→                      Upper
   736→                    </Label>
   737→                    <Input
   738→                      id={`${code}-upper`}
   739→                      className="h-8 text-xs"
   740→                      inputMode="decimal"
   741→                      value={b.upper}
   742→                      onChange={(e) => set({ upper: e.target.value })}
   743→                    />
   744→                  </div>
   745→                  <div>
   746→                    <Label className="text-[11px] text-stone-500" htmlFor={`${code}-points`}>
   747→                      Points (3–21)
   748→                    </Label>
   749→                    <Input
   750→                      id={`${code}-points`}
   751→                      className="h-8 text-xs"
   752→                      inputMode="numeric"
   753→                      value={b.points}
   754→                      onChange={(e) => set({ points: e.target.value })}
   755→                    />
   756→                  </div>
   757→                </div>
   758→              </div>
   759→            )
   760→          })}
   761→        </div>
   762→
   763→        {bothExplorable && (
   764→          <label className="flex items-center gap-2 text-xs text-stone-700">
   765→            <input
   766→              type="checkbox"
   767→              checked={includeGrid}
   768→              onChange={(e) => setIncludeGrid(e.target.checked)}
   769→              className="h-3.5 w-3.5"
   770→            />
   771→            Also evaluate the joint deterministic parameter grid (both axes reuse these bounds)
   772→          </label>
   773→        )}
   774→
   775→        {formError && <p className="text-sm text-red-700">{formError}</p>}
   776→        {create.isError && <p className="text-sm text-red-700">{apiErrorText(create.error)}</p>}
   777→
   778→        <Button
   779→          size="sm"
   780→          onClick={() => {
   781→            setFormError(null)
   782→            create.mutate()
   783→          }}
   784→          disabled={create.isPending}
   785→        >
   786→          {create.isPending ? (
   787→            <Loader2 className="mr-1.5 h-3.5 w-3.5 animate-spin" aria-hidden />
   788→          ) : (
   789→            <Play className="mr-1.5 h-3.5 w-3.5" aria-hidden />
   790→          )}
   791→          Run analysis
   792→        </Button>
   793→      </CardContent>
   794→    </Card>
   795→  )
   796→}

   822→  // offered to the engine below.

   824→    ? scenarioId

   829→    queryKey: ['sensitivity-runs', effectiveScenarioId],

   831→    enabled: Boolean(effectiveScenarioId),

   834→    (r) => r.run_status === 'succeeded' || r.run_status === 'completed_with_limitations',

   839→    queryKey: ['sensitivity-analyses', effectiveRunId],

   848→    analysis && analysis.source_execution_run_id === effectiveRunId ? analysis : null

   896→            value={null}
   897→            onValueChange={(field) => {
   898→              setSelectedField(field)
   899→              setScenarioId(null)
   900→              setRunId(null)
