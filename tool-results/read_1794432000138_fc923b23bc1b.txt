   107→/** Thrown for any non-2xx API response; carries the parsed error envelope. */
   108→export class ApiError extends Error {
   109→  readonly status: number
   110→  readonly code: string
   111→  readonly details: ApiErrorDetail[]
   112→  readonly requestId: string | undefined
   113→
   114→  constructor(params: {
   115→    status: number
   116→    code: string
   117→    message: string
   118→    details?: ApiErrorDetail[]
   119→    requestId?: string
   120→  }) {
   121→    super(params.message)
   122→    this.name = 'ApiError'
   123→    this.status = params.status
   124→    this.code = params.code
   125→    this.details = params.details ?? []
   126→    this.requestId = params.requestId
   127→  }
   128→}

   130→/** Human-readable summary of any thrown error for toasts. */
   131→export function describeError(err: unknown): { title: string; description?: string } {
   132→  if (err instanceof ApiError) {
   133→    const firstDetail = err.details[0]
   134→    const description = firstDetail
   135→      ? `${firstDetail.field}: ${firstDetail.reason}${err.requestId ? ` · request ${err.requestId.slice(0, 8)}` : ''}`
   136→      : err.requestId
   137→        ? `request ${err.requestId.slice(0, 8)}`
   138→        : undefined
   139→    return { title: err.message, description }
   140→  }
   141→  if (err instanceof Error) {
   142→    return { title: 'Network error', description: err.message }
   143→  }
   144→  return { title: 'Unexpected error' }
   145→}

   147→function withPort(path: string): string {
   148→  const joiner = path.includes('?') ? '&' : '?'
   149→  return `${path}${joiner}${PORT_QUERY}`
   150→}

   152→/**
   153→ * Exchange the HttpOnly refresh cookie for a fresh access+refresh cookie pair
   154→ * (rotating refresh: the presented token is revoked server-side; reusing a
   155→ * superseded token revokes the whole session family).
   156→ *
   157→ * Single-flight: several concurrent 401s share ONE rotation call — parallel
   158→ * rotations would present already-rotated tokens and revoke the family.
   159→ */
   160→let refreshInFlight: Promise<boolean> | null = null
   161→
   162→export async function refreshSession(): Promise<boolean> {
   163→  if (!refreshInFlight) {
   164→    refreshInFlight = authRequest('/refresh')
   165→      .then(() => true)
   166→      .catch(() => false)
   167→      .finally(() => {
   168→        refreshInFlight = null
   169→      })
   170→  }
   171→  return refreshInFlight
   172→}

   174→/** POST to a Next.js /api/auth/* route handler (session cookies live here). */
   175→async function authRequest<T>(path: string, body?: unknown): Promise<T> {
   176→  const response = await fetch(`${AUTH_BASE}${path}`, {
   177→    method: 'POST',
   178→    headers: body !== undefined ? { 'Content-Type': 'application/json' } : undefined,
   179→    body: body !== undefined ? JSON.stringify(body) : undefined,
   180→    cache: 'no-store',
   181→  })
   182→  const text = await response.text()
   183→  let payload: unknown = null
   184→  if (text) {
   185→    try {
   186→      payload = JSON.parse(text)
   187→    } catch {
   188→      payload = null
   189→    }
   190→  }
   191→  if (!response.ok) {
   192→    const envelope = payload as ApiErrorBody | null
   193→    const error =
   194→      envelope && typeof envelope === 'object' && 'error' in envelope ? envelope.error : undefined
   195→    throw new ApiError({
   196→      status: response.status,
   197→      code: error?.code ?? `HTTP_${response.status}`,
   198→      message: error?.message ?? `Request failed with status ${response.status}.`,
   199→      details: error?.details ?? [],
   200→      requestId: error?.request_id,
   201→    })
   202→  }
   203→  return payload as T
   204→}
   205→
   206→interface RequestOptions {
   207→  method?: 'GET' | 'POST' | 'PATCH' | 'PUT' | 'DELETE'
   208→  body?: unknown
   209→  /** Multipart form body (file uploads). Content-Type is set by the browser. */
   210→  formData?: FormData
   211→  /** Skip the 401→refresh retry (health checks, anonymous endpoints). */
   212→  anonymous?: boolean
   213→}
   214→
   215→interface RequestInternals {
   216→  /** Prevent recursive 401→refresh retries (used by refreshSession itself). */
   217→  skipAuthRetry?: boolean
   218→}
   219→
   220→async function request<T>(
   221→  path: string,
   222→  options: RequestOptions = {},
   223→  internals: RequestInternals = {},
   224→): Promise<T> {
   225→  const headers: Record<string, string> = { Accept: 'application/json' }
   226→  if (options.body !== undefined) headers['Content-Type'] = 'application/json'
   227→
   228→  let response: Response
   229→  try {
   230→    response = await fetch(withPort(path), {
   231→      method: options.method ?? 'GET',
   232→      headers,
   233→      body:
   234→        options.formData !== undefined
   235→          ? options.formData
   236→          : options.body !== undefined
   237→            ? JSON.stringify(options.body)
   238→            : undefined,
   239→      cache: 'no-store',
   240→    })
   241→  } catch (err) {
   242→    throw new ApiError({
   243→      status: 0,
   244→      code: 'NETWORK_ERROR',
   245→      message: 'Could not reach the CropTwist API.',
   246→      details: [],
   247→    })
   248→  }
   249→
   250→  const text = await response.text()
   251→  let payload: unknown = null
   252→  if (text) {
   253→    try {
   254→      payload = JSON.parse(text)
   255→    } catch {
   256→      payload = null
   257→    }
   258→  }
   259→
   260→  if (!response.ok) {
   261→    const envelope = payload as ApiErrorBody | null
   262→    const error = envelope && typeof envelope === 'object' && 'error' in envelope ? envelope.error : undefined
   263→    // Rotating-refresh retry (P0-2/P1): an access token may have simply
   264→    // expired while the session is still valid. Try ONE cookie refresh, then
   265→    // retry the original request once. Never recurse from the refresh call
   266→    // itself (it goes through authRequest, not here).
   267→    if (response.status === 401 && !options.anonymous && !internals.skipAuthRetry) {
   268→      const refreshed = await refreshSession()
   269→      if (refreshed) {
   270→        return request<T>(path, options, { skipAuthRetry: true })
   271→      }
   272→    }
   273→    throw new ApiError({
   274→      status: response.status,
   275→      code: error?.code ?? `HTTP_${response.status}`,
   276→      message: error?.message ?? `Request failed with status ${response.status}.`,
   277→      details: error?.details ?? [],
   278→      requestId: error?.request_id,
   279→    })
   280→  }
   281→
   282→  return payload as T
   283→}

   432→const JOB_TERMINAL_STATUSES = new Set(['succeeded', 'failed', 'cancelled'])
   433→
   434→/**
   435→ * Poll an ingestion/import job until it reaches a terminal state.
   436→ *
   437→ * The backend executes data jobs on a background worker pool (202 + status
   438→ * 'queued'); when workers are disabled it still executes inline (the response
   439→ * is already terminal, returned as-is). Progress callbacks receive every
   440→ * intermediate snapshot (status/progress for UI feedback).
   441→ */
   442→export async function pollJobUntilDone(
   443→  jobId: string,
   444→  opts: { onProgress?: (job: DataJob) => void; intervalMs?: number; timeoutMs?: number } = {},
   445→): Promise<DataJob> {
   446→  const interval = opts.intervalMs ?? 1000
   447→  const deadline = Date.now() + (opts.timeoutMs ?? 180_000)
   448→  let current = await getJob(jobId)
   449→  while (!JOB_TERMINAL_STATUSES.has(current.status)) {
   450→    opts.onProgress?.(current)
   451→    if (Date.now() > deadline) {
   452→      throw new Error('Timed out waiting for the job to finish — check its status later.')
   453→    }
   454→    await new Promise((resolve) => setTimeout(resolve, interval))
   455→    current = await getJob(jobId)
   456→  }
   457→  return current
   458→}

   489→export type GeometryParseResult =
   490→  | { ok: true; geometry: import('./types').GeoJSONGeometry }
   491→  | { ok: false; error: string }
   492→
   493→/** Parse + shallow-validate a GeoJSON Polygon / MultiPolygon from user input. */
   494→export function parseGeometryInput(text: string): GeometryParseResult {
   495→  let parsed: unknown
   496→  try {
   497→    parsed = JSON.parse(text)
   498→  } catch {
   499→    return { ok: false, error: 'The text is not valid JSON. Check for missing quotes, brackets or trailing commas.' }
   500→  }
   501→  if (typeof parsed !== 'object' || parsed === null || Array.isArray(parsed)) {
   502→    return { ok: false, error: 'Expected a GeoJSON object like {"type":"Polygon","coordinates":[...]}' }
   503→  }
   504→  const obj = parsed as Record<string, unknown>
   505→  if (obj.type === 'Polygon') {
   506→    if (!Array.isArray(obj.coordinates) || obj.coordinates.length === 0) {
   507→      return { ok: false, error: 'Polygon coordinates must be a non-empty array of rings.' }
   508→    }
   509→    return { ok: true, geometry: parsed as import('./types').GeoJSONPolygon }
   510→  }
   511→  if (obj.type === 'MultiPolygon') {
   512→    if (!Array.isArray(obj.coordinates) || obj.coordinates.length === 0) {
   513→      return { ok: false, error: 'MultiPolygon coordinates must be a non-empty array of polygons.' }
   514→    }
   515→    return { ok: true, geometry: parsed as import('./types').GeoJSONMultiPolygon }
   516→  }
   517→  return { ok: false, error: `Unsupported geometry type "${String(obj.type)}". Use Polygon or MultiPolygon.` }
   518→}

   537→/** Client-side guard mirroring backend job validation. Returns an error message or null. */
   538→export function validateJobWindow(
   539→  start: string,
   540→  end: string,
   541→  bounds?: { min?: string; max?: string; maxYears?: number },
   542→): string | null {
   543→  const dateRe = /^\d{4}-\d{2}-\d{2}$/
   544→  if (!dateRe.test(start) || !dateRe.test(end)) return 'Enter both dates as YYYY-MM-DD.'
   545→  const min = bounds?.min ?? COVERAGE_START
   546→  const max = bounds?.max ?? COVERAGE_END
   547→  if (start < min || end > max) {
   548→    return `Dates must fall inside the dataset coverage ${min} → ${max}.`
   549→  }
   550→  if (end < start) return 'The end date must be on or after the start date.'
   551→  const years = (Date.parse(end) - Date.parse(start)) / (365.25 * 24 * 3600 * 1000)
   552→  const maxYears = bounds?.maxYears ?? MAX_JOB_YEARS
   553→  if (years > maxYears) return `A single job covers at most ${maxYears} years. Split the range.`
   554→  return null
   555→}

   990→// --- laboratory CSV import ------------------------------------------------------
   991→// Backend contract: POST /soil-samples/import (org-level) requires a `mapping`
   992→// form field whose JSON carries default_field_id — the field binding arrives
   993→// there, so the component's (fieldId, file) signature is preserved.
   994→
   995→export async function importSoilLabCsv(fieldId: string, file: File): Promise<SoilLabImportResponse> {
   996→  const formData = new FormData()
   997→  formData.append('file', file)
   998→  formData.append('mapping', JSON.stringify({ default_field_id: fieldId }))
   999→  const data = await request<SoilLabImportResponse>('/api/v1/soil-samples/import', {
  1000→    method: 'POST',
  1001→    formData,
  1002→  })
  1003→  // The backend's idempotent "already_imported" reply omits the counters;
  1004→  // normalize so callers always see numbers (re-import of identical content
  1005→  // creates nothing — 0 is the truthful value there).
  1006→  return {
  1007→    ...data,
  1008→    created_samples: data.created_samples ?? 0,
  1009→    created_measurements: data.created_measurements ?? 0,
  1010→    errors: data.errors ?? [],
  1011→  }
  1012→}
