     1→"""Field state assembly service (Milestone 7.1).
     2→
     3→Assembles an immutable, checksummed field-state snapshot BY REFERENCE from
     4→already-versioned artifacts. It never duplicates domain data and never
     5→computes agronomy: every dimension either pins an existing artifact or is
     6→reported honestly as missing / not_evaluated / not_implemented.
     7→
     8→Immutability contract (same idiom as water plans):
     9→- assembly always creates a NEW row; nothing is rewritten;
    10→- idempotency keys are field-scoped and replay-safe (202 pattern);
    11→- referenced artifacts are protected by FK RESTRICT (F-13);
    12→- the checksum covers every scientifically material stored input;
    13→- `created_by`/`created_at` are provenance and are excluded from the hash.
    14→
    15→Temporal discipline (F-02): the state carries an explicit reference period;
    16→each dimension's coverage of that period is computed and recorded in
    17→`period_compatibility`. A pinned environmental run that does not overlap the
    18→period at all is REFUSED (it cannot qualify the state); a water plan with
    19→zero overlap is allowed but recorded as `period_mismatch` with a warning —
    20→never silently mixed.
    21→"""

    44→def _as_utc(value: Any, field_name: str) -> datetime:
    45→    """Parse an ISO datetime/date into an aware UTC datetime.
    46→
    47→    Naive datetimes are interpreted as UTC (platform rule: TIMESTAMPTZ
    48→    everywhere; naive input can only come from unzoned JSON). Plain dates
    49→    are interpreted as UTC midnight of that date.
    50→    """
    51→    if isinstance(value, datetime):
    52→        parsed = value
    53→    elif isinstance(value, date_cls):
    54→        parsed = datetime(value.year, value.month, value.day)
    55→    elif isinstance(value, str):
    56→        text = value.strip()
    57→        if text.endswith("Z"):
    58→            text = text[:-1] + "+00:00"
    59→        try:
    60→            parsed = datetime.fromisoformat(text)
    61→        except ValueError:
    62→            try:
    63→                d = date_cls.fromisoformat(text)
    64→            except ValueError:
    65→                raise ValidationFailed(
    66→                    f"{field_name} must be an ISO-8601 date or datetime.",
    67→                    details=[{"field": field_name, "reason": value}],
    68→                ) from None
    69→            parsed = datetime(d.year, d.month, d.day)
    70→    else:
    71→        raise ValidationFailed(
    72→            f"{field_name} must be an ISO-8601 date or datetime.",
    73→            details=[{"field": field_name, "reason": str(value)}],
    74→        )
    75→    if parsed.tzinfo is None:
    76→        parsed = parsed.replace(tzinfo=dt_timezone.utc)
    77→    return parsed.astimezone(dt_timezone.utc)
    78→

    81→    return value.astimezone(dt_timezone.utc).isoformat() if value else None

   116→    db: Session, field: dbm.Field, requested: Any, availability: dict[str, Any]
   117→) -> dbm.FeatureRun | None:
   118→    if requested is None:
   119→        run = _latest_succeeded_feature_run(db, field.id)
   120→        if run is None:
   121→            availability["environmental"] = {
   122→                "status": fsc.MISSING,
   123→                "reason": "No succeeded feature run exists for this field; "
   124→                "create one via POST /fields/{id}/feature-runs.",
   125→                "snapshot_id": None,

   129→        select(dbm.FeatureRun).where(
   130→            dbm.FeatureRun.id == requested, dbm.FeatureRun.field_id == field.id

   133→    if run is None:
   134→        raise NotFound(f"Feature run {requested} not found for this field.")
   135→    if run.status != "succeeded":
   136→        raise ValidationFailed(
   137→            "Pinned feature run has not succeeded; a state cannot pin a "
   138→            "non-succeeded run.",
   139→            details=[{"field": "environmental_feature_run_id", "reason": run.status}],

   173→def _resolve_rotation_run(
   174→    db: Session, field: dbm.Field, requested: Any, strategy_key: Any
   175→) -> tuple[dbm.RotationRun | None, str | None]:
   176→    if requested is None:
   177→        if strategy_key:
   178→            raise ValidationFailed(
   179→                "rotation_strategy_key requires rotation_run_id.",
   180→                details=[{"field": "rotation_strategy_key"}],
   181→            )
   182→        # No default: rotation runs are optimization experiments, not a
   183→        # general state (M7.0 F-06) — they are pinned explicitly only.
   184→        return None, None
   185→    run = db.scalar(
   186→        select(dbm.RotationRun).where(
   187→            dbm.RotationRun.id == requested, dbm.RotationRun.field_id == field.id
   188→        )
   189→    )
   190→    if run is None:
   191→        raise NotFound(f"Rotation run {requested} not found for this field.")
   192→    if run.status != "succeeded":
   193→        raise ValidationFailed(
   194→            "Pinned rotation run has not succeeded.",
   195→            details=[{"field": "rotation_run_id", "reason": run.status}],
   196→        )
   197→    if strategy_key is not None:
   198→        keys = {s.strategy_key for s in run.strategies}
   199→        if strategy_key not in keys:
   200→            raise ValidationFailed(
   201→                "rotation_strategy_key is not part of the pinned rotation run.",
   202→                details=[{"field": "rotation_strategy_key", "reason": str(strategy_key)}],
   203→            )
   204→    return run, strategy_key

   207→def _crop_history_ids_asof(db: Session, field_id: uuid_mod.UUID, asof: datetime) -> list[dict[str, Any]]:
   208→    """Pin crop-history rows whose sowing_date is on/before the reference time.
   209→
   210→    Rows are immutable (no update path) so pinning ids + key attributes by
   211→    value is sufficient for reconstruction without duplicating the table.
   212→    """
   213→    rows = (
   214→        db.execute(
   215→            select(dbm.CropHistory)
   216→            .where(dbm.CropHistory.field_id == field_id)
   217→            .order_by(dbm.CropHistory.sowing_date.asc().nullslast(), dbm.CropHistory.created_at.asc())
   218→        )
   219→        .scalars()
   220→        .all()
   221→    )
   222→    pinned = []
   223→    for row in rows:
   224→        if row.sowing_date is None:
   225→            continue  # undated rows cannot be placed as-of T — never guessed
   226→        sowing_dt = datetime(
   227→            row.sowing_date.year, row.sowing_date.month, row.sowing_date.day, tzinfo=dt_timezone.utc
   228→        )
   229→        if sowing_dt > asof:
   230→            continue
   231→        pinned.append(
   232→            {
   233→                "id": str(row.id),
   234→                "crop_label": row.crop_label,
   235→                "season_label": row.season_label,
   236→                "sowing_date": row.sowing_date.isoformat(),
   237→                "harvest_date": row.harvest_date.isoformat() if row.harvest_date else None,
   238→                "source": row.source,
   239→            }
   240→        )
   241→    return pinned

   244→def _seasons_interval(plan: dbm.WaterPlan | None) -> tuple[datetime, datetime] | None:
   245→    if not plan or not plan.seasons:
   246→        return None
   247→    starts: list[datetime] = []
   248→    ends: list[datetime] = []
   249→    for season in plan.seasons:
   250→        try:
   251→            s = date_cls.fromisoformat(str(season.get("start_date")))
   252→            e = date_cls.fromisoformat(str(season.get("end_date")))
   253→        except (TypeError, ValueError):
   254→            continue
   255→        starts.append(datetime(s.year, s.month, s.day, tzinfo=dt_timezone.utc))
   256→        ends.append(
   257→            datetime(e.year, e.month, e.day, tzinfo=dt_timezone.utc).replace(
   258→                hour=23, minute=59, second=59
   259→            )
   260→        )
   261→    if not starts or not ends:
   262→        return None
   263→    return min(starts), max(ends)

   266→def _compat_record(
   267→    artifact_start: datetime | None,
   268→    artifact_end: datetime | None,
   269→    period_start: datetime,
   270→    period_end: datetime,
   271→) -> dict[str, Any]:
   272→    if artifact_start is None or artifact_end is None:
   273→        return {"status": fsc.NOT_EVALUATED, "reason": "artifact has no recorded period"}
   274→    fraction = fsc.period_overlap_fraction(artifact_start, artifact_end, period_start, period_end)
   275→    if fraction is None:
   276→        return {"status": fsc.NOT_EVALUATED, "reason": "artifact has no recorded period"}
   277→    record: dict[str, Any] = {
   278→        "artifact_period": {"start": _iso(artifact_start), "end": _iso(artifact_end)},
   279→        "overlap_fraction": fraction,
   280→    }
   281→    if fraction <= 0.0:
   282→        record["status"] = fsc.OUTSIDE_PERIOD
   283→    elif fraction >= 1.0:
   284→        record["status"] = fsc.COVERS_PERIOD
   285→    else:
   286→        record["status"] = fsc.PARTIAL_COVER
   287→    return record

   290→def _state_manifest(db: Session, run: dbm.FeatureRun | None, field_id: uuid_mod.UUID) -> tuple[dict[str, Any], str]:
   291→    """The state's observation manifest: a verbatim copy of the pinned run's
   292→    manifest when available; otherwise an explicit, honestly empty document.
   293→
   294→    Pre-M7.1 runs have no stored manifest — the state never fabricates one;
   295→    it records the gap with reason `unavailable_pre_m7_1_run`.
   296→    """
   297→    if run is not None and run.observation_manifest is not None:
   298→        document = dict(run.observation_manifest)
   299→        stored_hash = run.observation_manifest_sha256
   300→        recomputed = sha256_of_document(document)
   301→        if stored_hash and stored_hash != recomputed:
   302→            raise Conflict(
   303→                "Stored observation manifest of the pinned feature run does not "
   304→                "match its checksum; the run artifact is not intact."
   305→            )
   306→        return document, recomputed
   307→    document = build_manifest(
   308→        [],
   309→        field_id=str(field_id),
   310→        purpose="field_state_environmental",
   311→        processing_configuration={
   312→            "manifest_status": "unavailable_pre_m7_1_run"
   313→            if run is not None
   314→            else "no_environmental_run_pinned",
   315→        },
   316→    )
   317→    return document, sha256_of_document(document)

   320→def _evidence_summary(
   321→    *,
   322→    run: dbm.FeatureRun | None,
   323→    plan: dbm.WaterPlan | None,
   324→    rotation: dbm.RotationRun | None,
   325→    strategy_key: str | None,
   326→    crop_history: list[dict[str, Any]],
   327→    manifest_document: dict[str, Any],
   328→    availability: dict[str, Any],
   329→) -> dict[str, Any]:
   330→    evidence: dict[str, Any] = {}
   331→    if run is not None:
   332→        source_products = []
   333→        for variable, selection in sorted((run.source_selection or {}).items()):
   334→            if isinstance(selection, dict):
   335→                source_products.append(
   336→                    {
   337→                        "variable": variable,
   338→                        "product_id": selection.get("dataset_product_id"),
   339→                        "product_code": selection.get("product_code"),
   340→                        "product_version": selection.get("product_version"),
   341→                        "is_synthetic": selection.get("is_synthetic"),
   342→                    }
   343→                )
   344→        evidence["environmental"] = {
   345→            "feature_run_id": str(run.id),
   346→            "feature_engine_version": run.feature_engine_version,
   347→            "feature_catalog_version": run.feature_catalog_version,
   348→            "is_synthetic": run.is_synthetic,
   349→            "source_products": source_products,
   350→            "observation_count": manifest_document.get("observation_count", 0),
   351→            "synthetic_observation_count": manifest_document.get("synthetic_observation_count", 0),
   352→        }
   353→    else:
   354→        evidence["environmental"] = {"feature_run_id": None}
   355→
   356→    if plan is not None:
   357→        evidence["water"] = {
   358→            "water_plan_id": str(plan.id),
   359→            "water_engine_version": plan.water_engine_version,
   360→            "methods_version": plan.methods_version,
   361→            "season_labels": sorted(s.get("label") for s in (plan.seasons or []) if s.get("label")),
   362→            "checksum_sha256": plan.checksum_sha256,
   363→        }
   364→    else:
   365→        evidence["water"] = {"water_plan_id": None}
   366→
   367→    if rotation is not None:
   368→        evidence["rotation"] = {
   369→            "rotation_run_id": str(rotation.id),
   370→            "rotation_engine_version": rotation.rotation_engine_version,
   371→            "solver_name": rotation.solver_name,
   372→            "solver_version": rotation.solver_version,
   373→            "strategy_key": strategy_key,
   374→            "solver_status": rotation.solver_status,
   375→        }
   376→    else:
   377→        evidence["rotation"] = {"rotation_run_id": None}
   378→
   379→    evidence["crop_history"] = {
   380→        "count": len(crop_history),
   381→        "crop_labels": sorted({c["crop_label"] for c in crop_history}),
   382→        "season_labels": sorted({c["season_label"] for c in crop_history if c["season_label"]}),
   383→    }
   384→    evidence["soil"] = {"assessment_run_id": None, "status": fsc.NOT_IMPLEMENTED}
   385→    evidence["management"] = {"management_plan_id": None, "status": fsc.NOT_IMPLEMENTED}
   386→    return evidence

   437→def state_checksum_payload(state: dbm.FieldState) -> dict[str, Any]:
   438→    """The exact checksum input for a persisted field state.
   439→
   440→    Everything scientifically material, nothing ephemeral: timestamps of
   441→    assembly (created_at) and the actor (created_by) are provenance, not
   442→    content, and are excluded. All other stored fields participate.
   443→    """
   444→    return {
   445→        "assembly_version": (state.source_versions or {}).get("field_state_assembly_version"),
   446→        "manifest_format_version": state.manifest_format_version,
   447→        "field_id": str(state.field_id),
   448→        "boundary_version": state.boundary_version,
   449→        "state_reference_time": _iso(state.state_reference_time),
   450→        "period_start": _iso(state.period_start),
   451→        "period_end": _iso(state.period_end),
   452→        "timezone": state.timezone,
   453→        "calendar_context": state.calendar_context,
   454→        "environmental_feature_run_id": (
   455→            str(state.environmental_feature_run_id) if state.environmental_feature_run_id else None
   456→        ),
   457→        "water_plan_id": str(state.water_plan_id) if state.water_plan_id else None,
   458→        "rotation_run_id": str(state.rotation_run_id) if state.rotation_run_id else None,
   459→        "rotation_strategy_key": state.rotation_strategy_key,
   460→        "crop_history_ids": sorted(state.crop_history_ids or []),
   461→        "knowledge_pack_refs": sorted(
   462→            state.knowledge_pack_refs or [], key=lambda r: _canonical(r)
   463→        ),
   464→        "observation_manifest_sha256": state.observation_manifest_sha256,
   465→        "domain_availability": state.domain_availability,
   466→        "evidence_summary": state.evidence_summary,
   467→        "quality_summary": state.quality_summary,
   468→        "period_compatibility": state.period_compatibility,
   469→    }
   470→
   471→
   472→def compute_state_checksum(state: dbm.FieldState) -> str:
   473→    return hashlib.sha256(_canonical(state_checksum_payload(state)).encode("utf-8")).hexdigest()

   476→# ------------------------------------------------------------- assembly
   477→def assemble_field_state(
   478→    db: Session,
   479→    *,
   480→    field: dbm.Field,
   481→    payload: dict[str, Any],
   482→    created_by: Any = None,
   483→) -> dbm.FieldState:
   484→    """Create (or idempotently return) a field state for `field`."""
   485→    idempotency_key = payload.get("idempotency_key")
   486→    if idempotency_key:
   487→        existing = db.scalar(
   488→            select(dbm.FieldState).where(
   489→                dbm.FieldState.idempotency_key == idempotency_key,
   490→                dbm.FieldState.field_id == field.id,
   491→            )
   492→        )
   493→        if existing is not None:
   494→            return existing
   495→
   496→    warnings: list[str] = []
   497→    not_evaluated: dict[str, Any] = {}
   498→
   499→    label = str(payload.get("label") or f"Field state {field.name}").strip() or "field state"
   500→    if len(label) > fsc.STATE_LABEL_MAX:
   501→        label = label[: fsc.STATE_LABEL_MAX]
   502→
   503→    reference_raw = payload.get("state_reference_time")
   504→    reference_time = (
   505→        _as_utc(reference_raw, "state_reference_time") if reference_raw else datetime.now(dt_timezone.utc)
   506→    )
   507→
   508→    availability = fsc.baseline_domain_availability()
   509→
   510→    # ---- resolve dimensions -------------------------------------------------
   511→    run = _resolve_environmental_run(db, field, payload.get("environmental_feature_run_id"), availability)
   512→    plan = _resolve_water_plan(db, field, payload.get("water_plan_id"), availability)
   513→    rotation, strategy_key = _resolve_rotation_run(
   514→        db, field, payload.get("rotation_run_id"), payload.get("rotation_strategy_key")
   515→    )
   516→    crop_history = _crop_history_ids_asof(db, field.id, reference_time)
   517→    if not crop_history:
   518→        availability["crop_history"] = {
   519→            "status": fsc.MISSING,
   520→            "reason": "No dated crop-history rows on or before the reference time.",
   521→            "snapshot_id": None,
   522→        }
   523→    else:
   524→        availability["crop_history"] = {
   525→            "status": fsc.AVAILABLE,
   526→            "reason": None,
   527→            "snapshot_id": None,
   528→        }
   529→
   530→    if run is not None:
   531→        availability["environmental"] = {
   532→            "status": fsc.AVAILABLE,
   533→            "reason": None,
   534→            "snapshot_id": str(run.id),
   535→        }
   536→    if plan is not None:
   537→        availability["water"] = {
   538→            "status": fsc.AVAILABLE,
   539→            "reason": None,
   540→            "snapshot_id": str(plan.id),
   541→        }
   542→    if rotation is not None:
   543→        availability["rotation"] = {
   544→            "status": fsc.AVAILABLE,
   545→            "reason": None,
   546→            "snapshot_id": str(rotation.id),
   547→        }
   548→
   549→    # ---- period -------------------------------------------------------------
   550→    period_start_raw = payload.get("period_start")
   551→    period_end_raw = payload.get("period_end")
   552→    if period_start_raw is not None and period_end_raw is None:
   553→        raise ValidationFailed(
   554→            "period_end is required when period_start is given.",
   555→            details=[{"field": "period_end"}],
   556→        )
   557→    if period_end_raw is not None and period_start_raw is None:
   558→        raise ValidationFailed(
   559→            "period_start is required when period_end is given.",
   560→            details=[{"field": "period_start"}],
   561→        )
   562→    if period_start_raw is not None:
   563→        period_start = _as_utc(period_start_raw, "period_start")
   564→        period_end = _as_utc(period_end_raw, "period_end")
   565→    elif run is not None:
   566→        # Default: the environmental run's own window — the only artifact
   567→        # whose period is always explicit and machine-comparable.
   568→        period_start = run.period_start
   569→        period_end = run.period_end
   570→    else:
   571→        raise ValidationFailed(
   572→            "period_start/period_end are required when no environmental "
   573→            "feature run is pinned (the state must declare the period it "
   574→            "qualifies).",
   575→            details=[{"field": "period_start"}],
   576→        )
   577→    if period_end < period_start:
   578→        raise ValidationFailed(
   579→            "period_end must be on or after period_start.",
   580→            details=[{"field": "period_end"}],
   581→        )
   582→
   583→    # ---- period compatibility (F-02) — recorded, never silently mixed -------
   584→    period_compatibility: dict[str, Any] = {}
   585→    if run is not None:
   586→        record = _compat_record(run.period_start, run.period_end, period_start, period_end)
   587→        period_compatibility["environmental"] = record
   588→        if record["status"] == fsc.OUTSIDE_PERIOD:
   589→            raise ValidationFailed(
   590→                "The pinned feature run's period does not overlap the state "
   591→                "period at all — an environmental snapshot cannot qualify a "
   592→                "period it does not touch.",
   593→                details=[
   594→                    {"field": "period_start", "reason": record["artifact_period"]["start"]},
   595→                    {"field": "period_end", "reason": record["artifact_period"]["end"]},
   596→                ],
   597→            )
   598→        if record["status"] == fsc.PARTIAL_COVER:
   599→            availability["environmental"]["status"] = fsc.PERIOD_MISMATCH
   600→            warnings.append(
   601→                "Environmental feature run only partially covers the state "
   602→                "period (overlap fraction recorded in period_compatibility)."
   603→            )
   604→    else:
   605→        period_compatibility["environmental"] = {
   606→            "status": fsc.NOT_EVALUATED,
   607→            "reason": "no environmental run pinned",
   608→        }
   609→
   610→    if plan is not None:
   611→        interval = _seasons_interval(plan)
   612→        record = _compat_record(interval[0] if interval else None, interval[1] if interval else None, period_start, period_end)
   613→        period_compatibility["water"] = record
   614→        if record["status"] == fsc.OUTSIDE_PERIOD:
   615→            availability["water"] = {
   616→                "status": fsc.PERIOD_MISMATCH,
   617→                "reason": "Water plan seasons lie entirely outside the state period; "
   618→                "the plan is pinned as provenance but does not qualify this period.",
   619→                "snapshot_id": str(plan.id),
   620→            }
   621→            warnings.append(
   622→                "Water plan seasons do not overlap the state period — recorded "
   623→                "as period_mismatch, never silently mixed."
   624→            )
   625→        elif record["status"] == fsc.PARTIAL_COVER:
   626→            warnings.append(
   627→                "Water plan seasons only partially cover the state period."
   628→            )
   629→    else:
   630→        period_compatibility["water"] = {
   631→            "status": fsc.NOT_EVALUATED,
   632→            "reason": "no water plan pinned",
   633→        }
   634→
   635→    if rotation is not None:
   636→        seasons = rotation.seasons or []
   637→        if seasons:
   638→            try:
   639→                starts = [
   640→                    datetime(*date_cls.fromisoformat(str(s.get("start_date"))).timetuple()[:3], tzinfo=dt_timezone.utc)
   641→                    for s in seasons
   642→                    if s.get("start_date")
   643→                ]
   644→                ends = [
   645→                    datetime(*date_cls.fromisoformat(str(s.get("end_date"))).timetuple()[:3], tzinfo=dt_timezone.utc)
   646→                    .replace(hour=23, minute=59, second=59)
   647→                    for s in seasons
   648→                    if s.get("end_date")
   649→                ]
   650→            except (TypeError, ValueError):
   651→                starts, ends = [], []
   652→            if starts and ends:
   653→                record = _compat_record(min(starts), max(ends), period_start, period_end)
   654→                period_compatibility["rotation"] = record
   655→                if record["status"] != fsc.COVERS_PERIOD:
   656→                    warnings.append(
   657→                        "Rotation run seasons do not fully cover the state period."
   658→                    )
   659→            else:
   660→                period_compatibility["rotation"] = {
   661→                    "status": fsc.NOT_EVALUATED,
   662→                    "reason": "rotation run seasons unparseable",
   663→                }
   664→        else:
   665→            period_compatibility["rotation"] = {
   666→                "status": fsc.NOT_EVALUATED,
   667→                "reason": "rotation run has no seasons",
   668→            }
   669→    else:
   670→        period_compatibility["rotation"] = {
   671→            "status": fsc.NOT_EVALUATED,
   672→            "reason": "no rotation run pinned",
   673→        }
   674→
   675→    period_compatibility["crop_history"] = {
   676→        "status": fsc.NOT_EVALUATED,
   677→        "reason": "crop history spans arbitrary past seasons; each row carries "
   678→        "its own dates — as-of rule: sowing_date <= state_reference_time",
   679→    }
   680→    period_compatibility["soil"] = {"status": fsc.NOT_EVALUATED, "reason": fsc.SOIL_NOT_IMPLEMENTED_REASON}
   681→    period_compatibility["management"] = {
   682→        "status": fsc.NOT_EVALUATED,
   683→        "reason": fsc.MANAGEMENT_NOT_IMPLEMENTED_REASON,
   684→    }
   685→
   686→    # ---- calendar context ----------------------------------------------------
   687→    calendar_context: dict[str, Any] = {
   688→        "farm_timezone": field.farm.timezone if field.farm else None,
   689→        "hemisphere": run.hemisphere if run is not None else None,
   690→        "season_labels": sorted(
   691→            {
   692→                *(s.get("label") for s in ((plan.seasons or []) if plan is not None else []) if s.get("label")),
   693→                *(c["season_label"] for c in crop_history if c["season_label"]),
   694→            }
   695→        ),
   696→    }
   697→
   698→    # ---- manifest ------------------------------------------------------------
   699→    manifest_document, manifest_hash = _state_manifest(db, run, field.id)
   700→
   701→    # ---- knowledge pack set resolved at assembly time (F-08) -----------------
   702→    from app.knowledge.service import load_knowledge
   703→
   704→    knowledge = load_knowledge(db)
   705→    pack_refs = [dict(ref) for ref in knowledge.pack_refs]
   706→
   707→    # ---- summaries + versions ------------------------------------------------
   708→    evidence_summary = _evidence_summary(
   709→        run=run,
   710→        plan=plan,
   711→        rotation=rotation,
   712→        strategy_key=strategy_key,
   713→        crop_history=crop_history,
   714→        manifest_document=manifest_document,
   715→        availability=availability,
   716→    )
   717→    quality_summary = _quality_summary(
   718→        run=run, plan=plan, rotation=rotation, manifest_document=manifest_document, db=db
   719→    )
   720→    source_versions = {
   721→        "field_state_assembly_version": fsc.FIELD_STATE_ASSEMBLY_VERSION,
   722→        "manifest_format_version": MANIFEST_FORMAT_VERSION,
   723→        "feature_engine_version": run.feature_engine_version if run else None,
   724→        "feature_catalog_version": run.feature_catalog_version if run else None,
   725→        "water_engine_version": plan.water_engine_version if plan else None,
   726→        "water_methods_version": plan.methods_version if plan else None,
   727→        "rotation_engine_version": rotation.rotation_engine_version if rotation else None,
   728→    }
   729→
   730→    # ---- persist (new row; nothing rewritten) ---------------------------------
   731→    state = dbm.FieldState(
   732→        field_id=field.id,
   733→        status="running",
   734→        label=label,
   735→        boundary_version=field.boundary_version,
   736→        state_reference_time=reference_time,
   737→        period_start=period_start,
   738→        period_end=period_end,
   739→        timezone=field.farm.timezone if field.farm else "UTC",
   740→        calendar_context=calendar_context,
   741→        environmental_feature_run_id=run.id if run is not None else None,
   742→        water_plan_id=plan.id if plan is not None else None,
   743→        rotation_run_id=rotation.id if rotation is not None else None,
   744→        rotation_strategy_key=strategy_key,
   745→        crop_history_ids=[c["id"] for c in crop_history],
   746→        knowledge_pack_refs=pack_refs,
   747→        observation_manifest=manifest_document,
   748→        observation_manifest_sha256=manifest_hash,
   749→        manifest_format_version=MANIFEST_FORMAT_VERSION,
   750→        domain_availability=availability,
   751→        evidence_summary=evidence_summary,
   752→        quality_summary=quality_summary,
   753→        period_compatibility=period_compatibility,
   754→        source_versions=source_versions,
   755→        request_parameters={
   756→            k: (v.isoformat() if isinstance(v, (datetime, date_cls)) else v)
   757→            for k, v in (payload.items() if isinstance(payload, dict) else [])
   758→        },
   759→        warnings=warnings,
   760→        not_evaluated=not_evaluated,
   761→        idempotency_key=idempotency_key,
   762→        created_by=getattr(created_by, "id", None),
   763→        started_at=datetime.now(dt_timezone.utc),
   764→    )
   765→    db.add(state)
   766→    try:
   767→        db.commit()
   768→    except Exception:
   769→        db.rollback()
   770→        state.status = "failed"
   771→        state.error_summary = "assembly failed"
   772→        db.add(state)
   773→        db.commit()
   774→        raise
   775→    try:
   776→        state.checksum_sha256 = compute_state_checksum(state)
   777→        state.status = "succeeded"
   778→        state.completed_at = datetime.now(dt_timezone.utc)
   779→        db.commit()
   780→    except Exception as exc:  # noqa: BLE001 — job failure accounting
   781→        db.rollback()
   782→        state.status = "failed"
   783→        state.error_summary = f"{type(exc).__name__}: {exc}"
   784→        db.commit()
   785→        raise
   786→    return state

   806→def verify_field_state(db: Session, state: dbm.FieldState) -> dict[str, Any]:
   807→    """Independently re-verify a state from its stored content.
   808→
   809→    Checks: (1) state checksum recomputed from stored fields; (2) manifest
   810→    checksum recomputed from the stored manifest document; (3) referenced
   811→    artifacts still exist and (where the artifact has its own checksum) are
   812→    intact; (4) the pinned boundary version row exists in the append-only
   813→    boundary history.
   814→    """
   815→    checksum_valid = (
   816→        state.checksum_sha256 is not None
   817→        and state.checksum_sha256 == compute_state_checksum(state)
   818→    )
   819→    manifest_valid = state.observation_manifest_sha256 is not None and (
   820→        state.observation_manifest_sha256
   821→        == sha256_of_document(state.observation_manifest or {})
   822→    )
   823→
   824→    referenced: dict[str, Any] = {}
   825→    if state.environmental_feature_run_id is not None:
   826→        run = db.get(dbm.FeatureRun, state.environmental_feature_run_id)
   827→        referenced["feature_run"] = {
   828→            "exists": run is not None,
   829→            "intact": run is not None and run.status == "succeeded",
   830→        }
   831→    if state.water_plan_id is not None:
   832→        plan = db.get(dbm.WaterPlan, state.water_plan_id)
   833→        intact = False
   834→        if plan is not None and plan.status == "succeeded":
   835→            if plan.checksum_sha256 is None:
   836→                intact = True  # legacy plan without checksum — existence only
   837→            else:
   838→                from app.water.service import _plan_snapshot_checksum
   839→
   840→                intact = plan.checksum_sha256 == _plan_snapshot_checksum(plan.result or {}, plan)
   841→        referenced["water_plan"] = {"exists": plan is not None, "intact": intact}
   842→    if state.rotation_run_id is not None:
   843→        rotation = db.get(dbm.RotationRun, state.rotation_run_id)
   844→        referenced["rotation_run"] = {"exists": rotation is not None, "intact": rotation is not None}
   845→
   846→    boundary_row = db.scalar(
   847→        select(dbm.FieldBoundaryVersion).where(
   848→            dbm.FieldBoundaryVersion.field_id == state.field_id,
   849→            dbm.FieldBoundaryVersion.version_number == state.boundary_version,
   850→        )
   851→    )
   852→    return {
   853→        "state_id": str(state.id),
   854→        "checksum_valid": checksum_valid,
   855→        "manifest_checksum_valid": manifest_valid,
   856→        "referenced_artifacts": referenced,
   857→        "boundary_version_record_exists": boundary_row is not None,
   858→        "verified_at": datetime.now(dt_timezone.utc).isoformat(),
   859→    }

   862→def compare_field_states(db: Session, state_ids: list[uuid_mod.UUID]) -> dict[str, Any]:
   863→    """Compare 2+ states — compatibility is assessed, never implied."""
   864→    if len(state_ids) < 2:
   865→        raise ValidationFailed(
   866→            "At least two state ids are required to compare.",
   867→            details=[{"field": "state_ids"}],
   868→        )
   869→    states: list[dbm.FieldState] = []
   870→    for sid in state_ids:
   871→        state = get_field_state(db, sid)
   872→        if state is None:
   873→            raise NotFound(f"Field state {sid} not found.")
   874→        states.append(state)
   875→
   876→    field_ids = {str(s.field_id) for s in states}
   877→    same_field = len(field_ids) == 1
   878→    if not same_field:
   879→        raise ValidationFailed(
   880→            "Field states from different fields cannot be compared for "
   881→            "temporal/spatial compatibility.",
   882→            details=[{"field": "state_ids", "reason": ",".join(sorted(field_ids))}],
   883→        )
   884→
   885→    dimension_refs = ("environmental_feature_run_id", "water_plan_id", "rotation_run_id")
   886→    dimensions: dict[str, Any] = {}
   887→    for dim in dimension_refs:
   888→        values = [getattr(s, dim) for s in states]
   889→        dimensions[dim] = {
   890→            "consistent": len({str(v) for v in values if v is not None}) <= 1 and all(v is not None for v in values),
   891→            "values": [str(v) if v else None for v in values],
   892→        }
   893→    dimensions["boundary_version"] = {
   894→        "consistent": len({s.boundary_version for s in states}) == 1,
   895→        "values": [s.boundary_version for s in states],
   896→    }
   897→    dimensions["observation_manifest_sha256"] = {
   898→        "consistent": len({s.observation_manifest_sha256 for s in states}) == 1,
   899→        "values": [s.observation_manifest_sha256 for s in states],
   900→    }
   901→
   902→    pair_overlaps: list[dict[str, Any]] = []
   903→    for i in range(len(states)):
   904→        for j in range(i + 1, len(states)):
   905→            a, b = states[i], states[j]
   906→            fraction = fsc.period_overlap_fraction(a.period_start, a.period_end, b.period_start, b.period_end)
   907→            pair_overlaps.append(
   908→                {
   909→                    "state_a": str(a.id),
   910→                    "state_b": str(b.id),
   911→                    "period_overlap_fraction": fraction,
   912→                    "compatible_periods": (fraction or 0.0) > 0.0,
   913→                }
   914→            )
   915→
   916→    return {
   917→        "state_ids": [str(s.id) for s in states],
   918→        "same_field": same_field,
   919→        "field_id": str(states[0].field_id),
   920→        "boundary_versions_consistent": dimensions["boundary_version"]["consistent"],
   921→        "dimension_comparison": dimensions,
   922→        "pairwise_period_overlap": pair_overlaps,
   923→        "note": (
   924→            "Comparison reports facts about the stored snapshots; states with "
   925→            "different boundary versions or non-overlapping periods are NOT "
   926→            "interchangeable and must not be treated as a time series without "
   927→            "explicit qualification."
   928→        ),
   929→    }
