     1→"""Scenario domain service (Milestone 7.2).
     2→
     3→Implements the scenario lifecycle (create → validate → finalize → archive),
     4→version/branch lineage, definition-level comparison and checksum
     5→verification on top of the pure contracts and validation engine.
     6→
     7→Hard architectural rules (M7.2 spec §2):
     8→
     9→- the parent Field State is the authoritative baseline and is NEVER mutated;
    10→- a scenario is a DEFINITION — no endpoint here executes anything;
    11→- finalized scenarios are immutable (application transitions + DB trigger);
    12→- every reference boundary resolves tenant ownership server-side;
    13→- climate projection / weather forecast / soil / management are reported
    14→  unavailable and never fabricated.
    15→"""

    59→# ------------------------------------------------------------- helpers
    60→def _as_utc(value: Any, field_name: str) -> datetime:
    61→    """Parse an ISO datetime/date into an aware UTC datetime (M7.1 idiom:
    62→    naive → UTC, plain dates → UTC midnight)."""
    63→    if isinstance(value, datetime):
    64→        parsed = value
    65→    elif isinstance(value, date_cls):
    66→        parsed = datetime(value.year, value.month, value.day)
    67→    elif isinstance(value, str):
    68→        text = value.strip()
    69→        if text.endswith("Z"):
    70→            text = text[:-1] + "+00:00"
    71→        try:
    72→            parsed = datetime.fromisoformat(text)
    73→        except ValueError:
    74→            try:
    75→                d = date_cls.fromisoformat(text)
    76→            except ValueError:
    77→                raise ValidationFailed(
    78→                    f"{field_name} must be an ISO-8601 date or datetime.",
    79→                    details=[{"field": field_name, "reason": str(value)}],
    80→                ) from None
    81→            parsed = datetime(d.year, d.month, d.day)
    82→    else:
    83→        raise ValidationFailed(
    84→            f"{field_name} must be an ISO-8601 date or datetime.",
    85→            details=[{"field": field_name, "reason": str(value)}],
    86→        )
    87→    if parsed.tzinfo is None:
    88→        parsed = parsed.replace(tzinfo=dt_timezone.utc)
    89→    return parsed.astimezone(dt_timezone.utc)
    90→
    91→
    92→def _iso(value: datetime | None) -> str | None:
    93→    return value.astimezone(dt_timezone.utc).isoformat() if value else None

    96→def _transition(scenario: dbm.Scenario, target: str) -> None:
    97→    """Apply a lifecycle transition, enforcing the documented legal table."""
    98→    allowed = sc.LEGAL_TRANSITIONS.get(scenario.scenario_status, frozenset())
    99→    if target not in allowed:
   100→        raise Conflict(
   101→            f"Illegal scenario status transition "
   102→            f"'{scenario.scenario_status}' → '{target}'.",
   103→            code="SCENARIO_INVALID_STATE_TRANSITION",
   104→            details=[
   105→                {"field": "scenario_status", "from": scenario.scenario_status, "to": target}
   106→            ],
   107→        )
   108→    scenario.scenario_status = target

   111→def _require_mutable(scenario: dbm.Scenario, action: str) -> None:
   112→    if scenario.scenario_status in ("finalized", "archived"):
   113→        raise Conflict(
   114→            f"Scenario is {scenario.scenario_status} and immutable; {action} is not "
   115→            "permitted. Create a new version instead.",
   116→            code="SCENARIO_IMMUTABLE",
   117→            details=[{"scenario_status": scenario.scenario_status}],
   118→        )

   121→def _resolve_parent_state(db: Session, field: dbm.Field, state_id: Any) -> dbm.FieldState:
   122→    """Resolve the parent Field State with full reference integrity.
   123→
   124→    Tenant ownership is already established for `field` by the API layer;
   125→    the parent state must additionally belong to THAT field — a state from
   126→    any other field (same tenant or not) is refused without leaking its
   127→    existence.
   128→    """
   129→    try:
   130→        parsed = uuid_mod.UUID(str(state_id))
   131→    except (TypeError, ValueError):
   132→        raise ValidationFailed(
   133→            "parent_field_state_id must be a UUID.",
   134→            details=[{"field": "parent_field_state_id"}],
   135→        ) from None
   136→    state = db.get(dbm.FieldState, parsed)
   137→    if state is None or state.field_id != field.id:
   138→        raise ValidationFailed(
   139→            "parent_field_state_id must reference a Field State of the same field.",
   140→            details=[{"field": "parent_field_state_id", "reason": str(state_id)}],
   141→            code="PARENT_STATE_MISMATCH",
   142→        )
   143→    if state.status != "succeeded":
   144→        raise ValidationFailed(
   145→            "The parent Field State has not succeeded and cannot be used as a baseline.",
   146→            details=[{"field": "parent_field_state_id", "status": state.status}],
   147→            code="PARENT_STATE_NOT_SUCCEEDED",
   148→        )
   149→    if not state.checksum_sha256:
   150→        raise ValidationFailed(
   151→            "The parent Field State has no checksum and cannot be referenced.",
   152→            details=[{"field": "parent_field_state_id"}],
   153→            code="PARENT_STATE_NOT_VERIFIED",
   154→        )
   155→    return state

   158→def _resolve_parent_scenario(
   159→    db: Session, field: dbm.Field, scenario_id: Any
   160→) -> dbm.Scenario:
   161→    try:
   162→        parsed = uuid_mod.UUID(str(scenario_id))
   163→    except (TypeError, ValueError):
   164→        raise ValidationFailed(
   165→            "parent_scenario_id must be a UUID.",
   166→            details=[{"field": "parent_scenario_id"}],

   168→    parent = db.get(dbm.Scenario, parsed)
   169→    if parent is None or parent.field_id != field.id:
   170→        raise ValidationFailed(
   171→            "parent_scenario_id must reference a scenario of the same field.",
   172→            details=[{"field": "parent_scenario_id", "reason": str(scenario_id)}],
   173→            code="PARENT_SCENARIO_MISMATCH",
   174→        )
   175→    return parent

   178→def _build_configuration(
   179→    scenario_type: str, raw_config: dict[str, Any] | None
   180→) -> tuple[Any, dict[str, Any]]:
   181→    """Parse + canonicalize the configuration document."""
   182→    if not isinstance(raw_config, dict):
   183→        raise ValidationFailed(
   184→            "configuration must be an object.",
   185→            details=[{"field": "configuration"}],
   186→        )
   187→    payload = {"schema_version": sc.SCENARIO_CONFIG_SCHEMA_VERSION, **raw_config}
   188→    if payload.get("type") not in (None, scenario_type):
   189→        raise ValidationFailed(
   190→            "configuration.type must match the scenario type.",
   191→            details=[{"field": "configuration.type"}],
   192→        )
   193→    payload["type"] = scenario_type
   194→    try:
   195→        model = sc.parse_configuration(scenario_type, payload)
   196→    except PydanticValidationError as exc:
   197→        details = [
   198→            {"field": ".".join(str(p) for p in err["loc"]), "reason": err["msg"]}
   199→            for err in exc.errors()
   200→        ]
   201→        raise ValidationFailed(
   202→            "Scenario configuration failed contract validation.",
   203→            details=details,
   204→            code="SCENARIO_CONFIGURATION_INVALID",
   205→        ) from None
   206→    return model, sc.canonicalize_configuration(model)

   209→def _checksum_for(
   210→    *,
   211→    scenario_type: str,
   212→    canonical_configuration: dict[str, Any],
   213→    parent_state: dbm.FieldState,
   214→    timezone: str,
   215→) -> tuple[str, str]:
   216→    payload = sc.scenario_checksum_payload(
   217→        scenario_type=scenario_type,
   218→        configuration_schema_version=sc.SCENARIO_CONFIG_SCHEMA_VERSION,
   219→        canonical_configuration=canonical_configuration,
   220→        parent_field_state_id=str(parent_state.id),
   221→        parent_snapshot_checksum=parent_state.checksum_sha256 or "",
   222→        boundary_version=parent_state.boundary_version,
   223→        timezone=timezone,
   224→    )
   225→    return sc.compute_scenario_checksum(payload), canonical_json(payload)

   228→# ------------------------------------------------------------- create
   229→def create_scenario(
   230→    db: Session,
   231→    *,
   232→    field: dbm.Field,
   233→    payload: dict[str, Any],
   234→    created_by: Any = None,
   235→) -> dbm.Scenario:
   236→    """Create a draft scenario definition (or idempotently return the
   237→    existing one for a repeated idempotency key)."""
   238→    idempotency_key = payload.get("idempotency_key")
   239→    if idempotency_key:
   240→        existing = db.scalar(
   241→            select(dbm.Scenario).where(
   242→                dbm.Scenario.idempotency_key == idempotency_key,
   243→                dbm.Scenario.field_id == field.id,
   244→            )
   245→        )
   246→        if existing is not None:
   247→            return existing
   248→
   249→    scenario_type = payload.get("scenario_type") or (payload.get("configuration") or {}).get("type")
   250→    if not scenario_type:
   251→        raise ValidationFailed(
   252→            "scenario_type is required.",
   253→            details=[{"field": "scenario_type"}],
   254→        )
   255→    if scenario_type not in sc.IMPLEMENTED_SCENARIO_TYPES:
   256→        capability = sc.SCENARIO_TYPE_CAPABILITIES.get(scenario_type)
   257→        reason = (
   258→            capability.unavailable_reason if capability is not None else "Unknown scenario type."
   259→        )
   260→        raise ValidationFailed(
   261→            f"Scenario type '{scenario_type}' is not implemented and cannot be created.",
   262→            details=[{"field": "scenario_type", "reason": reason}],
   263→            code="SCENARIO_TYPE_NOT_IMPLEMENTED",
   264→        )
   265→
   266→    name = str(payload.get("name") or "").strip()
   267→    if not name:
   268→        capability = sc.SCENARIO_TYPE_CAPABILITIES[scenario_type]
   269→        name = f"{capability.display_name} scenario"
   270→    if len(name) > sc.SCENARIO_NAME_MAX:
   271→        name = name[: sc.SCENARIO_NAME_MAX]
   272→    description = payload.get("description")
   273→    if description is not None:
   274→        description = str(description).strip() or None
   275→
   276→    parent_scenario = None
   277→    lineage_kind = sc.LineageKind.ROOT.value
   278→    scenario_version = 1
   279→    if payload.get("parent_scenario_id"):
   280→        parent_scenario = _resolve_parent_scenario(db, field, payload.get("parent_scenario_id"))
   281→        lineage_kind = sc.LineageKind.BRANCH.value
   282→
   283→    model, canonical_config = _build_configuration(scenario_type, payload.get("configuration"))
   284→    config_dict = model.model_dump(mode="json")
   285→
   286→    # The parent reference lives INSIDE the configuration document (M7.2
   287→    # spec §6 example); resolve it with full integrity checks.
   288→    parent_state = _resolve_parent_state(db, field, model.parent_field_state_id)
   289→
   290→    # Temporal scope: hypothetical types carry their own period; the observed
   291→    # reference inherits the parent state's period verbatim.
   292→    if scenario_type == sc.ScenarioType.HISTORICAL_OBSERVED.value:
   293→        period_start, period_end = parent_state.period_start, parent_state.period_end
   294→    else:
   295→        period = config_dict.get("period") or {}
   296→        period_start = _as_utc(period.get("start"), "configuration.period.start")
   297→        period_end = _as_utc(period.get("end"), "configuration.period.end")
   298→
   299→    timezone = parent_state.timezone or "UTC"
   300→    checksum, _canonical_payload = _checksum_for(
   301→        scenario_type=scenario_type,
   302→        canonical_configuration=canonical_config,
   303→        parent_state=parent_state,
   304→        timezone=timezone,
   305→    )
   306→
   307→    capability = sc.SCENARIO_TYPE_CAPABILITIES[scenario_type]
   308→    scenario = dbm.Scenario(
   309→        field_id=field.id,
   310→        name=name,
   311→        description=description,
   312→        scenario_type=scenario_type,
   313→        scenario_status=sc.ScenarioStatus.DRAFT.value,
   314→        lineage_kind=lineage_kind,
   315→        scenario_version=scenario_version,
   316→        parent_field_state_id=parent_state.id,
   317→        parent_snapshot_checksum=parent_state.checksum_sha256,
   318→        boundary_version=parent_state.boundary_version,
   319→        parent_scenario_id=parent_scenario.id if parent_scenario is not None else None,
   320→        configuration_schema_version=sc.SCENARIO_CONFIG_SCHEMA_VERSION,
   321→        canonical_configuration=canonical_config,
   322→        configuration_checksum=checksum,
   323→        period_start=period_start,
   324→        period_end=period_end,
   325→        timezone=timezone,
   326→        evidence_classification=capability.evidence_classification.value,
   327→        assumption_metadata=(config_dict.get("assumption_metadata") or {}),
   328→        validation_summary={},
   329→        created_by=getattr(created_by, "id", None),
   330→        idempotency_key=idempotency_key,
   331→    )
   332→    db.add(scenario)
   333→    db.commit()
   334→    db.refresh(scenario)
   335→    return scenario
   336→

   343→def list_scenarios(
   344→    db: Session,
   345→    field_id: uuid_mod.UUID,
   346→    *,
   347→    scenario_type: str | None = None,
   348→    scenario_status: str | None = None,

   353→    if scenario_status:

   356→        db.execute(query.order_by(dbm.Scenario.created_at.desc())).scalars().all()
   357→    )

   360→# ------------------------------------------------------------- edit / delete
   361→def patch_scenario(
   362→    db: Session,
   363→    scenario: dbm.Scenario,
   364→    payload: dict[str, Any],
   365→) -> dbm.Scenario:
   366→    """Edit a non-finalized definition. Any edit resets validation: the
   367→    previous verdict referred to a configuration that no longer exists."""
   368→    _require_mutable(scenario, "PATCH")
   369→    if scenario.scenario_status not in EDITABLE_STATUSES:
   370→        raise Conflict(
   371→            f"Scenario in status '{scenario.scenario_status}' cannot be edited.",
   372→            code="SCENARIO_INVALID_STATE_TRANSITION",
   373→        )
   374→
   375→    if "name" in payload:
   376→        name = str(payload.get("name") or "").strip()
   377→        if not name:
   378→            raise ValidationFailed("name cannot be empty.", details=[{"field": "name"}])
   379→        scenario.name = name[: sc.SCENARIO_NAME_MAX]
   380→    if "description" in payload:
   381→        scenario.description = str(payload.get("description") or "").strip() or None
   382→
   383→    if "configuration" in payload:
   384→        raw_config = payload.get("configuration")
   385→        if not isinstance(raw_config, dict) or not raw_config:
   386→            raise ValidationFailed(
   387→                "configuration must be a non-empty object.",
   388→                details=[{"field": "configuration"}],
   389→            )
   390→        model, canonical_config = _build_configuration(scenario.scenario_type, raw_config)
   391→        config_dict = model.model_dump(mode="json")
   392→        checksum, _ = _checksum_for(
   393→            scenario_type=scenario.scenario_type,
   394→            canonical_configuration=canonical_config,
   395→            parent_state=db.get(dbm.FieldState, scenario.parent_field_state_id),
   396→            timezone=scenario.timezone,
   397→        )
   398→        if checksum != scenario.configuration_checksum:
   399→            scenario.canonical_configuration = canonical_config
   400→            scenario.configuration_checksum = checksum
   401→            scenario.assumption_metadata = config_dict.get("assumption_metadata") or {}
   402→            if scenario.scenario_type != sc.ScenarioType.HISTORICAL_OBSERVED.value:
   403→                period = config_dict.get("period") or {}
   404→                scenario.period_start = _as_utc(period.get("start"), "configuration.period.start")
   405→                scenario.period_end = _as_utc(period.get("end"), "configuration.period.end")
   406→
   407→    # Any edit resets validation state.
   408→    if scenario.scenario_status != sc.ScenarioStatus.DRAFT.value:
   409→        _transition(scenario, sc.ScenarioStatus.DRAFT.value)
   410→    scenario.validation_summary = {}
   411→    db.commit()
   412→    db.refresh(scenario)
   413→    return scenario

   416→def delete_scenario(db: Session, scenario: dbm.Scenario) -> None:
   417→    """Only drafts can be hard-deleted; everything else must be archived so
   418→    historical references and lineage stay intact."""
   419→    if scenario.scenario_status != sc.ScenarioStatus.DRAFT.value:
   420→        raise Conflict(
   421→            "Only draft scenarios can be deleted; finalized/archived scenarios "
   422→            "must be archived instead to preserve lineage and checksums.",
   423→            code="SCENARIO_DELETE_RESTRICTED",
   424→        )
   425→    children = db.scalar(
   426→        select(dbm.Scenario.id).where(dbm.Scenario.parent_scenario_id == scenario.id).limit(1)
   427→    )
   428→    if children is not None:
   429→        raise Conflict(
   430→            "Scenario has child scenarios in the lineage and cannot be deleted.",
   431→            code="SCENARIO_DELETE_RESTRICTED",
   432→        )
   433→    db.delete(scenario)
   434→    db.commit()

   438→def _parent_state_view(db: Session, state: dbm.FieldState) -> sv.ParentStateView:
   439→    boundary_row = db.scalar(
   440→        select(dbm.FieldBoundaryVersion.id).where(
   441→            dbm.FieldBoundaryVersion.field_id == state.field_id,
   442→            dbm.FieldBoundaryVersion.version_number == state.boundary_version,
   443→        )
   444→    )
   445→    manifest = state.observation_manifest or {}
   446→    field = db.get(dbm.Field, state.field_id)
   447→    farm = db.get(dbm.Farm, field.farm_id) if field is not None else None
   448→    return sv.ParentStateView(
   449→        field_id=str(state.field_id),
   450→        organization_id=str(farm.organization_id) if farm is not None else "",
   451→        status=state.status,
   452→        checksum=state.checksum_sha256,
   453→        boundary_version=state.boundary_version,
   454→        boundary_record_exists=boundary_row is not None,
   455→        period_start=state.period_start,
   456→        period_end=state.period_end,
   457→        timezone=state.timezone,
   458→        domain_availability=state.domain_availability or {},
   459→        observation_count=int(manifest.get("observation_count") or 0),
   460→        synthetic_observation_count=int(manifest.get("synthetic_observation_count") or 0),
   461→    )

   476→def validate_scenario(
   477→    db: Session, scenario: dbm.Scenario, *, created_by: Any = None
   478→) -> dict[str, Any]:
   479→    """Run the pure validation engine and persist an immutable record.
   480→
   481→    The verdict is about the DEFINITION's integrity only — never about
   482→    execution or outcomes (nothing executes in M7.2).
   483→    """
   484→    _require_mutable(scenario, "validate")
   485→    if scenario.scenario_status not in VALIDATE_ALLOWED_STATUSES:
   486→        raise Conflict(
   487→            f"Scenario in status '{scenario.scenario_status}' cannot be "
   488→            "validated; edit it (PATCH) to return to draft first.",
   489→            code="SCENARIO_INVALID_STATE_TRANSITION",
   490→        )
   491→
   492→    parent_state = db.get(dbm.FieldState, scenario.parent_field_state_id)
   493→    parent_view = _parent_state_view(db, parent_state) if parent_state is not None else None
   494→    requesting_org = ""
   495→    if parent_state is not None:
   496→        field = db.get(dbm.Field, parent_state.field_id)
   497→        farm = db.get(dbm.Farm, field.farm_id) if field is not None else None
   498→        requesting_org = str(farm.organization_id) if farm is not None else ""
   499→    result = sv.validate_scenario(
   500→        _scenario_view(scenario),
   501→        parent_view,
   502→        requesting_organization_id=requesting_org,
   503→    )
   504→
   505→    # Draft → validation_pending → result (documented transition path).
   506→    _transition(scenario, sc.ScenarioStatus.VALIDATION_PENDING.value)
   507→    lifecycle = sc.VALIDATION_RESULT_TO_LIFECYCLE.get(result.status, sc.ScenarioStatus.INVALID.value)
   508→    _transition(scenario, lifecycle)
   509→
   510→    document = result.to_document()
   511→    document["configuration_checksum"] = scenario.configuration_checksum
   512→    record = dbm.ScenarioValidationRecord(
   513→        scenario_id=scenario.id,
   514→        scenario_version=scenario.scenario_version,
   515→        validation_rule_version=sc.SCENARIO_VALIDATION_RULES_VERSION,
   516→        validation_status=result.status,
   517→        validation_result=document,
   518→        configuration_checksum=scenario.configuration_checksum,
   519→        created_by=getattr(created_by, "id", None),
   520→    )

   527→    }
   528→    db.commit()
   529→    db.refresh(scenario)
   530→    return validation_payload(record, scenario)

   585→# ------------------------------------------------------------- versions / lineage
   586→def create_scenario_version(
   587→    db: Session,
   588→    scenario: dbm.Scenario,
   589→    payload: dict[str, Any],
   590→    *,
   591→    created_by: Any = None,
   592→) -> dbm.Scenario:
   593→    """Create a new VERSION of a finalized scenario (spec §7: a changed
   594→    definition is a new version — the original row is never touched)."""
   595→    if scenario.scenario_status != sc.ScenarioStatus.FINALIZED.value:
   596→        raise Conflict(
   597→            "Only finalized scenarios can be versioned; edit drafts directly "
   598→            "with PATCH.",
   599→            code="SCENARIO_INVALID_STATE_TRANSITION",
   600→        )
   601→    field = db.get(dbm.Field, scenario.field_id)
   602→    new_type = payload.get("scenario_type") or scenario.scenario_type
   603→    if new_type != scenario.scenario_type:
   604→        raise ValidationFailed(
   605→            "A new version keeps the scenario type; create a branch (POST "
   606→            "/fields/{id}/scenarios with parent_scenario_id) for a different type.",
   607→            details=[{"field": "scenario_type"}],
   608→            code="SCENARIO_TYPE_MISMATCH",
   609→        )
   610→    configuration = payload.get("configuration")
   611→    if configuration is None:
   612→        configuration = scenario.canonical_configuration
   613→    model, canonical_config = _build_configuration(scenario.scenario_type, configuration)
   614→    config_dict = model.model_dump(mode="json")
   615→    if str(model.parent_field_state_id) != str(scenario.parent_field_state_id):
   616→        raise ValidationFailed(
   617→            "A new version keeps the parent Field State of its predecessor.",
   618→            details=[{"field": "configuration.parent_field_state_id"}],
   619→            code="PARENT_STATE_MISMATCH",
   620→        )
   621→    checksum, _ = _checksum_for(
   622→        scenario_type=scenario.scenario_type,
   623→        canonical_configuration=canonical_config,
   624→        parent_state=db.get(dbm.FieldState, scenario.parent_field_state_id),
   625→        timezone=scenario.timezone,
   626→    )
   627→    if scenario_type_hypothetical(scenario.scenario_type):
   628→        period = config_dict.get("period") or {}
   629→        period_start = _as_utc(period.get("start"), "configuration.period.start")
   630→        period_end = _as_utc(period.get("end"), "configuration.period.end")
   631→    else:
   632→        period_start, period_end = scenario.period_start, scenario.period_end
   633→
   634→    version = dbm.Scenario(
   635→        field_id=scenario.field_id,
   636→        name=str(payload.get("name") or scenario.name)[: sc.SCENARIO_NAME_MAX],
   637→        description=payload.get("description", scenario.description),
   638→        scenario_type=scenario.scenario_type,
   639→        scenario_status=sc.ScenarioStatus.DRAFT.value,
   640→        lineage_kind=sc.LineageKind.REVISION.value,
   641→        scenario_version=scenario.scenario_version + 1,
   642→        parent_field_state_id=scenario.parent_field_state_id,
   643→        parent_snapshot_checksum=scenario.parent_snapshot_checksum,
   644→        boundary_version=scenario.boundary_version,
   645→        parent_scenario_id=scenario.id,
   646→        configuration_schema_version=sc.SCENARIO_CONFIG_SCHEMA_VERSION,
   647→        canonical_configuration=canonical_config,
   648→        configuration_checksum=checksum,
   649→        period_start=period_start,
   650→        period_end=period_end,
   651→        timezone=scenario.timezone,
   652→        evidence_classification=scenario.evidence_classification,
   653→        assumption_metadata=config_dict.get("assumption_metadata") or {},
   654→        validation_summary={},
   655→        created_by=getattr(created_by, "id", None),
   656→        idempotency_key=payload.get("idempotency_key"),
   657→    )
   658→    # `field` is resolved to keep the tenancy chain explicit (same farm).
   659→    assert field is not None and field.id == scenario.field_id
   660→    db.add(version)
   661→    db.commit()
   662→    db.refresh(version)
   663→    return version

   673→def _version_chain(db: Session, scenario: dbm.Scenario) -> list[dbm.Scenario]:
   674→    """All members of the revision chain containing `scenario` (ancestors +
   675→    self + descendants), ordered by scenario_version."""
   676→    root = scenario
   677→    guard = 0
   678→    while root.parent_scenario_id is not None and root.lineage_kind == sc.LineageKind.REVISION.value:
   679→        parent = db.get(dbm.Scenario, root.parent_scenario_id)
   680→        if parent is None or parent.lineage_kind != sc.LineageKind.REVISION.value:
   681→            break
   682→        root = parent
   683→        guard += 1
   684→        if guard > 1000:  # pragma: no cover — cycle guard
   685→            break
   686→    chain = [root]
   687→    current = root
   688→    guard = 0
   689→    while True:
   690→        child = db.scalar(
   691→            select(dbm.Scenario)
   692→            .where(
   693→                dbm.Scenario.parent_scenario_id == current.id,
   694→                dbm.Scenario.lineage_kind == sc.LineageKind.REVISION.value,
   695→            )
   696→            .order_by(dbm.Scenario.scenario_version.asc())
   697→        )
   698→        if child is None:
   699→            break
   700→        chain.append(child)
   701→        current = child
   702→        guard += 1
   703→        if guard > 1000:  # pragma: no cover — cycle guard
   704→            break
   705→    return chain

   731→def get_lineage(db: Session, scenario: dbm.Scenario) -> dict[str, Any]:
   732→    """Ancestors (any lineage kind) + direct children + revision chain."""
   733→    ancestors: list[dict[str, Any]] = []
   734→    current = scenario
   735→    guard = 0
   736→    while current.parent_scenario_id is not None and guard < 1000:
   737→        parent = db.get(dbm.Scenario, current.parent_scenario_id)
   738→        if parent is None:
   739→            break
   740→        ancestors.append(
   741→            {
   742→                "id": str(parent.id),
   743→                "name": parent.name,
   744→                "scenario_type": parent.scenario_type,
   745→                "scenario_version": parent.scenario_version,
   746→                "lineage_kind": parent.lineage_kind,
   747→                "scenario_status": parent.scenario_status,
   748→                "configuration_checksum": parent.configuration_checksum,
   749→            }
   750→        )
   751→        current = parent
   752→        guard += 1
   753→    children = (
   754→        db.execute(
   755→            select(dbm.Scenario)
   756→            .where(dbm.Scenario.parent_scenario_id == scenario.id)
   757→            .order_by(dbm.Scenario.created_at.asc())
   758→        )
   759→        .scalars()
   760→        .all()
   761→    )
   762→    chain = _version_chain(db, scenario)
   763→    return {
   764→        "scenario_id": str(scenario.id),
   765→        "lineage_kind": scenario.lineage_kind,
   766→        "scenario_version": scenario.scenario_version,
   767→        "ancestors": ancestors,
   768→        "children": [
   769→            {
   770→                "id": str(c.id),
   771→                "name": c.name,
   772→                "scenario_type": c.scenario_type,
   773→                "scenario_version": c.scenario_version,
   774→                "lineage_kind": c.lineage_kind,
   775→                "scenario_status": c.scenario_status,
   776→                "configuration_checksum": c.configuration_checksum,
   777→            }
   778→            for c in children
   779→        ],
   780→        "revision_chain_length": len(chain),
   781→        "note": (
   782→            "Revisions (new versions) and branches share one parent link, "
   783→            "distinguished by lineage_kind; archived scenarios remain part "
   784→            "of the lineage and stay checksum-verifiable."
   785→        ),
   786→    }

   789→# ------------------------------------------------------------- comparison
   790→def _diff_documents(
   791→    left: Any, right: Any, path: str = ""
   792→) -> list[dict[str, Any]]:
   793→    """Structural diff of two canonical documents (leaf-level)."""
   794→    diffs: list[dict[str, Any]] = []
   795→    if isinstance(left, dict) and isinstance(right, dict):
   796→        for key in sorted(set(left) | set(right)):
   797→            child_path = f"{path}.{key}" if path else key
   798→            if key not in left:
   799→                diffs.append({"path": child_path, "kind": "added", "left": None, "right": right[key]})
   800→            elif key not in right:
   801→                diffs.append({"path": child_path, "kind": "removed", "left": left[key], "right": None})
   802→            else:
   803→                diffs.extend(_diff_documents(left[key], right[key], child_path))
   804→    elif isinstance(left, list) and isinstance(right, list):
   805→        if left != right:
   806→            diffs.append({"path": path or "$", "kind": "changed", "left": left, "right": right})
   807→    elif left != right:
   808→        diffs.append({"path": path or "$", "kind": "changed", "left": left, "right": right})
   809→    return diffs

   812→def compare_scenarios(
   813→    db: Session, scenario: dbm.Scenario, against: dbm.Scenario
   814→) -> dict[str, Any]:
   815→    """Definition-level comparison ONLY (spec §10). No simulated outputs
   816→    exist in M7.2, so none are compared — the response says so explicitly."""
   817→    parent_a = db.get(dbm.FieldState, scenario.parent_field_state_id)
   818→    parent_b = db.get(dbm.FieldState, against.parent_field_state_id)
   819→
   820→    overlap = None
   821→    if parent_a is not None and parent_b is not None:
   822→        overlap = fsc.period_overlap_fraction(
   823→            scenario.period_start, scenario.period_end, against.period_start, against.period_end
   824→        )
   825→
   826→    config_diff = _diff_documents(
   827→        scenario.canonical_configuration or {}, against.canonical_configuration or {}
   828→    )
   829→    assumption_paths = [
   830→        d["path"]
   831→        for d in config_diff
   832→        if d["path"].startswith("adjustments.") or d["path"].startswith("period")
   833→    ]
   834→
   835→    parents_compatible = (
   836→        scenario.parent_field_state_id == against.parent_field_state_id
   837→        and scenario.parent_snapshot_checksum == against.parent_snapshot_checksum
   838→        and scenario.boundary_version == against.boundary_version
   839→    )
   840→
   841→    unavailable = {
   842→        dim: entry
   843→        for dim, entry in sc.SCENARIO_DIMENSION_AVAILABILITY.items()
   844→    }
   845→
   846→    return {
   847→        "scenario_a": {
   848→            "id": str(scenario.id),
   849→            "name": scenario.name,
   850→            "scenario_type": scenario.scenario_type,
   851→            "evidence_classification": scenario.evidence_classification,
   852→            "evidence_label": sc.evidence_label(scenario.evidence_classification),
   853→            "scenario_version": scenario.scenario_version,
   854→            "scenario_status": scenario.scenario_status,
   855→            "configuration_checksum": scenario.configuration_checksum,
   856→            "parent_field_state_id": str(scenario.parent_field_state_id),
   857→            "parent_snapshot_checksum": scenario.parent_snapshot_checksum,
   858→            "boundary_version": scenario.boundary_version,
   859→            "period": {"start": _iso(scenario.period_start), "end": _iso(scenario.period_end)},
   860→        },
   861→        "scenario_b": {
   862→            "id": str(against.id),
   863→            "name": against.name,
   864→            "scenario_type": against.scenario_type,
   865→            "evidence_classification": against.evidence_classification,
   866→            "evidence_label": sc.evidence_label(against.evidence_classification),
   867→            "scenario_version": against.scenario_version,
   868→            "scenario_status": against.scenario_status,
   869→            "configuration_checksum": against.configuration_checksum,
   870→            "parent_field_state_id": str(against.parent_field_state_id),
   871→            "parent_snapshot_checksum": against.parent_snapshot_checksum,
   872→            "boundary_version": against.boundary_version,
   873→            "period": {"start": _iso(against.period_start), "end": _iso(against.period_end)},
   874→        },
   875→        "parents": {
   876→            "same_field": scenario.field_id == against.field_id,
   877→            "same_parent_state": scenario.parent_field_state_id == against.parent_field_state_id,
   878→            "parent_checksums_equal": (
   879→                scenario.parent_snapshot_checksum == against.parent_snapshot_checksum
   880→            ),
   881→            "boundary_versions": [scenario.boundary_version, against.boundary_version],
   882→            "boundary_versions_equal": scenario.boundary_version == against.boundary_version,
   883→        },
   884→        "temporal_comparison": {
   885→            "period_overlap_fraction": overlap,
   886→            "compatible_periods": (overlap or 0.0) > 0.0,
   887→        },
   888→        "configuration_diff": config_diff,
   889→        "changed_assumptions": assumption_paths,
   890→        "identical_configuration": scenario.configuration_checksum == against.configuration_checksum,
   891→        "evidence_classes_differ": (
   892→            scenario.evidence_classification != against.evidence_classification
   893→        ),
   894→        "unavailable_dimensions": unavailable,
   895→        "warnings": [
   896→            *(
   897→                [
   898→                    "Scenarios reference DIFFERENT parent Field States — their "
   899→                    "configurations are not grounded in the same baseline evidence."
   900→                ]
   901→                if scenario.parent_field_state_id != against.parent_field_state_id
   902→                else []
   903→            ),
   904→            *(
   905→                ["Scenarios pin different boundary versions — spatial references differ."]
   906→                if scenario.boundary_version != against.boundary_version
   907→                else []
   908→            ),
   909→            *(
   910→                [
   911→                    "Scenarios have different evidence classifications (observed vs "
   912→                    "hypothetical vs not_available) — they are NOT equivalent evidence."
   913→                ]
   914→                if scenario.evidence_classification != against.evidence_classification
   915→                else []
   916→            ),
   917→            *(
   918→                ["The scenario periods do not overlap — differences are not temporally comparable."]
   919→                if overlap is not None and overlap <= 0.0
   920→                else []
   921→            ),
   922→        ],
   923→        "compatibility": {
   924→            "parents_compatible": parents_compatible,
   925→            "note": (
   926→                "Comparisons across incompatible parents or boundary versions "
   927→                "carry an explicit warning and must not be read as a like-for-"
   928→                "like contrast."
   929→            ),
   930→        },
   931→        "note": (
   932→            "Definition-level comparison only. No simulated environmental "
   933→            "outputs exist in M7.2 (M7.3 not implemented), so none are "
   934→            "compared and no score differences are fabricated."
   935→        ),
   936→    }

   939→# ------------------------------------------------------------- verification
   940→def verify_scenario(db: Session, scenario: dbm.Scenario) -> dict[str, Any]:
   941→    """Independently re-verify the stored definition from its own content."""
   942→    payload = sc.scenario_checksum_payload(
   943→        scenario_type=scenario.scenario_type,
   944→        configuration_schema_version=scenario.configuration_schema_version,
   945→        canonical_configuration=scenario.canonical_configuration or {},
   946→        parent_field_state_id=str(scenario.parent_field_state_id),
   947→        parent_snapshot_checksum=scenario.parent_snapshot_checksum,
   948→        boundary_version=scenario.boundary_version,
   949→        timezone=scenario.timezone,
   950→    )
   951→    recomputed = sc.compute_scenario_checksum(payload)
   952→    checksum_valid = scenario.configuration_checksum == recomputed
   953→
   954→    parent_state = db.get(dbm.FieldState, scenario.parent_field_state_id)
   955→    parent_intact = (
   956→        parent_state is not None
   957→        and parent_state.status == "succeeded"
   958→        and parent_state.checksum_sha256 == scenario.parent_snapshot_checksum
   959→    )
   960→    boundary_row = None
   961→    if parent_state is not None:
   962→        boundary_row = db.scalar(
   963→            select(dbm.FieldBoundaryVersion.id).where(
   964→                dbm.FieldBoundaryVersion.field_id == parent_state.field_id,
   965→                dbm.FieldBoundaryVersion.version_number == scenario.boundary_version,
   966→            )
   967→        )
   968→    latest_record = db.scalar(
   969→        select(dbm.ScenarioValidationRecord)
   970→        .where(dbm.ScenarioValidationRecord.scenario_id == scenario.id)
   971→        .order_by(dbm.ScenarioValidationRecord.created_at.desc())
   972→        .limit(1)
   973→    )
   974→    return {
   975→        "scenario_id": str(scenario.id),
   976→        "configuration_checksum_valid": checksum_valid,
   977→        "expected_checksum": recomputed,
   978→        "stored_checksum": scenario.configuration_checksum,
   979→        "parent_state_exists": parent_state is not None,
   980→        "parent_snapshot_checksum_matches": parent_intact,
   981→        "boundary_version_record_exists": boundary_row is not None,
   982→        "latest_validation": (
   983→            {
   984→                "record_id": str(latest_record.id),
   985→                "validation_status": latest_record.validation_status,
   986→                "configuration_checksum": latest_record.configuration_checksum,
   987→                "matches_current_configuration": (
   988→                    latest_record.configuration_checksum == scenario.configuration_checksum
   989→                ),
   990→            }
   991→            if latest_record is not None
   992→            else None
   993→        ),
   994→        "verified_at": _iso(datetime.now(dt_timezone.utc)),
   995→        "note": (
   996→            "Verification recomputes the definition checksum from stored "
   997→            "content and re-checks the parent reference. It says nothing "
   998→            "about simulation outcomes (none exist in M7.2)."
   999→        ),
  1000→    }
