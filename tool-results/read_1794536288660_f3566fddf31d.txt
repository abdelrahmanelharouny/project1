   119→/**
   120→ * Farms with their fields flattened via parallel queries — small workspaces,
   121→ * so this stays cheap. Exposed so tabs that need a global field picker
   122→ * (Observations / Data Passport) can render one list.
   123→ */
   124→export function useFarmsWithFields(enabled = true): {
   125→  farms: FarmWithFields[]
   126→  allFields: Field[]
   127→  isLoading: boolean
   128→  hasError: boolean
   129→} {
   130→  const farmsQuery = useFarms(enabled)
   131→  const farms = useMemo(() => farmsQuery.data?.items ?? [], [farmsQuery.data])
   132→
   133→  const fieldQueries = useQueries({
   134→    queries: farms.map((farm) => ({
   135→      queryKey: ['fields', farm.id],
   136→      queryFn: () => listFields(farm.id),
   137→      enabled,
   138→      staleTime: 15_000,
   139→    })),
   140→  })
   141→
   142→  const farmsWithFields = useMemo<FarmWithFields[]>(() => {
   143→    return farms.map((farm, index) => ({
   144→      farm,
   145→      fields: fieldQueries[index]?.data?.items ?? [],
   146→      fieldsQuery: fieldQueries[index],
   147→    }))
   148→  }, [farms, fieldQueries])
   149→
   150→  const allFields = useMemo(() => farmsWithFields.flatMap((entry) => entry.fields), [farmsWithFields])
   151→  const isLoading = farmsQuery.isLoading || fieldQueries.some((q) => q.isLoading)
   152→  const hasError = farmsQuery.isError || fieldQueries.some((q) => q.isError)
   153→
   154→  return { farms: farmsWithFields, allFields, isLoading, hasError }
   155→}
