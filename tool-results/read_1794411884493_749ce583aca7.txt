    33→def _crop(code, family=None, *, legume=False, cover=False, kc=None, duration=120,
    34→          temp_opt=(15, 28), temp_min_abs=5, ph=None, windows=None, traits_extra=None):
    35→    traits: dict[str, TraitValue] = {}
    36→
    37→    def t(code_, value=None, low=None, high=None, unit=None):
    38→        traits[code_] = TraitValue(
    39→            trait_code=code_, value_numeric=value, value_text=None, value_low=low,
    40→            value_high=high, unit=unit, region_code=None,
    41→            evidence_class="published_reference", confidence="moderate",
    42→            validation_status="reference", source_key="s", citation_text="citation",
    43→            pack_code="kp", pack_version=PACK_VERSION,
    44→        )
    45→
    46→    if duration is not None:
    47→        t("duration_days_total", value=duration, unit="d")
    48→    if kc is not None:
    49→        t("stage_init_days", value=20, unit="d")
    50→        t("stage_dev_days", value=30, unit="d")
    51→        t("stage_mid_days", value=40, unit="d")
    52→        t("stage_late_days", value=30, unit="d")
    53→    if temp_opt:
    54→        t("temp_opt_min_degC", value=temp_opt[0], unit="degC")
    55→        t("temp_opt_max_degC", value=temp_opt[1], unit="degC")
    56→    if temp_min_abs is not None:
    57→        t("temp_min_abs_degC", value=temp_min_abs, unit="degC")
    58→    if ph:
    59→        t("ph_opt_min", value=ph[0], unit="pH")
    60→        t("ph_opt_max", value=ph[1], unit="pH")
    61→    for code_, value in (traits_extra or {}).items():
    62→        t(code_, value=value)
    63→    return KnowledgeCrop(
    64→        crop_code=code, pack_code="kp", pack_version=PACK_VERSION, family_key=family,
    65→        scientific_name=None, common_name=code, crop_category=None,
    66→        life_cycle_type="annual", is_legume=legume, is_cover_crop_capable=cover,
    67→        traits=traits,
    68→        calendar_windows=tuple(windows or ()),
    69→        water_params=WaterParams(
    70→            kc_initial=kc[0], kc_mid=kc[1], kc_late=kc[2],
    71→            evidence_class="published_reference", source_key="s",
    72→            citation_text="citation", pack_code="kp", pack_version=PACK_VERSION,
    73→        ) if kc else None,
    74→    )

    99→def _evidence(**kw):
   100→    base = dict(
   101→        feature_run_id="run-1", is_synthetic=False,
   102→        season_precipitation_mm=120.0, season_mean_temp_degC=18.0,
   103→        season_mean_tmax_degC=None, season_hot_days_35c=2.0, season_et0_mm=420.0,
   104→        # Deepening F-07: the estimator refuses partial ET0 coverage (a month
   105→        # dropped from the total would under-report the requirement), so the
   106→        # fixture provides EVERY month of the May 1 - Sep 15 season.
   107→        monthly_et0=(
   108→            (2025, 5, 150.0), (2025, 6, 140.0), (2025, 7, 130.0),
   109→            (2025, 8, 120.0), (2025, 9, 90.0),
   110→        ),
   111→        coverage_ratio=1.0,
   112→    )
   113→    base.update(kw)
   114→    return ClimateEvidence(**base)

   117→def _farm(seasons=None, history=(), *, hemisphere="north", country="EG",
   118→          soil=None, climate=None, exclusions=(), equipment=(), allocation=None,
   119→          margins=None, synthetic=False):

   123→        hemisphere=hemisphere, seasons=tuple(seasons), soil=soil,
   124→        climate_by_season=climate or {s.label: _evidence() for s in seasons},
   125→        climate_run_refs={s.label: "run-1" for s in seasons},
   126→        history=tuple(history), market_exclusions=tuple(exclusions),
   127→        equipment_limits=tuple(equipment), water_allocation_m3=allocation,
   128→        margin_index_by_crop=dict(margins or {}),
   129→        synthetic_evidence_in_use=synthetic, soil_evidence_class=None,

   133→def _knowledge(crops, rules=()):

   293→        farm=farm, water_allocation_m3=allocation,
   294→        baseline_assignment=tuple((s.label, None) for s in seasons),
   295→        synthetic_evidence=farm.synthetic_evidence_in_use,
   296→    )

   543→class TestExplanations:
   544→    def test_explanation_blocks_complete(self):
   545→        crops = [
   546→            _crop("barley", "poaceae", duration=120, windows=[_window("05-01", "06-01")], kc=(0.3, 1.15, 0.25)),
   547→            _crop("soybean", "fabaceae", duration=120, windows=[_window("05-01", "06-01")], kc=(0.4, 1.15, 0.5), legume=True),
   548→        ]
   549→        rules = [_rule("r_ref", "barley", "barley", status="reference")]
   550→        problem = _problem(crops, rules)
   551→        from app.rotation.optimizer import optimize_strategy, PROFILE_WEIGHTS, build_baseline_solution
   552→        from app.rotation.constraints import check_all
   553→        from app.rotation import objectives as obj
   554→        from app.rotation.explain import build_explanations
   555→
   556→        sols = [build_baseline_solution(problem, "baseline")]
   557→        for key in ("water_priority", "soil_health_priority"):
   558→            res = optimize_strategy(problem, key, dict(PROFILE_WEIGHTS[key]))
   559→            assert isinstance(res, StrategySolution)
   560→            sols.append(res)
   561→        evaluated = []
   562→        for s in sols:
   563→            criteria = obj.evaluate_criteria(problem, dict(s.assignment))
   564→            evaluated.append(StrategySolution(
   565→                strategy_key=s.strategy_key, strategy_kind=s.strategy_kind,
   566→                is_baseline=s.is_baseline, profile_weights=s.profile_weights,
   567→                weights_effective={}, skipped_weights={}, assignment=s.assignment,
   568→                objective_value=s.objective_value, criteria=criteria,
   569→                constraints_checked=tuple(check_all(problem, dict(s.assignment))),
   570→                evidence_coverage=None,
   571→            ))
   572→        obj.normalize_within_set(evaluated)
   573→        final = []
   574→        for s in evaluated:
   575→            objective, eff, skip = obj.weighted_objective(s)
   576→            final.append(StrategySolution(
   577→                strategy_key=s.strategy_key, strategy_kind=s.strategy_kind,
   578→                is_baseline=s.is_baseline, profile_weights=s.profile_weights,
   579→                weights_effective=eff, skipped_weights=skip, assignment=s.assignment,
   580→                objective_value=objective, criteria=s.criteria,
   581→                constraints_checked=s.constraints_checked,
   582→                evidence_coverage=obj.evidence_coverage_of(s),
   583→            ))
   584→        obj.compute_pareto(final)
   585→        for s in final:
   586→            exp = build_explanations(problem, s, final)
   587→            for block in (
   588→                "why_generated", "recommendation", "constraints_checked",

   590→                "expected_benefits", "risks_trade_offs",
   591→                "local_expert_validation_required",
   592→            ):
   593→                assert block in exp, f"missing explanation block {block}"
   594→            assert exp["evidence_references"]
   595→            assert any("screening estimate" in l for l in exp["data_limitations"])
   596→        soil = next(s for s in final if s.strategy_key == "soil_health_priority")
   597→        exp = build_explanations(problem, soil, final)
   598→        assert exp["local_expert_validation_required"]
