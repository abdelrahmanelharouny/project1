    71→/** Thrown for any non-2xx API response; carries the parsed error envelope. */
    72→export class ApiError extends Error {
    73→  readonly status: number
    74→  readonly code: string
    75→  readonly details: ApiErrorDetail[]
    76→  readonly requestId: string | undefined
    77→
    78→  constructor(params: {
    79→    status: number
    80→    code: string
    81→    message: string
    82→    details?: ApiErrorDetail[]
    83→    requestId?: string
    84→  }) {
    85→    super(params.message)
    86→    this.name = 'ApiError'
    87→    this.status = params.status
    88→    this.code = params.code
    89→    this.details = params.details ?? []
    90→    this.requestId = params.requestId
    91→  }
    92→}

    94→/** Human-readable summary of any thrown error for toasts. */
    95→export function describeError(err: unknown): { title: string; description?: string } {
    96→  if (err instanceof ApiError) {
    97→    const firstDetail = err.details[0]
    98→    const description = firstDetail
    99→      ? `${firstDetail.field}: ${firstDetail.reason}${err.requestId ? ` · request ${err.requestId.slice(0, 8)}` : ''}`
   100→      : err.requestId
   101→        ? `request ${err.requestId.slice(0, 8)}`
   102→        : undefined
   103→    return { title: err.message, description }
   104→  }
   105→  if (err instanceof Error) {
   106→    return { title: 'Network error', description: err.message }
   107→  }
   108→  return { title: 'Unexpected error' }
   109→}

   111→function withPort(path: string): string {

   113→  return `${path}${joiner}${PORT_QUERY}`

   116→export function getToken(): string | null {
   117→  if (typeof window === 'undefined') return null
   118→  return window.localStorage.getItem(TOKEN_STORAGE_KEY)
   119→}

   121→export function setToken(token: string | null): void {
   122→  if (typeof window === 'undefined') return
   123→  if (token) {
   124→    window.localStorage.setItem(TOKEN_STORAGE_KEY, token)
   125→  } else {
   126→    window.localStorage.removeItem(TOKEN_STORAGE_KEY)
   127→    window.localStorage.removeItem(REFRESH_STORAGE_KEY)
   128→  }
   129→}

   136→function setRefreshToken(token: string | null): void {
   137→  if (typeof window === 'undefined') return
   138→  if (token) {
   139→    window.localStorage.setItem(REFRESH_STORAGE_KEY, token)
   140→  } else {
   141→    window.localStorage.removeItem(REFRESH_STORAGE_KEY)

   145→/**
   146→ * Exchange the stored refresh token for a fresh access+refresh pair
   147→ * (rotating refresh: the presented token is revoked server-side; reusing a
   148→ * superseded token revokes the whole session family).
   149→ */
   150→export async function refreshSession(): Promise<boolean> {
   151→  const refresh = getRefreshToken()
   152→  if (!refresh) return false
   153→  try {
   154→    const payload = await request<AuthResponse>(
   155→      '/api/v1/auth/refresh',
   156→      { method: 'POST', body: { refresh_token: refresh }, anonymous: true },
   157→      { skipAuthRetry: true },
   158→    )
   159→    if (!payload?.token) return false
   160→    setToken(payload.token)
   161→    setRefreshToken(payload.refresh_token ?? null)
   162→    return true
   163→  } catch {
   164→    // Invalid/expired/reused refresh — drop the stored session.
   165→    setToken(null)
   166→    return false
   167→  }
   168→}

   170→interface RequestOptions {
   171→  method?: 'GET' | 'POST' | 'PATCH' | 'PUT' | 'DELETE'
   172→  body?: unknown
   173→  /** Multipart form body (file uploads). Content-Type is set by the browser. */
   174→  formData?: FormData
   175→  /** Skip attaching the bearer token (health, dev/seed-demo). */
   176→  anonymous?: boolean
   177→}
   178→
   179→interface RequestInternals {
   180→  /** Prevent recursive 401→refresh retries (used by refreshSession itself). */
   181→  skipAuthRetry?: boolean
   182→}
   183→
   184→async function request<T>(
   185→  path: string,
   186→  options: RequestOptions = {},
   187→  internals: RequestInternals = {},
   188→): Promise<T> {
   189→  const headers: Record<string, string> = { Accept: 'application/json' }
   190→  if (options.body !== undefined) headers['Content-Type'] = 'application/json'
   191→  if (!options.anonymous) {
   192→    const token = getToken()
   193→    if (token) headers.Authorization = `Bearer ${token}`
   194→  }
   195→
   196→  let response: Response
   197→  try {
   198→    response = await fetch(withPort(path), {
   199→      method: options.method ?? 'GET',
   200→      headers,
   201→      body:
   202→        options.formData !== undefined
   203→          ? options.formData
   204→          : options.body !== undefined
   205→            ? JSON.stringify(options.body)
   206→            : undefined,
   207→      cache: 'no-store',
   208→    })
   209→  } catch (err) {
   210→    throw new ApiError({
   211→      status: 0,
   212→      code: 'NETWORK_ERROR',
   213→      message: 'Could not reach the FieldShift API.',
   214→      details: [],
   215→    })
   216→  }
   217→
   218→  const text = await response.text()
   219→  let payload: unknown = null
   220→  if (text) {
   221→    try {
   222→      payload = JSON.parse(text)
   223→    } catch {
   224→      payload = null
   225→    }
   226→  }
   227→
   228→  if (!response.ok) {
   229→    const envelope = payload as ApiErrorBody | null
   230→    const error = envelope && typeof envelope === 'object' && 'error' in envelope ? envelope.error : undefined
   231→    // Rotating-refresh retry (P0-2): an access token may have simply expired
   232→    // while the session is still valid. Try ONE refresh, then retry the
   233→    // original request once. Never recurse from the refresh call itself.
   234→    if (
   235→      response.status === 401 &&
   236→      !options.anonymous &&
   237→      !internals.skipAuthRetry &&
   238→      !path.includes('/auth/refresh')
   239→    ) {
   240→      const refreshed = await refreshSession()
   241→      if (refreshed) {
   242→        return request<T>(path, options, { skipAuthRetry: true })
   243→      }
   244→    }
   245→    throw new ApiError({
   246→      status: response.status,
   247→      code: error?.code ?? `HTTP_${response.status}`,
   248→      message: error?.message ?? `Request failed with status ${response.status}.`,
   249→      details: error?.details ?? [],
   250→      requestId: error?.request_id,
   251→    })
   252→  }
   253→
   254→  return payload as T
   255→}

   400→const JOB_TERMINAL_STATUSES = new Set(['succeeded', 'failed', 'cancelled'])
   401→
   402→/**
   403→ * Poll an ingestion/import job until it reaches a terminal state.
   404→ *
   405→ * The backend executes data jobs on a background worker pool (202 + status
   406→ * 'queued'); when workers are disabled it still executes inline (the response
   407→ * is already terminal, returned as-is). Progress callbacks receive every
   408→ * intermediate snapshot (status/progress for UI feedback).
   409→ */
   410→export async function pollJobUntilDone(
   411→  jobId: string,
   412→  opts: { onProgress?: (job: DataJob) => void; intervalMs?: number; timeoutMs?: number } = {},
   413→): Promise<DataJob> {
   414→  const interval = opts.intervalMs ?? 1000
   415→  const deadline = Date.now() + (opts.timeoutMs ?? 180_000)
   416→  let current = await getJob(jobId)
   417→  while (!JOB_TERMINAL_STATUSES.has(current.status)) {
   418→    opts.onProgress?.(current)
   419→    if (Date.now() > deadline) {
   420→      throw new Error('Timed out waiting for the job to finish — check its status later.')
   421→    }
   422→    await new Promise((resolve) => setTimeout(resolve, interval))
   423→    current = await getJob(jobId)
   424→  }
   425→  return current
   426→}

   457→export type GeometryParseResult =
   458→  | { ok: true; geometry: import('./types').GeoJSONGeometry }
   459→  | { ok: false; error: string }
   460→
   461→/** Parse + shallow-validate a GeoJSON Polygon / MultiPolygon from user input. */
   462→export function parseGeometryInput(text: string): GeometryParseResult {
   463→  let parsed: unknown
   464→  try {
   465→    parsed = JSON.parse(text)
   466→  } catch {
   467→    return { ok: false, error: 'The text is not valid JSON. Check for missing quotes, brackets or trailing commas.' }
   468→  }
   469→  if (typeof parsed !== 'object' || parsed === null || Array.isArray(parsed)) {
   470→    return { ok: false, error: 'Expected a GeoJSON object like {"type":"Polygon","coordinates":[...]}' }
   471→  }
   472→  const obj = parsed as Record<string, unknown>
   473→  if (obj.type === 'Polygon') {
   474→    if (!Array.isArray(obj.coordinates) || obj.coordinates.length === 0) {
   475→      return { ok: false, error: 'Polygon coordinates must be a non-empty array of rings.' }
   476→    }
   477→    return { ok: true, geometry: parsed as import('./types').GeoJSONPolygon }
   478→  }
   479→  if (obj.type === 'MultiPolygon') {
   480→    if (!Array.isArray(obj.coordinates) || obj.coordinates.length === 0) {
   481→      return { ok: false, error: 'MultiPolygon coordinates must be a non-empty array of polygons.' }
   482→    }
   483→    return { ok: true, geometry: parsed as import('./types').GeoJSONMultiPolygon }
   484→  }
   485→  return { ok: false, error: `Unsupported geometry type "${String(obj.type)}". Use Polygon or MultiPolygon.` }
   486→}

   505→/** Client-side guard mirroring backend job validation. Returns an error message or null. */
   506→export function validateJobWindow(
   507→  start: string,
   508→  end: string,
   509→  bounds?: { min?: string; max?: string; maxYears?: number },
   510→): string | null {
   511→  const dateRe = /^\d{4}-\d{2}-\d{2}$/
   512→  if (!dateRe.test(start) || !dateRe.test(end)) return 'Enter both dates as YYYY-MM-DD.'
   513→  const min = bounds?.min ?? COVERAGE_START
   514→  const max = bounds?.max ?? COVERAGE_END
   515→  if (start < min || end > max) {
   516→    return `Dates must fall inside the dataset coverage ${min} → ${max}.`
   517→  }
   518→  if (end < start) return 'The end date must be on or after the start date.'
   519→  const years = (Date.parse(end) - Date.parse(start)) / (365.25 * 24 * 3600 * 1000)
   520→  const maxYears = bounds?.maxYears ?? MAX_JOB_YEARS
   521→  if (years > maxYears) return `A single job covers at most ${maxYears} years. Split the range.`
   522→  return null
   523→}

   958→// --- laboratory CSV import ------------------------------------------------------
   959→// Backend contract: POST /soil-samples/import (org-level) requires a `mapping`
   960→// form field whose JSON carries default_field_id — the field binding arrives
   961→// there, so the component's (fieldId, file) signature is preserved.
   962→
   963→export async function importSoilLabCsv(fieldId: string, file: File): Promise<SoilLabImportResponse> {
   964→  const formData = new FormData()
   965→  formData.append('file', file)
   966→  formData.append('mapping', JSON.stringify({ default_field_id: fieldId }))
   967→  const data = await request<SoilLabImportResponse>('/api/v1/soil-samples/import', {
   968→    method: 'POST',
   969→    formData,
   970→  })
   971→  // The backend's idempotent "already_imported" reply omits the counters;
   972→  // normalize so callers always see numbers (re-import of identical content
   973→  // creates nothing — 0 is the truthful value there).
   974→  return {
   975→    ...data,
   976→    created_samples: data.created_samples ?? 0,
   977→    created_measurements: data.created_measurements ?? 0,
   978→    errors: data.errors ?? [],
   979→  }
   980→}
