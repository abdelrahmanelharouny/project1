   293→    unit: Mapped[str] = mapped_column(String(30), nullable=False)

   297→    time_support: Mapped[str] = mapped_column(String(20), nullable=False, default="daily")

   300→    quality_status: Mapped[str] = mapped_column(String(20), nullable=False, default="valid")

   305→    uncertainty_value: Mapped[float | None] = mapped_column(Float, nullable=True)
   306→    uncertainty_type: Mapped[str | None] = mapped_column(String(40), nullable=True)
   307→    is_synthetic: Mapped[bool] = mapped_column(Boolean, nullable=False, default=False)
   308→    # Job that produced this row (M3 audit): completes the job→observation→
   309→    # feature provenance chain. Nullable for legacy rows; jobs are never
   310→    # hard-deleted.
   311→    ingestion_job_id: Mapped[uuid.UUID | None] = mapped_column(
   312→        UUID(as_uuid=True), ForeignKey("ingestion_jobs.id", ondelete="SET NULL"), nullable=True
   313→    )
   314→    retrieved_at: Mapped[datetime] = mapped_column(DateTime(timezone=True), nullable=False)
   315→    processing_version: Mapped[str] = mapped_column(String(40), nullable=False, default="v1")
   316→    metadata: Mapped[dict[str, Any]] = mapped_column(JSONB, nullable=False, default=dict)
   317→    created_at: Mapped[datetime] = mapped_column(
   318→        DateTime(timezone=True), nullable=False, server_default=func.now()
   319→    )
   320→
   321→    field: Mapped[Field] = relationship(back_populates="observations")
   322→    dataset_product: Mapped[DatasetProduct] = relationship(back_populates="observations")

   324→        back_populates="observation", cascade="all, delete-orphan"
   325→    )
   326→
   327→    __table_args__ = (

   329→            "quality_status IN ('valid','suspect','invalid','missing','estimated','not_assessed')",

   331→        ),

   333→            "time_support IN ('instant','hourly','daily','pentadal','dekadal','monthly','seasonal','annual')",

   335→        ),
   336→        CheckConstraint(
   337→            "spatial_support_type IN ('point','grid_cell','polygon_average','field_intersection','station','lab_sample')",
   338→            name="ck_observation_spatial_support",

   340→        # Natural-key dedupe: makes provider ingestion idempotent on retry.

   342→            "field_id",
   343→            "dataset_product_id",
   344→            "variable_code",
   345→            "observed_at",
   346→            "time_support",
   347→            "spatial_support_type",
   348→            name="uq_observation_natural_key",
   349→        ),
   350→        Index("ix_observations_field_variable_time", field_id, variable_code, observed_at),
   351→        Index("ix_observations_ingestion_job", ingestion_job_id),
   352→        Index("ix_observations_dataset_product", dataset_product_id),
