    70→function fmt(value: number | null | undefined, digits = 1): string {
    71→  if (value === null || value === undefined || Number.isNaN(value)) return '—'
    72→  return value.toLocaleString(undefined, {
    73→    minimumFractionDigits: digits,
    74→    maximumFractionDigits: digits,
    75→  })
    76→}

   535→function AllocationBadge({ allocation }: { allocation: WaterPlanSeason['allocation'] }) {
   536→  if (allocation.status === 'unknown') {
   537→    return (
   538→      <Badge variant="outline" className="border-stone-300 bg-stone-100 text-stone-600">
   539→        allocation unknown
   540→      </Badge>
   541→    )
