     1→"""Integration: clean-database migration runs (Master protocol step 3).
     2→
     3→Runs alembic in a subprocess against a scratch database:
     4→drop -> create -> upgrade head -> downgrade base -> upgrade head again.
     5→"""
     6→
     7→import os
     8→import subprocess

    13→
    14→ROOT = os.path.dirname(os.path.dirname(os.path.dirname(os.path.abspath(__file__))))
    15→SCRATCH_DB = "fieldshift_mig_test"
    16→SCRATCH_URL = f"postgresql+psycopg2://postgres@127.0.0.1:5433/{SCRATCH_DB}"

    18→
    19→def _psql(dbname: str, sql: str) -> None:
    20→    conn = psycopg2.connect(host="127.0.0.1", port=5433, user="postgres", dbname=dbname)
    21→    conn.autocommit = True
    22→    conn.cursor().execute(sql)
    23→    conn.close()
    24→
    25→
    26→def _alembic(*args: str) -> subprocess.CompletedProcess:
    27→    env = dict(os.environ)
    28→    env["FIELDSHIFT_DATABASE_URL"] = SCRATCH_URL
    29→    return subprocess.run(
    30→        [os.path.join(ROOT, ".venv", "bin", "alembic"), *args],
    31→        cwd=ROOT, env=env, capture_output=True, text=True, timeout=180,
    32→    )
    33→
    34→
    35→@pytest.fixture()
    36→def scratch_db():
    37→    conn = psycopg2.connect(host="127.0.0.1", port=5433, user="postgres", dbname="postgres")
    38→    conn.autocommit = True
    39→    cur = conn.cursor()
    40→    cur.execute(f"DROP DATABASE IF EXISTS {SCRATCH_DB} WITH (FORCE)")
    41→    cur.execute(f"CREATE DATABASE {SCRATCH_DB} ENCODING 'UTF8'")
    42→    conn.close()
    43→    yield SCRATCH_DB
    44→    conn = psycopg2.connect(host="127.0.0.1", port=5433, user="postgres", dbname="postgres")
    45→    conn.autocommit = True
    46→    conn.cursor().execute(f"DROP DATABASE IF EXISTS {SCRATCH_DB} WITH (FORCE)")

   423→    def test_execution_triggers_freeze_result_rows(self, scratch_db):
   424→        from sqlalchemy.exc import DBAPIError
   425→
   426→        assert _alembic("upgrade", "head").returncode == 0
   427→        engine = create_engine(SCRATCH_URL)
   428→        run, scenario = self._seed_execution_chain(engine)
   429→
   430→        # content tamper attempts rejected
   431→        tamper_attempts = [
   432→            ("UPDATE scenario_execution_runs SET run_status = 'draft' WHERE id = :id", {"id": run}),
   433→            ("UPDATE scenario_execution_runs SET output_checksum = :n WHERE id = :id", {"id": run, "n": "0" * 64}),
   434→            ("UPDATE scenario_execution_runs SET warnings = '[\"x\"]' WHERE id = :id", {"id": run}),
   435→        ]
   436→        rejected = 0
   437→        for statement, params in tamper_attempts:
   438→            try:
   439→                with engine.begin() as conn:
   440→                    conn.execute(text(statement), params)
   441→            except DBAPIError:
   442→                rejected += 1
   443→        assert rejected == 3, f"trigger rejected {rejected}/3 tamper attempts"
   444→
   445→        # result-bearing runs cannot be deleted
   446→        try:
