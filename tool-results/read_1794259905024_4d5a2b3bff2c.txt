    81→export class ApiError extends Error {
    82→  readonly status: number
    83→  readonly code: string
    84→  readonly details: ApiErrorDetail[]
    85→  readonly requestId: string | undefined
    86→
    87→  constructor(params: {
    88→    status: number
    89→    code: string
    90→    message: string
    91→    details?: ApiErrorDetail[]
    92→    requestId?: string
    93→  }) {
    94→    super(params.message)
    95→    this.name = 'ApiError'
    96→    this.status = params.status
    97→    this.code = params.code
    98→    this.details = params.details ?? []
    99→    this.requestId = params.requestId
   100→  }
   101→}

   103→/** Human-readable summary of any thrown error for toasts. */
   104→export function describeError(err: unknown): { title: string; description?: string } {
   105→  if (err instanceof ApiError) {
   106→    const firstDetail = err.details[0]
   107→    const description = firstDetail
   108→      ? `${firstDetail.field}: ${firstDetail.reason}${err.requestId ? ` · request ${err.requestId.slice(0, 8)}` : ''}`
   109→      : err.requestId
   110→        ? `request ${err.requestId.slice(0, 8)}`
   111→        : undefined
   112→    return { title: err.message, description }
   113→  }
   114→  if (err instanceof Error) {
   115→    return { title: 'Network error', description: err.message }
   116→  }
   117→  return { title: 'Unexpected error' }
   118→}

   120→function withPort(path: string): string {
   121→  const joiner = path.includes('?') ? '&' : '?'
   122→  return `${path}${joiner}${PORT_QUERY}`

   125→/**
   126→ * Exchange the HttpOnly refresh cookie for a fresh access+refresh cookie pair
   127→ * (rotating refresh: the presented token is revoked server-side; reusing a
   128→ * superseded token revokes the whole session family).
   129→ *
   130→ * Single-flight: several concurrent 401s share ONE rotation call — parallel
   131→ * rotations would present already-rotated tokens and revoke the family.
   132→ */
   133→let refreshInFlight: Promise<boolean> | null = null
   134→
   135→export async function refreshSession(): Promise<boolean> {
   136→  if (!refreshInFlight) {
   137→    refreshInFlight = authRequest('/refresh')
   138→      .then(() => true)
   139→      .catch(() => false)
   140→      .finally(() => {
   141→        refreshInFlight = null
   142→      })
   143→  }
   144→  return refreshInFlight
   145→}

   147→/** POST to a Next.js /api/auth/* route handler (session cookies live here). */
   148→async function authRequest<T>(path: string, body?: unknown): Promise<T> {
   149→  const response = await fetch(`${AUTH_BASE}${path}`, {
   150→    method: 'POST',
   151→    headers: body !== undefined ? { 'Content-Type': 'application/json' } : undefined,
   152→    body: body !== undefined ? JSON.stringify(body) : undefined,
   153→    cache: 'no-store',
   154→  })
   155→  const text = await response.text()
   156→  let payload: unknown = null
   157→  if (text) {
   158→    try {
   159→      payload = JSON.parse(text)
   160→    } catch {
   161→      payload = null
   162→    }
   163→  }
   164→  if (!response.ok) {
   165→    const envelope = payload as ApiErrorBody | null
   166→    const error =
   167→      envelope && typeof envelope === 'object' && 'error' in envelope ? envelope.error : undefined
   168→    throw new ApiError({
   169→      status: response.status,
   170→      code: error?.code ?? `HTTP_${response.status}`,
   171→      message: error?.message ?? `Request failed with status ${response.status}.`,
   172→      details: error?.details ?? [],
   173→      requestId: error?.request_id,
   174→    })
   175→  }
   176→  return payload as T
   177→}
   178→
   179→interface RequestOptions {
   180→  method?: 'GET' | 'POST' | 'PATCH' | 'PUT' | 'DELETE'
   181→  body?: unknown
   182→  /** Multipart form body (file uploads). Content-Type is set by the browser. */
   183→  formData?: FormData
   184→  /** Skip the 401→refresh retry (health checks, anonymous endpoints). */
   185→  anonymous?: boolean
   186→}
   187→
   188→interface RequestInternals {
   189→  /** Prevent recursive 401→refresh retries (used by refreshSession itself). */
   190→  skipAuthRetry?: boolean
   191→}
   192→
   193→async function request<T>(
   194→  path: string,
   195→  options: RequestOptions = {},
   196→  internals: RequestInternals = {},
   197→): Promise<T> {
   198→  const headers: Record<string, string> = { Accept: 'application/json' }
   199→  if (options.body !== undefined) headers['Content-Type'] = 'application/json'
   200→
   201→  let response: Response
   202→  try {
   203→    response = await fetch(withPort(path), {
   204→      method: options.method ?? 'GET',
   205→      headers,
   206→      body:
   207→        options.formData !== undefined
   208→          ? options.formData
   209→          : options.body !== undefined
   210→            ? JSON.stringify(options.body)
   211→            : undefined,
   212→      cache: 'no-store',
   213→    })
   214→  } catch (err) {
   215→    throw new ApiError({
   216→      status: 0,
   217→      code: 'NETWORK_ERROR',
   218→      message: 'Could not reach the FieldShift API.',
   219→      details: [],
   220→    })
   221→  }
   222→
   223→  const text = await response.text()
   224→  let payload: unknown = null
   225→  if (text) {
   226→    try {
   227→      payload = JSON.parse(text)
   228→    } catch {
   229→      payload = null
   230→    }
   231→  }
   232→
   233→  if (!response.ok) {
   234→    const envelope = payload as ApiErrorBody | null
   235→    const error = envelope && typeof envelope === 'object' && 'error' in envelope ? envelope.error : undefined
   236→    // Rotating-refresh retry (P0-2/P1): an access token may have simply
   237→    // expired while the session is still valid. Try ONE cookie refresh, then
   238→    // retry the original request once. Never recurse from the refresh call
   239→    // itself (it goes through authRequest, not here).
   240→    if (response.status === 401 && !options.anonymous && !internals.skipAuthRetry) {
   241→      const refreshed = await refreshSession()
   242→      if (refreshed) {
   243→        return request<T>(path, options, { skipAuthRetry: true })
   244→      }
   245→    }
   246→    throw new ApiError({
   247→      status: response.status,
   248→      code: error?.code ?? `HTTP_${response.status}`,
   249→      message: error?.message ?? `Request failed with status ${response.status}.`,
   250→      details: error?.details ?? [],
   251→      requestId: error?.request_id,
   252→    })
   253→  }
   254→
   255→  return payload as T
   256→}

   405→const JOB_TERMINAL_STATUSES = new Set(['succeeded', 'failed', 'cancelled'])
   406→
   407→/**
   408→ * Poll an ingestion/import job until it reaches a terminal state.
   409→ *
   410→ * The backend executes data jobs on a background worker pool (202 + status
   411→ * 'queued'); when workers are disabled it still executes inline (the response
   412→ * is already terminal, returned as-is). Progress callbacks receive every
   413→ * intermediate snapshot (status/progress for UI feedback).
   414→ */
   415→export async function pollJobUntilDone(
   416→  jobId: string,
   417→  opts: { onProgress?: (job: DataJob) => void; intervalMs?: number; timeoutMs?: number } = {},
   418→): Promise<DataJob> {
   419→  const interval = opts.intervalMs ?? 1000
   420→  const deadline = Date.now() + (opts.timeoutMs ?? 180_000)
   421→  let current = await getJob(jobId)
   422→  while (!JOB_TERMINAL_STATUSES.has(current.status)) {
   423→    opts.onProgress?.(current)
   424→    if (Date.now() > deadline) {
   425→      throw new Error('Timed out waiting for the job to finish — check its status later.')
   426→    }
   427→    await new Promise((resolve) => setTimeout(resolve, interval))
   428→    current = await getJob(jobId)
   429→  }
   430→  return current
   431→}

   462→export type GeometryParseResult =
   463→  | { ok: true; geometry: import('./types').GeoJSONGeometry }
   464→  | { ok: false; error: string }
   465→
   466→/** Parse + shallow-validate a GeoJSON Polygon / MultiPolygon from user input. */
   467→export function parseGeometryInput(text: string): GeometryParseResult {
   468→  let parsed: unknown
   469→  try {
   470→    parsed = JSON.parse(text)
   471→  } catch {
   472→    return { ok: false, error: 'The text is not valid JSON. Check for missing quotes, brackets or trailing commas.' }
   473→  }
   474→  if (typeof parsed !== 'object' || parsed === null || Array.isArray(parsed)) {
   475→    return { ok: false, error: 'Expected a GeoJSON object like {"type":"Polygon","coordinates":[...]}' }
   476→  }
   477→  const obj = parsed as Record<string, unknown>
   478→  if (obj.type === 'Polygon') {
   479→    if (!Array.isArray(obj.coordinates) || obj.coordinates.length === 0) {
   480→      return { ok: false, error: 'Polygon coordinates must be a non-empty array of rings.' }
   481→    }
   482→    return { ok: true, geometry: parsed as import('./types').GeoJSONPolygon }
   483→  }
   484→  if (obj.type === 'MultiPolygon') {
   485→    if (!Array.isArray(obj.coordinates) || obj.coordinates.length === 0) {
   486→      return { ok: false, error: 'MultiPolygon coordinates must be a non-empty array of polygons.' }
   487→    }
   488→    return { ok: true, geometry: parsed as import('./types').GeoJSONMultiPolygon }
   489→  }
   490→  return { ok: false, error: `Unsupported geometry type "${String(obj.type)}". Use Polygon or MultiPolygon.` }
   491→}

   510→/** Client-side guard mirroring backend job validation. Returns an error message or null. */
   511→export function validateJobWindow(
   512→  start: string,
   513→  end: string,
   514→  bounds?: { min?: string; max?: string; maxYears?: number },
   515→): string | null {
   516→  const dateRe = /^\d{4}-\d{2}-\d{2}$/
   517→  if (!dateRe.test(start) || !dateRe.test(end)) return 'Enter both dates as YYYY-MM-DD.'
   518→  const min = bounds?.min ?? COVERAGE_START
   519→  const max = bounds?.max ?? COVERAGE_END
   520→  if (start < min || end > max) {
   521→    return `Dates must fall inside the dataset coverage ${min} → ${max}.`
   522→  }
   523→  if (end < start) return 'The end date must be on or after the start date.'
   524→  const years = (Date.parse(end) - Date.parse(start)) / (365.25 * 24 * 3600 * 1000)
   525→  const maxYears = bounds?.maxYears ?? MAX_JOB_YEARS
   526→  if (years > maxYears) return `A single job covers at most ${maxYears} years. Split the range.`
   527→  return null
   528→}

   963→// --- laboratory CSV import ------------------------------------------------------
   964→// Backend contract: POST /soil-samples/import (org-level) requires a `mapping`
   965→// form field whose JSON carries default_field_id — the field binding arrives
   966→// there, so the component's (fieldId, file) signature is preserved.
   967→
   968→export async function importSoilLabCsv(fieldId: string, file: File): Promise<SoilLabImportResponse> {
   969→  const formData = new FormData()
   970→  formData.append('file', file)
   971→  formData.append('mapping', JSON.stringify({ default_field_id: fieldId }))
   972→  const data = await request<SoilLabImportResponse>('/api/v1/soil-samples/import', {
   973→    method: 'POST',
   974→    formData,
   975→  })
   976→  // The backend's idempotent "already_imported" reply omits the counters;
   977→  // normalize so callers always see numbers (re-import of identical content
   978→  // creates nothing — 0 is the truthful value there).
   979→  return {
   980→    ...data,
   981→    created_samples: data.created_samples ?? 0,
   982→    created_measurements: data.created_measurements ?? 0,
   983→    errors: data.errors ?? [],
   984→  }
   985→}
