    51→    51→def _season_climate_evidence(
    52→    52→    run: dbm.FeatureRun,
    53→    53→    season_label: str,
    54→    54→    start: date,
    55→    55→    end: date,
    56→    56→) -> tuple[WaterEvidence | None, list[str]]:
    57→    57→    """Month-of-year climatology of ET0 and precipitation for one season window
    58→    58→    from a single feature run (the same historical-proxy basis as rotation)."""
    59→    59→    import calendar as cal
    60→    60→
    61→    61→    monthly: dict[str, dict[int, list[tuple[int, float]]]] = {}
    62→    62→    for row in run.values:
    63→    63→        if row.aggregation_level != "monthly" or row.value is None:
    64→    64→            continue
    65→    65→        monthly.setdefault(row.feature_code, {}).setdefault(
    66→    66→            row.period_start.month, []
    67→    67→        ).append((row.period_start.year, row.value))
    68→    68→
    69→    69→    def collect_month_means(code: str) -> list[tuple[int, int, float]]:
    70→    70→        table = monthly.get(code, {})
    71→    71→        out: list[tuple[int, int, float]] = []
    72→    72→        y, m = start.year, start.month
    73→    73→        while (y, m) <= (end.year, end.month):
    74→    74→            entries = table.get(m)
    75→    75→            if entries:
    76→    76→                out.append((y, m, sum(v for _yr, v in entries) / len(entries)))
    77→    77→            m += 1
    78→    78→            if m == 13:
    79→    79→                y, m = y + 1, 1
    80→    80→        return out
    81→    81→
    82→    82→    et0 = collect_month_means("reference_et0_total_monthly")
    83→    83→    precip = collect_month_means("precipitation_total_monthly")
    84→    84→
    85→    85→    warnings: list[str] = []
    86→    86→    if not et0:
    87→    87→        warnings.append(
    88→    88→            f"Season '{season_label}': no monthly ET0 feature values in the evidence "
    89→    89→            "run — crop ETc will not be evaluable (missing evidence is never zero)."
    90→    90→        )
    91→    91→    if not precip:
    92→    92→        warnings.append(
    93→    93→            f"Season '{season_label}': no monthly precipitation feature values in the "
    94→    94→            "evidence run — the rainfall credit will be unknown (conservative "
    95→    95→            "requirement, disclosed)."
    96→    96→        )
    97→    97→
    98→    98→    years_used = sorted({
    99→    99→        y
   100→   100→        for table in monthly.values()
   101→   101→        for entries in table.values()
   102→   102→        for y, _v in entries
   103→   103→    })
   104→   104→    evidence = WaterEvidence(
   105→   105→        feature_run_id=str(run.id),
   106→   106→        is_synthetic=bool(run.is_synthetic),
   107→   107→        monthly_et0=tuple(et0),
   108→   108→        monthly_precip=tuple(precip),
   109→   109→        coverage_ratio=1.0 if et0 else 0.0,
   110→   110→        years_used=tuple(years_used),
   111→   111→        basis=(
   112→   112→            f"month-of-year climatology from feature run {str(run.id)[:8]} "
   113→   113→            f"({f'years {years_used[0]}-{years_used[-1]}' if years_used else 'no years'}) "
   114→   114→            "— a historical proxy, NOT a forecast for the planning season."
   115→   115→        ),
   116→   116→    )
   117→   117→    return evidence, warnings

   120→   120→def _efficiency_from_payload(
   121→   121→    payload_eff: dict[str, Any] | None,
   122→   122→) -> EfficiencyInput | None:
   123→   123→    """Efficiency from an explicit farmer value or a method class (the FAO
   124→   124→    TM1 Annex I Table 8 indicative values). A method class value is a
   125→   125→    REFERENCE choice, recorded with its citation."""
   126→   126→    if payload_eff is None:
   127→   127→        return None
   128→   128→    if not isinstance(payload_eff, dict):
   129→   129→        raise ValidationFailed(
   130→   130→            "irrigation_efficiency must be an object.",
   131→   131→            details=[{"field": "irrigation_efficiency"}],
   132→   132→        )
   133→   133→    value = payload_eff.get("value")
   134→   134→    method_class = payload_eff.get("method_class")
   135→   135→    if value is not None:
   136→   136→        try:
   137→   137→            value = float(value)
   138→   138→        except (TypeError, ValueError):
   139→   139→            raise ValidationFailed(
   140→   140→                "irrigation_efficiency.value must be a number.",
   141→   141→                details=[{"field": "irrigation_efficiency"}],
   142→   142→            ) from None
   143→   143→        if not 0 < value <= 1:
   144→   144→            raise ValidationFailed(
   145→   145→                "irrigation_efficiency.value must be within (0, 1].",
   146→   146→                details=[{"field": "irrigation_efficiency", "reason": str(value)}],
   147→   147→            )
   148→   148→        return EfficiencyInput(
   149→   149→            value=value,
   150→   150→            evidence_class="farmer_reported_record",
   151→   151→            citation_text="Farmer-supplied field application efficiency.",
   152→   152→            method_class=method_class,
   153→   153→        )
   154→   154→    if method_class is not None:
   155→   155→        for entry in IRRIGATION_EFFICIENCY_CLASSES:
   156→   156→            if entry["method_class"] == method_class:
   157→   157→                return EfficiencyInput(
   158→   158→                    value=float(entry["efficiency"]),
   159→   159→                    evidence_class=entry["evidence_class"],
   160→   160→                    citation_text=entry["citation_text"],
   161→   161→                    method_class=method_class,
   162→   162→                )
   163→   163→        raise ValidationFailed(
   164→   164→            f"Unknown irrigation method class '{method_class}'.",
   165→   165→            details=[{
   166→   166→                "field": "irrigation_efficiency",
   167→   167→                "reason": ", ".join(e["method_class"] for e in IRRIGATION_EFFICIENCY_CLASSES),
   168→   168→            }],
   169→   169→        )
   170→   170→    raise ValidationFailed(
   171→   171→        "irrigation_efficiency needs 'value' (farmer-measured) or 'method_class'.",
   172→   172→        details=[{"field": "irrigation_efficiency"}],
   173→   173→    )

   176→   176→def build_soil_water_inputs(
   177→   177→    db: Session,
   178→   178→    field: dbm.Field,
   179→   179→    payload: dict[str, Any],
   180→   180→) -> tuple[SoilWaterInputs | None, list[str], dict[str, Any]]:
   181→   181→    """Assemble soil water-holding inputs from the requested source.
   182→   182→
   183→   183→    - farmer_measured: explicit θFC/θWP (+ optional Zr/p) from the request
   184→   184→      (evidence class farmer_reported_record);
   185→   185→    - ptf_texture: Saxton & Rawls (2006) over the field's latest soil
   186→   186→      observations (sand/clay/SOC), with the single-layer disclosure;
   187→   187→    - none: no soil inputs — the FAO-33 no-storage fallback applies."""
   188→   188→    source = payload.get("soil_source") or "none"
   189→   189→    warnings: list[str] = []
   190→   190→    meta: dict[str, Any] = {"soil_source": source}
   191→   191→    if source not in SOIL_SOURCES:
   192→   192→        raise ValidationFailed(
   193→   193→            f"Unknown soil_source '{source}'.",
   194→   194→            details=[{"field": "soil_source", "reason": ", ".join(SOIL_SOURCES)}],
   195→   195→        )
   196→   196→    if source == "none":
   197→   197→        return None, warnings, meta
   198→   198→
   199→   199→    if source == "farmer_measured":
   200→   200→        fs = payload.get("farmer_soil") or {}
   201→   201→        theta_fc = fs.get("theta_fc")
   202→   202→        theta_wp = fs.get("theta_wp")
   203→   203→        if theta_fc is None or theta_wp is None:
   204→   204→            raise ValidationFailed(
   205→   205→                "soil_source 'farmer_measured' requires farmer_soil.theta_fc and "
   206→   206→                "farmer_soil.theta_wp (m3/m3).",
   207→   207→                details=[{"field": "farmer_soil"}],
   208→   208→            )
   209→   209→        try:
   210→   210→            theta_fc, theta_wp = float(theta_fc), float(theta_wp)
   211→   211→        except (TypeError, ValueError):
   212→   212→            raise ValidationFailed(
   213→   213→                "farmer_soil.theta_fc/theta_wp must be numbers.",
   214→   214→                details=[{"field": "farmer_soil"}],
   215→   215→            ) from None
   216→   216→        if not (0 < theta_wp < theta_fc <= 0.6):
   217→   217→            raise ValidationFailed(
   218→   218→                "farmer_soil contents must satisfy 0 < θWP < θFC ≤ 0.6 (m3/m3).",
   219→   219→                details=[{"field": "farmer_soil", "reason": f"{theta_wp} !< {theta_fc}"}],
   220→   220→            )
   221→   221→        zr = fs.get("zr_m")
   222→   222→        p_table = fs.get("p_table")
   223→   223→        inputs = SoilWaterInputs(
   224→   224→            theta_fc=theta_fc,
   225→   225→            theta_fc_evidence_class="farmer_reported_record",
   226→   226→            theta_wp=theta_wp,
   227→   227→            theta_wp_evidence_class="farmer_reported_record",
   228→   228→            zr_m=float(zr) if zr is not None else None,
   229→   229→            zr_evidence_class="farmer_reported_record" if zr is not None else None,
   230→   230→            p_table=float(p_table) if p_table is not None else None,
   231→   231→            p_evidence_class="farmer_reported_record" if p_table is not None else None,
   232→   232→            source_label="farmer_measured",
   233→   233→            source_notes=(
   234→   234→                "Soil water-holding characteristics supplied by the farmer "
   235→   235→                "(farmer_reported_record) — not independently verified.",
   236→   236→            ),
   237→   237→        )
   238→   238→        meta["farmer_soil"] = fs
   239→   239→        return inputs, warnings, meta
   240→   240→
   241→   241→    # ptf_texture
   242→   242→    rows = db.execute(
   243→   243→        select(dbm.Observation)
   244→   244→        .where(
   245→   245→            dbm.Observation.field_id == field.id,
   246→   246→            dbm.Observation.variable_code.in_(SOIL_VARIABLES),
   247→   247→            dbm.Observation.value.is_not(None),
   248→   248→        )
   249→   249→        .order_by(dbm.Observation.observed_at.desc())
   250→   250→        .limit(50)
   251→   251→    ).scalars().all()
   252→   252→    latest: dict[str, dbm.Observation] = {}
   253→   253→    for obs in rows:
   254→   254→        latest.setdefault(obs.variable_code, obs)
   255→   255→    sand = latest.get("soil_sand_content")
   256→   256→    clay = latest.get("soil_clay_content")
   257→   257→    soc = latest.get("soil_organic_carbon")
   258→   258→    if sand is None or clay is None:
   259→   259→        warnings.append(
   260→   260→            "soil_source 'ptf_texture' requested but the field has no sand/clay "
   261→   261→            "observations — the PTF estimate is refused (never extrapolated) and "
   262→   262→            "the plan falls back to the no-storage FAO-33 method."
   263→   263→        )
   264→   264→        return None, warnings, {**meta, "ptf": "refused_missing_texture"}
   265→   265→    sand_frac = float(sand.value) / 1000.0  # g/kg → fraction
   266→   266→    clay_frac = float(clay.value) / 1000.0
   267→   267→    om_percent = (
   268→   268→        soil_mod.organic_carbon_to_om_percent(float(soc.value)) if soc is not None else None
   269→   269→    )
   270→   270→    notes = [
   271→   271→        "θFC/θWP ESTIMATED by the Saxton & Rawls (2006) pedotransfer function over "
   272→   272→        "texture (evidence class global_soil_prediction; PTF version "
   273→   273→        f"{PTF_VERSION}); a texture-based model estimate, not a measurement.",
   274→   274→        "The PTF inputs are the field's latest soil observations. If they come from "
   275→   275→        "SoilGrids, they describe the 0-5 cm layer ASSUMED representative of the "
   276→   276→        "root zone — a single-layer extrapolation explicitly disclosed (backlog S-6).",
   277→   277→    ]
   278→   278→    if om_percent is None:
   279→   279→        notes.append(
   280→   280→            "No organic-carbon observation available — the PTF OM term is set to 0 "
   281→   281→            "(a disclosed simplification, typically a small effect)."
   282→   282→        )
   283→   283→        om_used = 0.0
   284→   284→    else:
   285→   285→        om_used = om_percent
   286→   286→        notes.append(
   287→   287→            "Organic matter derived from soil organic carbon via the van Bemmelen "
   288→   288→            "factor (OM ≈ SOC × 1.724)."
   289→   289→        )
   290→   290→    try:
   291→   291→        est = soil_mod.saxton_rawls_2006(sand_frac, clay_frac, om_used)
   292→   292→    except soil_mod.PTFDomainError as exc:
   293→   293→        warnings.append(
   294→   294→            f"The field's texture is outside the Saxton-Rawls validity domain "
   295→   295→            f"({exc}) — the PTF estimate is refused and the plan falls back to the "
   296→   296→            "no-storage FAO-33 method."
   297→   297→        )
   298→   298→        return None, warnings, {**meta, "ptf": "refused_domain"}
   299→   299→    meta["ptf"] = {
   300→   300→        "ptf_version": PTF_VERSION,
   301→   301→        "sand_fraction": sand_frac,
   302→   302→        "clay_fraction": clay_frac,
   303→   303→        "om_percent": round(om_used, 4),
   304→   304→        "theta_fc": est["theta_fc_33kPa"],
   305→   305→        "theta_wp": est["theta_wp_1500kPa"],
   306→   306→        "input_evidence_classes": {
   307→   307→            "sand": sand.evidence_class,
   308→   308→            "clay": clay.evidence_class,
   309→   309→            "organic_carbon": soc.evidence_class if soc else None,
   310→   310→        },
   311→   311→    }
   312→   312→    inputs = SoilWaterInputs(
   313→   313→        theta_fc=est["theta_fc_33kPa"],
   314→   314→        theta_fc_evidence_class="global_soil_prediction",
   315→   315→        theta_wp=est["theta_wp_1500kPa"],
   316→   316→        theta_wp_evidence_class="global_soil_prediction",
   317→   317→        zr_m=None,   # Zr comes from the crop knowledge (pack) in the planner caller
   318→   318→        zr_evidence_class=None,
   319→   319→        p_table=None,  # p comes from the crop knowledge (pack)
   320→   320→        p_evidence_class=None,
   321→   321→        source_label="ptf_texture",
   322→   322→        source_notes=tuple(notes),
   323→   323→    )
   324→   324→    return inputs, warnings, meta

   327→   327→def start_water_plan(
   328→   328→    db: Session,
   329→   329→    *,
   330→   330→    field: dbm.Field,
   331→   331→    payload: dict[str, Any],
   332→   332→    created_by: Any = None,
   333→   333→) -> dbm.WaterPlan:
   334→   334→    """Validate, execute inline and persist a water plan (202-style job)."""
   335→   335→    idempotency_key = payload.get("idempotency_key")
   336→   336→    if idempotency_key:
   337→   337→        existing = db.scalar(
   338→   338→            select(dbm.WaterPlan).where(
   339→   339→                dbm.WaterPlan.idempotency_key == idempotency_key,
   340→   340→                dbm.WaterPlan.field_id == field.id,
   341→   341→            )
   342→   342→        )
   343→   343→        if existing is not None:
   344→   344→            return existing
   345→   345→
   346→   346→    raw_seasons = payload.get("seasons") or []
   347→   347→    if not raw_seasons:
   348→   348→        raise ValidationFailed(
   349→   349→            "At least one season is required.",
   350→   350→            details=[{"field": "seasons"}],
   351→   351→        )
   352→   352→    season_specs = _validate_seasons(raw_seasons)
   353→   353→    if len(season_specs) > MAX_SEASONS:
   354→   354→        raise ValidationFailed(
   355→   355→            f"At most {MAX_SEASONS} seasons per water plan.",
   356→   356→            details=[{"field": "seasons"}],
   357→   357→        )
   358→   358→
   359→   359→    # crop assignments: per-season crop_code required (knowledge-base crops)
   360→   360→    assignments_in = payload.get("crop_assignments") or []
   361→   361→    if not isinstance(assignments_in, list) or len(assignments_in) != len(season_specs):
   362→   362→        raise ValidationFailed(
   363→   363→            "crop_assignments must list one crop_code per season.",
   364→   364→            details=[{"field": "crop_assignments"}],
   365→   365→        )
   366→   366→    sowings_in = payload.get("sowing_dates") or {}
   367→   367→
   368→   368→    pe_method = payload.get("effective_rainfall_method") or "fao_aglw_1986"
   369→   369→    if pe_method not in EFFECTIVE_RAINFALL_METHODS:
   370→   370→        raise ValidationFailed(
   371→   371→            f"Unknown effective_rainfall_method '{pe_method}'.",
   372→   372→            details=[{"field": "effective_rainfall_method"}],
   373→   373→        )
   374→   374→    pe_fraction = payload.get("pe_fraction")
   375→   375→    if pe_method == "constant_fraction":
   376→   376→        if pe_fraction is None:
   377→   377→            raise ValidationFailed(
   378→   378→                "effective_rainfall_method 'constant_fraction' requires pe_fraction.",
   379→   379→                details=[{"field": "pe_fraction"}],
   380→   380→            )
   381→   381→        pe_fraction = float(pe_fraction)
   382→   382→        if not 0 <= pe_fraction <= 1:
   383→   383→            raise ValidationFailed(
   384→   384→                "pe_fraction must be within [0, 1].",
   385→   385→                details=[{"field": "pe_fraction"}],
   386→   386→            )
   387→   387→    elif pe_fraction is not None:
   388→   388→        raise ValidationFailed(
   389→   389→            "pe_fraction is only valid with the constant_fraction method.",
   390→   390→            details=[{"field": "pe_fraction"}],
   391→   391→        )
   392→   392→
   393→   393→    efficiency = _efficiency_from_payload(payload.get("irrigation_efficiency"))
   394→   394→
   395→   395→    allocation = payload.get("water_allocation_m3")
   396→   396→    if allocation is not None:
   397→   397→        try:
   398→   398→            allocation = float(allocation)
   399→   399→        except (TypeError, ValueError):
   400→   400→            raise ValidationFailed(
   401→   401→                "water_allocation_m3 must be a number.",
   402→   402→                details=[{"field": "water_allocation_m3"}],
   403→   403→            ) from None
   404→   404→        if allocation <= 0:
   405→   405→            raise ValidationFailed(
   406→   406→                "water_allocation_m3 must be positive.",
   407→   407→                details=[{"field": "water_allocation_m3"}],
   408→   408→            )
   409→   409→
   410→   410→    knowledge = load_knowledge(db)
   411→   411→    if not knowledge.crops:
   412→   412→        raise ValidationFailed(
   413→   413→            "No active knowledge pack with crops — import a knowledge pack first.",
   414→   414→            details=[{"field": "knowledge"}],
   415→   415→        )
   416→   416→    wk_map = water_knowledge_map(knowledge)
   417→   417→
   418→   418→    soil_inputs, soil_warnings, soil_meta = build_soil_water_inputs(db, field, payload)
   419→   419→
   420→   420→    area_ha = float(field.area_ha)
   421→   421→    label = str(payload.get("label") or "Seasonal water plan")[:120]
   422→   422→
   423→   423→    run = dbm.WaterPlan(
   424→   424→        field_id=field.id,
   425→   425→        status="running",
   426→   426→        label=label,
   427→   427→        seasons=[
   428→   428→            {"label": s.label, "start_date": s.start_date.isoformat(), "end_date": s.end_date.isoformat()}
   429→   429→            for s in season_specs
   430→   430→        ],
   431→   431→        crop_assignments=list(assignments_in),
   432→   432→        methods={
   433→   433→            "water_engine_version": WATER_ENGINE_VERSION,
   434→   434→            "methods_version": WATER_METHODS_VERSION,
   435→   435→            "effective_rainfall_method": pe_method,
   436→   436→            "pe_fraction": pe_fraction,
   437→   437→            "irrigation_efficiency": (
   438→   438→                {
   439→   439→                    "value": efficiency.value,
   440→   440→                    "evidence_class": efficiency.evidence_class,
   441→   441→                    "method_class": efficiency.method_class,
   442→   442→                }
   443→   443→                if efficiency
   444→   444→                else None
   445→   445→            ),
   446→   446→            "soil_source": soil_meta.get("soil_source", "none"),
   447→   447→            "ptf": soil_meta.get("ptf"),
   448→   448→        },
   449→   449→        soil_inputs={

   608→   608→def _plan_snapshot_checksum(plan_result: dict[str, Any], run: dbm.WaterPlan) -> str:
   609→   609→    import hashlib
   610→   610→    import json
   611→   611→
   612→   612→    payload = {
   613→   613→        "engine": run.water_engine_version,
   614→   614→        "methods": run.methods_version,
   615→   615→        "seasons": run.seasons,
   616→   616→        "assignments": run.crop_assignments,
   617→   617→        "methods_cfg": run.methods,
   618→   618→        "soil": run.soil_inputs,
   619→   619→        "allocation": run.water_allocation_m3,
   620→   620→        "totals": [
   621→   621→            s["totals"] for s in plan_result.get("seasons", [])
   622→   622→        ],
   623→   623→        "packs": run.knowledge_pack_refs,
   624→   624→    }
   625→   625→    canonical = json.dumps(payload, sort_keys=True, separators=(",", ":"), default=str)
   626→   626→    return hashlib.sha256(canonical.encode("utf-8")).hexdigest()
