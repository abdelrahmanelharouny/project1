   410→class FeatureDefinition(Base, TimestampMixin):
   411→    """Versioned feature definition (Milestone 4).
   412→
   413→    Feature codes, units, coverage policy and formula references are data, not
   414→    code smell: every computed value links to the definition version it was
   415→    produced with, so historical feature values remain interpretable when the
   416→    catalog evolves. Thresholds used by count-style indicators cite their
   417→    source convention (WMO / ETCCDI) and are explicitly NOT crop-specific.
   418→    """
   419→
   420→    __tablename__ = "feature_definitions"
   421→
   422→    id: Mapped[uuid.UUID] = mapped_column(UUID(as_uuid=True), primary_key=True, default=_uuid)
