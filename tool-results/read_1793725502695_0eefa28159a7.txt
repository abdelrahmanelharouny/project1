    70→    70→function fmt(value: number | null | undefined, digits = 1): string {
    71→    71→  if (value === null || value === undefined || Number.isNaN(value)) return '—'
    72→    72→  return value.toLocaleString(undefined, {
    73→    73→    minimumFractionDigits: digits,
    74→    74→    maximumFractionDigits: digits,
    75→    75→  })
    76→    76→}
