    69→function money(cents: number | null, currency: string, locale: string): string {
    70→  if (cents === null) return '—'
    71→  try {
    72→    return new Intl.NumberFormat(locale, { style: 'currency', currency }).format(cents / 100)
    73→  } catch {
    74→    return `${(cents / 100).toFixed(2)} ${currency}`
    75→  }
    76→}

    78→function statusBadgeClass(status: string): string {
    79→  switch (status) {
    80→    case 'active':
    81→      return 'border-emerald-300 bg-emerald-50 text-emerald-800'
    82→    case 'trialing':
    83→      return 'border-stone-300 bg-stone-100 text-stone-700'
    84→    case 'past_due':
    85→      return 'border-amber-300 bg-amber-50 text-amber-900'
    86→    case 'grace':
    87→      return 'border-amber-200 bg-amber-50/60 text-amber-800'
    88→    case 'canceled':
    89→      return 'border-red-200 bg-red-50 text-red-800'
    90→    default:
    91→      return 'border-stone-200 bg-stone-50 text-stone-500'
    92→  }
    93→}

    95→function statusLabel(t: T, status: string): string {
    96→  switch (status) {
    97→    case 'active':
    98→      return t('billing.statusActive')
    99→    case 'trialing':
   100→      return t('billing.statusTrialing')
   101→    case 'past_due':
   102→      return t('billing.statusPastDue')
   103→    case 'grace':
   104→      return t('billing.statusGrace')
   105→    case 'canceled':
   106→      return t('billing.statusCanceled')
   107→    default:
   108→      return t('billing.statusUnknown')
   109→  }
   110→}

   141→function toastMutationError(t: T, err: unknown): void {
   142→  if (err instanceof ApiError && err.status === 403) {
   143→    toast.error(t('errors.forbiddenTitle'), { description: t('errors.forbiddenBody') })
   144→    return
   145→  }
   146→  const d = describeError(err)
   147→  toast.error(d.title, { description: d.description })
   148→}
   149→
   150→function InlineError({ err, t }: { err: unknown; t: T }) {
   151→  const forbidden = err instanceof ApiError && err.status === 403
   152→  return (
   153→    <div className="rounded-md border border-amber-200 bg-amber-50 p-3 text-sm text-amber-900">
   154→      <p className="font-semibold">{forbidden ? t('errors.forbiddenTitle') : t('common.loadError')}</p>
   155→      <p className="mt-1 text-xs leading-relaxed">
   156→        {forbidden ? t('errors.forbiddenBody') : describeError(err).title}
   157→      </p>
   158→    </div>
   159→  )
   160→}

   279→      invalidateBilling()

   282→      setCancelOpen(false)

   284→    },

   317→    return (

   319→        <Skeleton className="h-8 w-48" />

   323→    )

   325→  if (!org) return <NoOrgEmptyState t={t} />
   326→

   328→    subscription !== null && (subscription.provider === 'sandbox' || subscription.provider_mode === 'test')
   329→
   330→  return (
   331→    <div className="space-y-4">
   332→      <header>
   333→        <h2 className="text-lg font-semibold text-stone-900">{t('billing.title')}</h2>
   334→        <p className="mt-0.5 text-sm text-stone-500">{t('billing.intro')}</p>
   335→      </header>
   336→
   337→      {subQuery.isError ? (
   338→        <InlineError err={subQuery.error} t={t} />
   339→      ) : subQuery.isLoading ? (
   340→        <Skeleton className="h-40 w-full" />
   341→      ) : (
   342→        <SubscriptionCard
   343→          t={t}
   344→          subscription={subscription}
   345→          onCancel={() => setCancelOpen(true)}
   346→          onResume={() => setResumeOpen(true)}
   347→        />
   348→      )}
   349→
   350→      {/* Plans */}
   351→      <Card>
   352→        <CardHeader className="pb-2">
   353→          <div className="flex flex-wrap items-center justify-between gap-3">
   354→            <div>
   355→              <CardTitle className="text-base">{t('billing.plansTitle')}</CardTitle>
   356→              <CardDescription className="text-sm text-stone-500">{t('billing.plansDesc')}</CardDescription>
   357→            </div>
   358→            <div className="flex items-center gap-2">
   359→              <Switch
   360→                id="p3-billing-period"
   361→                checked={period === 'yearly'}
   362→                onCheckedChange={(checked) => setPeriod(checked ? 'yearly' : 'monthly')}
   363→                aria-label={period === 'yearly' ? t('billing.yearly') : t('billing.monthly')}

   366→                {period === 'yearly' ? t('billing.yearly') : t('billing.monthly')}
   367→              </Label>
   368→            </div>
   369→          </div>
   370→        </CardHeader>
   371→        <CardContent>

   373→            <div className="grid gap-3 md:grid-cols-2 xl:grid-cols-3">
   374→              <Skeleton className="h-56 w-full" />

   379→            <InlineError err={plansQuery.error} t={t} />

   385→                  t={t}
   386→                  plan={plan}
   387→                  period={period}
   388→                  locale={locale}
   389→                  isCurrent={liveSubscription && subscription?.plan?.code === plan.code}
   390→                  hasLiveSubscription={liveSubscription}
   391→                  busy={startM.isPending || changeM.isPending}

   483→                  <SelectContent>
   484→                    <SelectItem value="succeeded">{t('billing.outcomeSucceeded')}</SelectItem>
   485→                    <SelectItem value="declined">{t('billing.outcomeDeclined')}</SelectItem>
   486→                  </SelectContent>
   487→                </Select>
   488→              </div>
   489→              <Button
   490→                variant="outline"
   491→                disabled={sandboxM.isPending}
   492→                onClick={() => sandboxM.mutate({ organization_id: org.id, outcome: sandboxOutcome })}

   494→                {sandboxM.isPending ? t('common.loading') : t('billing.sandboxPay')}
   495→              </Button>
   496→            </div>
   497→          </CardContent>
   498→        </Card>
   499→      )}
   500→
   501→      {/* Confirmations */}
   502→      <AlertDialog open={cancelOpen} onOpenChange={setCancelOpen}>
   503→        <AlertDialogContent>
   504→          <AlertDialogHeader>
   505→            <AlertDialogTitle>{t('billing.cancelDialogTitle')}</AlertDialogTitle>
   506→            <AlertDialogDescription>
   507→              {t('billing.cancelDialogBody', { date: formatDay(subscription?.current_period_end ?? null) })}

   509→          </AlertDialogHeader>

   519→              {t('billing.cancelConfirm')}

   524→
   525→      <AlertDialog open={resumeOpen} onOpenChange={setResumeOpen}>

   527→          <AlertDialogHeader>
   528→            <AlertDialogTitle>{t('billing.resumeDialogTitle')}</AlertDialogTitle>
   529→            <AlertDialogDescription>{t('billing.resumeDialogBody')}</AlertDialogDescription>
   530→          </AlertDialogHeader>
   531→          <AlertDialogFooter>
   532→            <AlertDialogCancel>{t('common.cancel')}</AlertDialogCancel>
   533→            <AlertDialogAction
   534→              onClick={(e) => {
   535→                e.preventDefault()
   536→                if (orgId) resumeM.mutate(orgId)

   573→                    organization_id: orgId,

   584→          </AlertDialogFooter>

   673→function PlanCard({
   674→  t,
   675→  plan,
   676→  period,
   677→  locale,
   678→  isCurrent,
   679→  hasLiveSubscription,
   680→  busy,
   681→  onSelect,
   682→}: {
   683→  t: T
   684→  plan: PlanCatalogItem
   685→  period: 'monthly' | 'yearly'
   686→  locale: string
   687→  isCurrent: boolean | null
   688→  hasLiveSubscription: boolean
   689→  busy: boolean
   690→  onSelect: () => void
   691→}) {
   692→  const price = plan.prices.find((p) => p.billing_period === period) ?? null
   693→  const monthly = plan.prices.find((p) => p.billing_period === 'monthly')?.amount_cents ?? null
   694→  const yearly = plan.prices.find((p) => p.billing_period === 'yearly')?.amount_cents ?? null
   695→  let savingsPct: number | null = null
   696→  if (monthly !== null && yearly !== null && monthly > 0 && yearly >= 0) {
   697→    const pctValue = Math.round((1 - yearly / 12 / monthly) * 100)
   698→    if (pctValue > 0) savingsPct = pctValue
   699→  }
   700→  const selfServe = price !== null && price.amount_cents !== null
   701→
   702→  return (
   703→    <div
   704→      className={`flex flex-col rounded-lg border p-4 ${
   705→        isCurrent ? 'border-emerald-300 bg-emerald-50/40' : 'border-stone-200 bg-white'
   706→      }`}
   707→    >
   708→      <div className="flex items-center justify-between gap-2">
   709→        <p className="font-semibold text-stone-900">{plan.name}</p>
   710→        {isCurrent ? (
   711→          <Badge variant="outline" className="border-emerald-300 bg-white text-emerald-800">
   712→            {t('billing.currentPlanBadge')}
   713→          </Badge>
   714→        ) : null}
   715→      </div>
   716→      {plan.description ? <p className="mt-1 text-xs leading-relaxed text-stone-500">{plan.description}</p> : null}
   717→
   718→      <div className="mt-3 flex items-baseline gap-2">
   719→        {selfServe ? (
   720→          <>
   721→            <span className="text-xl font-semibold text-stone-900">
   722→              {money(price?.amount_cents ?? null, price?.currency_code ?? 'USD', locale)}
   723→            </span>
   724→            <span className="text-xs text-stone-500">
   725→              / {period === 'yearly' ? t('billing.yearly').toLowerCase() : t('billing.monthly').toLowerCase()}
   726→            </span>
   727→          </>
   728→        ) : (
   729→          <span className="text-sm font-medium text-stone-600">{t('billing.priceUnavailable')}</span>
   730→        )}
   731→      </div>
   732→      <div className="mt-0.5 flex items-center gap-2 text-xs text-stone-500">
   733→        {savingsPct !== null && period === 'yearly' ? (
   734→          <Badge variant="outline" className="border-emerald-200 bg-emerald-50 text-emerald-800">
   735→            {t('billing.savings', { pct: savingsPct })}
   736→          </Badge>
   737→        ) : null}
   738→        {plan.trial_days ? (
   739→          <span>
   740→            {plan.trial_days}d · {t('billing.statusTrialing').toLowerCase()}
   741→          </span>
   742→        ) : null}
   743→      </div>
   744→
   745→      <ul className="mt-3 flex-1 space-y-1">
   746→        {ENTITLEMENT_KEYS.map(({ key, labelKey }) => {
   747→          const value = plan.entitlements[key]
   748→          if (value === undefined) return null
   749→          if (typeof value === 'boolean') {
   750→            return (
   751→              <li key={key} className="flex items-center gap-1.5 text-xs">
   752→                {value ? (
   753→                  <Check className="h-3.5 w-3.5 text-emerald-700" aria-hidden />
   754→                ) : (
   755→                  <X className="h-3.5 w-3.5 text-stone-400" aria-hidden />
   756→                )}
   757→                <span className={value ? 'text-stone-700' : 'text-stone-400'}>{t(`billing.${labelKey}`)}</span>
   758→              </li>
   759→            )
   760→          }
   761→          return (
   762→            <li key={key} className="flex items-center gap-1.5 text-xs text-stone-700">
   763→              <Check className="h-3.5 w-3.5 text-emerald-700" aria-hidden />
   764→              <span>
   765→                {t(`billing.${labelKey}`)}:{' '}
   766→                <span className="font-medium">{value === -1 ? t('common.unlimited') : value}</span>
   767→              </span>
   768→            </li>
   769→          )
   770→        })}
   771→      </ul>
   772→
   773→      <div className="mt-4">
   774→        {isCurrent ? (
   775→          <Button variant="outline" className="w-full" disabled>
   776→            {t('billing.currentPlanBadge')}
   777→          </Button>
   778→        ) : selfServe ? (
   779→          <Button
   780→            className="w-full bg-emerald-700 text-white hover:bg-emerald-800"
   781→            disabled={busy}
   782→            onClick={onSelect}
   783→          >
   784→            {hasLiveSubscription ? t('billing.changeAction') : t('billing.startAction')}
   785→          </Button>
   786→        ) : (
   787→          <Tooltip>
   788→            <TooltipTrigger asChild>
   789→              <a
   790→                href="mailto:sales@croptwist.example?subject=Plan%20quote"
   791→                className="inline-flex h-9 w-full items-center justify-center gap-1.5 rounded-md border border-stone-300 bg-white px-3 text-sm font-medium text-stone-700 transition-colors hover:bg-stone-50"
   792→              >
   793→                {t('billing.contactSales')}
   794→                <ExternalLink className="h-3.5 w-3.5" aria-hidden />
   795→              </a>
   796→            </TooltipTrigger>
   797→            <TooltipContent className="max-w-56">
   798→              <p>{t('billing.contactSalesTip')}</p>
   799→            </TooltipContent>
   800→          </Tooltip>
   801→        )}
   802→      </div>
   803→    </div>
   804→  )
   805→}

   807→function UsageRow({ t, item }: { t: T; item: { key: string; limit: number; unlimited: boolean; used: number } }) {
   808→  const labelKey = ENTITLEMENT_KEYS.find((e) => e.key === item.key)?.labelKey
   809→  const label = labelKey ? t(`billing.${labelKey}`) : item.key
   810→  const pctValue =
   811→    item.unlimited || item.limit <= 0 ? null : Math.min(100, Math.round((item.used / item.limit) * 100))
   812→  return (
   813→    <div>
   814→      <div className="flex items-baseline justify-between gap-2 text-sm">
   815→        <span className="text-stone-700">{label}</span>
   816→        {item.unlimited ? (
   817→          <span className="text-xs font-medium text-stone-500">{t('common.unlimited')}</span>
   818→        ) : (
   819→          <span className="text-xs text-stone-600">
   820→            {t('billing.usageOf', { used: item.used, limit: item.limit })}
   821→          </span>
   822→        )}
   823→      </div>
   824→      {!item.unlimited && pctValue !== null ? <Progress className="mt-1 h-2" value={pctValue} aria-label={label} /> : null}
   825→    </div>
   826→  )
