    63→function apiErrorText(err: unknown): string {
    64→  if (err instanceof ApiError) {
    65→    const detail = err.details[0]
    66→    const detailText = detail?.reason ? `${detail.field}: ${detail.reason}` : detail?.field
    67→    return detailText ? `${err.message} (${detailText})` : err.message
    68→  }
    69→  return err instanceof Error ? err.message : 'Unexpected error'
    70→}
    71→
    72→function shortHash(hash: string | null | undefined): string {
    73→  return hash ? `${hash.slice(0, 12)}…` : '—'
    74→}
    75→
    76→function shortIso(iso: string | null | undefined): string {
    77→  return iso ? iso.replace('T', ' ').slice(0, 16) : '—'
    78→}
    79→
    80→function fmt(value: number | null | undefined, digits = 3): string {
    81→  if (value === null || value === undefined) return '—'
    82→  if (Number.isInteger(value)) return String(value)
    83→  return value.toFixed(digits)
    84→}

   139→function TransformationPreview({ scenario }: { scenario: Scenario }) {
   140→  const config = scenario.canonical_configuration as Record<string, unknown> | null
   141→  const adjustments =
   142→    (config?.adjustments as Record<string, number | null> | undefined) ?? {}
   143→  const entries = useMemo(
   144→    () =>
   145→      Object.entries(adjustments).filter(([, v]) => v !== null && v !== undefined) as Array<
   146→        [string, number]
   147→      >,
   148→    [adjustments],
   149→  )
   150→  if (entries.length === 0) return null
   151→  const labels: Record<string, string> = {
   152→    temperature_delta_c: 'Temperature',
   153→    precipitation_multiplier: 'Precipitation',
   154→    precipitation_delta_mm: 'Precipitation shift',
   155→    water_allocation_multiplier: 'Irrigation allocation',
   156→    seasonal_timing_shift_days: 'Seasonal timing shift',
   157→    rainfall_period_shift_days: 'Rainfall period shift',
   158→  }
   159→  const fmtValue = (key: string, v: number) => {
   160→    switch (key) {
   161→      case 'temperature_delta_c':
   162→        return `${v > 0 ? '+' : ''}${v} °C`
   163→      case 'precipitation_multiplier':
   164→        return `×${v}`
   165→      case 'precipitation_delta_mm':
   166→        return `${v > 0 ? '+' : ''}${v} mm`
   167→      case 'water_allocation_multiplier':
   168→        return `×${v}`
   169→      default:
   170→        return `${v > 0 ? '+' : ''}${v} days`
   171→    }
   172→  }
   173→  return (
   174→    <div className="rounded-md border border-amber-200 bg-amber-50 p-3">
   175→      <p className="text-xs font-medium text-amber-900">
   176→        <FlaskConical className="mr-1 inline h-3.5 w-3.5" aria-hidden />
   177→        Hypothetical assumptions to apply (never observed data)
   178→      </p>
   179→      <ul className="mt-1.5 flex flex-wrap gap-x-4 gap-y-1 text-xs text-amber-900">
   180→        {entries.map(([key, value]) => (
   181→          <li key={key} className="font-mono">
   182→            {labels[key] ?? key}: <strong>{fmtValue(key, value)}</strong>
   183→          </li>
   184→        ))}
   185→      </ul>
   186→    </div>
   187→  )
   188→}

   194→function ComparisonTable({ rows }: { rows: ScenarioDerivedFeature[] }) {
   195→  const [filter, setFilter] = useState<'all' | 'scenario_recalculated' | 'not_evaluated' | 'not_affected'>('all')
   196→  const filtered = useMemo(
   197→    () => (filter === 'all' ? rows : rows.filter((r) => r.evidence_status === filter)),
   198→    [rows, filter],
   199→  )
   200→  const filters: Array<{ key: typeof filter; label: string }> = [
   201→    { key: 'all', label: `All (${rows.length})` },
   202→    {
   203→      key: 'scenario_recalculated',
   204→      label: `Scenario-derived (${rows.filter((r) => r.evidence_status === 'scenario_recalculated').length})`,
   205→    },
   206→    {
   207→      key: 'not_evaluated',
   208→      label: `Not evaluated (${rows.filter((r) => r.evidence_status === 'not_evaluated').length})`,
   209→    },
   210→    {
   211→      key: 'not_affected',
   212→      label: `Not affected (${rows.filter((r) => r.evidence_status === 'not_affected').length})`,
   213→    },
   214→  ]
   215→  return (
   216→    <div>
   217→      <div className="mb-2 flex flex-wrap gap-1.5">
   218→        {filters.map((f) => (
   219→          <Button
   220→            key={f.key}
   221→            size="sm"
   222→            variant={filter === f.key ? 'default' : 'outline'}
   223→            className="min-h-8 px-2.5 text-xs"
   224→            onClick={() => setFilter(f.key)}
   225→          >
   226→            {f.label}
   227→          </Button>
   228→        ))}
   229→      </div>
   230→      <div className="max-h-96 overflow-auto rounded-md border border-stone-200">
   231→        <Table>
   232→          <TableHeader className="sticky top-0 bg-white">
   233→            <TableRow>
   234→              <TableHead className="min-w-52">Feature</TableHead>
   235→              <TableHead>Level / period</TableHead>
   236→              <TableHead className="text-right">Baseline</TableHead>
   237→              <TableHead className="text-right">Scenario</TableHead>
   238→              <TableHead className="text-right">Δ</TableHead>
   239→              <TableHead className="text-right">Δ %</TableHead>
   240→              <TableHead>Evidence</TableHead>
   241→            </TableRow>
   242→          </TableHeader>
   243→          <TableBody>
   244→            {filtered.map((r) => (
   245→              <TableRow key={r.id} className={r.evidence_status !== 'scenario_recalculated' ? 'bg-stone-50/60' : ''}>
   246→                <TableCell className="text-xs">
   247→                  <span className="font-medium text-stone-800">{r.feature_name || r.feature_code}</span>
   248→                  <span className="block font-mono text-[10px] text-stone-400">{r.feature_code}</span>
   249→                  {r.uncertainty_metadata?.reason && (
   250→                    <span className="mt-0.5 block max-w-72 text-[10px] leading-snug text-stone-500">
   251→                      {r.uncertainty_metadata.reason}
   252→                    </span>
   253→                  )}
   254→                </TableCell>
   255→                <TableCell className="text-xs text-stone-600">
   256→                  {r.aggregation_level}
   257→                  <span className="block text-[10px] text-stone-400">{r.period_label}</span>
   258→                </TableCell>
   259→                <TableCell className="text-right font-mono text-xs">
   260→                  {r.evidence_status === 'scenario_recalculated' ? fmt(r.baseline_value) : '—'}
   261→                </TableCell>
   262→                <TableCell className="text-right font-mono text-xs">
   263→                  {r.evidence_status === 'scenario_recalculated' ? fmt(r.scenario_value) : '—'}
   264→                </TableCell>
   265→                <TableCell className="text-right font-mono text-xs">
   266→                  {r.difference_value === null ? '—' : fmt(r.difference_value)}
   267→                </TableCell>
   268→                <TableCell className="text-right font-mono text-xs">
   269→                  {r.difference_percent === null ? '—' : `${fmt(r.difference_percent, 2)}%`}
   270→                </TableCell>
   271→                <TableCell>
   272→                  <EvidenceStatusBadge status={r.evidence_status} />
   273→                </TableCell>
   274→              </TableRow>
   275→            ))}
   276→            {filtered.length === 0 && (
   277→              <TableRow>
   278→                <TableCell colSpan={7} className="py-6 text-center text-xs text-stone-500">
   279→                  No rows for this filter.
   280→                </TableCell>
   281→              </TableRow>
   282→            )}
   283→          </TableBody>
   284→        </Table>
   285→      </div>
   286→      <p className="mt-1.5 text-[11px] text-stone-500">
   287→        Δ% is shown only when the baseline is non-zero and the comparison is valid. Scenario values are
   288→        hypothetical — never observations.
   289→      </p>
   290→    </div>
   291→  )
   292→}

   298→function EvidenceExplorer({ runId }: { runId: string }) {
   299→  const [observations, setObservations] = useState<ScenarioDerivedObservation[] | null>(null)
   300→  const [total, setTotal] = useState<number | null>(null)
   301→  const [error, setError] = useState<string | null>(null)
   302→  const [loading, setLoading] = useState(false)
   303→
   304→  const load = async () => {
   305→    setLoading(true)
   306→    setError(null)
   307→    try {
   308→      const data = await listExecutionObservations(runId, { limit: 100 })
   309→      setObservations(data.items)
   310→      setTotal(data.total_transformed)
   311→    } catch (err) {
   312→      setError(apiErrorText(err))
   313→    } finally {
   314→      setLoading(false)
   315→    }
   316→  }
   317→
   318→  return (
   319→    <div className="rounded-md border border-stone-200 p-3">
   320→      <div className="flex flex-wrap items-center justify-between gap-2">
   321→        <p className="text-xs font-medium text-stone-700">
   322→          Evidence explorer — transformed observations{' '}
   323→          {total !== null && <span className="font-normal text-stone-500">({total} total, first 100)</span>}
   324→        </p>
   325→        {observations === null ? (
   326→          <Button size="sm" variant="outline" className="min-h-8 text-xs" onClick={load} disabled={loading}>
   327→            {loading ? <Loader2 className="mr-1 h-3 w-3 animate-spin" aria-hidden /> : null}
   328→            Load evidence
   329→          </Button>
   330→        ) : null}
   331→      </div>
   332→      {error && <p className="mt-2 text-xs text-red-700">{error}</p>}
   333→      {observations && (
   334→        <div className="mt-2 max-h-80 overflow-auto rounded border border-stone-100">
   335→          <Table>
   336→            <TableHeader className="sticky top-0 bg-white">
   337→              <TableRow>
   338→                <TableHead>Variable</TableHead>
   339→                <TableHead className="text-right">Baseline</TableHead>
   340→                <TableHead className="text-right">Transformed</TableHead>
   341→                <TableHead>Unit</TableHead>
   342→                <TableHead>Transformation</TableHead>
   343→                <TableHead>Period</TableHead>
   344→                <TableHead>Evidence</TableHead>
   345→              </TableRow>
   346→            </TableHeader>
   347→            <TableBody>
   348→              {observations.map((o) => (
   349→                <TableRow key={o.id}>
   350→                  <TableCell className="font-mono text-[11px]">{o.variable_code}</TableCell>
   351→                  <TableCell className="text-right font-mono text-[11px]">{fmt(o.baseline_value)}</TableCell>
   352→                  <TableCell className="text-right font-mono text-[11px]">{fmt(o.transformed_value)}</TableCell>
   353→                  <TableCell className="text-[11px]">{o.unit}</TableCell>
   354→                  <TableCell className="font-mono text-[10px] text-stone-500">
   355→                    {o.transformation_code} {JSON.stringify(o.transformation_parameter)}
   356→                  </TableCell>
   357→                  <TableCell className="text-[10px] text-stone-500">{shortIso(o.period_start)}</TableCell>
   358→                  <TableCell>
   359→                    <ObsEvidenceBadge evidence={o.evidence_class} synthetic={o.is_synthetic} />
   360→                  </TableCell>
   361→                </TableRow>
   362→              ))}
   363→            </TableBody>
   364→          </Table>
   365→        </div>
   366→      )}
   367→    </div>
   368→  )
   369→}

   375→function RunDetail({ run }: { run: ScenarioExecutionRun }) {
   376→  const [comparison, setComparison] = useState<ScenarioExecutionCompareResponse | null>(null)
   377→  const [verify, setVerify] = useState<ScenarioExecutionVerifyResponse | null>(null)
   378→  const [error, setError] = useState<string | null>(null)
   379→  const [busy, setBusy] = useState<string | null>(null)
   380→
   381→  const runAction = async (key: string, fn: () => Promise<void>) => {
   382→    setBusy(key)
   383→    setError(null)
   384→    try {
   385→      await fn()
   386→    } catch (err) {
   387→      setError(apiErrorText(err))
   388→    } finally {
   389→      setBusy(null)
   390→    }
   391→  }
   392→
   393→  const isResult = run.run_status === 'succeeded' || run.run_status === 'completed_with_limitations'
   394→
   395→  return (
   396→    <div className="mt-3 space-y-3 rounded-md border border-stone-200 bg-stone-50/60 p-3">
   397→      {run.run_status === 'failed' && run.error_summary && (
   398→        <Alert variant="destructive">
   399→          <ShieldAlert className="h-4 w-4" aria-hidden />
   400→          <AlertTitle>Execution failed</AlertTitle>
   401→          <AlertDescription className="break-all">{run.error_summary}</AlertDescription>
   402→        </Alert>
   403→      )}
   404→      <div className="grid gap-x-4 gap-y-1 text-xs text-stone-600 sm:grid-cols-2 lg:grid-cols-3">
   405→        <div>
   406→          <span className="text-stone-400">Output checksum: </span>
   407→          <span className="break-all font-mono" title={run.output_checksum ?? ''}>
   408→            {shortHash(run.output_checksum)}
   409→          </span>
   410→        </div>
   411→        <div>
   412→          <span className="text-stone-400">Input fingerprint: </span>
   413→          <span className="break-all font-mono">{shortHash(run.input_fingerprint)}</span>
   414→        </div>
   415→        <div>
   416→          <span className="text-stone-400">Engines: </span>
   417→          {run.transformation_engine_version} · {run.feature_engine_version}
   418→        </div>
   419→        <div>
   420→          <span className="text-stone-400">Feature spec: </span>
   421→          {run.feature_catalog_version}
   422→        </div>
   423→        <div>
   424→          <span className="text-stone-400">Window: </span>
   425→          {shortIso(run.execution_configuration.period_start)} →{' '}
   426→          {shortIso(run.execution_configuration.period_end)} ({run.calendar_timezone})
   427→        </div>
   428→        <div>
   429→          <span className="text-stone-400">Parent snapshot: </span>
   430→          <span className="break-all font-mono">{shortHash(run.parent_snapshot_checksum)}</span>
   431→        </div>
   432→      </div>
   433→
   434→      {run.warnings.length > 0 && (
   435→        <div className="rounded-md border border-amber-200 bg-amber-50 p-2.5">
   436→          <p className="text-xs font-medium text-amber-900">Limitations &amp; disclosures</p>
   437→          <ul className="mt-1 list-disc space-y-0.5 pl-4 text-[11px] text-amber-900">
   438→            {run.warnings.map((w, i) => (
   439→              <li key={i} className="[overflow-wrap:anywhere]">
   440→                {w}
   441→              </li>
   442→            ))}
   443→          </ul>
   444→        </div>
   445→      )}
   446→
   447→      {Object.keys(run.not_evaluated_reasons).length > 0 && (
   448→        <details className="rounded-md border border-stone-200 bg-white p-2.5">
   449→          <summary className="cursor-pointer text-xs font-medium text-stone-700">
   450→            Not evaluated ({Object.keys(run.not_evaluated_reasons).length}) — with reasons
   451→          </summary>
   452→          <ul className="mt-1.5 space-y-1 text-[11px] text-stone-600">
   453→            {Object.entries(run.not_evaluated_reasons).map(([code, reason]) => (
   454→              <li key={code} className="[overflow-wrap:anywhere]">
   455→                <span className="font-mono text-stone-500">{code}</span> — {reason}
   456→              </li>
   457→            ))}
   458→          </ul>
   459→        </details>
   460→      )}
   461→
   462→      {error && (
   463→        <Alert variant="destructive">
   464→          <ShieldAlert className="h-4 w-4" aria-hidden />
   465→          <AlertDescription>{error}</AlertDescription>
   466→        </Alert>
   467→      )}
   468→
   469→      <div className="flex flex-wrap gap-2">
   470→        <Button
   471→          size="sm"
   472→          variant="outline"
   473→          className="min-h-8 text-xs"
   474→          disabled={!isResult || busy !== null}
   475→          onClick={() =>
   476→            runAction('compare', async () => {
   477→              const data = await compareScenarioExecution(run.run_id)
   478→              setComparison(data)
   479→            })
   480→          }
   481→        >
   482→          Baseline vs scenario
   483→        </Button>
   484→        <Button
   485→          size="sm"
   486→          variant="outline"
   487→          className="min-h-8 text-xs"
   488→          disabled={!isResult || busy !== null}
   489→          onClick={() =>
   490→            runAction('verify', async () => {
   491→              const data = await verifyScenarioExecution(run.run_id)
   492→              setVerify(data)
   493→            })
   494→          }
   495→        >
   496→          Verify checksums
   497→        </Button>
   498→      </div>
   499→
   500→      {verify && (
   501→        <div className="rounded-md border border-stone-200 bg-white p-2.5 text-xs">
   502→          <p className="flex items-center gap-1.5 font-medium text-stone-700">
   503→            {verify.output_checksum_valid ? (
   504→              <ShieldCheck className="h-3.5 w-3.5 text-emerald-700" aria-hidden />
   505→            ) : (
   506→              <ShieldAlert className="h-3.5 w-3.5 text-red-700" aria-hidden />
   507→            )}
   508→            Output checksum valid: {String(verify.output_checksum_valid)} · definition checksum valid:{' '}
   509→            {String(verify.scenario_definition_checksum_valid)} · parent intact:{' '}
   510→            {String(verify.parent_snapshot_intact)} · manifest pinned:{' '}
   511→            {String(verify.manifest_checksum_matches_run)}
   512→          </p>
   513→          <p className="mt-1 text-[11px] text-stone-500">{verify.note}</p>
   514→        </div>
   515→      )}
   516→
   517→      {comparison && (
   518→        <div className="space-y-3">
   519→          <p className="text-[11px] text-stone-500">{comparison.note}</p>
   520→          <ComparisonTable rows={comparison.rows} />
   521→          <EvidenceExplorer runId={run.run_id} />
   522→        </div>
   523→      )}
   524→    </div>
   525→  )
   526→}

   542→            <ChevronDown className="h-3.5 w-3.5 shrink-0 text-stone-400" aria-hidden />
   543→          )}
   544→          <span className="min-w-0 flex-1">
   545→            <span className="block break-all font-mono text-[11px] text-stone-500">
   546→              {run.run_id}
   547→            </span>
   548→            <span className="text-xs text-stone-600">
   549→              {shortIso(run.completed_at || run.started_at)} · {run.transformed_observation_count} transformed obs
   550→              · {run.derived_feature_count} feature rows
   551→            </span>

   562→// Main section (per finalized scenario)
   563→// ---------------------------------------------------------------------------
   564→
   565→const EXECUTABLE_TYPES = new Set(['user_defined_stress', 'historical_perturbation'])
   566→
   567→export function ScenarioExecutionSection({ scenario }: { scenario: Scenario }) {
   568→  const [runs, setRuns] = useState<ScenarioExecutionRun[] | null>(null)
   569→  const [busy, setBusy] = useState(false)
   570→  const [error, setError] = useState<string | null>(null)
   571→
   572→  const isExecutable =
   573→    scenario.scenario_status === 'finalized' && EXECUTABLE_TYPES.has(scenario.scenario_type)
   574→
   575→  const loadRuns = async () => {
   576→    setError(null)
   577→    try {
   578→      const data = await listScenarioExecutions(scenario.id)
   579→      setRuns(data.items)
   580→    } catch (err) {
   581→      setError(apiErrorText(err))
   582→    }
   583→  }
   584→
   585→  const execute = async () => {
   586→    setBusy(true)
   587→    setError(null)
   588→    try {
   589→      await executeScenario(scenario.id, {})
   590→      const data = await listScenarioExecutions(scenario.id)
   591→      setRuns(data.items)
   592→    } catch (err) {
   593→      setError(apiErrorText(err))
   594→    } finally {
   595→      setBusy(false)
   596→    }
   597→  }
   598→
   599→  return (
   600→    <div className="mt-3 space-y-2.5 rounded-md border border-stone-200 bg-white p-3">
   601→      <div className="flex flex-wrap items-center justify-between gap-2">
   602→        <p className="text-sm font-medium text-stone-800">Execute scenario (M7.3)</p>
   603→        <div className="flex gap-2">
   604→          <Button
   605→            size="sm"
   606→            variant="outline"
   607→            className="min-h-8 text-xs"
   608→            onClick={loadRuns}
   609→            disabled={busy}
   610→          >
   611→            Refresh executions
   612→          </Button>
   613→          <Button
   614→            size="sm"
   615→            className="min-h-8 text-xs"
   616→            disabled={!isExecutable || busy}
   617→            onClick={execute}
   618→            title={
   619→              isExecutable
   620→                ? 'Execute the configured transformations against the parent Field State'
   621→                : 'Only finalized stress/perturbation scenarios can execute'
   622→            }
   623→          >
   624→            {busy ? <Loader2 className="mr-1 h-3 w-3 animate-spin" aria-hidden /> : <Play className="mr-1 h-3 w-3" aria-hidden />}
   625→            Execute
   626→          </Button>
   627→        </div>
   628→      </div>
   629→
   630→      <TransformationPreview scenario={scenario} />
   631→
   632→      <Alert className="border-stone-200 bg-stone-50">
   633→        <Info className="h-4 w-4 text-stone-500" aria-hidden />
   634→        <AlertDescription className="text-[11px] leading-snug text-stone-600">
   635→          Execution applies the hypothetical transformations to the parent Field State&apos;s pinned
   636→          observations and recomputes affected environmental features. It is a{' '}
   637→          <strong>derived hypothetical calculation</strong> — results never enter the observation
   638→          record, parent artifacts stay immutable, and ET0-related features are never reused as
   639→          scenario outputs.
   640→        </AlertDescription>
   641→      </Alert>
   642→
   643→      {error && (
   644→        <Alert variant="destructive">
   645→          <CircleAlert className="h-4 w-4" aria-hidden />
   646→          <AlertDescription>{error}</AlertDescription>
   647→        </Alert>
   648→      )}
   649→
   650→      {runs !== null && runs.length === 0 && (
   651→        <p className="text-xs text-stone-500">No executions yet for this scenario.</p>
   652→      )}
   653→      {runs !== null && runs.length > 0 && (
   654→        <div className="space-y-2">
   655→          {runs.map((r) => (
   656→            <ExecutionRunRow key={r.run_id} run={r} />
   657→          ))}
   658→        </div>
   659→      )}
   660→    </div>
   661→  )
   662→}
